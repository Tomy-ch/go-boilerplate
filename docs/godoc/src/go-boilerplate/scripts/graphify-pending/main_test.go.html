    24  // captureLog は標準ロガーの出力を捕まえます。未処理の内訳は戻り値ではなくログにしか出ないため、
    25  // そこを検査するにはここを通します。
    26  func captureLog(t *testing.T) *bytes.Buffer {
    27  	t.Helper()
    28  	buffer := &bytes.Buffer{}
    29  	previous := log.Writer()
    30  	flags := log.Flags()
    31  	log.SetOutput(buffer)
    32  	log.SetFlags(0)
    33  	t.Cleanup(func() {
    34  		log.SetOutput(previous)
    35  		log.SetFlags(flags)
    36  	})
    37  
    38  	return buffer
    39  }

    41  // staticRead は内容を固定で返す読み出し手段を作ります。未登録のパスは存在しない扱いです。
    42  func staticRead(bodies map[string]string) func(string) ([]byte, error) {
    43  	return func(name string) ([]byte, error) {
    44  		body, ok := bodies[name]
    45  		if !ok {
    46  			return nil, os.ErrNotExist
    47  		}
    48  
    49  		return []byte(body), nil
    50  	}
    51  }

    53  // staticGit は git のサブコマンドごとに固定の出力を返す実行手段を作ります。
    54  func staticGit(log, diff string) func(...string) (string, error) {
    55  	return func(args ...string) (string, error) {
    56  		if len(args) > 0 && args[0] == "log" {
    57  			return log, nil
    58  		}
    59  
    60  		return diff, nil
    61  	}
    62  }
