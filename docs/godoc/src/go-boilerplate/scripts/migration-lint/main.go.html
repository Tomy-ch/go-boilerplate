     1  // Package main は database/migrations の連番を検査するツール。
     2  //
     3  //	-check duplicate: 同じ連番を持つマイグレーションが複数ないかを調べる。
     4  //	-check gap:       連番に欠番がないかを調べる。
     5  //
     6  // 検査対象は `<連番>_<名前>.<kind>.sql` の連番部分（最初の `_` より前）で、
     7  // up / down のどちらを見るかは -kind で切り替える。
     8  //
     9  // このツールは lefthook の pre-commit ゲートから呼ばれる。壊れ方が
    10  // 「何も検査しなくなる」方向に出るため、判定ロジックはシェルの中ではなく
    11  // テストの当たる Go 側に置く。
    12  //
    13  // マイグレーションが 1 件も無い場合は正常終了する。サンプル API の削除
    14  // (make setup-remove-sample-api) がマイグレーションごと消し得るため、
    15  // 「ファイルが無い」ことをここでの失敗にはしない。

    54  // run は、指定された検査をマイグレーションの連番に対して実行します。
    55  func run(args []string) error {
    56  	fs := flag.NewFlagSet("migration-lint", flag.ContinueOnError)
    57  	kind := fs.String("kind", "up", "検査するマイグレーションの種別 (up / down)")
    58  	check := fs.String("check", "duplicate", "検査内容 (duplicate / gap)")
    59  	dir := fs.String("dir", "database/migrations", "マイグレーションディレクトリ")
    60  
    61  	if err := fs.Parse(args); err != nil {
    62  		// ヘルプ要求は失敗ではないので 0 で終える。usage は flag が既に出力している。
    63  		if xerrors.Is(err, flag.ErrHelp) {
    64  			return nil
    65  		}
    66  
    67  		return xerrors.Wrap(err, "failed to parse flags")
    68  	}
    69  
    70  	versions, err := collectVersions(*dir, *kind)
    71  	if err != nil {
    72  		return err
    73  	}
    74  
    75  	var problem string
    76  
    77  	switch *check {
    78  	case "duplicate":
    79  		problem = reportDuplicates(*kind, versions)
    80  	case "gap":
    81  		problem = reportGaps(*kind, versions)
    82  	default:
    83  		return xerrors.Wrap(errUnknownCheck, *check)
    84  	}
    85  
    86  	if problem == "" {
    87  		return nil
    88  	}
    89  
    90  	// 報告文は欠番の一覧を含む複数行なので、エラーのメッセージへ畳み込まずそのまま出す。
    91  	// 畳み込むと sentinel の文言が最終行の末尾に付き、並べた連番が読めなくなる。
    92  	log.Print(problem)
    93  
    94  	return errNumberingProblem
    95  }

    97  // collectVersions は、指定種別のマイグレーションファイルから連番部分を昇順で返します。
    98  // 連番は先頭ゼロ埋めの固定幅なので、辞書順の並びは数値順と一致します。
    99  func collectVersions(dir, kind string) ([]string, error) {
   100  	matches, err := filepath.Glob(filepath.Join(dir, "*."+kind+".sql"))
   101  	if err != nil {
   102  		return nil, xerrors.Wrap(err, "failed to scan "+dir)
   103  	}
   104  
   105  	versions := make([]string, 0, len(matches))
   106  
   107  	for _, m := range matches {
   108  		base := filepath.Base(m)
   109  
   110  		version, _, found := strings.Cut(base, "_")
   111  		if !found {
   112  			return nil, xerrors.Wrap(errNoVersionPrefix, base)
   113  		}
   114  
   115  		versions = append(versions, version)
   116  	}
   117  
   118  	sort.Strings(versions)
   119  
   120  	return versions, nil
   121  }

   123  // reportDuplicates は、重複する連番があればその報告文を返します（無ければ空文字）。
   124  func reportDuplicates(kind string, versions []string) string {
   125  	seen := make(map[string]int, len(versions))
   126  	duplicates := make([]string, 0)
   127  
   128  	for _, v := range versions {
   129  		seen[v]++
   130  
   131  		if seen[v] == duplicateThreshold {
   132  			duplicates = append(duplicates, v)
   133  		}
   134  	}
   135  
   136  	if len(duplicates) == 0 {
   137  		return ""
   138  	}
   139  
   140  	return fmt.Sprintf("Duplicate migration numbers (%s): %s", kind, strings.Join(duplicates, " "))
   141  }
   142  
   143  // reportGaps は、連番に欠番があればその報告文を返します（無ければ空文字）。
   144  // 期待値は最小値から最大値までの連番を、最小値と同じ桁数へゼロ埋めしたものです。
   145  func reportGaps(kind string, versions []string) string {
   146  	if len(versions) == 0 {
   147  		return ""
   148  	}
   149  
   150  	expected, err := expectedSequence(versions)
   151  	if err != nil {
   152  		return fmt.Sprintf("Migration version is not numeric (%s): %v", kind, err)
   153  	}
   154  
   155  	if strings.Join(versions, "\n") == strings.Join(expected, "\n") {
   156  		return ""
   157  	}
   158  
   159  	return fmt.Sprintf(
   160  		"Migration version gap detected (%s)\nExisting :\n%s\nExpected :\n%s",
   161  		kind, strings.Join(versions, "\n"), strings.Join(expected, "\n"),
   162  	)
   163  }
   164  
   165  // expectedSequence は、昇順の連番列から「欠番の無い状態」を組み立てます。
   166  func expectedSequence(versions []string) ([]string, error) {
   167  	width := len(versions[0])
   168  
   169  	first, err := strconv.Atoi(versions[0])
   170  	if err != nil {
   171  		return nil, xerrors.Wrap(err, "invalid version "+versions[0])
   172  	}
   173  
   174  	last, err := strconv.Atoi(versions[len(versions)-1])
   175  	if err != nil {
   176  		return nil, xerrors.Wrap(err, "invalid version "+versions[len(versions)-1])
   177  	}
   178  
   179  	expected := make([]string, 0, last-first+1)
   180  	for i := first; i <= last; i++ {
   181  		expected = append(expected, fmt.Sprintf("%0*d", width, i))
   182  	}
   183  
   184  	return expected, nil
   185  }
