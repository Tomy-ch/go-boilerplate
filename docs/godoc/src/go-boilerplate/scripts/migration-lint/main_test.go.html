    14  // writeMigrations は、指定した名前の空マイグレーションファイルを持つ一時ディレクトリを作ります。
    15  func writeMigrations(t *testing.T, names ...string) string {
    16  	t.Helper()
    17  
    18  	dir := t.TempDir()
    19  
    20  	for _, name := range names {
    21  		require.NoError(t, os.WriteFile(filepath.Join(dir, name), []byte(""), 0o600))
    22  	}
    23  
    24  	return dir
    25  }
