     1  // Package main は package-lock.json のエントリが、その lockfile 自身の
     2  // .npmrc `min-release-age` を満たしているかを監査するツール。
     3  //
     4  //	audit: 各 package-lock.json を走査し、公開から min-release-age 日未満の
     5  //	       バージョンが記録されていないかを調べる。
     6  //
     7  // npm の `min-release-age` は依存解決（npm install / update）時にだけ効き、
     8  // lockfile を再現するだけの `npm ci` では評価されない。本リポジトリの CI と
     9  // イメージビルドは全て `npm ci` なので、cooldown を破って作られた lockfile は
    10  // CI 上では何の異常も示さずに通る。このツールはその死角を埋める。
    11  //
    12  // 検出＝バイパスの証拠になる。cooldown 下の `npm install` は窓内バージョンの
    13  // 解決自体を拒否するため、窓内エントリが lockfile に載る経路は
    14  // 「cooldown を明示的に外して解決した」以外に存在しない。誤検知はほぼ出ない。
    15  //
    16  // **本ツールは finding があっても正常終了する。** cooldown のバイパスは
    17  // テックリード / アーキテクトの専任判断であり、CRITICAL への即応など正当な
    18  // 行使を CI が塞ぐのは誤り。非ゼロ終了はツール自身の失敗（取得不能・解析不能）
    19  // に限る。ゲートではなく検知器であることを、ワークフロー側の設定ではなく
    20  // ツールの性質として持たせている。
    21  //
    22  // 守備範囲は「方針のドリフト」に限る。すなわち事故によるバイパス、規約の風化、
    23  // npm 側の挙動変化である。commit 権限を持つ攻撃者はこのツールもワークフローも
    24  // 同じ変更で削除できるため、技術的な防止手段ではない。成立するのは検知と
    25  // attribution までで、そこから先の抑止は組織側の運用に委ねる。
    26  // 強制の担保は CODEOWNERS による lockfile / .npmrc のレビュー必須化が担う。

   100  func run(args []string, client *http.Client, wd func() (string, error)) error {
   101  	if len(args) == 0 || args[0] != "audit" {
   102  		return xerrors.Wrap(errUsage, "❌ usage: npm-cooldown audit [--base=<git-ref>] [--summary-out=<path>] [--github]")
   103  	}
   104  
   105  	fs := flag.NewFlagSet("audit", flag.ContinueOnError)
   106  	base := fs.String("base", "", "この git ref からの差分で追加/変更されたエントリだけを対象にする（未指定なら全件）")
   107  	summaryOut := fs.String("summary-out", "", "Markdown サマリの出力先")
   108  	github := fs.Bool("github", false, "GitHub Actions の ::warning:: アノテーションを出力する")
   109  	if err := fs.Parse(args[1:]); err != nil {
   110  		// ヘルプ要求は失敗ではないので 0 で終える。usage は flag が既に出力している。
   111  		if xerrors.Is(err, flag.ErrHelp) {
   112  			return nil
   113  		}
   114  
   115  		return xerrors.Wrap(err, "❌ フラグを解釈できません")
   116  	}
   117  
   118  	root, err := wd()
   119  	if err != nil {
   120  		return xerrors.Wrap(err, "❌ getwd")
   121  	}
   122  
   123  	locks, err := lockfiles(root)
   124  	if err != nil {
   125  		return xerrors.Wrap(err, "❌ lockfile の走査")
   126  	}
   127  	if len(locks) == 0 {
   128  		log.Printf("ℹ️ package-lock.json が見つかりません")
   129  
   130  		return nil
   131  	}
   132

   134  	if err != nil {
   135  		return xerrors.Wrap(err, "❌ 監査")
   136  	}
   137  
   138  	report(findings, audited, *base, *github)
   139  	if *summaryOut != "" {
   140  		if writeErr := os.WriteFile(*summaryOut, []byte(summary(findings, audited, *base)), summaryPerm); writeErr != nil {
   141  			return xerrors.Wrap(writeErr, "❌ write summary")
   142  		}
   143  	}
   144  	if out := os.Getenv("GITHUB_OUTPUT"); out != "" {
   145  		if writeErr := appendOutput(out, len(findings), audited); writeErr != nil {
   146  			return xerrors.Wrap(writeErr, "❌ write GITHUB_OUTPUT")
   147  		}
   148  	}
   149  
   150  	// finding があっても nil を返す（パッケージ doc の設計意図）。
   151  	return nil
   152  }

   154  // lockfiles は node_modules / vendor 配下を除く package-lock.json を返す。
   155  func lockfiles(root string) ([]string, error) {
   156  	var found []string
   157  	err := filepath.WalkDir(root, func(path string, d os.DirEntry, err error) error {
   158  		if err != nil {
   159  			return err
   160  		}
   161  		rel, perr := lockPath(root, path, d)
   162  		if perr != nil {
   163  			return perr
   164  		}
   165  		if rel != "" {
   166  			found = append(found, rel)
   167  		}
   168  		return nil
   169  	})
   170  	if err != nil {
   171  		return nil, xerrors.Wrap(err, "walk")
   172  	}
   173  	sort.Strings(found)
   174  	return found, nil
   175  }

   177  // lockPath は走査中の 1 エントリを、収集すべき root からの相対パスへ変換する。対象外なら空文字を、
   178  // 枝ごと落とすディレクトリでは filepath.SkipDir を返す。相対化に失敗したものを絶対パスのまま
   179  // 収集すると、以降の root との結合が壊れた場所を指す。
   180  func lockPath(root, path string, d os.DirEntry) (string, error) {
   181  	if d.IsDir() {
   182  		switch d.Name() {
   183  		case "node_modules", "vendor", ".git":
   184  			return "", filepath.SkipDir
   185  		}
   186  		return "", nil
   187  	}
   188  	if d.Name() != "package-lock.json" {
   189  		return "", nil
   190  	}
   191  	return filepath.Rel(root, path)
   192  }

   194  // minReleaseAge は lockfile と同じディレクトリの .npmrc から min-release-age を読む。
   195  // 宣言が無ければ 0 を返す（＝そのlockfileには守るべき cooldown が無い）。
   196  func minReleaseAge(root, lock string) (int, error) {
   197  	p := filepath.Join(root, filepath.Dir(lock), ".npmrc")
   198  	b, err := os.ReadFile(p) //nolint:gosec // path は lockfile の同階層に限定
   199  	if os.IsNotExist(err) {
   200  		return 0, nil
   201  	}
   202  	if err != nil {
   203  		return 0, xerrors.Wrap(err, "read "+p)
   204  	}
   205  	for line := range strings.SplitSeq(string(b), "\n") {
   206  		line = strings.TrimSpace(line)
   207  		if strings.HasPrefix(line, "#") || strings.HasPrefix(line, ";") {
   208  			continue
   209  		}
   210  		k, v, ok := strings.Cut(line, "=")
   211  		if !ok || strings.TrimSpace(k) != "min-release-age" {
   212  			continue
   213  		}
   214  		n, cerr := strconv.Atoi(strings.TrimSpace(v))
   215  		if cerr != nil {
   216  			return 0, xerrors.Wrap(cerr, "parse min-release-age in "+p)
   217  		}
   218  		return n, nil
   219  	}
   220  	return 0, nil
   221  }

   223  // parseLock は lockfile(v2/v3) の packages から実パッケージのエントリを取り出す。
   224  // ルート（キーが空文字）と workspace リンクは対象外。
   225  func parseLock(data []byte) ([]entry, error) {
   226  	var doc struct {
   227  		Packages map[string]struct {
   228  			Version string `json:"version"`
   229  			Link    bool   `json:"link"`
   230  		} `json:"packages"`
   231  	}
   232  	if err := json.Unmarshal(data, &doc); err != nil {
   233  		return nil, xerrors.Wrap(err, "unmarshal lockfile")
   234  	}
   235  	var out []entry
   236  	for path, p := range doc.Packages {
   237  		if path == "" || p.Link || p.Version == "" {
   238  			continue
   239  		}
   240  		idx := strings.LastIndex(path, nodeModulesSep)
   241  		if idx < 0 {
   242  			continue // node_modules 配下でない＝workspace 自身
   243  		}
   244  		out = append(out, entry{path: path, name: path[idx+len(nodeModulesSep):], version: p.Version})
   245  	}
   246  	sort.Slice(out, func(i, j int) bool { return out[i].path < out[j].path })
   247  	return out, nil
   248  }

   250  // changedOnly は base ref 時点の lockfile と比較し、追加/変更された name@version だけを残す。
   251  // base に lockfile が存在しない（新規追加）場合は全件を対象にする。
   252  func changedOnly(ctx context.Context, root, lock, base string, cur []entry) ([]entry, error) {
   253  	cctx, cancel := context.WithTimeout(ctx, gitTimeout)
   254  	defer cancel()
   255  
   256  	cmd := exec.CommandContext(cctx, "git", "show", base+":"+lock) //nolint:gosec // base は CI 由来の ref、lock は走査結果
   257  	cmd.Dir = root
   258  	out, err := cmd.Output()
   259  	if err != nil {
   260  		// base 側に lockfile が無い＝新規追加。全件が「追加分」なのでそのまま返す。
   261  		return cur, nil
   262  	}
   263  	prev, err := parseLock(out)
   264  	if err != nil {
   265  		return nil, err
   266  	}
   267  	known := make(map[string]struct{}, len(prev))
   268  	for _, e := range prev {
   269  		known[e.key()] = struct{}{}
   270  	}
   271  	var diff []entry
   272  	for _, e := range cur {
   273  		if _, ok := known[e.key()]; !ok {
   274  			diff = append(diff, e)
   275  		}
   276  	}
   277  	return diff, nil
   278  }

   280  // auditAll は全 lockfile を監査し、finding と監査したエントリ総数を返す。
   281  func auditAll(ctx context.Context, client *http.Client, root string, locks []string, base string) ([]finding, int, error) {
   282  	var findings []finding
   283  	audited := 0
   284  
   285  	for _, lock := range locks {
   286  		fs, n, err := auditLock(ctx, client, root, lock, base)
   287  		if err != nil {
   288  			return nil, 0, err
   289  		}
   290  		findings = append(findings, fs...)
   291  		audited += n
   292  	}
   293  
   294  	sort.Slice(findings, func(i, j int) bool { return findings[i].ageDays < findings[j].ageDays })
   295  	return findings, audited, nil
   296  }

   298  // auditLock は lockfile 1 本を監査する。min-release-age の宣言が無ければ対象外として何も返さない。
   299  func auditLock(ctx context.Context, client *http.Client, root, lock, base string) ([]finding, int, error) {
   300  	minAge, err := minReleaseAge(root, lock)
   301  	if err != nil {
   302  		return nil, 0, err
   303  	}
   304  	if minAge <= 0 {
   305  		log.Printf("⏭️  %s: .npmrc に min-release-age が無いため対象外", lock)
   306  		return nil, 0, nil
   307  	}
   308  
   309  	entries, err := lockEntries(ctx, root, lock, base)
   310  	if err != nil {
   311  		return nil, 0, err
   312  	}
   313  	if len(entries) == 0 {
   314  		log.Printf("✅ %s: 対象エントリなし", lock)
   315  		return nil, 0, nil
   316  	}
   317  
   318  	times, err := publishTimes(ctx, client, entries)
   319  	if err != nil {
   320  		return nil, 0, err
   321  	}
   322  
   323  	var findings []finding
   324  	for _, e := range entries {
   325  		pub, ok := times[e.key()]
   326  		if !ok || pub.IsZero() {
   327  			continue // registry に公開日が無い（private / 取得不能）。判定不能は報告しない
   328  		}
   329  		if age := int(time.Since(pub).Hours() / hoursPerDay); age < minAge {
   330  			findings = append(findings, finding{lockfile: lock, entry: e, published: pub, ageDays: age, minAge: minAge})
   331  		}
   332  	}
   333  	log.Printf("🔍 %s: %d 件を監査（min-release-age=%d）", lock, len(entries), minAge)
   334  	return findings, len(entries), nil
   335  }

   337  // lockEntries は監査対象のエントリを返す。base 指定時はそこからの追加/変更分に絞る。
   338  func lockEntries(ctx context.Context, root, lock, base string) ([]entry, error) {
   339  	data, err := os.ReadFile(filepath.Join(root, lock)) //nolint:gosec // lockfiles() の走査結果
   340  	if err != nil {
   341  		return nil, xerrors.Wrap(err, "read "+lock)
   342  	}
   343  	entries, err := parseLock(data)
   344  	if err != nil {
   345  		return nil, err
   346  	}
   347  	if base == "" {
   348  		return entries, nil
   349  	}
   350  	return changedOnly(ctx, root, lock, base, entries)
   351  }

   353  // publishTimes は name@version の公開日時を registry から並列取得する。
   354  // packument の .time にのみ公開日があるため version 文書では代替できない。
   355  func publishTimes(ctx context.Context, client *http.Client, entries []entry) (map[string]time.Time, error) {
   356  	byName := map[string][]entry{}
   357  	for _, e := range entries {
   358  		byName[e.name] = append(byName[e.name], e)
   359  	}
   360  
   361  	var (
   362  		mu      sync.Mutex
   363  		result  = map[string]time.Time{}
   364  		firstEr error
   365  		wg      sync.WaitGroup
   366  		sem     = make(chan struct{}, fetchWorkers)
   367  	)
   368  	for name, es := range byName {
   369  		wg.Add(1)
   370  		go func(name string, es []entry) {
   371  			defer wg.Done()
   372  			sem <- struct{}{}
   373  			defer func() { <-sem }()
   374  
   375  			times, err := packumentTimes(ctx, client, name)
   376  			mu.Lock()
   377  			defer mu.Unlock()
   378  			if err != nil {
   379  				if firstEr == nil {
   380  					firstEr = err
   381  				}
   382  				return
   383  			}
   384  			for _, e := range es {
   385  				if t, ok := times[e.version]; ok {
   386  					result[e.key()] = t
   387  				}
   388  			}
   389  		}(name, es)
   390  	}
   391  	wg.Wait()
   392  	if firstEr != nil {
   393  		return nil, firstEr
   394  	}
   395  	return result, nil
   396  }

   398  // packumentTimes は 1 パッケージの全バージョンの公開日時を返す。
   399  func packumentTimes(ctx context.Context, client *http.Client, name string) (map[string]time.Time, error) {
   400  	cctx, cancel := context.WithTimeout(ctx, fetchTimeout)
   401  	defer cancel()
   402  
   403  	req, err := http.NewRequestWithContext(cctx, http.MethodGet, registryBase+name, nil)
   404  	if err != nil {
   405  		return nil, xerrors.Wrap(err, "request "+name)
   406  	}
   407  	resp, err := client.Do(req)
   408  	if err != nil {
   409  		return nil, xerrors.Wrap(err, "fetch "+name)
   410  	}
   411  	defer func() { _ = resp.Body.Close() }()
   412  
   413  	if resp.StatusCode == http.StatusNotFound {
   414  		return map[string]time.Time{}, nil // private / 削除済み。判定不能として扱う
   415  	}
   416  	if resp.StatusCode != http.StatusOK {
   417  		return nil, xerrors.Wrap(errRegistryStatus, fmt.Sprintf("registry %s status=%d", name, resp.StatusCode))
   418  	}
   419  
   420  	var doc struct {
   421  		Time map[string]time.Time `json:"time"`
   422  	}
   423  	if err := json.NewDecoder(resp.Body).Decode(&doc); err != nil {
   424  		return nil, xerrors.Wrap(err, "decode packument "+name)
   425  	}
   426  	return doc.Time, nil
   427  }

   429  // report は人間向け出力と、GitHub Actions のアノテーションを出す。
   430  func report(findings []finding, audited int, base string, github bool) {
   431  	scope := "全件"
   432  	if base != "" {
   433  		scope = base + " からの追加/変更分"
   434  	}
   435  	if len(findings) == 0 {
   436  		log.Printf("✅ cooldown 違反なし（%s / %d 件を監査）", scope, audited)
   437  		return
   438  	}
   439  
   440  	log.Printf("⚠️ cooldown 違反 %d 件（%s / %d 件を監査）", len(findings), scope, audited)
   441  	for _, f := range findings {
   442  		log.Printf("  - %s %s: 公開 %d 日 (< %d) [%s]",
   443  			f.entry.name, f.entry.version, f.ageDays, f.minAge, f.lockfile)
   444  		if github {
   445  			log.Printf("::warning file=%s::%s@%s は公開 %d 日で min-release-age=%d を満たしていません。npm の cooldown を外して解決した lockfile の可能性があります",
   446  				f.lockfile, f.entry.name, f.entry.version, f.ageDays, f.minAge)
   447  		}
   448  	}
   449  }

   451  // summary は PR コメント用の Markdown を組み立てる。
   452  func summary(findings []finding, audited int, base string) string {
   453  	var b strings.Builder
   454  	scope := "全エントリ"
   455  	if base != "" {
   456  		scope = "この変更で追加/変更されたエントリ"
   457  	}
   458  	if len(findings) == 0 {
   459  		fmt.Fprintf(&b, "%s %d 件を監査し、cooldown 違反はありませんでした。\n", scope, audited)
   460  		return b.String()
   461  	}
   462  
   463  	fmt.Fprintf(&b, "%s %d 件のうち %d 件が min-release-age を満たしていません。\n\n", scope, audited, len(findings))
