    41  func GenerateCtxKey(name, typ, importPath, importAlias, outDir, testValue string) error {
    42  	if name == "" || typ == "" {
    43  		return errors.New("name and type are required")
    44  	}
    45  
    46  	if outDir == "" {
    47  		outDir = "."
    48  	}
    49  	outDir = filepath.Clean(outDir)
    50  
    51  	camel, err := toExportedName(name)
    52  	if err != nil {
    53  		return err
    54  	}
    55  
    56  	lower, err := toIdentifierLower(name)
    57  	if err != nil {
    58  		return err
    59  	}
    60  
    61  	// import handling with qualifier alignment
    62  	alias, err := resolveImportAlias(typ, importPath, importAlias)
    63  	if err != nil {
    64  		return err
    65  	}

    67  
    68  	typeExpr := typ
    69  
    70  	success, fail := resolveTestValue(typeExpr, lower, testValue)

    73  		NameLower:        lower,
    74  		NameFlat:         lower,
    75  		NameCamel:        camel,
    76  		Type:             typeExpr,
    77  		ImportPath:       importPath,
    78  		ImportAlias:      importAlias,
    79  		TestSuccessValue: success,
    80  		TestFailValue:    fail,
    81  	}
    82  
    83  	if err := os.MkdirAll(outDir, dirPerm); err != nil {
    84  		return err
    85  	}
    86  
    87  	if err := writeFile(filepath.Join(outDir, p.NameFlat+"_ctx.gen.go"), ctxTpl, p); err != nil {
    88  		return err

    90  
    91  	if err := writeFile(filepath.Join(outDir, p.NameFlat+"_ctx_test.go"), testTpl, p); err != nil {

    94

   120  func toExportedName(s string) (string, error) {
   121  	parts := regexp.MustCompile(`[^\p{L}\p{N}]+`).Split(s, -1)
   122  
   123  	var sb strings.Builder
   124  	for _, p := range parts {
   125  		if p == "" {
   126  			continue
   127  		}
   128  		runes := []rune(p)
   129  		sb.WriteString(strings.ToUpper(string(runes[0])) + string(runes[1:]))
   130  	}
   131  	out := sb.String()
   132  
   133  	if out == "" {
   134  		return "", fmt.Errorf("invalid name: %s", s)
   135  	}
   136  
   137  	if !isValidIdentifier(out) {
   138  		return "", fmt.Errorf("invalid identifier: %s", out)
   139  	}
   140  
   141  	return out, nil
   142  }

   144  func toIdentifierLower(s string) (string, error) {
   145  	// split on non-alnum, join, and lower
   146  	parts := regexp.MustCompile(`[^\p{L}\p{N}]+`).Split(s, -1)
   147  	var sb strings.Builder
   148  	for _, p := range parts {
   149  		if p == "" {
   150  			continue
   151  		}
   152  		sb.WriteString(strings.ToLower(p))
   153  	}
   154  	out := sb.String()
   155  	if out == "" {
   156  		return "", fmt.Errorf("invalid name: %s", s)
   157  	}
   158  	// ensure starts with a letter or '_'
   159  	runes := []rune(out)
   160  	if !unicode.IsLetter(runes[0]) && runes[0] != '_' {
   161  		out = "x" + out
   162  	}
   163  	if !isValidIdentifier(out) {
   164  		return "", fmt.Errorf("invalid identifier: %s", out)
   165  	}
   166  	return out, nil
   167  }

   169  func isValidIdentifier(s string) bool {
   170  	if s == "" {
   171  		return false
   172  	}
   173  	for i, r := range s {
   174  		if i == 0 {
   175  			if !unicode.IsLetter(r) && r != '_' {
   176  				return false
   177  			}
   178  		} else {
   179  			if !unicode.IsLetter(r) && !unicode.IsDigit(r) && r != '_' {
   180  				return false
   181  			}
   182  		}
   183  	}
   184  	return true
   185  }

   187  func sanitizeAlias(s string) string {
   188  	s = strings.ReplaceAll(s, "-", "_")
   189  	s = strings.ReplaceAll(s, ".", "_")
   190  
   191  	if !isValidIdentifier(s) {
   192  		return "pkg"
   193  	}
   194  
   195  	if token.Lookup(s).IsKeyword() {
   196  		return "pkg"
   197  	}
   198

   202  func resolveImportAlias(typ, importPath, importAlias string) (string, error) {
   203  	if importPath == "" {
   204  		return importAlias, nil
   205  	}
   206  
   207  	qualifier := extractQualifier(typ)
   208  
   209  	// No alias provided: prefer qualifier, fallback to last segment
   210  	if importAlias == "" {
   211  		if qualifier != "" {
   212  			return qualifier, nil
   213  		}
   214  		return sanitizeAlias(lastSegment(importPath)), nil
   215  	}
   216  
   217  	// Alias provided: validate against qualifier when present
   218  	if qualifier != "" && qualifier != importAlias {
   219  		return "", fmt.Errorf("type qualifier (%s) does not match alias (%s)", qualifier, importAlias)
   220  	}
   221  
   222  	return importAlias, nil
   223  }

   225  func extractQualifier(t string) string {
   226  	// find last occurrence of identifier followed by '.'
   227  	re := regexp.MustCompile(`[A-Za-z_][A-Za-z0-9_]*\.`)
   228  	matches := re.FindAllString(t, -1)
   229  	if len(matches) == 0 {
   230  		return ""
   231  	}
   232  	last := matches[len(matches)-1]
   233  	return strings.TrimSuffix(last, ".")
   234  }

   236  func lastSegment(path string) string {
   237  	parts := strings.Split(path, "/")
   238  	return parts[len(parts)-1]

   241  func resolveTestValue(t, nameLower, override string) (string, string) {
   242  	switch t {
   243  	case "string":
   244  		return `"test-` + nameLower + `"`, `""`
   245  	case "int":
   246  		return "123", "0"
   247  	case "bool":
   248  		return "true", "false"
   249  	default:
   250  		// 任意型は意味ある success 値を自動合成できない。
   251  		// 呼び出し側が -test-value を渡せば success に採用し、未指定なら従来通り zero 値にフォールバックする。
   252  		fail := "*new(" + t + ")"
   253  		if override != "" {
   254  			return override, fail
   255  		}
   256  		return fail, fail
   257  	}
