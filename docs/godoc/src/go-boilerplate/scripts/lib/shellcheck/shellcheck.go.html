     1  // Package shellcheck は shellcheck の起動と結果の解釈を 1 箇所へ集める。
     2  //
     3  // 呼び出し側は 2 つある。composite action から抽出した run ステップを検査する actions-shellcheck と、
     4  // ファイルとして置かれたシェルを検査する shell-lint である。どちらも渡し方が同じでなければ、同じ
     5  // スクリプトが経路によって違う判定を受けうる。
     6  //
     7  // 検査対象は常に標準入力から渡す。コマンドの引数を定数に保つためで、可変引数は gosec の G204 に当たる。
     8  // 方言は内容の shebang から shellcheck 自身が決めるので、渡し方は結果を変えない。
     9  package shellcheck
    10  
    11  import (
    12  	"context"
    13  	"fmt"
    14  	"os/exec"
    15  	"strings"
    16  	"time"
    17  
    18  	"go-boilerplate/pkg/xerrors"
    19  )

    21  const (
    22  	// Binary は起動する実行ファイル名。所在確認も同じ名前で行う。
    23  	Binary = "shellcheck"
    24  
    25  	// findingsExitCode は「指摘があった」ことを表す終了コード。失敗ではなく結果である。
    26  	findingsExitCode = 1
    27  
    28  	timeout = 30 * time.Second
    29  )

    31  var (
    32  	// ErrMissing は shellcheck が PATH に無いことを表す。
    33  	ErrMissing = xerrors.New("shellcheck が PATH にありません（mise install shellcheck）")
    34  
    35  	// ErrRun は shellcheck を起動できなかった、または指摘以外の理由で終了したことを表す。
    36  	ErrRun = xerrors.New("shellcheck の実行に失敗しました")
    37  )

    39  // Setup は shellcheck の所在を確かめ、走査の基点を返します。
    40  //
    41  // 順序に意味があります。基点が取れても shellcheck が無ければ 1 件も検査できないので、走査を始める
    42  // 前に落とします。両ツールが同じ前提で始まることを、この 1 箇所が保証します。
    43  func Setup(wd func() (string, error), lookPath func(string) (string, error)) (string, error) {
    44  	if _, err := lookPath(Binary); err != nil {
    45  		return "", xerrors.Wrap(ErrMissing, err.Error())
    46  	}
    47  
    48  	root, err := wd()
    49  	if err != nil {
    50  		return "", xerrors.Wrap(err, "getwd")
    51  	}
    52  
    53  	return root, nil
    54  }
    55  
    56  // Run は script を shellcheck に掛け、gcc 形式の出力を返します。
    57  //
    58  // 指摘があるときの終了コード 1 は失敗ではなく結果なので、そのまま出力を返します。区別しないと
    59  // 「指摘があった」と「起動できなかった」が同じ扱いになり、後者が黙って緑になります。
    60  func Run(ctx context.Context, script string) (string, error) {
    61  	cctx, cancel := context.WithTimeout(ctx, timeout)
    62  	defer cancel()
    63  
    64  	cmd := exec.CommandContext(cctx, Binary, "--norc", "--format=gcc", "-")
    65  	cmd.Stdin = strings.NewReader(script)
    66  	var stdout, stderr strings.Builder
    67  	cmd.Stdout = &stdout
    68  	cmd.Stderr = &stderr
    69  
    70  	err := cmd.Run()
    71  	if err == nil {
    72  		return stdout.String(), nil
    73  	}
    74  
    75  	var exitErr *exec.ExitError
    76  	if xerrors.As(err, &exitErr) && exitErr.ExitCode() == findingsExitCode {
    77  		return stdout.String(), nil
    78  	}
    79  
    80  	return "", xerrors.Wrap(ErrRun, fmt.Sprintf("%v: %s", err, strings.TrimSpace(stderr.String())))
    81  }
