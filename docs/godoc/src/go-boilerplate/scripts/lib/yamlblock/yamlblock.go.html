     1  // Package yamlblock は YAML のブロックスカラー（`key: |` / `key: >-`）の中身の判定を 1 箇所へ集める。
     2  //
     3  // 呼び出し側は 2 つある。`uses: owner/repo@<sha>` の取りこぼしを検出する pin-actions と、
     4  // `uses: docker://<image>:<tag>` の取りこぼしを検出する pin-images である。どちらも
     5  // 「厳格なパターンで潰した残り」を行単位で走査するため、ブロックスカラーの中身を外さなければ
     6  // `run:` スクリプトが `uses:` を含む文字列を出力するだけで検出が誤爆する。
     7  //
     8  // 片方だけが除外すると、同じ workflow が経路によって違う判定を受ける。
     9  package yamlblock
    10  
    11  import (
    12  	"regexp"
    13  	"strings"
    14  )
    15  
    16  // headerRe は値がブロックスカラー（`|` / `>`）で始まる行。
    17  // 字下げ指示子と chomp 指示子は YAML がどちらの順序も許すため（`|2-` / `|-2`）両方を受ける。
    18  var headerRe = regexp.MustCompile(`:[ \t]*[|>][+-]?\d?[+-]?[ \t]*(?:#.*)?$`)
    19  
    20  // ContentLines は data のうちブロックスカラーの中身に当たる行番号（1 始まり）を返す。
    21  //
    22  // 中身の範囲は字下げで決まる。ヘッダ行より深い字下げの行と、その途中の空行が中身で、
    23  // 字下げがヘッダ以下へ戻った行で終わる。ヘッダ行そのものは中身に含めない。
    24  func ContentLines(data string) map[int]bool {
    25  	content := map[int]bool{}
    26  	headerIndent := -1
    27  
    28  	for i, line := range strings.Split(data, "\n") {
    29  		indent := len(line) - len(strings.TrimLeft(line, " \t"))
    30  		if headerIndent >= 0 {
    31  			if strings.TrimSpace(line) == "" || indent > headerIndent {
    32  				content[i+1] = true
    33  				continue
    34  			}
    35  			headerIndent = -1
    36  		}
    37  		if headerRe.MatchString(line) {
    38  			headerIndent = indent
    39  		}
    40  	}
    41  
    42  	return content
    43  }
