     1  // Package ghfiles は `uses:` を書ける GitHub Actions の定義ファイルの列挙を 1 箇所へ集める。
     2  //
     3  // 呼び出し側は 2 つある。`uses: owner/repo@<sha>` を SHA へ固定する pin-actions と、
     4  // `uses: docker://<image>:<tag>` を digest へ固定する pin-images である。同じ行が両者から
     5  // 見えている必要は無いが、**同じファイル集合が見えている必要はある**——片方だけが見る
     6  // ディレクトリができると、そこに置かれた参照がどちらの網からも外れる。
     7  //
     8  // composite action は `uses: ./.github/actions/<group>/<name>` のように入れ子へ置けるため走査は
     9  // 再帰する。1 階層で打ち切ると入れ子の定義が検査されないまま通る。
    10  package ghfiles
    11  
    12  import (
    13  	"io/fs"
    14  	"os"
    15  	"path/filepath"
    16  	"slices"
    17  	"sort"
    18  
    19  	"go-boilerplate/pkg/xerrors"
    20  )
    21  
    22  // actionFilenames は composite action の定義ファイル名。
    23  var actionFilenames = []string{"action.yml", "action.yaml"}
    24  
    25  // workflowPatterns は root からの相対で workflow 定義を指すパターン。
    26  var workflowPatterns = []string{".github/workflows/*.yaml", ".github/workflows/*.yml"}
    27  
    28  // Collect は workflow 定義とリポジトリ内 composite action 定義のパスを名前順で返す。
    29  // `.github/actions` が無いリポジトリは空として扱い、それ以外の走査失敗は握り潰さず返す。
    30  func Collect(root string) ([]string, error) {
    31  	var files []string
    32  	for _, pat := range workflowPatterns {
    33  		m, err := filepath.Glob(filepath.Join(root, pat))
    34  		if err != nil {
    35  			return nil, xerrors.Wrap(err, "glob "+pat)
    36  		}
    37  		files = append(files, m...)
    38  	}
    39  
    40  	actionsDir := filepath.Join(root, ".github", "actions")
    41  	err := filepath.WalkDir(actionsDir, func(path string, d fs.DirEntry, err error) error {
    42  		if err != nil {
    43  			return err
    44  		}
    45  		if !d.IsDir() && slices.Contains(actionFilenames, d.Name()) {
    46  			files = append(files, path)
    47  		}
    48  		return nil
    49  	})
    50  	if err != nil && !xerrors.Is(err, os.ErrNotExist) {
    51  		return nil, xerrors.Wrap(err, "walk .github/actions")
    52  	}
    53  
    54  	sort.Strings(files)
    55  	return files, nil
    56  }
