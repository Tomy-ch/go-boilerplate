   175  // parseSSOT は SSOT の本文を IP・クラス定義・ジョブ宣言へ分解する。
   176  func parseSSOT(data string) (*ssot, error) {
   177  	s := &ssot{classes: map[string]*class{}, jobs: map[string]*jobSpec{}, reasons: map[string]string{}}
   178  	var cur *section
   179  
   180  	lines := strings.Split(data, "\n")
   181  	for i := 0; i < len(lines); i++ {
   182  		stripped, err := stripValueComment(lines[i], i)
   183  		if err != nil {
   184  			return nil, err
   185  		}
   186  		line := strings.TrimSpace(stripped)
   187  		if line == "" {
   188  			continue
   189  		}
   190  		if m := sectionRe.FindStringSubmatch(line); m != nil {
   191  			if cur, err = s.enterSection(m[1], m[2]+m[3]); err != nil {
   192  				return nil, err
   193  			}
   194  			continue
   195  		}
   196  		if cur == nil {
   197  			return nil, xerrors.Wrap(errSSOTSyntax, fmt.Sprintf("%d 行目: %q（セクションの外に値があります）", i+1, line))
   198  		}
   199  		key, values, next, err := parseAssignment(lines, i)
   200  		if err != nil {
   201  			return nil, err
   202  		}
   203  		i = next
   204  		if err := cur.assign(key, values); err != nil {
   205  			return nil, xerrors.Wrap(err, fmt.Sprintf("%d 行目", i+1))
   206  		}
   207  	}
   208  	return s, s.validate()
   209  }

   224  func (sec *section) assign(key string, values []string) error {
   225  	if sec.seen[key] {
   226  		return xerrors.Wrap(errSSOTDuplicate, "キー "+key)
   227  	}
   228  	sec.seen[key] = true
   229  	if sec.ip != "" {

   231  	}

   235  // stripValueComment は SSOT の 1 行から、クオートの外の `#` 以降を落とす。
   236  // クオートの中の `#` は値を途中で切ってしまうので、構文エラーとして返す。n は 0 起点の行番号。
   237  func stripValueComment(line string, n int) (string, error) {
   238  	inQuote := false
   239  	for i, r := range line {
   240  		switch {
   241  		case r == '"':
   242  			inQuote = !inQuote
   243  		case r == '#' && inQuote:
   244  			return "", xerrors.Wrap(errSSOTSyntax, fmt.Sprintf("%d 行目: 値の中に # は書けません", n+1))
   245  		case r == '#':
   246  			return line[:i], nil
   247  		}
   248  	}
   249  	return line, nil
   250  }

   260  // openSection は新しい `[class.x]` / `[job."x"]` / `[ip."x"]` を登録する。既出なら重複エラー。
   261  func (s *ssot) openSection(kind, name string) error {
   262  	if _, dup := s.classes[name]; dup && kind == "class" {
   263  		return xerrors.Wrap(errSSOTDuplicate, "class "+name)
   264  	}
   265  	if _, dup := s.jobs[name]; dup && kind == "job" {
   266  		return xerrors.Wrap(errSSOTDuplicate, "job "+name)
   267  	}
   268  	switch kind {
   269  	case "class":
   270  		s.classes[name] = &class{}
   271  	case "ip":
   272  		if slices.Contains(s.ips, name) {
   273  			return xerrors.Wrap(errSSOTDuplicate, "ip "+name)
   274  		}
   275  		if err := validateIP(name); err != nil {
   276  			return err
   277  		}
   278  		s.ips = append(s.ips, name)
   279  	default:
   280  		s.jobs[name] = &jobSpec{}
   281  		s.jobOrder = append(s.jobOrder, name)
   282  	}
   283  	return nil
   284  }

   286  // parseAssignment は lines[i] から始まる代入を読み、キー・値・消費した最終行を返す。
   287  // 配列は `]` が現れるまで後続行を取り込む。
   288  func parseAssignment(lines []string, i int) (string, []string, int, error) {
   289  	head, err := stripValueComment(lines[i], i)
   290  	if err != nil {
   291  		return "", nil, 0, err
   292  	}
   293  	line := strings.TrimSpace(head)
   294  	if m := scalarRe.FindStringSubmatch(line); m != nil {
   295  		return m[1], []string{m[2]}, i, nil
   296  	}
   297  	m := arrayHeadRe.FindStringSubmatch(line)
   298  	if m == nil {
   299  		return "", nil, 0, xerrors.Wrap(errSSOTSyntax, fmt.Sprintf("%d 行目: %q", i+1, line))
   300  	}
   301  	key, rest := m[1], m[2]
   302  	var values []string
   303  	for {
   304  		closed := strings.Contains(rest, "]")
   305  		for _, q := range quotedRe.FindAllStringSubmatch(rest, -1) {
   306  			values = append(values, q[1])
   307  		}
   308  		if closed {
   309  			return key, values, i, nil
   310  		}
   311  		i++
   312  		if i >= len(lines) {
   313  			return "", nil, 0, xerrors.Wrap(errSSOTSyntax, "配列が閉じられていません: "+key)
   314  		}
   315  		if rest, err = stripValueComment(lines[i], i); err != nil {
   316  			return "", nil, 0, err
   317  		}
   318  	}
   319  }

   321  // assign は読み取ったキーを、開いているクラスかジョブのセクションのフィールドへ載せる。
   322  func assign(c *class, j *jobSpec, key string, values []string) error {
   323  	if err := requireUnique(values); err != nil {
   324  		return xerrors.Wrap(err, key)
   325  	}
   326  	switch {
   327  	case c != nil && key == "includes":
   328  		c.includes = values
   329  	case c != nil && key == "hosts":
   330  		c.hosts = values
   331  	case j != nil && key == "classes":
   332  		j.classes = values
   333  	case j != nil && key == "extra":
   334  		j.extra = values
   335  	case j != nil && key == "egress_policy" && len(values) == 1:
   336  		j.policy = values[0]
   337  	case j != nil && key == "sudo" && len(values) == 1:
   338  		if strings.TrimSpace(values[0]) == "" {
   339  			return xerrors.Wrap(errSSOTReason, "sudo")
   340  		}
   341  		j.sudo = values[0]
   342  	default:
   343  		return xerrors.Wrap(errSSOTSyntax, "未知のキー "+key)
   344  	}
   345  	return nil
   346  }

   361  func requireUnique(values []string) error {
   362  	seen := map[string]bool{}
   363  	for _, v := range values {
   364  		if seen[v] {

   368  	}

   372  // validateHost はホストが bullfrog の allowed-domains に書けるドメイン名であることを確かめる。
   373  // ワイルドカードは先頭の `*.` だけを許す。bullfrog は英数字・`.`・`-`・`*` 以外を含む値を拒否して
   374  // ステップごと失敗するので、`_` も書けない。
   375  func validateHost(host string) error {
   376  	if !hostRe.MatchString(host) {
   377  		return xerrors.Wrap(errSSOTHost, host)
   378  	}
   379  	return nil
   380  }

   382  // validateIP は IP セクションの名前が IPv4 のアドレスか CIDR であることを確かめる。
   383  // bullfrog は数字・`.`・`/` 以外を含む allowed-ips を拒否してステップごと失敗するので、IPv6 も書けない。
   384  func validateIP(ip string) error {
   385  	if !ipRe.MatchString(ip) {
   386  		return xerrors.Wrap(errSSOTIP, ip)
   387  	}
   388  	if net.ParseIP(ip) != nil {
   389  		return nil
   390  	}
   391  	if _, _, err := net.ParseCIDR(ip); err == nil {
   392  		return nil
   393  	}
   394  	return xerrors.Wrap(errSSOTIP, ip)
   395  }

   397  // validate は SSOT 単体で判定できる不整合を返す。
   398  func (s *ssot) validate() error {
   399  	if base, ok := s.classes["base"]; ok {
   400  		s.baseHosts = base.hosts
   401  	}
   402  	for _, ip := range s.ips {
   403  		if _, ok := s.reasons[ip]; !ok {
   404  			return xerrors.Wrap(errSSOTReason, "ip "+ip)
   405  		}
   406  	}
   407  	for name, c := range s.classes {
   408  		for _, inc := range c.includes {
   409  			if _, ok := s.classes[inc]; !ok {
   410  				return xerrors.Wrap(errSSOTUnknownClass, name+" includes "+inc)
   411  			}
   412  		}
   413  		for _, h := range c.hosts {
   414  			if err := validateHost(h); err != nil {
   415  				return xerrors.Wrap(err, "class "+name)
   416  			}
   417  		}
   418  	}
   419  	for _, key := range s.jobOrder {
   420  		if err := s.validateJob(key, s.jobs[key]); err != nil {
   421  			return err
   422  		}
   423  	}
   424  	return nil
   425  }

   427  // validateJob は 1 ジョブの宣言の整合を返す。
   428  func (s *ssot) validateJob(key string, j *jobSpec) error {
   429  	switch j.policy {
   430  	case "", policyBlock:
   431  		j.policy = policyBlock
   432  	case policyAudit:
   433  		if len(j.classes) > 0 || len(j.extra) > 0 {
   434  			return xerrors.Wrap(errSSOTPolicy, key+": audit は許可リストを持ちません")
   435  		}
   436  	default:

   505  // scanWorkflow は 1 ファイルの通信制限ステップを読み、ジョブごとに 1 件ずつ返す。
   506  // name は workflow のファイル名。ステップを持つジョブは、その先頭に通信制限ステップを持たなければならない。
   507  func scanWorkflow(name string, lines []string) ([]step, error) {
   508  	jobs := workflowJobs(lines)
   509  	stepped := jobsWithSteps(lines, jobs)
   510  	var steps []step
   511  	found := map[string]bool{}
   512  
   513  	for i := 0; i < len(lines); i++ {
   514  		m := stepUsesRe.FindStringSubmatch(lines[i])
   515  		if m == nil {
   516  			if actionRefRe.MatchString(stripComment(lines[i])) {
   517  				return nil, xerrors.Wrap(errStepMalformed, fmt.Sprintf("%s %d 行目: %q", name, i+1, strings.TrimSpace(lines[i])))
   518  			}
   519  			continue
   520  		}
   521  		if jobs[i] == "" {
   522  			return nil, xerrors.Wrap(errWorkflowJobless, fmt.Sprintf("%s %d 行目", name, i+1))
   523  		}
   524  		key := name + ":" + jobs[i]
   525  		if found[key] {
   526  			return nil, xerrors.Wrap(errStepDuplicate, key)
   527  		}
   528  		found[key] = true
   529  		if first := stepped[jobs[i]]; !withinFirstStep(lines, first, i) {
   530  			return nil, xerrors.Wrap(errStepNotFirst, key)
   531  		}

   533  		st, err := readStep(name, lines, i+1, indent)
   534  		if err != nil {
   535  			return nil, err
   536  		}
   537  		if err := checkStepKeys(lines, stepped[jobs[i]], st.endLine, indent); err != nil {
   538  			return nil, xerrors.Wrap(err, key)
   539  		}
   540  		st.jobKey = key
   541  		steps = append(steps, st)
   542  		i = st.endLine - 1
   543  	}
   544  	for job := range stepped {
   545  		if !found[name+":"+job] {
   546  			return nil, xerrors.Wrap(errStepMissing, name+":"+job)
   547  		}
   548  	}

   567  	}
   568  	return out

   627  // readStep は uses の次の行から始まる with ブロックを読む。indent は uses キーの字下げ。
   628  // ブロックの中身は生成器が所有するので、egress-policy 以外の行は読み飛ばして差し替える。
   629  func readStep(name string, lines []string, withLine, indent int) (step, error) {
   630  	pad := strings.Repeat(" ", indent)
   631  	if withLine >= len(lines) || lines[withLine] != pad+"with:" {
   632  		return step{}, xerrors.Wrap(errStepMalformed, fmt.Sprintf("%s %d 行目: uses の次に with がありません", name, withLine+1))
   633  	}
   634  	st := step{withLine: withLine, endLine: withLine + 1, indent: indent}
   635  	inner := pad + strings.Repeat(" ", withIndentStep)
   636  	for j := withLine + 1; j < len(lines) && strings.HasPrefix(lines[j], inner); j++ {
   637  		if m := policyRe.FindStringSubmatch(lines[j]); m != nil && st.policy == "" {
   638  			st.policy = m[1]
   639  		}
   640  		st.endLine = j + 1
   641  	}
   642  	if blockResumes(lines, st.endLine, inner) {
   643  		return step{}, xerrors.Wrap(errStepMalformed, fmt.Sprintf("%s %d 行目: with ブロックの途中に空行かコメントがあります", name, st.endLine+1))
   644  	}
   645  	if st.policy == "" {
   646  		return step{}, xerrors.Wrap(errStepMalformed, fmt.Sprintf("%s %d 行目: egress-policy がありません", name, withLine+1))
   647  	}
   648  	return st, nil
   649  }

   745  func (s *ssot) rewriteFile(name, data string, used map[string]bool) (string, error) {
   746  	lines := strings.Split(data, "\n")

   748  	if err != nil {
