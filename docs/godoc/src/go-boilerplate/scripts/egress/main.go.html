   137  // parseSSOT は SSOT の本文をクラス定義とジョブ宣言へ分解する。
   138  func parseSSOT(data string) (*ssot, error) {
   139  	s := &ssot{classes: map[string]*class{}, jobs: map[string]*jobSpec{}}
   140  	var curClass *class
   141  	var curJob *jobSpec
   142  	var seen map[string]bool
   143  
   144  	lines := strings.Split(data, "\n")
   145  	for i := 0; i < len(lines); i++ {
   146  		line := strings.TrimSpace(stripComment(lines[i]))
   147  		if line == "" {
   148  			continue
   149  		}
   150  		if m := sectionRe.FindStringSubmatch(line); m != nil {
   151  			name := m[2] + m[3]
   152  			if err := s.openSection(m[1], name); err != nil {
   153  				return nil, err
   154  			}
   155  			curClass, curJob = s.classes[name], s.jobs[name]
   156  			seen = map[string]bool{}
   157  			continue
   158  		}
   159  		if curClass == nil && curJob == nil {
   160  			return nil, xerrors.Wrap(errSSOTSyntax, fmt.Sprintf("%d 行目: %q（セクションの外に値があります）", i+1, line))
   161  		}
   162  		key, values, next, err := parseAssignment(lines, i)
   163  		if err != nil {
   164  			return nil, err
   165  		}
   166  		i = next
   167  		if seen[key] {
   168  			return nil, xerrors.Wrap(errSSOTDuplicate, fmt.Sprintf("%d 行目: キー %q", i+1, key))
   169  		}
   170  		seen[key] = true
   171  		if err := assign(curClass, curJob, key, values); err != nil {
   172  			return nil, xerrors.Wrap(err, fmt.Sprintf("%d 行目", i+1))
   173  		}
   174  	}
   175  	return s, s.validate()
   176  }

   186  // openSection は新しい `[class.x]` / `[job."x"]` を登録する。既出なら重複エラー。
   187  func (s *ssot) openSection(kind, name string) error {
   188  	if _, dup := s.classes[name]; dup && kind == "class" {
   189  		return xerrors.Wrap(errSSOTDuplicate, "class "+name)
   190  	}
   191  	if _, dup := s.jobs[name]; dup && kind == "job" {
   192  		return xerrors.Wrap(errSSOTDuplicate, "job "+name)
   193  	}
   194  	if kind == "class" {
   195  		s.classes[name] = &class{}
   196  		return nil
   197  	}
   198  	s.jobs[name] = &jobSpec{}
   199  	s.jobOrder = append(s.jobOrder, name)
   200  	return nil
   201  }

   203  // parseAssignment は lines[i] から始まる代入を読み、キー・値・消費した最終行を返す。
   204  // 配列は `]` が現れるまで後続行を取り込む。
   205  func parseAssignment(lines []string, i int) (string, []string, int, error) {
   206  	line := strings.TrimSpace(stripComment(lines[i]))
   207  	if m := scalarRe.FindStringSubmatch(line); m != nil {
   208  		return m[1], []string{m[2]}, i, nil
   209  	}
   210  	m := arrayHeadRe.FindStringSubmatch(line)
   211  	if m == nil {
   212  		return "", nil, 0, xerrors.Wrap(errSSOTSyntax, fmt.Sprintf("%d 行目: %q", i+1, line))
   213  	}
   214  	key, rest := m[1], m[2]
   215  	var values []string
   216  	for {
   217  		closed := strings.Contains(rest, "]")
   218  		for _, q := range quotedRe.FindAllStringSubmatch(rest, -1) {
   219  			values = append(values, q[1])
   220  		}
   221  		if closed {
   222  			return key, values, i, nil
   223  		}
   224  		i++
   225  		if i >= len(lines) {
   226  			return "", nil, 0, xerrors.Wrap(errSSOTSyntax, "配列が閉じられていません: "+key)
   227  		}
   228  		rest = stripComment(lines[i])
   229  	}
   230  }

   232  // assign は読み取ったキーを、開いているセクションのフィールドへ載せる。
   233  func assign(c *class, j *jobSpec, key string, values []string) error {
   234  	if err := requireUnique(values); err != nil {
   235  		return xerrors.Wrap(err, key)
   236  	}
   237  	switch {
   238  	case c != nil && key == "includes":
   239  		c.includes = values
   240  	case c != nil && key == "hosts":
   241  		c.hosts = values
   242  	case j != nil && key == "classes":
   243  		j.classes = values
   244  	case j != nil && key == "extra":
   245  		j.extra = values
   246  	case j != nil && key == "egress_policy" && len(values) == 1:
   247  		j.policy = values[0]
   248  	default:
   249  		return xerrors.Wrap(errSSOTSyntax, "未知のキー "+key)
   250  	}
   251  	return nil
   252  }

   254  // requireUnique は同一リスト内の重複を弾く。重複は「どちらを直せばよいか」が読めなくなる。
   255  func requireUnique(values []string) error {
   256  	seen := map[string]bool{}
   257  	for _, v := range values {
   258  		if seen[v] {
   259  			return xerrors.Wrap(errSSOTDuplicate, v)
   260  		}
   261  		seen[v] = true
   262  	}
   263  	return nil
   264  }

   266  // validate は SSOT 単体で判定できる不整合を返す。
   267  func (s *ssot) validate() error {
   268  	if base, ok := s.classes["base"]; ok {
   269  		s.baseHosts = base.hosts
   270  	}
   271  	for name, c := range s.classes {
   272  		for _, inc := range c.includes {
   273  			if _, ok := s.classes[inc]; !ok {
   274  				return xerrors.Wrap(errSSOTUnknownClass, name+" includes "+inc)
   275  			}
   276  		}
   277  	}
   278  	for _, key := range s.jobOrder {
   279  		if err := s.validateJob(key, s.jobs[key]); err != nil {
   280  			return err
   281  		}
   282  	}
   283  	return nil
   284  }

   286  // validateJob は 1 ジョブの宣言の整合を返す。
   287  func (s *ssot) validateJob(key string, j *jobSpec) error {
   288  	switch j.policy {
   289  	case "", policyBlock:
   290  		j.policy = policyBlock
   291  	case policyAudit:
   292  		if len(j.classes) > 0 || len(j.extra) > 0 {
   293  			return xerrors.Wrap(errSSOTPolicy, key+": audit は allowed-endpoints を持ちません")
   294  		}
   295  	default:
   296  		return xerrors.Wrap(errSSOTPolicy, key+": "+j.policy)
   297  	}
   298  	for _, cn := range j.classes {
   299  		if cn == "base" {
   300  			return xerrors.Wrap(errSSOTBaseClass, key)
   301  		}
   302  		if _, ok := s.classes[cn]; !ok {
   303  			return xerrors.Wrap(errSSOTUnknownClass, key+" classes "+cn)
   304  		}
   305  	}
   306  	_, err := s.hostsFor(key)
   307  	return err
   308  }

   310  // hostsFor は 1 ジョブの許可リストを算出する。base → classes（includes 展開込み）→ extra の順で、
   311  // 重複は初出を残す。並びを SSOT の宣言順に固定しているため、生成物の差分はレビューで読める。
   312  func (s *ssot) hostsFor(key string) ([]string, error) {
   313  	j := s.jobs[key]
   314  	if j.policy == policyAudit {
   315  		return nil, nil
   316  	}
   317  	out := &orderedHosts{seen: map[string]bool{}}
   318  	out.add(s.baseHosts)
   319  	for _, cn := range j.classes {
   320  		if err := s.expandClass(cn, out, map[string]bool{}); err != nil {
   321  			return nil, xerrors.Wrap(err, key)
   322  		}
   323  	}
   324  	out.add(j.extra)
   325  	return out.list, nil
   326  }

   328  // expandClass は includes を先に展開してからクラス自身の hosts を足す。visiting は循環検出用。
   329  func (s *ssot) expandClass(name string, out *orderedHosts, visiting map[string]bool) error {
   330  	if visiting[name] {
   331  		return xerrors.Wrap(errSSOTClassCycle, name)
   332  	}
   333  	visiting[name] = true
   334  	defer delete(visiting, name)
   335  
   336  	c, ok := s.classes[name]
   337  	if !ok {
   338  		return xerrors.Wrap(errSSOTUnknownClass, name)
   339  	}
   340  	for _, inc := range c.includes {
   341  		if err := s.expandClass(inc, out, visiting); err != nil {
   342  			return err
   343  		}
   344  	}
   345  	out.add(c.hosts)
   346  	return nil
   347  }

   359  // scanWorkflow は 1 ファイルの harden-runner 記述を、ジョブ単位の egress-policy と
   360  // allowed-endpoints ブロックへ分解する。name は workflow のファイル名。
   361  func scanWorkflow(name string, lines []string) (map[string]string, []block, error) {
   362  	policies := map[string]string{}
   363  	var blocks []block
   364  	jobs := workflowJobs(lines)
   365  
   366  	for i := 0; i < len(lines); i++ {
   367  		if m := egressPolicyRe.FindStringSubmatch(lines[i]); m != nil {
   368  			if jobs[i] == "" {
   369  				return nil, nil, xerrors.Wrap(errWorkflowJobless, fmt.Sprintf("%s %d 行目", name, i+1))
   370  			}
   371  			policies[name+":"+jobs[i]] = m[1]
   372  			continue
   373  		}
   374  		m := endpointsKeyRe.FindStringSubmatch(lines[i])
   375  		if m == nil {
   376  			continue
   377  		}
   378  		if jobs[i] == "" {
   379  			return nil, nil, xerrors.Wrap(errWorkflowJobless, fmt.Sprintf("%s %d 行目", name, i+1))
   380  		}
   381  		b, err := readBlock(name, lines, i, len(m[1]))
   382  		if err != nil {
   383  			return nil, nil, err
   384  		}
   385  		b.jobKey = name + ":" + jobs[i]
   386  		blocks = append(blocks, b)
   387  		i = b.endLine - 1
   388  	}
   389  	return policies, blocks, nil
   390  }

   392  // workflowJobs は各行が属する `jobs:` 直下の job id を行番号順に返す。ジョブの外は空文字。
   393  func workflowJobs(lines []string) []string {
   394  	out := make([]string, len(lines))
   395  	inJobs, job := false, ""
   396  	for i, line := range lines {
   397  		switch {
   398  		case strings.HasPrefix(line, "#"):
   399  		case line != "" && !strings.HasPrefix(line, " "):
   400  			inJobs, job = strings.HasPrefix(line, "jobs:"), ""
   401  		case inJobs:
   402  			if m := jobIDRe.FindStringSubmatch(line); m != nil {
   403  				job = m[1]
   404  			}
   405  		}
   406  		out[i] = job
   407  	}
   408  	return out
   409  }

   411  // readBlock は keyLine から始まる折り畳みスカラーのホスト行を読む。
   412  // ホスト 1 個以外の行（コメント・複数トークン）は握り潰さず fail-close する
   413  // （fail-closed の範囲は scripts/README.md の egress 行）。
   414  func readBlock(name string, lines []string, keyLine, indent int) (block, error) {
   415  	b := block{keyLine: keyLine, endLine: keyLine + 1, indent: indent}
   416  	for j := keyLine + 1; j < len(lines); j++ {
   417  		line := lines[j]
   418  		if strings.TrimSpace(line) == "" {
   419  			break
   420  		}
   421  		if len(line)-len(strings.TrimLeft(line, " ")) <= indent {
   422  			break
   423  		}
   424  		host := strings.TrimSpace(line)
   425  		if strings.HasPrefix(host, "#") || strings.ContainsAny(host, " \t") {
   426  			return block{}, xerrors.Wrap(errBlockComment, fmt.Sprintf("%s %d 行目: %q", name, j+1, host))
   427  		}
   428  		b.hosts = append(b.hosts, host)
   429  		b.endLine = j + 1
   430  	}
   431  	return b, nil
   432  }

   435  func renderBlock(hosts []string, indent int) []string {
   436  	pad := strings.Repeat(" ", indent+blockIndentStep)

   438  	for _, h := range hosts {

   444  // applyOrCheck は SSOT を正本に allowed-endpoints を書き換える。dryRun=true は差分を非ゼロ終了で報告する。
   445  func applyOrCheck(root string, dryRun bool) error {
   446  	raw, err := os.ReadFile(filepath.Join(root, ssotFile)) //nolint:gosec // path is cwd + literal filename
   447  	if err != nil {
   448  		return xerrors.Wrap(err, "read "+ssotFile)
   449  	}
   450  	s, err := parseSSOT(string(raw))
   451  	if err != nil {
   452  		return err
   453  	}
   454  	files, err := filepath.Glob(filepath.Join(root, ".github", "workflows", "*.yaml"))
   455  	if err != nil {
   456  		return xerrors.Wrap(err, "glob workflows")
   457  	}
   458  	sort.Strings(files)
   459  
   460  	changes, used, err := s.planFiles(files)
   461  	if err != nil {
   462  		return err
   463  	}
   464  	if orphans := orphanJobs(s, used); len(orphans) > 0 {
   465  		return xerrors.Wrap(errJobOrphan, strings.Join(orphans, ", ")+"（"+ssotFile+" の該当エントリを削除してください）")
   466  	}
   467  	return writeChanges(root, changes, dryRun)

   491  // rewriteFile は 1 ファイルの allowed-endpoints ブロックを SSOT 由来の内容へ差し替えた本文を返す。
   492  func (s *ssot) rewriteFile(name, data string, used map[string]bool) (string, error) {
   493  	lines := strings.Split(data, "\n")
   494  	policies, blocks, err := scanWorkflow(name, lines)
   495  	if err != nil {
   496  		return "", err
   497  	}
   498  	byJob := map[string]bool{}
   499  	for _, b := range blocks {
   500  		byJob[b.jobKey] = true
   501  	}
   502  	for key, policy := range policies {
   503  		if err := s.checkPolicy(key, policy, byJob[key]); err != nil {
   504  			return "", err
   505  		}
   506  		used[key] = true
   507  	}
   508  	// 後ろから差し替えて、先行ブロックの行番号を保つ。
   509  	for _, v := range slices.Backward(blocks) {
   510  		b := v
   511  		hosts, err := s.hostsFor(b.jobKey)
   512  		if err != nil {
   513  			return "", err
   514  		}
   515  		lines = append(lines[:b.keyLine+1:b.keyLine+1],
   516  			append(renderBlock(hosts, b.indent), lines[b.endLine:]...)...)
   517  	}
   518  	return strings.Join(lines, "\n"), nil
   519  }

   522  func (s *ssot) checkPolicy(key, policy string, hasBlock bool) error {
   523  	j, ok := s.jobs[key]
   524  	if !ok {
   525  		return xerrors.Wrap(errJobMissing, key+"（"+ssotFile+" へ追加してください）")
   526  	}
   527  	if j.policy != policy {
   528  		return xerrors.Wrap(errPolicyMismatch, fmt.Sprintf("%s: workflow=%s SSOT=%s", key, policy, j.policy))
   529  	}

   532  			fmt.Sprintf("%s: egress-policy=%s に対して allowed-endpoints の在否が合いません", key, policy))

   551  func writeChanges(root string, changes map[string]string, dryRun bool) error {
   552  	paths := make([]string, 0, len(changes))
   553  	for f := range changes {
   554  		paths = append(paths, f)

   563  			}
   564  			return xerrors.Wrap(errEgressDrift, "make egress-apply してコミットしてください: "+strings.Join(rels, ", "))
   565  		}
   566  		log.Printf("✅ 全ジョブの allowed-endpoints が %s 通りです", ssotFile)
   567  
   568  		return nil
   569  	}
   570  	for _, f := range paths {
   571  		if err := os.WriteFile(f, []byte(changes[f]), filePerm); err != nil { //nolint:gosec // path from cwd glob
   572  			return xerrors.Wrap(err, "write "+relTo(root, f))
   573  		}
   574  		log.Printf("  updated %s", relTo(root, f))
   575  	}
   576  	log.Printf("✅ %d ファイルへ反映しました", len(paths))
   577  
   578  	return nil
   579  }
