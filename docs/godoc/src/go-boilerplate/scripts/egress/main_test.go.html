    61  // testWorkflow は allowed-endpoints ブロックを 1 つ持つ workflow の雛形を返す。
    62  func testWorkflow(jobID string, hosts ...string) string {
    63  	var b strings.Builder
    64  	b.WriteString("name: T\n\njobs:\n  " + jobID + ":\n    steps:\n")
    65  	b.WriteString("      - name: Harden the runner\n        uses: step-security/harden-runner@x\n")
    66  	b.WriteString("        with:\n          egress-policy: block\n          allowed-endpoints: >\n")
    67  	for _, h := range hosts {
    68  		b.WriteString("            " + h + "\n")
    69  	}
    70  	b.WriteString("\n      - name: Checkout\n")
    71  	return b.String()
    72  }
