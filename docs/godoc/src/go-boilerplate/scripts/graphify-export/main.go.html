   147  func run(inputPath, outputDir string) error {
   148  	source, err := loadGraph(inputPath)
   149  	if err != nil {
   150  		return xerrors.Wrap(err, "Graphify graph の読み込み")
   151  	}
   152  
   153  	model, err := normalize(source)
   154  	if err != nil {
   155  		return xerrors.Wrap(err, "semantic model の正規化")
   156  	}
   157  
   158  	docs, err := marshalDocuments(model)
   159  	if err != nil {
   160  		return xerrors.Wrap(err, "JSON の生成")
   161  	}
   162  	if err := writeDocuments(outputDir, docs); err != nil {
   163  		return xerrors.Wrap(err, "graphify-out への書き出し")
   164  	}
   165  
   166  	//nolint:gosec // 出力先は開発者が指定するローカルパスで、ログの宛先も標準エラーのみ。
   167  	log.Printf("Graphify export complete: %d nodes, %d edges -> %s", len(model.Nodes), len(model.Edges), outputDir)
   168  
   169  	return nil
   170  }

   172  func loadGraph(path string) (sourceGraph, error) {
   173  	file, err := os.Open(path) //nolint:gosec // 開発者が指定したローカル生成物を読む CLI。
   174  	if err != nil {
   175  		return sourceGraph{}, xerrors.Wrap(err, "open "+path)
   176  	}
   177  	defer func() { _ = file.Close() }()
   178  
   179  	decoder := json.NewDecoder(file)
   180  	decoder.UseNumber()
   181  	var graph sourceGraph
   182  	if err := decoder.Decode(&graph); err != nil {
   183  		return sourceGraph{}, xerrors.Wrap(err, "decode "+path)
   184  	}
   185  	if err := decoder.Decode(&struct{}{}); !xerrors.Is(err, io.EOF) {
   186  		return sourceGraph{}, xerrors.Wrap(errTrailingJSON, "decode "+path)
   187  	}
   188  
   189  	return graph, nil
   190  }

   192  func normalize(source sourceGraph) (semanticModel, error) {
   193  	if len(source.Links) > 0 && len(source.Edges) > 0 {
   194  		return semanticModel{}, xerrors.Wrap(errAmbiguousEdges, "links と edges が同時に存在します")
   195  	}
   196  	if len(source.Nodes) == 0 {
   197  		return semanticModel{}, xerrors.Wrap(errNoNodes, "node が1件もありません")
   198  	}
   199  
   200  	model := semanticModel{Nodes: make([]node, 0, len(source.Nodes))}
   201  	nodeIDs := make(map[string]struct{}, len(source.Nodes))
   202  	for _, raw := range source.Nodes {
   203  		normalized, err := normalizeNode(raw)
   204  		if err != nil {
   205  			return semanticModel{}, err
   206  		}
   207  		if _, exists := nodeIDs[normalized.ID]; exists {
   208  			return semanticModel{}, xerrors.Wrap(errDuplicateNodeID, fmt.Sprintf("node ID %q が重複しています", normalized.ID))
   209  		}
   210  		nodeIDs[normalized.ID] = struct{}{}
   211  		model.Nodes = append(model.Nodes, normalized)
   212  	}
   213  
   214  	rawEdges := source.Links
   215  	if len(rawEdges) == 0 {
   216  		rawEdges = source.Edges
   217  	}
   218  	model.Edges = make([]edge, 0, len(rawEdges))
   219  	for _, raw := range rawEdges {
   220  		normalized, err := normalizeEdge(raw)
   221  		if err != nil {
   222  			return semanticModel{}, err
   223  		}
   224  		if _, exists := nodeIDs[normalized.Source]; !exists {
   225  			return semanticModel{}, xerrors.Wrap(errDanglingEdge, fmt.Sprintf("edge source %q に対応する node がありません", normalized.Source))
   226  		}
   227  		if _, exists := nodeIDs[normalized.Target]; !exists {
   228  			return semanticModel{}, xerrors.Wrap(errDanglingEdge, fmt.Sprintf("edge target %q に対応する node がありません", normalized.Target))
   229  		}
   230  		model.Edges = append(model.Edges, normalized)
   231  	}
   232  
   233  	sort.Slice(model.Nodes, func(i, j int) bool { return model.Nodes[i].ID < model.Nodes[j].ID })
   234  	sort.Slice(model.Edges, func(i, j int) bool { return model.Edges[i].sortKey < model.Edges[j].sortKey })
   235  	model.Metadata = buildMetadata(source, len(model.Nodes), len(model.Edges))
   236  
   237  	return model, nil
   238  }

   240  func normalizeNode(raw map[string]any) (node, error) {
   241  	id, err := preferredString(raw, "id")
   242  	if err != nil {
   243  		return node{}, xerrors.Wrap(err, "node id が不正です")
   244  	}
   245  	if id == "" {
   246  		return node{}, xerrors.Wrap(errInvalidNode, "node id が不正です: 空文字です")
   247  	}
   248  	name, err := preferredString(raw, "label", "name")
   249  	if err != nil {
   250  		return node{}, xerrors.Wrap(err, fmt.Sprintf("node %q の name が不正です", id))
   251  	}
   252  	if name == "" {
   253  		name = id
   254  	}
   255  	kind, err := preferredString(raw, "file_type", "kind", "type")
   256  	if err != nil {
   257  		return node{}, xerrors.Wrap(err, fmt.Sprintf("node %q の kind が不正です", id))
   258  	}
   259  	if kind == "" {
   260  		kind = "unknown"
   261  	}
   262  	file, err := preferredString(raw, "source_file")
   263  	if err != nil {
   264  		return node{}, xerrors.Wrap(err, fmt.Sprintf("node %q の source_file が不正です", id))
   265  	}
   266  	position, err := preferredString(raw, "source_location")
   267  	if err != nil {
   268  		return node{}, xerrors.Wrap(err, fmt.Sprintf("node %q の source_location が不正です", id))
   269  	}
   270  
   271  	return node{
   272  		ID:       id,
   273  		Kind:     kind,
   274  		Name:     name,
   275  		Location: newLocation(file, position),
   276  		Attributes: cloneWithoutKeys(
   277  			raw, "id", "label", "name", "file_type", "kind", "type", "source_file", "source_location",
   278  		),
   279  	}, nil
   280  }

   282  func normalizeEdge(raw map[string]any) (edge, error) {
   283  	source, err := preferredString(raw, "source")
   284  	if err != nil {
   285  		return edge{}, xerrors.Wrap(err, "edge source が不正です")
   286  	}
   287  	if source == "" {
   288  		return edge{}, xerrors.Wrap(errInvalidEdge, "edge source が不正です: 空文字です")
   289  	}
   290  	target, err := preferredString(raw, "target")
   291  	if err != nil {
   292  		return edge{}, xerrors.Wrap(err, "edge target が不正です")
   293  	}
   294  	if target == "" {
   295  		return edge{}, xerrors.Wrap(errInvalidEdge, "edge target が不正です: 空文字です")
   296  	}
   297  	relation, err := preferredString(raw, "relation", "type", "kind")
   298  	if err != nil {
   299  		return edge{}, xerrors.Wrap(err, fmt.Sprintf("edge %q -> %q の relation が不正です", source, target))
   300  	}
   301  	if relation == "" {
   302  		return edge{}, xerrors.Wrap(errInvalidEdge, fmt.Sprintf("edge %q -> %q の relation が不正です: 空文字です", source, target))
   303  	}
   304  	file, err := preferredString(raw, "source_file")
   305  	if err != nil {
   306  		return edge{}, xerrors.Wrap(err, fmt.Sprintf("edge %q -> %q の source_file が不正です", source, target))
   307  	}
   308  	position, err := preferredString(raw, "source_location")
   309  	if err != nil {
   310  		return edge{}, xerrors.Wrap(err, fmt.Sprintf("edge %q -> %q の source_location が不正です", source, target))
   311  	}
   312  
   313  	normalized := edge{
   314  		Source:     source,
   315  		Target:     target,
   316  		Relation:   relation,
   317  		Location:   newLocation(file, position),
   318  		Attributes: cloneWithoutKeys(raw, "source", "target", "relation", "type", "kind", "source_file", "source_location"),
   319  	}
   320  	encoded, err := json.Marshal(normalized)
   321  	if err != nil {
   322  		return edge{}, xerrors.Wrap(err, fmt.Sprintf("edge %q -> %q の sort key", source, target))
   323  	}
   324  	normalized.sortKey = string(encoded)
   325  
   326  	return normalized, nil
   327  }

   329  func preferredString(attributes map[string]any, keys ...string) (string, error) {
   330  	for _, key := range keys {
   331  		value, exists := attributes[key]
   332  		if !exists || value == nil {
   333  			continue
   334  		}
   335  		text, ok := value.(string)
   336  		if !ok {
   337  			return "", xerrors.Wrap(errNotString, key+" は string ではありません")
   338  		}
   339  
   340  		return text, nil
   341  	}
   342  
   343  	return "", nil
   344  }

   346  func cloneWithoutKeys(source map[string]any, keys ...string) map[string]any {
   347  	omit := make(map[string]struct{}, len(keys))
   348  	for _, key := range keys {
   349  		omit[key] = struct{}{}
   350  	}
   351  	cloned := make(map[string]any)
   352  	for key, value := range source {
   353  		if _, skip := omit[key]; !skip {
   354  			cloned[key] = value
   355  		}
   356  	}
   357  	if len(cloned) == 0 {
   358  		return nil
   359  	}
   360  
   361  	return cloned
   362  }

   372  func buildMetadata(source sourceGraph, nodeCount, edgeCount int) metadata {

   407  func marshalDocuments(model semanticModel) (documents, error) {
   408  	marshal := func(value any) ([]byte, error) {
   409  		body, err := json.MarshalIndent(value, "", "  ")
   410  		if err != nil {
   411  			return nil, err
   412  		}
   413  
   414  		return append(body, '\n'), nil
   415  	}
   416  
   417  	nodes, err := marshal(nodeDocument{SchemaVersion: schemaVersion, Nodes: model.Nodes})
   418  	if err != nil {
   419  		return documents{}, err
   420  	}
   421  	edges, err := marshal(edgeDocument{SchemaVersion: schemaVersion, Edges: model.Edges})
   422  	if err != nil {
   423  		return documents{}, err
   424  	}
   425  	meta, err := marshal(model.Metadata)
   426  	if err != nil {
   427  		return documents{}, err
   428  	}
   429  
   430  	return documents{nodes: nodes, edges: edges, metadata: meta}, nil
   431  }

   433  func writeDocuments(outputDir string, docs documents) error {
   434  	if strings.TrimSpace(outputDir) == "" {
   435  		return xerrors.Wrap(errEmptyOutputDir, "output directory が空です")
   436  	}
   437  	if err := os.MkdirAll(outputDir, artifactDirMode); err != nil { //nolint:gosec // 開発者が指定したローカル生成物の出力先。
   438  		return err
   439  	}
   440  
   441  	writeAtomic := func(name string, body []byte) error {
   442  		temporary, err := os.CreateTemp(outputDir, "."+name+"-*")
   443  		if err != nil {
   444  			return err
   445  		}
   446  		temporaryPath := temporary.Name()
   447  		defer func() { _ = os.Remove(temporaryPath) }() //nolint:gosec // 直前に自分で作った一時ファイルを畳む。
   448  		if err := temporary.Chmod(artifactFileMode); err != nil {
   449  			_ = temporary.Close()
   450  			return err
   451  		}
   452  		if _, err := temporary.Write(body); err != nil {
   453  			_ = temporary.Close()
   454  			return err
   455  		}
   456  		if err := temporary.Close(); err != nil {
   457  			return err
   458  		}
   459  
   460  		//nolint:gosec // 開発者が指定したローカル生成物の出力先で、name は呼び出し側の固定値。
   461  		return os.Rename(temporaryPath, filepath.Join(outputDir, name))
   462  	}
   463  
   464  	if err := writeAtomic("nodes.json", docs.nodes); err != nil {
   465  		return err
   466  	}
   467  	if err := writeAtomic("edges.json", docs.edges); err != nil {
   468  		return err
   469  	}
   470  
   471  	return writeAtomic("metadata.json", docs.metadata)
   472  }
