   129  func TestReadLock(t *testing.T) {
   130  	t.Parallel()
   131  
   132  	t.Run("正常系", func(t *testing.T) {
   133  		t.Parallel()
   134  
   135  		t.Run("有効な行のみを repo@tag→SHA として読み込む", func(t *testing.T) {
   136  			t.Parallel()
   137  			dir := t.TempDir()
   138  			path := filepath.Join(dir, "actions-pin.toml")
   139  			body := "# comment\n" +
   140  				"\"actions/checkout@v7.0.0\" = \"" + shaCheckout + "\"\n" +
   141  				"invalid line\n" +
   142  				"\"actions/setup-go@v6\" = \"" + shaSetupGo + "\"\n"
   143  			require.NoError(t, os.WriteFile(path, []byte(body), 0o600))
   144  
   145  			lock, err := readLock(path)
   146  			require.NoError(t, err)
   147  			assert.Equal(t, map[string]string{
   148  				"actions/checkout@v7.0.0": shaCheckout,
   149  				"actions/setup-go@v6":     shaSetupGo,
   150  			}, lock)
   151  		})
   152  	})
   153  
   154  	t.Run("異常系", func(t *testing.T) {
   155  		t.Parallel()
   156  
   157  		t.Run("ファイルが存在しなければエラーを返す", func(t *testing.T) {
   158  			t.Parallel()
   159  			_, err := readLock(filepath.Join(t.TempDir(), "absent.toml"))
   160  			require.Error(t, err)
   161  		})
   162  	})
