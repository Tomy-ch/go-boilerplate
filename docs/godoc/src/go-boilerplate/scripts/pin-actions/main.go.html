    47  var (
    48  	// uses: [-] owner/repo[/sub]@<ref> [# <tag>]
    49  	// 空白は `[ \t]` に限定する（`\s` だと改行を食って行が結合する）。
    50  	// クオートは値に含めない。含めるとクオート文字が repo / tag に混入したキーで lockfile を引き、
    51  	// resolve がそれを GitHub のリポジトリ名として扱って必ず失敗する。厳密パターンから外れた行は
    52  	// detectLooseUses が拾い、クオートを外して書き直せば通る形で fail-close する。
    53  	usesRe = regexp.MustCompile(`(?m)^([ \t]*(?:-[ \t]*)?uses:[ \t]*)([^@\s'"]+)@([^\s#'"]+)(?:[ \t]*#[ \t]*(\S+))?[ \t]*$`)
    54  	lockRe = regexp.MustCompile(`^"([^"]+)"\s*=\s*"([0-9a-f]{40})"`)
    55  	// usesRe の取りこぼしを拾う緩いパターン。行頭にアンカーせず、クオートされたキー（`"uses":`）も
    56  	// 拾い、値は行末までまとめて捉えて呼び出し元が解釈する。直前の文字を見るのは `disuses:` のような
    57  	// 語末一致を弾くため。
    58  	looseUsesRe = regexp.MustCompile(`(?:^|[^A-Za-z0-9_-])["']?uses["']?[ \t]*:(.*)$`)
    59  	// 値がブロックスカラー（`|` / `>`）で始まる行。以降の深い字下げの行はその中身。
    60  	blockScalarRe = regexp.MustCompile(`:[ \t]*[|>][+-]?[0-9]*[ \t]*(?:#.*)?$`)
    61  )

   159  // parseUses は uses: 行の path / ref / コメント tag から ref を構築する。第2戻り値が false なら対象外。
   160  func parseUses(path, refStr, comment string) (ref, bool) {
   161  	if strings.HasPrefix(path, ".") {
   162  		return ref{}, false // ローカル参照
   163  	}
   164  	// コンテナイメージ参照。GitHub のリポジトリではないので git の ref として解決できず、
   165  	// owner/repo として分解すると `docker:/` のような無意味なキーになる。固定は digest を扱う
   166  	// pin-images の責務（docs/design/security.md）。
   167  	if strings.HasPrefix(path, dockerScheme) {
   168  		return ref{}, false
   169  	}
   170  	seg := strings.Split(path, "/")
   171  	if len(seg) < repoSegments {
   172  		return ref{}, false
   173  	}
   174  	tag := refStr
   175  	if comment != "" {
   176  		tag = comment // 既に SHA 固定済み。版はコメント側。
   177  	}
   178  	return ref{
   179  		repo: seg[0] + "/" + seg[1],
   180  		sub:  strings.Join(seg[repoSegments:], "/"),
   181  		tag:  tag,
   182  	}, true
   183  }

   185  // targetFiles は走査対象の workflow / composite action ファイルを返す。
   186  // composite action は `uses: ./.github/actions/<group>/<name>` のように入れ子へ置けるため再帰的に集める。
   187  // 1 階層の glob では入れ子のファイルが警告なく走査から消え、そこに書かれた外部参照が
   188  // 検疫・固定・drift 検査のいずれの対象からも外れる。
   189  func targetFiles(root string) ([]string, error) {
   190  	var files []string
   191  	for _, pat := range []string{
   192  		".github/workflows/*.yml", ".github/workflows/*.yaml",
   193  	} {
   194  		m, err := globFiles(root, pat)
   195  		if err != nil {
   196  			return nil, err
   197  		}
   198  		files = append(files, m...)
   199  	}
   200  
   201  	actionsDir := filepath.Join(root, ".github", "actions")
   202  	err := filepath.WalkDir(actionsDir, func(path string, d fs.DirEntry, err error) error {
   203  		if err != nil {
   204  			return err
   205  		}
   206  		if !d.IsDir() && (d.Name() == "action.yml" || d.Name() == "action.yaml") {
   207  			files = append(files, path)
   208  		}
   209  		return nil
   210  	})
   211  	if err != nil && !xerrors.Is(err, os.ErrNotExist) {
   212  		return nil, xerrors.Wrap(err, "walk .github/actions")
   213  	}
   214  
   215  	sort.Strings(files)
   216  	return files, nil
   217  }

   241  // detectLooseUses は固定対象として解釈できない uses: を、その値の説明とともに返す（重複除去・昇順）。
   242  //
   243  // usesRe は行頭にアンカーしたブロック記法にしか一致しないが、YAML は同じ内容を flow mapping
   244  // (`- {name: Checkout, uses: actions/checkout@v4}`)、クオートしたキー (`"uses": ...`)、値を次行へ送る
   245  // ブロックスカラー (`uses: >-`) でも等価に書ける。いずれも usesRe が一致ゼロになり、その状態は
   246  // 「固定漏れ無し」と区別が付かない。緩いパターンで補い、解釈できない値が残れば呼び出し元が
   247  // fail-close する。ローカル参照と版を持たない参照は誤検知を避けるため対象外。
   248  //
   249  // ブロックスカラーの中身は YAML の構造ではなく単なるテキストなので走査から外す。外さないと
   250  // `run:` スクリプトが uses: を含む文字列を出力するだけで検出が誤爆する。
   251  func detectLooseUses(data string) []string {
   252  	var found []string
   253  	blockIndent := -1
   254  	for line := range strings.SplitSeq(data, "\n") {
   255  		trimmed := strings.TrimSpace(line)
   256  		indent := len(line) - len(strings.TrimLeft(line, " \t"))
   257  		if blockIndent >= 0 {
   258  			if trimmed == "" || indent > blockIndent {
   259  				continue
   260  			}
   261  			blockIndent = -1
   262  		}
   263  		if trimmed == "" || strings.HasPrefix(trimmed, "#") || usesRe.MatchString(line) {
   264  			continue
   265  		}
   266  		if m := looseUsesRe.FindStringSubmatch(line); m != nil {
   267  			if note, loose := looseUsesValue(m[1]); loose {
   268  				found = append(found, note)
   269  			}
   270  		}
   271  		if blockScalarRe.MatchString(line) {
   272  			blockIndent = indent
   273  		}
   274  	}
   275  	sort.Strings(found)
   276  	return uniq(found)
   277  }

   279  // looseUsesValue は緩いパターンで捉えた uses: の値を分類する。第2戻り値が true なら固定対象として
   280  // 解釈できず、呼び出し元が fail-close すべきもの。第1戻り値はその値を指す報告用の文言。
   281  func looseUsesValue(rest string) (string, bool) {
   282  	v := rest
   283  	if i := strings.IndexAny(v, ",}#"); i >= 0 {
   284  		v = v[:i]
   285  	}
   286  	v = strings.Trim(strings.TrimSpace(v), `"'`)
   287  	if f := strings.Fields(v); len(f) > 0 {
   288  		v = f[0]
   289  	}
   290  	switch {
   291  	case v == "" || strings.HasPrefix(v, "|") || strings.HasPrefix(v, ">"):
   292  		return "uses: の値が同じ行にありません", true
   293  	case strings.HasPrefix(v, "*") || strings.HasPrefix(v, "&"):
   294  		return v + "（YAML の alias / anchor は解決できません）", true
   295  	case strings.HasPrefix(v, "."), !strings.Contains(v, "@"):
   296  		return "", false
   297  	default:
   298  		return v, true
   299  	}
   300  }

   302  // resolve は走査対象の参照を SHA へ解決して lockfile を書き直す。
   303  // apiBase は経過日数を問い合わせる GitHub API のルート URL（本番は githubAPIBase）。
   304  func resolve(root, apiBase string, files []string, minAgeDays int) error {
   305  	keys, err := collectKeys(root, files)
   306  	if err != nil {
   307  		return err
   308  	}
   309  	// lockfile 不在（初回）は空マップで続行するが、それ以外の読み込み失敗は握り潰さず fail-close する
   310  	// （既存ピンが lock から脱落し供給網ガードの維持保証が破れるのを防ぐ。applyOrCheck と対称）。
   311  	existing, err := readLock(filepath.Join(root, lockFile))
   312  	if !isIgnorableLockErr(err) {
   313  		return xerrors.Wrap(err, "read lockfile")
   314  	}
   315  
   316  	ctx := context.Background()
   317  	lock := map[string]string{}
   318  	var notes []string
   319  	for k, r := range keys {
   320  		sha, err := resolveSHA(ctx, r.repo, r.tag)
   321  		if err != nil {
   322  			return xerrors.Wrap(err, "resolve "+k)
   323  		}
   324  		ageFn := func() (int, error) { return refAgeDays(ctx, apiBase, r.repo, r.tag, sha) }
   325  		use, note, err := quarantine(ageFn, k, sha, minAgeDays, existing)
   326  		if err != nil {
   327  			return xerrors.Wrap(err, "age "+k)
   328  		}
   329  		if note != "" {
   330  			notes = append(notes, note)
   331  		}
   332  		if use == "" {
   333  			continue
   334  		}
   335  		lock[k] = use
   336  		log.Printf("  %s -> %s", k, use)
   337  	}
   338  	sort.Strings(notes)
   339  	for _, n := range notes {
   340  		log.Printf("  ⚠️ %s", n)
   341  	}
   342  
   343  	if err := writeLock(filepath.Join(root, lockFile), lock); err != nil {
   344  		return xerrors.Wrap(err, "write lockfile")
   345  	}
   346  	log.Printf("✅ %s に %d 件を書き出しました", lockFile, len(lock))
   347  
   348  	return nil
   349  }

   351  func collectKeys(root string, files []string) (map[string]ref, error) {
   352  	keys := map[string]ref{}
   353  	var loose []string
   354  	for _, f := range files {
   355  		data, err := os.ReadFile(f) //nolint:gosec // path from cwd glob
   356  		if err != nil {
   357  			return nil, xerrors.Wrap(err, "read "+rel(root, f))
   358  		}
   359  		for _, r := range fileRefs(string(data)) {
   360  			keys[r.key()] = r
   361  		}
   362  		for _, r := range detectLooseUses(string(data)) {
   363  			loose = append(loose, rel(root, f)+": "+r)
   364  		}
   365  	}
   366  	if len(loose) > 0 {
   367  		return nil, xerrors.Wrap(errLooseUses, strings.Join(loose, ", "))
   368  	}
   369  	return keys, nil
   370  }

   372  // quarantine は minAgeDays 未満の新しすぎる解決先を採用しない。第1戻り値（採用 SHA）が "" は skip（初回かつ新しすぎ）。
   373  // 経過日数は ageFn から取得し、その失敗は err で呼び出し元へ伝播する。
   374  // minAgeDays<=0 のときは ageFn を呼ばず候補をそのまま採用する。
   375  func quarantine(ageFn func() (int, error), key, candidate string, minAgeDays int, existing map[string]string) (string, string, error) {
   376  	if minAgeDays <= 0 {
   377  		return candidate, "", nil
   378  	}
   379  	age, err := ageFn()
   380  	if err != nil {
   381  		return "", "", err
   382  	}
   383  	if age >= minAgeDays {
   384  		return candidate, "", nil
   385  	}
   386  	if prev, ok := existing[key]; ok {
   387  		return prev, fmt.Sprintf("%s: 解決先が %d 日 (<%d) のため既存ピンを維持", key, age, minAgeDays), nil
   388  	}
   389  	return "", fmt.Sprintf("%s: 解決先が %d 日 (<%d)・既存ピン無しのため skip", key, age, minAgeDays), nil
   390  }

   392  // refAgeDays は解決先の経過日数を返す。tag に対応する Release の published_at と sha の commit 日時を
   393  // 常に両方取得し、新しい方から算出する。apiBase は GitHub API のルート URL（本番は githubAPIBase）。
   394  // Release が無い（404）ときは commit 日時のみで算出し、それ以外の非 200 はエラーにする。
   395  func refAgeDays(ctx context.Context, apiBase, repo, tag, sha string) (int, error) {
   396  	var release struct {
   397  		//nolint:tagliatelle // GitHub API のレスポンスフィールド名(published_at)に合わせる必要があるため
   398  		PublishedAt time.Time `json:"published_at"`
   399  	}
   400  	st, err := githubGet(ctx, apiBase+"/repos/"+repo+"/releases/tags/"+url.PathEscape(tag), &release)
   401  	if err != nil {
   402  		return 0, err
   403  	}
   404  	if st != http.StatusOK && st != http.StatusNotFound {
   405  		return 0, xerrors.Wrap(errGitHubAPIStatus, fmt.Sprintf("releases/tags/%s status=%d", tag, st))
   406  	}
   407  	var published time.Time
   408  	if st == http.StatusOK {
   409  		published = release.PublishedAt
   410  	}
   411  
   412  	var commit struct {
   413  		Commit struct {
   414  			Committer struct {
   415  				Date time.Time `json:"date"`
   416  			} `json:"committer"`
   417  		} `json:"commit"`
   418  	}
   419  	st, err = githubGet(ctx, apiBase+"/repos/"+repo+"/commits/"+url.PathEscape(sha), &commit)
   420  	if err != nil {
   421  		return 0, err
   422  	}
   423  	if st != http.StatusOK {
   424  		return 0, xerrors.Wrap(errGitHubAPIStatus, fmt.Sprintf("commits/%s status=%d", sha, st))
   425  	}
   426  
   427  	t, err := pickRefTime(published, commit.Commit.Committer.Date)
   428  	if err != nil {
   429  		return 0, err
   430  	}
   431  	return daysSince(t), nil
   432  }

   434  // pickRefTime は Release の公開日時と commit の日時のうち新しい方を返す。どちらも不明ならエラー。
   435  //
   436  // Release は tag 名にしか紐づかず、tag が別の commit へ付け替えられても published_at は据え置かれる。
   437  // published_at だけを見ると、この機構が想定する脅威そのもの（侵害されたアカウントによる moving tag の
   438  // 付け替え）に対して何年も前の日付が返り、検疫が常に素通りする。逆に commit の committer date は
   439  // GIT_COMMITTER_DATE で任意の過去日時へ設定できる。どちらか一方でも「新しい」と言う限り検疫が掛かる
   440  // よう新しい方を採る。日付偽装そのものに耐える保証ではなく、付け替えの検知は lockfile の差分レビューが
   441  // 担う（docs/design/security.md）。
   442  func pickRefTime(published, committed time.Time) (time.Time, error) {
   443  	switch {
   444  	case published.IsZero() && committed.IsZero():
   445  		return time.Time{}, errRefDateUnavailable
   446  	case published.After(committed):
   447  		return published, nil
   448  	default:
   449  		return committed, nil
   450  	}
   451  }

   482  // rewritePins は lock を元に uses: を固定した内容と、lock 未登録の参照キー一覧を返す。
   483  func rewritePins(data string, lock map[string]string) (string, []string) {
   484  	var missing []string
   485  	out := usesRe.ReplaceAllStringFunc(data, func(line string) string {
   486  		m := usesRe.FindStringSubmatch(line)
   487  		r, ok := parseUses(m[2], m[3], m[4])
   488  		if !ok {
   489  			return line
   490  		}
   491  		sha, found := lock[r.key()]
   492  		if !found {
   493  			missing = append(missing, r.key())
   494  			return line
   495  		}
   496  		path := r.repo
   497  		if r.sub != "" {
   498  			path += "/" + r.sub
   499  		}
   500  		return fmt.Sprintf("%s%s@%s # %s", m[1], path, sha, r.tag)
   501  	})
   502  	return out, missing
   503  }

   505  // planRewrites は全ファイルを読み切り、固定後の内容と fail-close 条件を確定させる。
   506  // 1 ファイルずつ書きながら進むと、未登録参照で中断したときに「exit 1 なのに作業ツリーは書き換え済み」
   507  // という中途半端な状態が残るため、判定と書き込みを分ける。
   508  func planRewrites(root string, files []string, lock map[string]string) (*rewritePlan, error) {
   509  	plan := &rewritePlan{changes: map[string]string{}, used: map[string]bool{}}
   510  	for _, f := range files {
   511  		data, err := os.ReadFile(f) //nolint:gosec // path from cwd glob
   512  		if err != nil {
   513  			return nil, xerrors.Wrap(err, "read "+rel(root, f))
   514  		}
   515  		for _, r := range detectLooseUses(string(data)) {
   516  			plan.loose = append(plan.loose, rel(root, f)+": "+r)
   517  		}
   518  		for _, r := range fileRefs(string(data)) {
   519  			plan.used[r.key()] = true
   520  		}
   521  		out, miss := rewritePins(string(data), lock)
   522  		plan.missing = append(plan.missing, miss...)
   523  		if out != string(data) {
   524  			plan.changes[f] = out
   525  		}
   526  	}
   527  	return plan, nil
   528  }

   530  // validate は書き込み前に中断すべき条件を返す。
   531  func (p *rewritePlan) validate(lock map[string]string) error {
   532  	if len(p.loose) > 0 {
   533  		sort.Strings(p.loose)
   534  		return xerrors.Wrap(errLooseUses, strings.Join(uniq(p.loose), ", "))
   535  	}
   536  	if len(p.missing) > 0 {
   537  		sort.Strings(p.missing)
   538  		return xerrors.Wrap(errLockMissingKey,
   539  			strings.Join(uniq(p.missing), ", ")+"（make pin-actions-resolve を実行してください）")
   540  	}
   541  	if orphans := orphanKeys(lock, p.used); len(orphans) > 0 {
   542  		return xerrors.Wrap(errLockOrphanKey,
   543  			strings.Join(orphans, ", ")+"（make pin-actions-resolve を実行するか該当行を削除してください）")
   544  	}
   545  	return nil
   546  }

   548  // orphanKeys は lockfile にあるがどの uses: からも参照されないキーを返す。
   549  // 孤児を残すと lockfile が現用インベントリの鏡でなくなり、レビューで lock の差分だけを読めば足りるという
   550  // 前提が崩れる。
   551  func orphanKeys(lock map[string]string, used map[string]bool) []string {
   552  	var orphans []string
   553  	for k := range lock {
   554  		if !used[k] {
   555  			orphans = append(orphans, k)

   557  	}
   558  	sort.Strings(orphans)
   559  	return orphans

   562  // applyOrCheck は lockfile を SSOT に uses: を固定する。dryRun=true は書き換えず drift を非ゼロ終了で報告する。
   563  func applyOrCheck(root string, files []string, dryRun bool) error {
   564  	lock, err := readLock(filepath.Join(root, lockFile))
   565  	if err != nil {
   566  		return xerrors.Wrap(err, "read lockfile（先に resolve を実行してください）")
   567  	}
   568  	plan, err := planRewrites(root, files, lock)
   569  	if err != nil {
   570  		return err
   571  	}
   572  	if err := plan.validate(lock); err != nil {
   573  		return err
   574  	}
   575  
   576  	paths := sortedChangePaths(plan.changes)
   577  
   578  	if dryRun {
   579  		if len(paths) > 0 {
   580  			return xerrors.Wrap(errPinDrift,
   581  				"make pin-actions-resolve && make pin-actions-apply してコミットしてください: "+
   582  					strings.Join(relAll(root, paths), ", "))
   583  		}
   584  		log.Printf("✅ 全アクションが lockfile 通りに固定されています")
   585  
   586  		return nil
   587  	}
   588  	for _, f := range paths {
   589  		if err := os.WriteFile(f, []byte(plan.changes[f]), filePerm); err != nil {
   590  			return xerrors.Wrap(err, "write "+rel(root, f))
   591  		}
   592  		log.Printf("  updated %s", rel(root, f))
   593  	}
   594  	log.Printf("✅ %d ファイルを固定しました", len(paths))
   595  
   596  	return nil
   597  }

   620  // resolveSHA は owner/repo の tag/branch を commit SHA へ解決する。annotated tag は ^{} で deref する。
   621  func resolveSHA(ctx context.Context, repo, tag string) (string, error) {
   622  	cctx, cancel := context.WithTimeout(ctx, lsRemoteTimeout)
   623  	defer cancel()
   624  	url := "https://github.com/" + repo
   625  	// --end-of-options 以降を確実に ref 名として扱わせ、`-` 始まりの tag のオプション誤解釈を防ぐ。
   626  	out, err := exec.CommandContext(cctx, "git", "ls-remote", url, "--end-of-options", tag, tag+"^{}").Output() //nolint:gosec // 参照名は workflow 由来
   627  	if err != nil {
   628  		return "", xerrors.Wrap(err, "git ls-remote")
   629  	}
   630  	return selectSHA(string(out), tag)
   631  }
   632  
   633  // selectSHA は git ls-remote の生出力から tag に対応する commit SHA を選ぶ。
   634  // 優先順位は annotated tag の deref (^{}) > 軽量 tag > branch head。いずれも無ければ未発見エラー。
   635  func selectSHA(out, tag string) (string, error) {
   636  	var tagSHA, derefSHA, headSHA string
   637  	for line := range strings.SplitSeq(strings.TrimSpace(out), "\n") {
   638  		parts := strings.Fields(line)
   639  		if len(parts) != lsRemoteCols {
   640  			continue
   641  		}
   642  		sha, name := parts[0], parts[1]
   643  		switch name {
   644  		case "refs/tags/" + tag + "^{}":
   645  			derefSHA = sha
   646  		case "refs/tags/" + tag:
   647  			tagSHA = sha
   648  		case "refs/heads/" + tag:
   649  			headSHA = sha
   650  		}
   651  	}
   652  	switch {
   653  	case derefSHA != "": // annotated tag は deref した commit を採用
   654  		return derefSHA, nil
   655  	case tagSHA != "":
   656  		return tagSHA, nil
   657  	case headSHA != "":
   658  		return headSHA, nil
   659  	default:
   660  		return "", xerrors.Wrap(errRefNotFound, fmt.Sprintf("%q", tag))
   661  	}
   662  }

   685  // readLock は lockfile を repo@tag→SHA として読む。空行とコメント行以外で代入として解釈できない行、
   686  // および既出キーの再定義はエラーにする。読み飛ばしや後勝ちの上書きは、そのエントリが「存在しない」
   687  // あるいは「行順で決まる」状態を警告なく作り、lockfile が SSOT として機能しなくなる。
   688  func readLock(path string) (map[string]string, error) {
   689  	f, err := os.Open(path) //nolint:gosec // path is constructed from cwd + literal filename
   690  	if err != nil {
   691  		return nil, err
   692  	}
   693  	defer func() { _ = f.Close() }()
   694  	lock := map[string]string{}
   695  	sc := bufio.NewScanner(f)
   696  	for lineNo := 1; sc.Scan(); lineNo++ {
   697  		line := strings.TrimSpace(sc.Text())
   698  		if line == "" || strings.HasPrefix(line, "#") {
   699  			continue
   700  		}
   701  		m := lockRe.FindStringSubmatch(line)
   702  		if m == nil {
   703  			return nil, xerrors.Wrap(errLockInvalidLine,
   704  				fmt.Sprintf("%d 行目: %q（make pin-actions-resolve を実行するか該当行を削除してください）", lineNo, line))
   705  		}
   706  		if _, dup := lock[m[1]]; dup {
   707  			return nil, xerrors.Wrap(errLockDuplicateKey,
   708  				fmt.Sprintf("%d 行目: %q（make pin-actions-resolve を実行するか重複行を削除してください）", lineNo, m[1]))
   709  		}
   710  		lock[m[1]] = m[2]
   711  	}
   712  	return lock, sc.Err()
   713  }
