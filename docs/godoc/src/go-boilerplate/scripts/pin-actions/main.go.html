     1  // Package main は GitHub Actions の `uses:` 参照を不変の commit SHA へ固定するツール。
     2  //
     3  //	resolve: .github/workflows/** と .github/actions/** の外部アクション参照を走査し、
     4  //	         tag/version を git ls-remote で SHA へ解決して lockfile (.github/actions-pin.toml) へ書き出す。
     5  //	apply:   lockfile を SSOT に各ファイルの `uses: owner/repo[/sub]@<ref>` を
     6  //	         `uses: owner/repo[/sub]@<sha> # <ref>` へ置換する。
     7  //	check:   apply と同じ判定を書き換えなしで行い、未固定/古い/未登録があれば非ゼロ終了する（CI / hook 用）。
     8  //
     9  // 既に固定済み (`@<sha> # <ref>`) の行は、コメントの <ref> を版として再解決するため idempotent。
    10  // ローカル参照 (`uses: ./...`) は @ を持たないため対象外。

    43  	// uses: [-] owner/repo[/sub]@<ref> [# <tag>]
    44  	// 空白は `[ \t]` に限定する（`\s` だと改行を食って行が結合する）。

    46  	lockRe = regexp.MustCompile(`^"([^"]+)"\s*=\s*"([0-9a-f]{40})"`)

    58  func main() {
    59  	log.SetFlags(0)
    60  	if len(os.Args) < minArgs {
    61  		log.Fatalf("❌ usage: pin-actions <resolve|apply|check>")
    62  	}
    63  	root, err := os.Getwd()
    64  	if err != nil {
    65  		log.Fatalf("❌ getwd: %v", err)
    66  	}
    67  	files, err := targetFiles(root)
    68  	if err != nil {
    69  		log.Fatalf("❌ %v", err)
    70  	}
    71  	switch os.Args[1] {
    72  	case "resolve":
    73  		fs := flag.NewFlagSet("resolve", flag.ExitOnError)
    74  		minAge := fs.Int("min-age-days", 0, "N 日未満のコミットは quarantine（0 で無効）")
    75  		_ = fs.Parse(os.Args[2:])
    76  		resolve(root, files, *minAge)
    77  	case "apply":
    78  		applyOrCheck(root, files, false)
    79  	case "check":
    80  		applyOrCheck(root, files, true)
    81  	default:
    82  		log.Fatalf("❌ usage: pin-actions <resolve|apply|check>")
    83  	}
    84  }

    86  // parseUses は uses: 行の path / ref / コメント tag から ref を構築する。第2戻り値が false なら対象外。
    87  func parseUses(path, refStr, comment string) (ref, bool) {
    88  	if strings.HasPrefix(path, ".") {
    89  		return ref{}, false // ローカル参照
    90  	}
    91  	seg := strings.Split(path, "/")
    92  	if len(seg) < repoSegments {
    93  		return ref{}, false
    94  	}
    95  	tag := refStr
    96  	if comment != "" {
    97  		tag = comment // 既に SHA 固定済み。版はコメント側。
    98  	}
    99  	return ref{
   100  		repo: seg[0] + "/" + seg[1],
   101  		sub:  strings.Join(seg[repoSegments:], "/"),
   102  		tag:  tag,
   103  	}, true
   104  }

   106  func targetFiles(root string) ([]string, error) {
   107  	var files []string
   108  	for _, pat := range []string{
   109  		".github/workflows/*.yml", ".github/workflows/*.yaml",
   110  		".github/actions/*/action.yml", ".github/actions/*/action.yaml",
   111  	} {
   112  		m, err := filepath.Glob(filepath.Join(root, pat))
   113  		if err != nil {
   114  			return nil, xerrors.Wrap(err, "glob "+pat)
   115  		}
   116  		files = append(files, m...)
   117  	}
   118  	sort.Strings(files)
   119  	return files, nil
   120  }

   122  func resolve(root string, files []string, minAgeDays int) {
   123  	keys := collectKeys(files)
   124  	existing, _ := readLock(filepath.Join(root, lockFile)) // 無ければ空
   125  
   126  	ctx := context.Background()
   127  	lock := map[string]string{}
   128  	var notes []string
   129  	for k, r := range keys {
   130  		sha, err := resolveSHA(ctx, r.repo, r.tag)
   131  		if err != nil {
   132  			log.Fatalf("❌ resolve %s: %v", k, err)
   133  		}
   134  		use, note := quarantine(ctx, r.repo, r.tag, k, sha, minAgeDays, existing)
   135  		if note != "" {
   136  			notes = append(notes, note)
   137  		}
   138  		if use == "" {
   139  			continue
   140  		}
   141  		lock[k] = use
   142  		log.Printf("  %s -> %s", k, use)
   143  	}
   144  	sort.Strings(notes)
   145  	for _, n := range notes {
   146  		log.Printf("  ⚠️ %s", n)
   147  	}
   148  
   149  	if err := writeLock(filepath.Join(root, lockFile), lock); err != nil {
   150  		log.Fatalf("❌ write lockfile: %v", err)
   151  	}
   152  	log.Printf("✅ %s に %d 件を書き出しました", lockFile, len(lock))
   153  }

   155  func collectKeys(files []string) map[string]ref {
   156  	keys := map[string]ref{}
   157  	for _, f := range files {
   158  		data, err := os.ReadFile(f) //nolint:gosec // path from cwd glob
   159  		if err != nil {
   160  			log.Fatalf("❌ read %s: %v", f, err)
   161  		}
   162  		for _, m := range usesRe.FindAllStringSubmatch(string(data), -1) {
   163  			if r, ok := parseUses(m[2], m[3], m[4]); ok {
   164  				keys[r.key()] = r
   165  			}
   166  		}
   167  	}
   168  	return keys
   169  }

   171  // quarantine は minAgeDays 未満の新しすぎる解決先を採用しない。第1戻り値 "" は skip（初回かつ新しすぎ）。
   172  func quarantine(ctx context.Context, repo, tag, key, candidate string, minAgeDays int, existing map[string]string) (string, string) {
   173  	if minAgeDays <= 0 {
   174  		return candidate, ""
   175  	}
   176  	age, err := refAgeDays(ctx, repo, tag, candidate)
   177  	if err != nil {
   178  		log.Fatalf("❌ age %s: %v", key, err)
   179  	}
   180  	if age >= minAgeDays {
   181  		return candidate, ""
   182  	}
   183  	if prev, ok := existing[key]; ok {
   184  		return prev, fmt.Sprintf("%s: 解決先が %d 日 (<%d) のため既存ピンを維持", key, age, minAgeDays)
   185  	}
   186  	return "", fmt.Sprintf("%s: 解決先が %d 日 (<%d)・既存ピン無しのため skip", key, age, minAgeDays)
   187  }

   189  // refAgeDays は解決先の経過日数を返す。タグに対応する Release があれば published_at（偽装困難）を、
   190  // 無ければ commit の committer date をフォールバックに使う。
   191  func refAgeDays(ctx context.Context, repo, tag, sha string) (int, error) {
   192  	var rel struct {
   193  		//nolint:tagliatelle // GitHub API のレスポンスフィールド名(published_at)に合わせる必要があるため
   194  		PublishedAt time.Time `json:"published_at"`
   195  	}
   196  	st, err := githubGet(ctx, "https://api.github.com/repos/"+repo+"/releases/tags/"+tag, &rel)
   197  	if err != nil {
   198  		return 0, err
   199  	}
   200  	if st == http.StatusOK && !rel.PublishedAt.IsZero() {
   201  		return daysSince(rel.PublishedAt), nil
   202  	}
   203  	if st != http.StatusOK && st != http.StatusNotFound {
   204  		return 0, xerrors.New(fmt.Sprintf("releases/tags/%s: %d", tag, st))
   205  	}
   206  
   207  	var commit struct {
   208  		Commit struct {
   209  			Committer struct {
   210  				Date time.Time `json:"date"`
   211  			} `json:"committer"`
   212  		} `json:"commit"`
   213  	}
   214  	st, err = githubGet(ctx, "https://api.github.com/repos/"+repo+"/commits/"+sha, &commit)
   215  	if err != nil {
   216  		return 0, err
   217  	}
   218  	if st != http.StatusOK {
   219  		return 0, xerrors.New(fmt.Sprintf("commits/%s: %d", sha, st))
   220  	}
   221  	return daysSince(commit.Commit.Committer.Date), nil
   222  }

   224  // githubGet は GitHub API を GET し、200 のとき out に JSON をデコードして HTTP ステータスを返す。
   225  func githubGet(ctx context.Context, url string, out any) (int, error) {
   226  	cctx, cancel := context.WithTimeout(ctx, lsRemoteTimeout)
   227  	defer cancel()
   228  	req, err := http.NewRequestWithContext(cctx, http.MethodGet, url, nil)
   229  	if err != nil {
   230  		return 0, err
   231  	}
   232  	req.Header.Set("Accept", "application/vnd.github+json")
   233  	if tok := os.Getenv("GITHUB_TOKEN"); tok != "" {
   234  		req.Header.Set("Authorization", "Bearer "+tok)
   235  	}
   236  	resp, err := http.DefaultClient.Do(req)
   237  	if err != nil {
   238  		return 0, err
   239  	}
   240  	defer func() { _ = resp.Body.Close() }()
   241  	if resp.StatusCode == http.StatusOK {
   242  		if err := json.NewDecoder(resp.Body).Decode(out); err != nil {
   243  			return resp.StatusCode, err
   244  		}
   245  	}
   246  	return resp.StatusCode, nil
   247  }

   253  // rewritePins は lock を元に uses: を固定した内容と、lock 未登録の参照キー一覧を返す。
   254  func rewritePins(data string, lock map[string]string) (string, []string) {
   255  	var missing []string
   256  	out := usesRe.ReplaceAllStringFunc(data, func(line string) string {
   257  		m := usesRe.FindStringSubmatch(line)
   258  		r, ok := parseUses(m[2], m[3], m[4])
   259  		if !ok {
   260  			return line
   261  		}
   262  		sha, found := lock[r.key()]
   263  		if !found {
   264  			missing = append(missing, r.key())
   265  			return line
   266  		}
   267  		path := r.repo
   268  		if r.sub != "" {
   269  			path += "/" + r.sub
   270  		}
   271  		return fmt.Sprintf("%s%s@%s # %s", m[1], path, sha, r.tag)
   272  	})
   273  	return out, missing
   274  }

   276  // applyOrCheck は lockfile を SSOT に uses: を固定する。dryRun=true は書き換えず drift を非ゼロ終了で報告する。
   277  func applyOrCheck(root string, files []string, dryRun bool) {
   278  	lock, err := readLock(filepath.Join(root, lockFile))
   279  	if err != nil {
   280  		log.Fatalf("❌ read lockfile（先に resolve を実行してください）: %v", err)
   281  	}
   282  	var missing, drifted []string
   283  	changed := 0
   284  	for _, f := range files {
   285  		data, err := os.ReadFile(f) //nolint:gosec // path from cwd glob
   286  		if err != nil {
   287  			log.Fatalf("❌ read %s: %v", f, err)
   288  		}
   289  		out, miss := rewritePins(string(data), lock)
   290  		missing = append(missing, miss...)
   291  		if out == string(data) {
   292  			continue
   293  		}
   294  		if dryRun {
   295  			drifted = append(drifted, rel(root, f))
   296  			continue
   297  		}
   298  		if err := os.WriteFile(f, []byte(out), filePerm); err != nil { //nolint:gosec // workflow ファイル権限
   299  			log.Fatalf("❌ write %s: %v", f, err)
   300  		}
   301  		changed++
   302  		log.Printf("  updated %s", rel(root, f))
   303  	}
   304  	report(missing, drifted, dryRun, changed)
   305  }

   307  func report(missing, drifted []string, dryRun bool, changed int) {
   308  	if len(missing) > 0 {
   309  		sort.Strings(missing)
   310  		log.Fatalf("❌ lockfile に未登録の参照があります（make pin-actions-resolve を実行してください）: %s",
   311  			strings.Join(uniq(missing), ", "))
   312  	}
   313  	if !dryRun {
   314  		log.Printf("✅ %d ファイルを固定しました", changed)
   315  		return
   316  	}
   317  	if len(drifted) > 0 {
   318  		sort.Strings(drifted)
   319  		log.Fatalf("❌ 未固定/古い参照があります（make pin-actions-resolve && make pin-actions-apply してコミットしてください）: %s",
   320  			strings.Join(drifted, ", "))
   321  	}
   322  	log.Printf("✅ 全アクションが lockfile 通りに固定されています")
   323  }

   325  // resolveSHA は owner/repo の tag/branch を commit SHA へ解決する。annotated tag は ^{} で deref する。
   326  func resolveSHA(ctx context.Context, repo, tag string) (string, error) {
   327  	cctx, cancel := context.WithTimeout(ctx, lsRemoteTimeout)
   328  	defer cancel()
   329  	url := "https://github.com/" + repo
   330  	out, err := exec.CommandContext(cctx, "git", "ls-remote", url, tag, tag+"^{}").Output() //nolint:gosec // 参照名は workflow 由来
   331  	if err != nil {
   332  		return "", xerrors.Wrap(err, "git ls-remote")
   333  	}
   334  	var tagSHA, derefSHA, headSHA string
   335  	for line := range strings.SplitSeq(strings.TrimSpace(string(out)), "\n") {
   336  		parts := strings.Fields(line)
   337  		if len(parts) != lsRemoteCols {
   338  			continue
   339  		}
   340  		sha, name := parts[0], parts[1]
   341  		switch name {
   342  		case "refs/tags/" + tag + "^{}":
   343  			derefSHA = sha
   344  		case "refs/tags/" + tag:
   345  			tagSHA = sha
   346  		case "refs/heads/" + tag:
   347  			headSHA = sha
   348  		}
   349  	}
   350  	switch {
   351  	case derefSHA != "": // annotated tag は deref した commit を採用
   352  		return derefSHA, nil
   353  	case tagSHA != "":
   354  		return tagSHA, nil
   355  	case headSHA != "":
   356  		return headSHA, nil
   357  	default:
   358  		return "", xerrors.New(fmt.Sprintf("ref %q が見つかりません", tag))
   359  	}
   360  }

   362  func writeLock(path string, lock map[string]string) error {
   363  	keys := make([]string, 0, len(lock))
   364  	for k := range lock {
   365  		keys = append(keys, k)
   366  	}
   367  	sort.Strings(keys)
   368  	var b strings.Builder
   369  	b.WriteString("# GitHub Actions の pin 対象 SHA（SSOT）。\n")
   370  	b.WriteString("# make pin-actions-resolve で解決・make pin-actions-apply で workflow へ反映する。\n")
   371  	for _, k := range keys {
   372  		fmt.Fprintf(&b, "%q = %q\n", k, lock[k])
   373  	}
   374  	return os.WriteFile(path, []byte(b.String()), filePerm)
   375  }

   377  func readLock(path string) (map[string]string, error) {
   378  	f, err := os.Open(path) //nolint:gosec // path is constructed from cwd + literal filename
   379  	if err != nil {
   380  		return nil, err
   381  	}
   382  	defer func() { _ = f.Close() }()
   383  	lock := map[string]string{}
   384  	sc := bufio.NewScanner(f)
   385  	for sc.Scan() {
   386  		if m := lockRe.FindStringSubmatch(strings.TrimSpace(sc.Text())); m != nil {
   387  			lock[m[1]] = m[2]
   388  		}
   389  	}
   390  	return lock, sc.Err()
   391  }

   400  func uniq(s []string) []string {
   401  	out := s[:0]
   402  	var prev string
   403  	for i, v := range s {
   404  		if i == 0 || v != prev {
   405  			out = append(out, v)
   406  		}
   407  		prev = v
   408  	}
   409  	return out
