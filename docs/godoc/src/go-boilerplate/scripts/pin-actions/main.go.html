     1  // pin-actions は GitHub Actions の `uses:` 参照を不変の commit SHA へ固定するツール。
     2  //
     3  //	resolve: .github/workflows/** と .github/actions/** の外部アクション参照を走査し、
     4  //	         tag/version を git ls-remote で SHA へ解決して lockfile (.github/actions-pin.toml) へ書き出す。
     5  //	apply:   lockfile を SSOT に各ファイルの `uses: owner/repo[/sub]@<ref>` を
     6  //	         `uses: owner/repo[/sub]@<sha> # <ref>` へ置換する。
     7  //	check:   apply と同じ判定を書き換えなしで行い、未固定/古い/未登録があれば非ゼロ終了する（CI / hook 用）。
     8  //
     9  // 既に固定済み (`@<sha> # <ref>`) の行は、コメントの <ref> を版として再解決するため idempotent。
    10  // ローカル参照 (`uses: ./...`) は @ を持たないため対象外。

    40  var (
    41  	// uses: [-] owner/repo[/sub]@<ref> [# <tag>]
    42  	// 空白は `[ \t]` に限定する（`\s` だと改行を食って行が結合する）。
    43  	usesRe = regexp.MustCompile(`(?m)^([ \t]*(?:-[ \t]*)?uses:[ \t]*)([^@\s]+)@([^\s#]+)(?:[ \t]*#[ \t]*(\S+))?[ \t]*$`)
    44  	lockRe = regexp.MustCompile(`^"([^"]+)"\s*=\s*"([0-9a-f]{40})"`)
    45  )

    56  func main() {
    57  	log.SetFlags(0)
    58  	if len(os.Args) < minArgs {
    59  		log.Fatalf("❌ usage: pin-actions <resolve|apply|check>")
    60  	}
    61  	root, err := os.Getwd()
    62  	if err != nil {
    63  		log.Fatalf("❌ getwd: %v", err)
    64  	}
    65  	files, err := targetFiles(root)
    66  	if err != nil {
    67  		log.Fatalf("❌ %v", err)
    68  	}
    69  	switch os.Args[1] {
    70  	case "resolve":
    71  		fs := flag.NewFlagSet("resolve", flag.ExitOnError)
    72  		minAge := fs.Int("min-age-days", 0, "N 日未満のコミットは quarantine（0 で無効）")
    73  		_ = fs.Parse(os.Args[2:])
    74  		resolve(root, files, *minAge)
    75  	case "apply":
    76  		applyOrCheck(root, files, false)
    77  	case "check":
    78  		applyOrCheck(root, files, true)
    79  	default:
    80  		log.Fatalf("❌ usage: pin-actions <resolve|apply|check>")
    81  	}
    82  }

    84  // parseUses は uses: 行の path / ref / コメント tag から ref を構築する。第2戻り値が false なら対象外。
    85  func parseUses(path, refStr, comment string) (ref, bool) {
    86  	if strings.HasPrefix(path, ".") {
    87  		return ref{}, false // ローカル参照
    88  	}
    89  	seg := strings.Split(path, "/")
    90  	if len(seg) < repoSegments {
    91  		return ref{}, false
    92  	}
    93  	tag := refStr
    94  	if comment != "" {
    95  		tag = comment // 既に SHA 固定済み。版はコメント側。
    96  	}
    97  	return ref{
    98  		repo: seg[0] + "/" + seg[1],
    99  		sub:  strings.Join(seg[repoSegments:], "/"),
   100  		tag:  tag,
   101  	}, true
   102  }

   104  func targetFiles(root string) ([]string, error) {
   105  	var files []string
   106  	for _, pat := range []string{
   107  		".github/workflows/*.yml", ".github/workflows/*.yaml",
   108  		".github/actions/*/action.yml", ".github/actions/*/action.yaml",
   109  	} {
   110  		m, err := filepath.Glob(filepath.Join(root, pat))
   111  		if err != nil {
   112  			return nil, fmt.Errorf("glob %s: %w", pat, err)
   113  		}
   114  		files = append(files, m...)
   115  	}
   116  	sort.Strings(files)
   117  	return files, nil
   118  }

   120  func resolve(root string, files []string, minAgeDays int) {
   121  	keys := collectKeys(files)
   122  	existing, _ := readLock(filepath.Join(root, lockFile)) // 無ければ空
   123  
   124  	ctx := context.Background()
   125  	lock := map[string]string{}
   126  	var notes []string
   127  	for k, r := range keys {
   128  		sha, err := resolveSHA(ctx, r.repo, r.tag)
   129  		if err != nil {
   130  			log.Fatalf("❌ resolve %s: %v", k, err)
   131  		}
   132  		use, note := quarantine(ctx, r.repo, r.tag, k, sha, minAgeDays, existing)
   133  		if note != "" {
   134  			notes = append(notes, note)
   135  		}
   136  		if use == "" {
   137  			continue
   138  		}
   139  		lock[k] = use
   140  		log.Printf("  %s -> %s", k, use)
   141  	}
   142  	sort.Strings(notes)
   143  	for _, n := range notes {
   144  		log.Printf("  ⚠️ %s", n)
   145  	}
   146  
   147  	if err := writeLock(filepath.Join(root, lockFile), lock); err != nil {
   148  		log.Fatalf("❌ write lockfile: %v", err)
   149  	}
   150  	log.Printf("✅ %s に %d 件を書き出しました", lockFile, len(lock))
   151  }

   153  func collectKeys(files []string) map[string]ref {
   154  	keys := map[string]ref{}
   155  	for _, f := range files {
   156  		data, err := os.ReadFile(f) //nolint:gosec // path from cwd glob
   157  		if err != nil {
   158  			log.Fatalf("❌ read %s: %v", f, err)
   159  		}
   160  		for _, m := range usesRe.FindAllStringSubmatch(string(data), -1) {
   161  			if r, ok := parseUses(m[2], m[3], m[4]); ok {
   162  				keys[r.key()] = r
   163  			}
   164  		}
   165  	}
   166  	return keys
   167  }

   169  // quarantine は minAgeDays 未満の新しすぎる解決先を採用しない。第1戻り値 "" は skip（初回かつ新しすぎ）。
   170  func quarantine(ctx context.Context, repo, tag, key, candidate string, minAgeDays int, existing map[string]string) (string, string) {
   171  	if minAgeDays <= 0 {
   172  		return candidate, ""
   173  	}
   174  	age, err := refAgeDays(ctx, repo, tag, candidate)
   175  	if err != nil {
   176  		log.Fatalf("❌ age %s: %v", key, err)
   177  	}
   178  	if age >= minAgeDays {
   179  		return candidate, ""
   180  	}
   181  	if prev, ok := existing[key]; ok {
   182  		return prev, fmt.Sprintf("%s: 解決先が %d 日 (<%d) のため既存ピンを維持", key, age, minAgeDays)
   183  	}
   184  	return "", fmt.Sprintf("%s: 解決先が %d 日 (<%d)・既存ピン無しのため skip", key, age, minAgeDays)
   185  }

   187  // refAgeDays は解決先の経過日数を返す。タグに対応する Release があれば published_at（偽装困難）を、
   188  // 無ければ commit の committer date をフォールバックに使う。
   189  func refAgeDays(ctx context.Context, repo, tag, sha string) (int, error) {
   190  	var rel struct {
   191  		PublishedAt time.Time `json:"published_at"`
   192  	}
   193  	st, err := githubGet(ctx, "https://api.github.com/repos/"+repo+"/releases/tags/"+tag, &rel)
   194  	if err != nil {
   195  		return 0, err
   196  	}
   197  	if st == http.StatusOK && !rel.PublishedAt.IsZero() {
   198  		return daysSince(rel.PublishedAt), nil
   199  	}
   200  	if st != http.StatusOK && st != http.StatusNotFound {
   201  		return 0, fmt.Errorf("releases/tags/%s: %d", tag, st)
   202  	}
   203  
   204  	var commit struct {
   205  		Commit struct {
   206  			Committer struct {
   207  				Date time.Time `json:"date"`
   208  			} `json:"committer"`
   209  		} `json:"commit"`
   210  	}
   211  	st, err = githubGet(ctx, "https://api.github.com/repos/"+repo+"/commits/"+sha, &commit)
   212  	if err != nil {
   213  		return 0, err
   214  	}
   215  	if st != http.StatusOK {
   216  		return 0, fmt.Errorf("commits/%s: %d", sha, st)
   217  	}
   218  	return daysSince(commit.Commit.Committer.Date), nil
   219  }

   221  // githubGet は GitHub API を GET し、200 のとき out に JSON をデコードして HTTP ステータスを返す。
   222  func githubGet(ctx context.Context, url string, out any) (int, error) {
   223  	cctx, cancel := context.WithTimeout(ctx, lsRemoteTimeout)
   224  	defer cancel()
   225  	req, err := http.NewRequestWithContext(cctx, http.MethodGet, url, nil)
   226  	if err != nil {
   227  		return 0, err
   228  	}
   229  	req.Header.Set("Accept", "application/vnd.github+json")
   230  	if tok := os.Getenv("GITHUB_TOKEN"); tok != "" {
   231  		req.Header.Set("Authorization", "Bearer "+tok)
   232  	}
   233  	resp, err := http.DefaultClient.Do(req)
   234  	if err != nil {
   235  		return 0, err
   236  	}
   237  	defer func() { _ = resp.Body.Close() }()
   238  	if resp.StatusCode == http.StatusOK {
   239  		if err := json.NewDecoder(resp.Body).Decode(out); err != nil {
   240  			return resp.StatusCode, err
   241  		}
   242  	}
   243  	return resp.StatusCode, nil
   244  }

   250  // rewritePins は lock を元に uses: を固定した内容と、lock 未登録の参照キー一覧を返す。
   251  func rewritePins(data string, lock map[string]string) (string, []string) {
   252  	var missing []string
   253  	out := usesRe.ReplaceAllStringFunc(data, func(line string) string {
   254  		m := usesRe.FindStringSubmatch(line)
   255  		r, ok := parseUses(m[2], m[3], m[4])
   256  		if !ok {
   257  			return line
   258  		}
   259  		sha, found := lock[r.key()]
   260  		if !found {
   261  			missing = append(missing, r.key())
   262  			return line
   263  		}
   264  		path := r.repo
   265  		if r.sub != "" {
   266  			path += "/" + r.sub
   267  		}
   268  		return fmt.Sprintf("%s%s@%s # %s", m[1], path, sha, r.tag)
   269  	})
   270  	return out, missing
   271  }

   273  // applyOrCheck は lockfile を SSOT に uses: を固定する。dryRun=true は書き換えず drift を非ゼロ終了で報告する。
   274  func applyOrCheck(root string, files []string, dryRun bool) {
   275  	lock, err := readLock(filepath.Join(root, lockFile))
   276  	if err != nil {
   277  		log.Fatalf("❌ read lockfile（先に resolve を実行してください）: %v", err)
   278  	}
   279  	var missing, drifted []string
   280  	changed := 0
   281  	for _, f := range files {
   282  		data, err := os.ReadFile(f) //nolint:gosec // path from cwd glob
   283  		if err != nil {
   284  			log.Fatalf("❌ read %s: %v", f, err)
   285  		}
   286  		out, miss := rewritePins(string(data), lock)
   287  		missing = append(missing, miss...)
   288  		if out == string(data) {
   289  			continue
   290  		}
   291  		if dryRun {
   292  			drifted = append(drifted, rel(root, f))
   293  			continue
   294  		}
   295  		if err := os.WriteFile(f, []byte(out), filePerm); err != nil { //nolint:gosec // workflow ファイル権限
   296  			log.Fatalf("❌ write %s: %v", f, err)
   297  		}
   298  		changed++
   299  		log.Printf("  updated %s", rel(root, f))
   300  	}
   301  	report(missing, drifted, dryRun, changed)
   302  }

   304  func report(missing, drifted []string, dryRun bool, changed int) {
   305  	if len(missing) > 0 {
   306  		sort.Strings(missing)
   307  		log.Fatalf("❌ lockfile に未登録の参照があります（make pin-actions-resolve を実行してください）: %s",
   308  			strings.Join(uniq(missing), ", "))
   309  	}
   310  	if !dryRun {
   311  		log.Printf("✅ %d ファイルを固定しました", changed)
   312  		return
   313  	}
   314  	if len(drifted) > 0 {
   315  		sort.Strings(drifted)
   316  		log.Fatalf("❌ 未固定/古い参照があります（make pin-actions-resolve && make pin-actions-apply してコミットしてください）: %s",
   317  			strings.Join(drifted, ", "))
   318  	}
   319  	log.Printf("✅ 全アクションが lockfile 通りに固定されています")
   320  }

   322  // resolveSHA は owner/repo の tag/branch を commit SHA へ解決する。annotated tag は ^{} で deref する。
   323  func resolveSHA(ctx context.Context, repo, tag string) (string, error) {
   324  	cctx, cancel := context.WithTimeout(ctx, lsRemoteTimeout)
   325  	defer cancel()
   326  	url := "https://github.com/" + repo
   327  	out, err := exec.CommandContext(cctx, "git", "ls-remote", url, tag, tag+"^{}").Output() //nolint:gosec // 参照名は workflow 由来
   328  	if err != nil {
   329  		return "", fmt.Errorf("git ls-remote: %w", err)
   330  	}
   331  	var tagSHA, derefSHA, headSHA string
   332  	for _, line := range strings.Split(strings.TrimSpace(string(out)), "\n") {
   333  		parts := strings.Fields(line)
   334  		if len(parts) != lsRemoteCols {
   335  			continue
   336  		}
   337  		sha, name := parts[0], parts[1]
   338  		switch name {
   339  		case "refs/tags/" + tag + "^{}":
   340  			derefSHA = sha
   341  		case "refs/tags/" + tag:
   342  			tagSHA = sha
   343  		case "refs/heads/" + tag:
   344  			headSHA = sha
   345  		}
   346  	}
   347  	switch {
   348  	case derefSHA != "": // annotated tag は deref した commit を採用
   349  		return derefSHA, nil
   350  	case tagSHA != "":
   351  		return tagSHA, nil
   352  	case headSHA != "":
   353  		return headSHA, nil
   354  	default:
   355  		return "", fmt.Errorf("ref %q が見つかりません", tag)
   356  	}
   357  }

   359  func writeLock(path string, lock map[string]string) error {
   360  	keys := make([]string, 0, len(lock))
   361  	for k := range lock {
   362  		keys = append(keys, k)
   363  	}
   364  	sort.Strings(keys)
   365  	var b strings.Builder
   366  	b.WriteString("# GitHub Actions の pin 対象 SHA（SSOT）。\n")
   367  	b.WriteString("# make pin-actions-resolve で解決・make pin-actions-apply で workflow へ反映する。\n")
   368  	for _, k := range keys {
   369  		fmt.Fprintf(&b, "%q = %q\n", k, lock[k])
   370  	}
   371  	return os.WriteFile(path, []byte(b.String()), filePerm)
   372  }

   374  func readLock(path string) (map[string]string, error) {
   375  	f, err := os.Open(path) //nolint:gosec // path is constructed from cwd + literal filename
   376  	if err != nil {
   377  		return nil, err
   378  	}
   379  	defer func() { _ = f.Close() }()
   380  	lock := map[string]string{}
   381  	sc := bufio.NewScanner(f)
   382  	for sc.Scan() {
   383  		if m := lockRe.FindStringSubmatch(strings.TrimSpace(sc.Text())); m != nil {
   384  			lock[m[1]] = m[2]
   385  		}
   386  	}
   387  	return lock, sc.Err()
   388  }

   397  func uniq(s []string) []string {
   398  	out := s[:0]
   399  	var prev string
   400  	for i, v := range s {
   401  		if i == 0 || v != prev {
   402  			out = append(out, v)
   403  		}
   404  		prev = v
   405  	}
   406  	return out
   407  }
