   164  // writeScripts は相対パスと内容の対からテスト用のツリーを作り、その基点を返します。
   165  func writeScripts(t *testing.T, files map[string]string) string {
   166  	t.Helper()
   167  
   168  	root := t.TempDir()
   169  	for name, body := range files {
   170  		path := filepath.Join(root, filepath.FromSlash(name))
   171  		require.NoError(t, os.MkdirAll(filepath.Dir(path), 0o750))
   172  		require.NoError(t, os.WriteFile(path, []byte(body), 0o600))
   173  	}
   174  
   175  	return root
   176  }

   183  func requireShellcheck(t *testing.T) {
   184  	t.Helper()
   185  
   186  	if _, err := exec.LookPath(shellcheck.Binary); err == nil {
   187  		return
   188  	}
   189  	if os.Getenv(requireShellcheckEnv) != "" {
   190  		t.Fatalf("shellcheck が PATH にありません（%s 指定時は skip しません）", requireShellcheckEnv)
   191  	}
   192  	t.Skip("shellcheck が PATH にありません")
   193  }
