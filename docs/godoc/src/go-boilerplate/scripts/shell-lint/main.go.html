     2  //
     3  // composite action の中のシェルは actions-shellcheck が見るが、ファイルとして置かれたシェルは
     4  // どのゲートにも掛かっていなかった。フックやセットアップの入口はそこに居る —— PreToolUse フック、
     5  // SessionStart フック、スキル同梱スクリプト、コンテナの init。いずれも編集のたび、あるいは
     6  // セッションのたびに走るのに、壊れても CI は緑を返していた。
     7  //
     8  // 内容をそのまま渡すので、指摘の行・列は写し戻さずに使える（起動の詳細は scripts/lib/shellcheck）。

    29  var skippedDirs = []string{".git", "node_modules", "vendor", "tmp"}

    41  // run はリポジトリ内のシェルスクリプトを shellcheck に掛け、結果を報告します。
    42  // wd は走査の基点となるディレクトリの取得手段、lookPath は shellcheck の所在確認手段です。
    43  func run(ctx context.Context, wd func() (string, error), lookPath func(string) (string, error)) error {
    44  	root, err := shellcheck.Setup(wd, lookPath)
    45  	if err != nil {
    46  		return err
    47  	}
    48  
    49  	scripts, err := shellScripts(root)
    50  	if err != nil {
    51  		return err
    52  	}
    53  
    54  	var findings []string
    55  	for _, script := range scripts {
    56  		// script は shellScripts が root 配下を walk して得た相対パスだけを取る。
    57  		body, err := os.ReadFile(filepath.Join(root, script)) //nolint:gosec // G304: 走査結果のみ
    58  		if err != nil {
    59  			return xerrors.Wrap(err, "read")
    60  		}
    61  
    62  		out, err := shellcheck.Run(ctx, string(body))
    63  		if err != nil {
    64  			return err
    65  		}
    66  		findings = append(findings, prefixFindings(script, out)...)
    67  	}
    68  
    69  	if len(findings) > 0 {
    70  		log.Print(strings.Join(findings, "\n"))
    71  		return xerrors.Wrap(errFindings, fmt.Sprintf("%d 件", len(findings)))
    72  	}
    73  
    74  	log.Printf("✅ シェルスクリプト %d ファイルを shellcheck で検査しました", len(scripts))
    75  
    76  	return nil
    77  }

    79  // shellScripts は root 配下の `*.sh` をリポジトリ相対パスで昇順に返します。
    80  func shellScripts(root string) ([]string, error) {
    81  	var scripts []string
    82  
    83  	err := filepath.WalkDir(root, func(path string, entry fs.DirEntry, err error) error {
    84  		if err != nil {
    85  			return err
    86  		}
    87  		if entry.IsDir() {
    88  			if slices.Contains(skippedDirs, entry.Name()) {
    89  				return fs.SkipDir
    90  			}
    91  
    92  			return nil
    93  		}
    94  		if !strings.HasSuffix(entry.Name(), shellSuffix) {
    95  			return nil
    96  		}
    97  
    98  		rel, err := filepath.Rel(root, path)
    99  		if err != nil {
   100  			return xerrors.Wrap(err, "rel")
   101  		}
   102  		scripts = append(scripts, filepath.ToSlash(rel))
   103  
   104  		return nil
   105  	})
   106  	if err != nil {
   107  		return nil, xerrors.Wrap(err, "walk")
   108  	}
   109  
   110  	slices.Sort(scripts)
   111  
   112  	return scripts, nil
   113  }

   115  // prefixFindings は shellcheck の出力を 1 行 1 指摘へ整え、先頭をリポジトリ相対パスへ差し替えます。
   116  // stdin で渡しているため shellcheck 自身は入力を `-` としか呼べず、どのファイルの指摘か言えません。
   117  func prefixFindings(script, out string) []string {
   118  	trimmed := strings.TrimSpace(out)
   119  	if trimmed == "" {
   120  		return nil
   121  	}
   122  
   123  	var findings []string
   124  	for line := range strings.SplitSeq(trimmed, "\n") {
   125  		_, rest, found := strings.Cut(line, ":")
   126  		if !found {
   127  			continue
   128  		}
   129  		findings = append(findings, script+":"+rest)
   130  	}
   131  
   132  	return findings
   133  }
