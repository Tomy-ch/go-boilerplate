     1  // Package main は、boilerplate を自分のリポジトリとして初期化する際の
     2  // git / gh 操作を担うツール。
     3  //
     4  //	preflight           初期化してよい状態かを確かめる（v0.0.0 タグがあれば中止）
     5  //	bootstrap           タグを作り直し、develop / staging / production を用意してデフォルトブランチを移す
     6  //	prune-release-notes v0.0.0.md 以外のリリースノートを削除する
     7  //
     8  // ラベル・ルールセット・ワークフローの初期化は他の make ターゲットが持つため、
     9  // 全体の連鎖は setup-repository.mk に残る。ここが持つのは git / gh の手順だけ。
    10  //
    11  // 実体が make のレシピではなくここに在るのは、タグの一括削除やデフォルトブランチの
    12  // 移動を含み、分岐を実地で確かめようとすると本当にリポジトリを壊すしかないため。
    13  // 手順の組み立てを純粋関数へ寄せ、実行層を runner として差し替え可能にして、テストで固定する。
    14  //
    15  // git / gh はホストの認証情報を使うため、ツールランナーではなくホストで実行する
    16  // （cmd/db-slot と同じ扱い）。

    60  	allowFail bool

    63  // runner は、手順を実際に走らせる実行層。タグの一括削除や origin への push を伴う
    64  // bootstrap の手順を、実リポジトリを壊さずに検証できるよう関数値で保持する。
    65  type runner struct {
    66  	// run は、手順を 1 つ実行する。
    67  	run func(s step) error
    68  	// output は、コマンドの標準出力を取り出す。
    69  	output func(name string, args ...string) (string, error)
    70  	// branchExists は、ローカルブランチの有無を返す。
    71  	branchExists func(branch string) bool
    72  }

    74  // main はエラーを終了コードへ変換するだけに留め、判断は execute が持ちます。

    76  // コードがそのぶん増える。名前が run でないのは、1 コマンドを起動する run が
    77  // 同じパッケージに既に在るため。
    78  func main() {
    79  	log.SetFlags(0)
    80  
    81  	if err := execute(hostRunner(), os.Args[1:]); err != nil {
    82  		log.Fatalf("%v", err)
    83  	}
    84  }

    86  // execute は、サブコマンドを選んで実行します。
    87  func execute(r runner, args []string) error {
    88  	if len(args) == 0 {
    89  		return errUsage
    90  	}
    91  
    92  	switch args[0] {
    93  	case "preflight":
    94  		return runPreflight()
    95  	case "bootstrap":
    96  		return runBootstrap(r)
    97  	case "prune-release-notes":
    98  		return runPruneReleaseNotes()
    99  	default:
   100  		return errUnknownSubcommand
   101  	}
   102  }

   108  // parseLines は、コマンド出力を空行を除いた行の並びへ変換します。
   109  func parseLines(out string) []string {
   110  	lines := make([]string, 0)
   111  
   112  	for line := range strings.SplitSeq(out, "\n") {
   113  		if t := strings.TrimSpace(line); t != "" {
   114  			lines = append(lines, t)
   115  		}
   116  	}
   117  
   118  	return lines
   119  }

   121  // tagDeletionSteps は、既存タグをローカルとリモートの両方から消す手順を返します。
   122  // リモートに存在しないタグの削除は失敗しても続行します（ローカルにだけ在るタグは正常な状態）。
   123  func tagDeletionSteps(tags []string) []step {
   124  	steps := make([]step, 0, len(tags)*stepsPerTag)
   125  
   126  	for _, tag := range tags {
   127  		steps = append(steps,
   128  			step{name: "git", args: []string{"tag", "-d", tag}},
   129  			step{name: "git", args: []string{"push", "origin", ":refs/tags/" + tag}, allowFail: true},
   130  		)
   131  	}
   132  
   133  	return steps
   134  }

   137  func initialTagSteps() []step {
   138  	return []step{
   139  		{name: "git", args: []string{"tag", "-a", initialTag, "-m", "Initial boilerplate tag"}},
   140  		{name: "git", args: []string{"push", "origin", initialTag}},
   141  	}
   142  }

   144  // branchCreationSteps は、まだ無いブランチだけを作る手順と、既存のためスキップした名前を返します。
   145  func branchCreationSteps(existing []string) ([]step, []string) {
   146  	have := make(map[string]bool, len(existing))
   147  	for _, b := range existing {
   148  		have[b] = true
   149  	}
   150  
   151  	steps := make([]step, 0, len(managedBranches))
   152  	skipped := make([]string, 0)
   153  
   154  	for _, b := range managedBranches {
   155  		if have[b] {
   156  			skipped = append(skipped, b)
   157  
   158  			continue
   159  		}
   160  
   161  		steps = append(steps, step{name: "git", args: []string{"branch", b}})
   162  	}
   163  
   164  	return steps, skipped
   165  }

   167  // branchPushStep は、用意したブランチをまとめて push する手順を返します。
   168  func branchPushStep() step {
   169  	return step{name: "git", args: append([]string{"push", "origin"}, managedBranches...)}
   170  }

   172  // defaultBranchStep は、GitHub 上のデフォルトブランチを移す手順を返します。
   173  func defaultBranchStep(repo string) step {
   174  	return step{name: "gh", args: []string{"api", "-X", "PATCH", "repos/" + repo, "-f", "default_branch=" + defaultBranch}}
   175  }

   177  // isReleaseBranch は、初期化時に破棄してよいリリースブランチかを返します。
   178  // 判定は前方一致ではなく部分一致（hotfix/release/... のような名前も対象に含める）。
   179  func isReleaseBranch(name string) bool {
   180  	return strings.Contains(name, "release/")
   181  }

   183  // originalBranchCleanupSteps は、初期化前に居たブランチがリリースブランチだった場合に
   184  // それを削除する手順を返します。リリースブランチでなければ何もしません。
   185  // リモートに未 push のブランチもあり得るため、リモート削除の失敗は許容します。
   186  func originalBranchCleanupSteps(original string) []step {
   187  	if !isReleaseBranch(original) {
   188  		return nil
   189  	}
   190  
   191  	return []step{
   192  		{name: "git", args: []string{"branch", "-D", original}},
   193  		{name: "git", args: []string{"push", "origin", "--delete", original}, allowFail: true},
   194  	}
   195  }

   197  // releaseNotesToDelete は、リリースノートのうち削除すべきものを返します。
   198  func releaseNotesToDelete(entries []string) []string {
   199  	keep := initialTag + ".md"
   200  	targets := make([]string, 0, len(entries))
   201  
   202  	for _, e := range entries {
   203  		if e != keep {
   204  			targets = append(targets, e)
   205  		}
   206  	}
   207  
   208  	return targets
   209  }

   225  func runBootstrap(r runner) error {
   226  	if err := resetTags(r); err != nil {
   227  		return err
   228  	}
   229  
   230  	if err := createBranches(r); err != nil {

   237  func resetTags(r runner) error {
   238  	log.Printf("🔧 タグの初期化を開始します...")
   239  
   240  	out, err := r.output("git", "tag")
   241  	if err != nil {
   242  		return err
   243  	}
   244  
   245  	tags := parseLines(out)
   246  	if len(tags) == 0 {
   247  		log.Printf("🟡 削除対象のタグが存在しません。")
   248  	} else {
   249  		if err := r.runAll(tagDeletionSteps(tags)); err != nil {
   250  			return err
   251  		}
   252  
   253  		log.Printf("🧹 すべてのタグを削除しました。")
   254  	}
   255  
   256  	log.Printf("✅ タグの初期化を終了します。")
   257  	log.Printf("🔧 %s のタグ打ちを開始します...", initialTag)
   258  
   259  	if err := r.runAll(initialTagSteps()); err != nil {
   260  		return err
   261  	}
   262  
   263  	log.Printf("✅ %s のタグ打ちが完了しました。", initialTag)
   264  
   265  	return nil
   266  }

   268  func createBranches(r runner) error {
   269  	log.Printf("🔧 ブランチ作成を開始します...")
   270  
   271  	existing := make([]string, 0, len(managedBranches))
   272  
   273  	for _, b := range managedBranches {
   274  		if r.branchExists(b) {
   275  			existing = append(existing, b)
   276  		}
   277  	}
   278  
   279  	steps, skipped := branchCreationSteps(existing)
   280  
   281  	for _, b := range skipped {
   282  		log.Printf("🟡 ブランチ 【%s】 は既に存在します。作成処理をスキップします。", b)
   283  	}
   284  
   285  	if err := r.runAll(steps); err != nil {
   286  		return err
   287  	}
   288  
   289  	if err := r.run(branchPushStep()); err != nil {
   290  		return err
   291  	}
   292  
   293  	log.Printf("✅ ブランチの作成を終了します。")
   294  
   295  	return nil
   296  }

   298  func moveDefaultBranch(r runner) error {
   299  	log.Printf("🔧 デフォルトブランチの設定を開始します...")
   300  
   301  	repo, err := r.output("gh", "repo", "view", "--json", "name,owner", "-q", `.owner.login + "/" + .name`)
   302  	if err != nil {
   303  		return err
   304  	}
   305  
   306  	if err := r.run(defaultBranchStep(strings.TrimSpace(repo))); err != nil {
   307  		return err
   308  	}
   309  
   310  	if err := r.run(step{name: "git", args: []string{"fetch", "--prune"}}); err != nil {
   311  		return err
   312  	}
   313  
   314  	// 元居たブランチ名は switch する前に控える。
   315  	original, err := r.output("git", "branch", "--show-current")
   316  	if err != nil {
   317  		return err
   318  	}
   319  
   320  	if err := r.run(step{name: "git", args: []string{"switch", defaultBranch}}); err != nil {
   321  		return err
   322  	}
   323  
   324  	if err := r.runAll(originalBranchCleanupSteps(strings.TrimSpace(original))); err != nil {
   325  		return err
   326  	}
   327  
   328  	log.Printf("✅ デフォルトブランチの設定を終了します。")
   329  
   330  	return nil
   331  }

   333  func runPruneReleaseNotes() error {
   334  	log.Printf("🔧 リリースノートの初期化を開始します...")
   335  
   336  	entries, err := os.ReadDir(releaseNoteDir)
   337  	if err != nil {
   338  		if os.IsNotExist(err) {
   339  			log.Printf("🟡 %s ディレクトリが存在しないためスキップします。", releaseNoteDir)
   340  
   341  			return nil
   342  		}
   343  
   344  		return xerrors.Wrap(err, "failed to read "+releaseNoteDir)
   345  	}
   346  
   347  	names := make([]string, 0, len(entries))
   348  
   349  	for _, e := range entries {
   350  		if !e.IsDir() {
   351  			names = append(names, e.Name())
   352  		}
   353  	}
   354  
   355  	for _, name := range releaseNotesToDelete(names) {
   356  		if err := os.Remove(filepath.Join(releaseNoteDir, name)); err != nil {
   357  			return xerrors.Wrap(err, "failed to remove "+name)
   358  		}
   359  	}
   360  
   361  	log.Printf("🧹 %s.md 以外のリリースノートを削除しました。", initialTag)
   362  	log.Printf("✅ リリースノートの初期化を終了します。")
   363  
   364  	return nil
   365  }

   385  func tagExists(tag string) bool {
   386  	ctx, cancel := context.WithTimeout(context.Background(), commandTimeout)
   387  	defer cancel()
   388  
   389  	//nolint:gosec // tag は本ファイル内の定数

   401  func run(s step) error {
   402  	ctx, cancel := context.WithTimeout(context.Background(), commandTimeout)
   403  	defer cancel()
   404  
   405  	cmd := exec.CommandContext(ctx, s.name, s.args...) //nolint:gosec // 引数は本ファイル内で組み立てた固定手順
   406  	cmd.Stdout = os.Stdout
   407  	cmd.Stderr = os.Stderr
   408  
   409  	if err := cmd.Run(); err != nil && !s.allowFail {
   410  		return xerrors.Wrap(err, "failed: "+s.String())
   411  	}
   412  
   413  	return nil
   414  }

   416  func output(name string, args ...string) (string, error) {
   417  	ctx, cancel := context.WithTimeout(context.Background(), commandTimeout)
   418  	defer cancel()
   419  
   420  	out, err := exec.CommandContext(ctx, name, args...).Output() //nolint:gosec // 引数は本ファイル内の固定値
   421  	if err != nil {
   422  		return "", xerrors.Wrap(err, "failed: "+name+" "+strings.Join(args, " "))
   423  	}
   424  
   425  	return string(out), nil
