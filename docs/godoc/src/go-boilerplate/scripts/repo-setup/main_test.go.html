    33  // fakeRunner は、git / gh を一切起動せず呼び出しの並びだけを控える実行器。
    34  // bootstrap の手順はタグの一括削除と origin への push を含むため、
    35  // 実リポジトリと GitHub へ触れずに骨格を検証する目的で実行層をここへ差し替えます。
    36  type fakeRunner struct {
    37  	// calls は、run / output が呼ばれた順の「実行ファイル 引数...」表記。
    38  	calls []string
    39  	// outputs は、output が返す標準出力（キーは calls と同じ表記）。
    40  	outputs map[string]string
    41  	// failOn は、この表記の呼び出しだけを失敗させます（空なら常に成功）。
    42  	// allowFail は実行層の責務なのでここでは解釈しません。
    43  	failOn string
    44  	// branches は、branchExists が true を返すブランチ。
    45  	branches map[string]bool
    46  }
    47  
    48  // runner は、この実行器を差し込んだ runner を返します。
    49  func (f *fakeRunner) runner() runner {
    50  	return runner{run: f.run, output: f.output, branchExists: f.branchExists}
    51  }

    53  func (f *fakeRunner) run(s step) error {
    54  	call := s.String()
    55  	f.calls = append(f.calls, call)
    56  
    57  	if call == f.failOn {
    58  		return errFakeCommand
    59  	}
    60  
    61  	return nil
    62  }

    64  func (f *fakeRunner) output(name string, args ...string) (string, error) {
    65  	call := step{name: name, args: args}.String()
    66  	f.calls = append(f.calls, call)
    67  
    68  	if call == f.failOn {
    69  		return "", errFakeCommand
    70  	}
    71  
    72  	return f.outputs[call], nil
    73  }
