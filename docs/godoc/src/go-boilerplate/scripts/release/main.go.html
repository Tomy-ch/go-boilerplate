    95  func execute(r runner, args []string) error {
    96  	if len(args) == 0 {
    97  		return errUsage
    98  	}
    99  
   100  	var err error
   101  
   102  	switch args[0] {
   103  	case "tag":
   104  		err = runTag(r, args[1:])
   105  	case "branch":
   106  		err = runBranch(r, args[1:])
   107  	default:
   108  		return errUnknownSubcommand
   109  	}

   111  	// ヘルプ要求は失敗ではないので 0 で終える。usage は flag が既に出力している。
   112  	if xerrors.Is(err, errHelpRequested) {
   113  		return nil
   114  	}
   115  
   116  	return err
   117  }

   125  // parseVersion は、`vX.Y.Z` 形式の文字列を version へ変換します。
   126  func parseVersion(s string) (version, bool) {
   127  	m := semverPattern.FindStringSubmatch(strings.TrimSpace(s))
   128  	if m == nil {
   129  		return version{}, false
   130  	}
   131  
   132  	major, _ := strconv.Atoi(m[1])
   133  	minor, _ := strconv.Atoi(m[2])
   134  	patch, _ := strconv.Atoi(m[3])
   135  
   136  	return version{major: major, minor: minor, patch: patch}, true
   137  }

   139  // latestVersion は、`git tag` の出力から最新のリリースバージョンを選びます。
   140  // 比較は数値ごとに行うため、v1.9.0 より v1.10.0 が新しいと判定されます。
   141  func latestVersion(tagOutput string) (version, bool) {
   142  	versions := make([]version, 0)
   143  
   144  	for line := range strings.SplitSeq(tagOutput, "\n") {
   145  		if v, ok := parseVersion(line); ok {
   146  			versions = append(versions, v)
   147  		}
   148  	}
   149  
   150  	if len(versions) == 0 {
   151  		return version{}, false
   152  	}
   153  
   154  	sort.Slice(versions, func(i, j int) bool {
   155  		a, b := versions[i], versions[j]
   156  		if a.major != b.major {
   157  			return a.major > b.major
   158  		}
   159  
   160  		if a.minor != b.minor {
   161  			return a.minor > b.minor
   162  		}
   163  
   164  		return a.patch > b.patch
   165  	})
   166  
   167  	return versions[0], true
   168  }

   170  // bump は、指定の粒度で次のバージョンを返します。
   171  func bump(v version, kind string) (version, error) {
   172  	switch kind {
   173  	case "patch":
   174  		return version{major: v.major, minor: v.minor, patch: v.patch + 1}, nil
   175  	case "minor":
   176  		return version{major: v.major, minor: v.minor + 1}, nil
   177  	case "major":
   178  		return version{major: v.major + 1}, nil
   179  	default:
   180  		return version{}, xerrors.Wrap(errUnknownBump, kind)
   181  	}
   182  }

   222  // resolveNext は、最新タグを取得して次バージョン（現在, 次）を決めます。
   223  func resolveNext(r runner, bumpKind string) (version, version, error) {
   224  	if err := r.run(step{name: "git", args: []string{"fetch", "--tags", "origin"}}); err != nil {
   225  		return version{}, version{}, err
   226  	}
   227  
   228  	out, err := r.output("git", "tag")
   229  	if err != nil {
   230  		return version{}, version{}, err
   231  	}
   232  
   233  	current, ok := latestVersion(out)
   234  	if !ok {
   235  		return version{}, version{}, errNoTag
   236  	}
   237  
   238  	next, err := bump(current, bumpKind)
   239  	if err != nil {
   240  		return version{}, version{}, err
   241  	}
   242  
   243  	return current, next, nil
   244  }

   260  func runTag(r runner, args []string) error {
   261  	fs := flag.NewFlagSet("tag", flag.ContinueOnError)
   262  	bumpKind := fs.String("bump", "", "patch / minor / major")
   263  
   264  	if err := parseFlags(fs, args); err != nil {
   265  		return err
   266  	}
   267  
   268  	current, next, err := resolveNext(r, *bumpKind)
   269  	if err != nil {
   270  		return err
   271  	}
   272  
   273  	log.Printf("🔖 タグから最新タグバージョンを取得: %s", current)
   274  	log.Printf("➡️ 次のリリースバージョンを作成: %s", next)
   275  
   276  	// production を最新へ合わせてからリリースノートの有無を見る。順序は入れ替えないこと。
   277  	// タグは production HEAD に打つので、確かめるべきは「production にノートがあるか」であり、
   278  	// switch 前の作業ツリー（別ブランチ）にノートがあるかではない。
   279  	log.Printf("🔄 productionブランチの最新を取得中...")
   280  
   281  	if err := r.runAll(syncProductionSteps()); err != nil {
   282  		return err
   283  	}
   284  
   285  	log.Printf("✅ 最新のproductionを取得完了")
   286  
   287  	note := notePath(next.String())
   288  	if _, err := os.Stat(note); err != nil {
   289  		return xerrors.Wrap(errNoReleaseNote, "❌ "+note)
   290  	}
   291  
   292  	if err := r.runAll(tagSteps(next.String(), note)); err != nil {
   293  		return err
   294  	}
   295  
   296  	log.Printf("✅ タグを打ちました %s on production HEAD", next)
   297  
   298  	return nil
   299  }

   301  func runBranch(r runner, args []string) error {
   302  	fs := flag.NewFlagSet("branch", flag.ContinueOnError)
   303  	bumpKind := fs.String("bump", "", "patch / minor / major")
   304  	prefix := fs.String("prefix", "release", "ブランチ名の接頭辞 (release / hotfix)")
   305  	base := fs.String("base", "production", "分岐元ブランチ")
   306  
   307  	if err := parseFlags(fs, args); err != nil {
   308  		return err
   309  	}
   310  
   311  	log.Printf("🔄 最新のタグを取得中...")
   312  
   313  	current, next, err := resolveNext(r, *bumpKind)
   314  	if err != nil {
   315  		if xerrors.Is(err, errNoTag) {
   316  			return errNoTagForBranch
   317  		}
   318  
   319  		return err
   320  	}
   321  
   322  	log.Printf("✅ 最新のタグを取得完了")
   323  
   324  	branch := *prefix + "/" + next.String()
   325  
   326  	log.Printf("🔖 タグから最新リリースバージョンを取得: 【 %s 】", current)
   327  	log.Printf("➡️ 次のリリースバージョンを作成: 【 %s 】", next)
   328  	log.Printf("🌱 ブランチを作成: %s → 【 %s 】", *base, branch)
   329  
   330  	if r.remoteBranchExists(branch) {
   331  		return xerrors.Wrap(errBranchExists, "❌ ブランチ【 "+branch+" 】")
   332  	}
   333  
   334  	status, err := r.output("git", "status", "--porcelain")
   335  	if err != nil {
   336  		return err
   337  	}
   338  
   339  	if strings.TrimSpace(status) != "" {
   340  		_ = r.run(step{name: "git", args: []string{"status", "--short"}})
   341  
   342  		return errDirtyWorktree
   343  	}
   344  
   345  	if err := r.runAll(branchSteps(*base, branch)); err != nil {
   346  		return err
   347  	}
   348  
   349  	log.Printf("✅ デフォルトブランチを %s に切り替えて、プッシュしました。", branch)
   350  
   351  	return nil
   352  }

   360  func remoteBranchExists(branch string) bool {
   361  	ctx, cancel := context.WithTimeout(context.Background(), commandTimeout)
   362  	defer cancel()
   363  
   364  	//nolint:gosec // branch はタグ由来のバージョン文字列
   365  	return exec.CommandContext(ctx, "git", "ls-remote", "--exit-code", "--heads", "origin", branch).Run() == nil

   368  func run(s step) error {
   369  	ctx, cancel := context.WithTimeout(context.Background(), commandTimeout)
   370  	defer cancel()
   371  
   372  	cmd := exec.CommandContext(ctx, s.name, s.args...) //nolint:gosec // 引数は本ファイル内で組み立てた固定手順
   373  	cmd.Stdout = os.Stdout
   374  	cmd.Stderr = os.Stderr
   375  
   376  	if err := cmd.Run(); err != nil {
   377  		return xerrors.Wrap(err, "failed: "+s.String())
   378  	}
   379  
   380  	return nil
   381  }

   383  // runAll は、手順を並び順に実行します。失敗した時点で残りは実行しません。
   384  func (r runner) runAll(steps []step) error {
   385  	for _, s := range steps {

   394  func output(name string, args ...string) (string, error) {
   395  	ctx, cancel := context.WithTimeout(context.Background(), commandTimeout)
   396  	defer cancel()
   397  
   398  	out, err := exec.CommandContext(ctx, name, args...).Output() //nolint:gosec // 引数は本ファイル内の固定値
   399  	if err != nil {
   400  		return "", xerrors.Wrap(err, "failed: "+name+" "+strings.Join(args, " "))

   403  	return string(out), nil
