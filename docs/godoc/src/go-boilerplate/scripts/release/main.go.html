   100  func execute(r runner, args []string) error {
   101  	if len(args) == 0 {
   102  		return errUsage
   103  	}
   104  
   105  	var err error
   106  
   107  	switch args[0] {
   108  	case "tag":
   109  		err = runTag(r, args[1:])
   110  	case "branch":
   111  		err = runBranch(r, args[1:])
   112  	default:
   113  		return errUnknownSubcommand
   114  	}
   115  
   116  	// ヘルプ要求は失敗ではないので 0 で終える。usage は flag が既に出力している。
   117  	if xerrors.Is(err, errHelpRequested) {
   118  		return nil
   119  	}
   120  
   121  	return err
   122  }

   130  // parseVersion は、`vX.Y.Z` 形式の文字列を version へ変換します。
   131  func parseVersion(s string) (version, bool) {
   132  	m := semverPattern.FindStringSubmatch(strings.TrimSpace(s))
   133  	if m == nil {
   134  		return version{}, false
   135  	}
   136  
   137  	major, _ := strconv.Atoi(m[1])
   138  	minor, _ := strconv.Atoi(m[2])
   139  	patch, _ := strconv.Atoi(m[3])
   140  
   141  	return version{major: major, minor: minor, patch: patch}, true
   142  }

   144  // latestVersion は、`git tag` の出力から最新のリリースバージョンを選びます。
   145  // 比較は数値ごとに行うため、v1.9.0 より v1.10.0 が新しいと判定されます。
   146  func latestVersion(tagOutput string) (version, bool) {
   147  	versions := make([]version, 0)
   148  
   149  	for line := range strings.SplitSeq(tagOutput, "\n") {
   150  		if v, ok := parseVersion(line); ok {
   151  			versions = append(versions, v)
   152  		}
   153  	}
   154  
   155  	if len(versions) == 0 {
   156  		return version{}, false
   157  	}
   158  
   159  	sort.Slice(versions, func(i, j int) bool {
   160  		a, b := versions[i], versions[j]
   161  		if a.major != b.major {
   162  			return a.major > b.major
   163  		}
   164  
   165  		if a.minor != b.minor {
   166  			return a.minor > b.minor
   167  		}
   168  
   169  		return a.patch > b.patch
   170  	})
   171  
   172  	return versions[0], true
   173  }

   175  // bump は、指定の粒度で次のバージョンを返します。
   176  func bump(v version, kind string) (version, error) {
   177  	switch kind {
   178  	case "patch":
   179  		return version{major: v.major, minor: v.minor, patch: v.patch + 1}, nil
   180  	case "minor":
   181  		return version{major: v.major, minor: v.minor + 1}, nil
   182  	case "major":
   183  		return version{major: v.major + 1}, nil
   184  	default:
   185  		return version{}, xerrors.Wrap(errUnknownBump, kind)
   186  	}
   187  }

   227  // resolveNext は、最新タグを取得して次バージョン（現在, 次）を決めます。
   228  func resolveNext(r runner, bumpKind string) (version, version, error) {
   229  	if err := r.run(step{name: "git", args: []string{"fetch", "--tags", "origin"}}); err != nil {
   230  		return version{}, version{}, err
   231  	}
   232  
   233  	out, err := r.output("git", "tag")
   234  	if err != nil {
   235  		return version{}, version{}, err
   236  	}
   237  
   238  	current, ok := latestVersion(out)
   239  	if !ok {
   240  		return version{}, version{}, errNoTag
   241  	}
   242  
   243  	next, err := bump(current, bumpKind)
   244  	if err != nil {
   245  		return version{}, version{}, err
   246  	}
   247  
   248  	return current, next, nil
   249  }

   251  // parseFlags は、サブコマンドのフラグを解釈します。ヘルプ要求は失敗ではないため、通常のエラーと
   252  // 区別できるよう errHelpRequested を返します。ここを一括りにすると `-h` が異常終了になり、
   253  // 同じリポジトリの他ツール（tool-cooldown / go-cooldown）と終了コードが食い違います。
   254  func parseFlags(fs *flag.FlagSet, args []string) error {
   255  	err := fs.Parse(args)
   256  	switch {
   257  	case err == nil:
   258  		return nil
   259  	case xerrors.Is(err, flag.ErrHelp):
   260  		return errHelpRequested
   261  	default:
   262  		return xerrors.Wrap(err, "failed to parse flags")
   263  	}
   264  }

   266  func runTag(r runner, args []string) error {
   267  	fs := flag.NewFlagSet("tag", flag.ContinueOnError)
   268  	bumpKind := fs.String("bump", "", "patch / minor / major")
   269  
   270  	if err := parseFlags(fs, args); err != nil {
   271  		return err
   272  	}
   273  
   274  	current, next, err := resolveNext(r, *bumpKind)
   275  	if err != nil {
   276  		return err
   277  	}
   278  
   279  	log.Printf("🔖 タグから最新タグバージョンを取得: %s", current)
   280  	log.Printf("➡️ 次のリリースバージョンを作成: %s", next)
   281  
   282  	// production を最新へ合わせてからリリースノートの有無を見る。順序は入れ替えないこと。
   283  	// タグは production HEAD に打つので、確かめるべきは「production にノートがあるか」であり、
   284  	// switch 前の作業ツリー（別ブランチ）にノートがあるかではない。
   285  	log.Printf("🔄 productionブランチの最新を取得中...")
   286  
   287  	if err := r.runAll(syncProductionSteps()); err != nil {
   288  		return err
   289  	}
   290  
   291  	log.Printf("✅ 最新のproductionを取得完了")
   292  
   293  	note := notePath(next.String())
   294  	if _, err := os.Stat(note); err != nil {
   295  		return xerrors.Wrap(errNoReleaseNote, "❌ "+note)
   296  	}
   297  
   298  	if err := r.runAll(tagSteps(next.String(), note)); err != nil {
   299  		return err
   300  	}
   301  
   302  	log.Printf("✅ タグを打ちました %s on production HEAD", next)
   303  
   304  	return nil
   305  }

   307  func runBranch(r runner, args []string) error {
   308  	fs := flag.NewFlagSet("branch", flag.ContinueOnError)
   309  	bumpKind := fs.String("bump", "", "patch / minor / major")
   310  	prefix := fs.String("prefix", "release", "ブランチ名の接頭辞 (release / hotfix)")
   311  	base := fs.String("base", "production", "分岐元ブランチ")
   312  
   313  	if err := parseFlags(fs, args); err != nil {
   314  		return err
   315  	}
   316  
   317  	log.Printf("🔄 最新のタグを取得中...")
   318  
   319  	current, next, err := resolveNext(r, *bumpKind)
   320  	if err != nil {
   321  		if xerrors.Is(err, errNoTag) {
   322  			return errNoTagForBranch
   323  		}
   324  
   325  		return err
   326  	}
   327  
   328  	log.Printf("✅ 最新のタグを取得完了")
   329  
   330  	branch := *prefix + "/" + next.String()
   331  
   332  	log.Printf("🔖 タグから最新リリースバージョンを取得: 【 %s 】", current)
   333  	log.Printf("➡️ 次のリリースバージョンを作成: 【 %s 】", next)
   334  	log.Printf("🌱 ブランチを作成: %s → 【 %s 】", *base, branch)
   335  
   336  	if r.remoteBranchExists(branch) {
   337  		return xerrors.Wrap(errBranchExists, "❌ ブランチ【 "+branch+" 】")
   338  	}
   339  
   340  	status, err := r.output("git", "status", "--porcelain")
   341  	if err != nil {
   342  		return err
   343  	}
   344  
   345  	if strings.TrimSpace(status) != "" {
   346  		_ = r.run(step{name: "git", args: []string{"status", "--short"}})
   347  
   348  		return errDirtyWorktree
   349  	}
   350  
   351  	if err := r.runAll(branchSteps(*base, branch)); err != nil {
   352  		return err
   353  	}
   354  
   355  	log.Printf("✅ デフォルトブランチを %s に切り替えて、プッシュしました。", branch)
   356  
   357  	return nil
   358  }

   366  func remoteBranchExists(branch string) bool {
   367  	ctx, cancel := context.WithTimeout(context.Background(), commandTimeout)
   368  	defer cancel()
   369  
   370  	//nolint:gosec // branch はタグ由来のバージョン文字列
   371  	return exec.CommandContext(ctx, "git", "ls-remote", "--exit-code", "--heads", "origin", branch).Run() == nil

   374  func run(s step) error {
   375  	ctx, cancel := context.WithTimeout(context.Background(), commandTimeout)
   376  	defer cancel()
   377  
   378  	cmd := exec.CommandContext(ctx, s.name, s.args...) //nolint:gosec // 引数は本ファイル内で組み立てた固定手順
   379  	cmd.Stdout = os.Stdout
   380  	cmd.Stderr = os.Stderr
   381  
   382  	if err := cmd.Run(); err != nil {
   383  		return xerrors.Wrap(err, "failed: "+s.String())
   384  	}
   385  
   386  	return nil
   387  }

   389  // runAll は、手順を並び順に実行します。失敗した時点で残りは実行しません。
   390  func (r runner) runAll(steps []step) error {
   391  	for _, s := range steps {
   392  		if err := r.run(s); err != nil {
   393  			return err
   394  		}
   395  	}
   396  
   397  	return nil
   398  }

   400  func output(name string, args ...string) (string, error) {
   401  	ctx, cancel := context.WithTimeout(context.Background(), commandTimeout)
   402  	defer cancel()
   403  
   404  	out, err := exec.CommandContext(ctx, name, args...).Output() //nolint:gosec // 引数は本ファイル内の固定値
   405  	if err != nil {
   406  		return "", xerrors.Wrap(err, "failed: "+name+" "+strings.Join(args, " "))
   407  	}
   408  
   409  	return string(out), nil
   410  }
