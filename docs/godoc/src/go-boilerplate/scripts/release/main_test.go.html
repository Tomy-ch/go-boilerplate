    35  // fakeRunner は、git / gh を一切起動せず呼び出しの並びだけを控える実行器。
    36  // tag / branch の手順はタグの push と GitHub Release の作成を含み、走らせてしまうと
    37  // 取り消せないため、中止条件と手順の順序はここへ差し替えた実行層の上で検証します。
    38  type fakeRunner struct {
    39  	// calls は、run / output が呼ばれた順の「実行ファイル 引数...」表記。
    40  	calls []string
    41  	// outputs は、output が返す標準出力（キーは calls と同じ表記）。
    42  	outputs map[string]string
    43  	// failOn は、この表記の呼び出しだけを失敗させます（空なら常に成功）。
    44  	failOn string
    45  	// remoteBranches は、remoteBranchExists が true を返すブランチ。
    46  	remoteBranches map[string]bool
    47  }
    48  
    49  // runner は、この実行器を差し込んだ runner を返します。
    50  func (f *fakeRunner) runner() runner {
    51  	return runner{run: f.run, output: f.output, remoteBranchExists: f.remoteBranchExists}
    52  }

    54  func (f *fakeRunner) run(s step) error {
    55  	call := s.String()
    56  	f.calls = append(f.calls, call)
    57  
    58  	if call == f.failOn {
    59  		return errFakeCommand
    60  	}
    61  
    62  	return nil
    63  }
    64  
    65  func (f *fakeRunner) output(name string, args ...string) (string, error) {
    66  	call := step{name: name, args: args}.String()
    67  	f.calls = append(f.calls, call)
    68  
    69  	if call == f.failOn {
    70  		return "", errFakeCommand
    71  	}
    72  
    73  	return f.outputs[call], nil
    74  }
