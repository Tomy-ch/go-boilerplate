   105  // wd は走査の基点となるディレクトリの取得手段、lookPath は shellcheck の所在確認手段で、

   107  func run(ctx context.Context, wd func() (string, error), lookPath func(string) (string, error)) error {
   108  	if _, err := lookPath(shellcheckBin); err != nil {
   109  		return xerrors.Wrap(errShellcheckMissing, err.Error())
   110  	}
   111  
   112  	root, err := wd()
   113  	if err != nil {
   114  		return xerrors.Wrap(err, "getwd")
   115  	}
   116  
   117  	files, steps, err := collectSteps(os.DirFS(root))
   118  	if err != nil {
   119  		return err
   120  	}
   121  
   122  	res, err := check(ctx, steps)
   123  	if err != nil {
   124  		return err
   125  	}
   126  
   127  	for _, s := range res.skipped {
   128  		log.Printf("  ⏭️ %s", s)
   129  	}
   130  
   131  	for _, f := range res.findings {
   132  		log.Print(f)
   133  	}
   134  
   135  	if len(res.findings) > 0 {
   136  		return xerrors.Wrap(errFindings, fmt.Sprintf("%d 件（検査 %d ステップ）", len(res.findings), res.checked))
   137  	}
   138  
   139  	log.Printf("✅ composite action %d ファイルの run を %d ステップ検査しました（対象外 shell: %d ステップ）",
   140  		len(files), res.checked, len(res.skipped))
   141  
   142  	return nil
   143  }

   145  func collectSteps(fsys fs.FS) ([]string, []step, error) {

   165  // actionFiles は .github/actions 配下の action 定義ファイルをパス順に返す。
   166  //
   167  // WalkDir はリンク先を辿らない DirEntry で再帰要否を決めるため、シンボリックリンクを素通しにすると
   168  // リンク先の action がどのゲートにも掛からないまま緑で通る。リンク先を解決して扱いを決める。
   169  func actionFiles(fsys fs.FS) ([]string, error) {
   170  	var files []string
   171  	err := fs.WalkDir(fsys, actionsDir, func(path string, entry fs.DirEntry, err error) error {
   172  		if err != nil {
   173  			if xerrors.Is(err, fs.ErrNotExist) {
   174  				return fs.SkipAll
   175  			}
   176  			return err
   177  		}
   178  		if entry.Type()&fs.ModeSymlink != 0 {
   179  			return appendSymlink(fsys, path, entry.Name(), &files)
   180  		}
   181  		if !entry.IsDir() && isActionFile(entry.Name()) {
   182  			files = append(files, path)
   183  		}
   184  		return nil
   185  	})
   186  	if err != nil {
   187  		return nil, xerrors.Wrap(err, "walk "+actionsDir)
   188  	}
   189  	sort.Strings(files)
   190  	return files, nil
   191  }

   193  // appendSymlink はシンボリックリンクの実体を見て、action 定義ファイルなら files へ加える。
   194  //
   195  // ディレクトリへのリンクは、辿ると循環し得るうえ fs.FS には実体の同一性を判定する手段が無いため
   196  // 受け付けない。解決できないリンクも、実体がファイルだったのかディレクトリだったのかを言えず、
   197  // 対象外に寄せると走査が黙って縮むため同じく受け付けない。どちらも人手で解消する。
   198  func appendSymlink(fsys fs.FS, path, name string, files *[]string) error {
   199  	info, err := fs.Stat(fsys, path)
   200  	if err != nil {
   201  		return xerrors.Wrap(errActionSymlinkUnresolved, path)
   202  	}
   203  	if info.IsDir() {
   204  		return xerrors.Wrap(errActionSymlinkDir, path)
   205  	}
   206  	if isActionFile(name) {
   207  		*files = append(*files, path)
   208  	}
   209  	return nil
   210  }

   216  func parseAction(file string, data []byte) ([]step, error) {
   217  	var doc yaml.Node
   218  	if err := yaml.Unmarshal(data, &doc); err != nil {
   219  		return nil, xerrors.Wrap(err, "parse "+file)
   220  	}
   221  	if err := requireSingleDocument(file, data); err != nil {
   222  		return nil, err
   223  	}
   224  	want, err := countRunSteps(file, data)
   225  	if err != nil {
   226  		return nil, err
   227  	}
   228  	steps, err := extractSteps(file, data, &doc)
   229  	if err != nil {
   230  		return nil, err
   231  	}
   232  	if len(steps) != want {
   233  		return nil, xerrors.Wrap(errStepCountMismatch, fmt.Sprintf("%s: 抽出 %d / 期待 %d", file, len(steps), want))
   234  	}
   235  	return steps, nil
   236  }

   238  // requireSingleDocument は action 定義ファイルが単一の YAML ドキュメントであることを確かめる。
   239  //
   240  // yaml.Unmarshal は --- で区切られた 2 番目以降のドキュメントをエラー無しで捨てるため、
   241  // 抽出側もデコードして数える側も揃って見落とす。件数差の突き合わせでも検知できないので、
   242  // ここで落とす。GitHub Actions は action 定義に複数ドキュメントを認めていない。
   243  func requireSingleDocument(file string, data []byte) error {
   244  	dec := yaml.NewDecoder(bytes.NewReader(data))
   245  	var first any
   246  	if err := dec.Decode(&first); err != nil {
   247  		if xerrors.Is(err, io.EOF) {
   248  			return nil
   249  		}
   250  		return xerrors.Wrap(err, "parse "+file)
   251  	}
   252  	var second any
   253  	switch err := dec.Decode(&second); {
   254  	case err == nil:
   255  		return xerrors.Wrap(errMultipleDocuments, file)
   256  	case xerrors.Is(err, io.EOF):
   257  		return nil
   258  	default:
   259  		return xerrors.Wrap(err, "parse "+file)
   260  	}
   261  }

   263  // countRunSteps はデコード結果から run ステップ数を数える。件数を using の値と無関係に数えるのは、
   264  // 綴りを取り違えた action（using: Composite など）を「対象外」に寄せると、run を持つのに 1 件も
   265  // 検査されない状態が緑で通るため。
   266  func countRunSteps(file string, data []byte) (int, error) {
   267  	var doc any
   268  	if err := yaml.Unmarshal(data, &doc); err != nil {
   269  		return 0, xerrors.Wrap(err, "decode "+file)
   270  	}
   271  	steps := fieldValue(fieldValue(doc, "runs"), "steps")
   272  	if steps == nil {
   273  		return 0, nil
   274  	}
   275  	// リストとして読めない形を「対象外」に寄せると、検査範囲が黙って縮んだまま緑になる。
   276  	list, ok := steps.([]any)
   277  	if !ok {
   278  		return 0, xerrors.Wrap(errStepsNotSequence, file)
   279  	}
   280  	count := 0
   281  	for _, item := range list {
   282  		if _, ok := fieldValue(item, "run").(string); ok {
   283  			count++
   284  		}
   285  	}
   286  	return count, nil
   287  }

   297  func extractSteps(file string, data []byte, doc *yaml.Node) ([]step, error) {
   298  	runs := mapValue(documentRoot(doc), "runs")
   299  	if using := mapValue(runs, "using"); using == nil || using.Value != compositeUsing {
   300  		return nil, nil
   301  	}
   302  	stepsNode := mapValue(runs, "steps")
   303  	if stepsNode == nil {
   304  		return nil, nil
   305  	}
   306  	var steps []step
   307  	for _, node := range stepsNode.Content {
   308  		run := mapValue(node, "run")
   309  		if run == nil {
   310  			continue
   311  		}
   312  		// ブロック折り畳み（>）は隣接する非空行を空白へ畳むため、本文の行と action 定義ファイルの行が
   313  		// 1 対 1 で対応しない。指摘の位置を写し戻せないうえ、畳まれた行がソースに無い構文を作って
   314  		// 誤検知も生むため受け付けない。
   315  		if run.Style == yaml.FoldedStyle {
   316  			return nil, xerrors.Wrap(errFoldedRun, fmt.Sprintf("%s:%d", file, run.Line))
   317  		}
   318  		shell := mapValue(node, "shell")
   319  		if shell == nil {
   320  			return nil, xerrors.Wrap(errNoShell, fmt.Sprintf("%s:%d", file, run.Line))
   321  		}
   322  		firstLine := bodyFirstLine(run)
   323  		steps = append(steps, step{
   324  			file:      file,
   325  			shell:     shell.Value,
   326  			script:    run.Value,
   327  			firstLine: firstLine,
   328  			colBase:   bodyColumnBase(data, run, firstLine),
   329  		})
   330  	}
   331  	return steps, nil
   332  }

   341  func mapValue(node *yaml.Node, key string) *yaml.Node {
   342  	node = resolveAlias(node)
   343  	if node == nil || node.Kind != yaml.MappingNode {
   344  		return nil
   345  	}
   346  	var merged *yaml.Node
   347  	for i := 0; i+1 < len(node.Content); i += 2 {
   348  		switch node.Content[i].Value {
   349  		case key:
   350  			return resolveAlias(node.Content[i+1])
   351  		case mergeKey:
   352  			merged = node.Content[i+1]
   353  		}
   354  	}
   355  	return mergeValue(merged, key)
   356  }

   358  func mergeValue(merged *yaml.Node, key string) *yaml.Node {
   359  	merged = resolveAlias(merged)
   360  	if merged == nil {
   361  		return nil
   362  	}
   363  	if merged.Kind != yaml.SequenceNode {
   364  		return mapValue(merged, key)
   365  	}
   366  	for _, node := range merged.Content {
   367  		if v := mapValue(node, key); v != nil {
   368  			return v
   369  		}
   370  	}
   371  	return nil
   372  }

   388  func bodyColumnBase(data []byte, run *yaml.Node, firstLine int) int {
   389  	switch run.Style {
   390  	case yaml.LiteralStyle:
   391  		// ブロックスカラーの本文はインデントを剥がした形で得られるため、剥がされた幅を足し戻す。
   392  		return blockIndentWidth(data, firstLine, run.Value)
   393  	case yaml.SingleQuotedStyle, yaml.DoubleQuotedStyle:
   394  		// 引用符付きスカラーは範囲の先頭が開き引用符を指すため、その 1 文字分を読み飛ばす。
   395  		return run.Column
   396  	default:
   397  		return run.Column - firstColumn
   398  	}
   399  }
   400  
   401  // blockIndentWidth はブロックスカラーから剥がされたインデント幅を返す。
   402  //
   403  // 幅を決めるのは最初の非空行だが、その行の空白をそのまま採ると明示インデント指示子（run: |2 など）
   404  // を書いた本文で列がずれる。指示子は剥がす幅を親ノード基準で固定するので、本文がそれより深く
   405  // 書かれていれば余りが値側へ残り、生の行の空白は剥がされた幅より広くなる。
   406  // 生の行の空白から値に残った空白を引けば、指示子の有無に依らず剥がされた幅そのものが出る。
   407  // 本文が空行で始まる場合に幅 0 を採るとそのステップの全指摘がずれるため、両側とも最初の非空行を見る。
   408  func blockIndentWidth(data []byte, firstLine int, value string) int {
   409  	lines := strings.Split(string(data), "\n")
   410  	if firstLine < firstBodyIndex || firstLine > len(lines) {
   411  		return 0
   412  	}
   413  	raw, ok := firstIndentWidth(lines[firstLine-firstBodyIndex:])
   414  	if !ok {
   415  		return 0
   416  	}
   417  	kept, _ := firstIndentWidth(strings.Split(value, "\n"))
   418  	return raw - kept
   419  }

   421  // firstIndentWidth は最初の非空行のインデント幅を返す。第2戻り値が false なら非空行が無い。
   422  func firstIndentWidth(lines []string) (int, bool) {
   423  	for _, line := range lines {
   424  		trimmed := strings.TrimLeft(line, " \t")
   425  		if trimmed == "" {
   426  			continue
   427  		}
   428  		return len(line) - len(trimmed), true
   429  	}
   430  	return 0, false
   431  }

   433  func check(ctx context.Context, steps []step) (result, error) {
   434  	var res result
   435  	for _, s := range steps {
   436  		shebang, ok := shebangs[shellDialect(s.shell)]
   437  		if !ok {
   438  			res.skipped = append(res.skipped,
   439  				fmt.Sprintf("%s:%d: shell=%q は shellcheck の対象外のため検査しません", s.file, s.firstLine, s.shell))
   440  			continue
   441  		}
   442  		script, err := maskExpressions(s.script)
   443  		if err != nil {
   444  			return result{}, xerrors.Wrap(err, fmt.Sprintf("%s:%d", s.file, s.firstLine))
   445  		}
   446  		out, err := runShellcheck(ctx, shebang, script)
   447  		if err != nil {
   448  			return result{}, err
   449  		}
   450  		res.checked++
   451  		res.findings = append(res.findings, remapFindings(s, out)...)
   452  	}
   453  	return res, nil
   454  }

   456  // shellDialect は shell 指定から shellcheck へ渡す方言名を取り出す。env 経由の指定（shell: env FOO=bar bash）
   457  // では KEY=VALUE 形の変数代入がインタプリタ名の手前に並ぶため、それらを読み飛ばして最初のコマンドを採る。
   458  func shellDialect(shell string) string {
   459  	if strings.Contains(shell, exprOpen) {
   460  		return ""
   461  	}
   462  	for field := range strings.FieldsSeq(shell) {
   463  		if fieldBase(field) == envCommand || isAssignment(field) {
   464  			continue
   465  		}
   466  		return fieldBase(field)
   467  	}
   468  	return ""
   469  }

   471  func isAssignment(field string) bool {
   472  	name, _, ok := strings.Cut(field, "=")
   473  	if !ok || name == "" {
   474  		return false
   475  	}
   476  	for i, r := range name {
   477  		if r == '_' || unicode.IsLetter(r) || (i > 0 && unicode.IsDigit(r)) {
   478  			continue
   479  		}
   480  		return false
   481  	}
   482  	return true
   483  }

   492  func maskExpressions(script string) (string, error) {
   493  	var masked strings.Builder
   494  	for {
   495  		open := strings.Index(script, exprOpen)
   496  		if open < 0 {
   497  			masked.WriteString(script)
   498  			return masked.String(), nil
   499  		}
   500  		masked.WriteString(script[:open])
   501  		rest := script[open+len(exprOpen):]
   502  		end := exprEnd(rest)
   503  		if end < 0 {
   504  			return "", errUnterminatedExpr
   505  		}

   512  // exprEnd は式の開始直後から見た閉じ位置を返す。閉じが無ければ -1。
   513  //
   514  // 式の中の ' は文字列リテラルの境界なので、その内側の }} で式を打ち切らないよう引用符を追う。
   515  // ただし ' が奇数個だと引用符の状態が戻らず、自分の }} を読み飛ばして後続の別の式まで
   516  // 巻き込む。GitHub Expressions の式は入れ子にならないため、閉じを見つける前に次の ${{ が
   517  // 来ることは引用符の対応が壊れている証拠であり、そこで未閉じとして扱う。
   518  // 巻き込んだ区間は maskExpressions が空行へ潰すので、fail-open にするとその間のシェルが
   519  // 検査対象から丸ごと消える。
   520  func exprEnd(expr string) int {
   521  	quoted := false
   522  	for i := range len(expr) {
   523  		switch {
   524  		case strings.HasPrefix(expr[i:], exprOpen):
   525  			return -1
   526  		case expr[i] == '\'':
   527  			quoted = !quoted
   528  		case !quoted && strings.HasPrefix(expr[i:], exprClose):
   529  			return i
   530  		}
   531  	}
   532  	return -1
   533  }
