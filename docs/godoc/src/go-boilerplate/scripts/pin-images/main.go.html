     1  // Package main は Dockerfile の `FROM` base image を不変の digest へ固定するツール。
     2  //
     3  //	resolve: docker/*/Dockerfile の FROM を走査し、image:tag を registry の現在 digest へ
     4  //	         解決して lockfile (docker/images-pin.toml) へ書き出す。
     5  //	apply:   lockfile を SSOT に各 FROM を `FROM image:tag@sha256:... [AS stage]` へ固定する。
     6  //	check:   apply と同じ判定を書き換えなしで行い、未固定/未登録/drift があれば非ゼロ終了する（CI / hook 用）。
     7  //
     8  // tag は版の SSOT として FROM 行に残し、digest を lock 側で管理する（tag 再割り当て攻撃の遮断）。
     9  //
    10  // supply-chain cooldown: --min-age-days 未満（＝公開直後）の digest は採用しない。
    11  // mutable tag は「N 日前に何を指していたか」を registry へ問えないため、step-back 先は
    12  // git tag のような版履歴ではなく自分の前回 lock。前回 lock が無い初回は退行先が無く、
    13  // 出来立ての未検証 digest を掴む危険があるため、resolve は tag のまま残さず非ゼロ終了する
    14  // （緊急ブートストラップのみ --min-age-days=0 で明示採用）。
    15  //
    16  // apply/check は fail-closed：管理対象の FROM は image:tag@sha256:... へ固定され、かつ lock(SSOT)
    17  // に登録済みでなければならない。lock 未登録の image は「未登録」、digest 無し（tag のみ）は
    18  // 「未固定」として非ゼロ終了する（tag のみ運用は一切許容しない）。

    63  func main() {
    64  	log.SetFlags(0)
    65  	if len(os.Args) < minArgs {
    66  		log.Fatalf("❌ usage: pin-images <resolve|apply|check>")
    67  	}
    68  	root, err := os.Getwd()
    69  	if err != nil {
    70  		log.Fatalf("❌ getwd: %v", err)
    71  	}
    72  	files, err := targetFiles(root)
    73  	if err != nil {
    74  		log.Fatalf("❌ %v", err)
    75  	}
    76  	switch os.Args[1] {
    77  	case "resolve":
    78  		fs := flag.NewFlagSet("resolve", flag.ExitOnError)
    79  		minAge := fs.Int("min-age-days", 0, "N 日未満の新しすぎる digest は quarantine（0 で無効）")
    80  		_ = fs.Parse(os.Args[2:])
    81  		resolve(root, files, *minAge)
    82  	case "apply":
    83  		applyOrCheck(root, files, false)
    84  	case "check":
    85  		applyOrCheck(root, files, true)
    86  	default:
    87  		log.Fatalf("❌ usage: pin-images <resolve|apply|check>")
    88  	}

   100  // parseFrom は FROM の ref を image:tag へ分解する。第2戻り値が false なら対象外
   101  // （tag 無し＝ビルドステージ参照 / scratch、あるいは registry port を tag と誤認する形）。
   102  func parseFrom(ref string) (imageRef, bool) {
   103  	name, _, _ := strings.Cut(ref, "@") // 既存の @digest を捨てる
   104  	image, tag, ok := lastColonSplit(name)
   105  	if !ok || tag == "" || strings.Contains(tag, "/") {
   106  		return imageRef{}, false // tag が無い / 最後の ':' が registry:port だった
   107  	}
   108  	return imageRef{image: image, tag: tag}, true
   109  }

   111  // lastColonSplit は最後の ':' で分割する（tag に ':' は含まれない前提）。
   112  func lastColonSplit(s string) (string, string, bool) {
   113  	i := strings.LastIndex(s, ":")
   114  	if i < 0 {
   115  		return "", "", false
   116  	}
   117  	return s[:i], s[i+1:], true
   118  }

   120  func resolve(root string, files []string, minAgeDays int) {
   121  	keys := collectKeys(files)
   122  	existing, _ := readLock(filepath.Join(root, lockFile)) // 無ければ空
   123  
   124  	ctx := context.Background()
   125  	lock := map[string]string{}
   126  	var notes, skipped []string
   127  	for k, r := range keys {
   128  		digest, err := resolveDigest(ctx, r.key())
   129  		if err != nil {
   130  			log.Fatalf("❌ resolve %s: %v", k, err)
   131  		}
   132  		use, note := quarantine(ctx, r, k, digest, minAgeDays, existing)
   133  		if note != "" {
   134  			notes = append(notes, note)
   135  		}
   136  		if use == "" {
   137  			skipped = append(skipped, k)
   138  			continue
   139  		}
   140  		lock[k] = use
   141  		log.Printf("  %s -> %s", k, use)
   142  	}
   143  	sort.Strings(notes)
   144  	for _, n := range notes {
   145  		log.Printf("  ⚠️ %s", n)
   146  	}
   147  
   148  	if err := writeLock(filepath.Join(root, lockFile), lock); err != nil {
   149  		log.Fatalf("❌ write lockfile: %v", err)
   150  	}
   151  	log.Printf("✅ %s に %d 件を書き出しました", lockFile, len(lock))
   152  
   153  	// 退行先の無い出来立て image（fresh + 既存ピン無し）は tag のまま残さず失敗させる。
   154  	// tag のみ運用を許すと未検証 digest を掴むため、check も同状態を弾く（fail-closed）。
   155  	if len(skipped) > 0 {
   156  		sort.Strings(skipped)
   157  		log.Fatalf("❌ 退行先の無い出来立て image は採用できません（%d 日未満・既存ピン無し）。"+
   158  			"aged 後に再実行するか、緊急時のみ --min-age-days=0 で明示採用してください: %s",
   159  			minAgeDays, strings.Join(skipped, ", "))
   160  	}
   161  }

   163  func collectKeys(files []string) map[string]imageRef {
   164  	keys := map[string]imageRef{}
   165  	for _, f := range files {
   166  		data, err := os.ReadFile(f) //nolint:gosec // path from cwd glob
   167  		if err != nil {
   168  			log.Fatalf("❌ read %s: %v", f, err)
   169  		}
   170  		for _, m := range fromRe.FindAllStringSubmatch(string(data), -1) {
   171  			if r, ok := parseFrom(m[2]); ok {
   172  				keys[r.key()] = r
   173  			}
   174  		}
   175  	}
   176  	return keys
   177  }

   179  // quarantine は minAgeDays 未満の新しすぎる digest を採用しない。第1戻り値 "" は skip。
   180  func quarantine(ctx context.Context, r imageRef, key, candidate string, minAgeDays int, existing map[string]string) (string, string) {
   181  	if minAgeDays <= 0 {
   182  		return candidate, ""
   183  	}
   184  	age, err := digestAgeDays(ctx, r.key())
   185  	if err != nil {
   186  		log.Fatalf("❌ age %s: %v", key, err)
   187  	}
   188  	if age >= minAgeDays {
   189  		return candidate, ""
   190  	}
   191  	if prev, ok := existing[key]; ok {
   192  		return prev, fmt.Sprintf("%s: 現 digest が %d 日 (<%d) のため既存ピンを維持", key, age, minAgeDays)
   193  	}
   194  	return "", fmt.Sprintf("%s: 現 digest が %d 日 (<%d)・既存ピン無しのため tag のまま skip", key, age, minAgeDays)
   195  }

   197  // resolveDigest は image:tag の現在の index digest を registry から取得する。
   198  func resolveDigest(ctx context.Context, ref string) (string, error) {
   199  	out, err := inspect(ctx, ref)
   200  	if err != nil {
   201  		return "", err
   202  	}
   203  	m := digestRe.FindStringSubmatch(out)
   204  	if m == nil {
   205  		return "", xerrors.New(ref + ": Digest 行を解析できません")
   206  	}
   207  	return m[1], nil
   208  }

   210  // digestAgeDays は image:tag が指す digest の image config created から経過日数を返す。
   211  // 公式イメージのビルド日は publish 日にほぼ一致する。マルチアーキの場合は最も古い created を採る。
   212  func digestAgeDays(ctx context.Context, ref string) (int, error) {
   213  	created, err := earliestCreated(ctx, ref)
   214  	if err != nil {
   215  		return 0, err
   216  	}
   217  	return int(time.Since(created).Hours() / hoursPerDay), nil
   218  }

   220  func earliestCreated(ctx context.Context, ref string) (time.Time, error) {
   221  	// マルチアーキ (index) は .Image が map、単一アーキは struct。前者→後者でフォールバックする。
   222  	// inspect 自体のエラー（429 等）は握り潰さず伝播し、テンプレート不一致のときだけ次を試す。
   223  	var lastErr error
   224  	for _, tmpl := range []string{
   225  		`{{ range .Image }}{{ println .Created }}{{ end }}`,
   226  		`{{ println .Image.Created }}`,
   227  	} {
   228  		out, err := inspect(ctx, ref, "--format", tmpl)
   229  		if err != nil {
   230  			lastErr = err
   231  			continue
   232  		}
   233  		if t, ok := minCreated(out); ok {
   234  			return t, nil
   235  		}
   236  	}
   237  	if lastErr != nil {
   238  		return time.Time{}, lastErr
   239  	}
   240  	return time.Time{}, xerrors.New(ref + ": created を解析できません")
   241  }

   243  // minCreated は inspect 出力の各行（"2006-01-02 15:04:05.9 +0000 UTC"）から最古の時刻を返す。
   244  func minCreated(out string) (time.Time, bool) {
   245  	var earliest time.Time
   246  	found := false
   247  	for line := range strings.SplitSeq(out, "\n") {
   248  		fields := strings.Fields(line)
   249  		if len(fields) < createdTimeCols {
   250  			continue
   251  		}
   252  		// "date time zone" の 3 フィールドだけを RFC 化して parse（末尾 "UTC" 等は捨てる）。
   253  		t, err := time.Parse("2006-01-02 15:04:05.999999999 -0700",
   254  			strings.Join(fields[:createdTimeCols], " "))
   255  		if err != nil {
   256  			continue
   257  		}
   258  		if !found || t.Before(earliest) {
   259  			earliest, found = t, true
   260  		}
   261  	}
   262  	return earliest, found
   263  }

   265  func inspect(ctx context.Context, ref string, extra ...string) (string, error) {
   266  	cctx, cancel := context.WithTimeout(ctx, inspectTimeout)
   267  	defer cancel()
   268  	args := append([]string{"buildx", "imagetools", "inspect", ref}, extra...)
   269  	cmd := exec.CommandContext(cctx, "docker", args...) //nolint:gosec // ref は Dockerfile 由来
   270  	var stderr strings.Builder
   271  	cmd.Stderr = &stderr
   272  	out, err := cmd.Output()
   273  	if err != nil {
   274  		return "", xerrors.Wrap(err, "docker buildx imagetools inspect "+ref+": "+strings.TrimSpace(stderr.String()))
   275  	}
   276  	return string(out), nil
   277  }

   279  // rewritePins は lock を元に FROM を digest 固定した内容と、lock 未登録の image キー一覧を返す。
   280  // lock に無い image は「未登録」として報告し、行は書き換えない（digest は剥がさない）。
   281  func rewritePins(data string, lock map[string]string) (string, []string) {
   282  	var missing []string
   283  	out := fromRe.ReplaceAllStringFunc(data, func(line string) string {
   284  		m := fromRe.FindStringSubmatch(line)
   285  		r, ok := parseFrom(m[2])
   286  		if !ok {
   287  			return line
   288  		}
   289  		digest, found := lock[r.key()]
   290  		if !found {
   291  			missing = append(missing, r.key())
   292  			return line
   293  		}
   294  		return m[1] + r.key() + "@" + digest + m[3]
   295  	})
   296  	return out, missing
   297  }

   299  // applyOrCheck は lockfile を SSOT に FROM を digest 固定する。dryRun=true は書き換えず
   300  // 未固定/未登録/drift を非ゼロ終了で報告する。tag のみへ戻す正規化はしない（fail-closed）。
   301  func applyOrCheck(root string, files []string, dryRun bool) {
   302  	lock, err := readLock(filepath.Join(root, lockFile))
   303  	if err != nil {
   304  		log.Fatalf("❌ read lockfile（先に make pin-images-resolve を実行してください）: %v", err)
   305  	}
   306  	var missing, drifted []string
   307  	changed := 0
   308  	for _, f := range files {
   309  		data, err := os.ReadFile(f) //nolint:gosec // path from cwd glob
   310  		if err != nil {
   311  			log.Fatalf("❌ read %s: %v", f, err)
   312  		}
   313  		out, miss := rewritePins(string(data), lock)
   314  		missing = append(missing, miss...)
   315  		if out == string(data) {
   316  			continue
   317  		}
   318  		if dryRun {
   319  			drifted = append(drifted, rel(root, f))
   320  			continue
   321  		}
   322  		if err := os.WriteFile(f, []byte(out), filePerm); err != nil { //nolint:gosec // Dockerfile 権限
   323  			log.Fatalf("❌ write %s: %v", f, err)
   324  		}
   325  		changed++
   326  		log.Printf("  updated %s", rel(root, f))
   327  	}
   328  	report(missing, drifted, dryRun, changed)
   329  }

   331  func report(missing, drifted []string, dryRun bool, changed int) {
   332  	if len(missing) > 0 {
   333  		sort.Strings(missing)
   334  		log.Fatalf("❌ lockfile に未登録の base image があります（make pin-images-resolve を実行してください）: %s",
   335  			strings.Join(uniq(missing), ", "))
   336  	}
   337  	if !dryRun {
   338  		log.Printf("✅ %d ファイルを固定しました", changed)
   339  		return
   340  	}
   341  	if len(drifted) > 0 {
   342  		sort.Strings(drifted)
   343  		log.Fatalf("❌ FROM が未固定か lockfile と不一致です（make pin-images-resolve && make pin-images-apply してコミットしてください）: %s",
   344  			strings.Join(drifted, ", "))
   345  	}
   346  	log.Printf("✅ 全 base image が lockfile 通りに固定されています")
   347  }

   349  func writeLock(path string, lock map[string]string) error {
   350  	keys := make([]string, 0, len(lock))
   351  	for k := range lock {
   352  		keys = append(keys, k)
   353  	}
   354  	sort.Strings(keys)
   355  	var b strings.Builder
   356  	b.WriteString("# Docker base image の pin 対象 digest（SSOT）。\n")
   357  	b.WriteString("# make pin-images-resolve で解決・make pin-images-apply で Dockerfile へ反映する。\n")
   358  	for _, k := range keys {
   359  		fmt.Fprintf(&b, "%q = %q\n", k, lock[k])
   360  	}
   361  	return os.WriteFile(path, []byte(b.String()), filePerm)
   362  }

   364  func readLock(path string) (map[string]string, error) {
   365  	f, err := os.Open(path) //nolint:gosec // path is constructed from cwd + literal filename
   366  	if err != nil {
   367  		return nil, err
   368  	}
   369  	defer func() { _ = f.Close() }()
   370  	lock := map[string]string{}
   371  	sc := bufio.NewScanner(f)
   372  	for sc.Scan() {
   373  		if m := lockRe.FindStringSubmatch(strings.TrimSpace(sc.Text())); m != nil {
   374  			lock[m[1]] = m[2]
   375  		}
   376  	}
   377  	return lock, sc.Err()
   378  }

   388  func uniq(s []string) []string {
   389  	out := s[:0]
   390  	var prev string
   391  	for i, v := range s {
   392  		if i == 0 || v != prev {
   393  			out = append(out, v)
   394  		}
   395  		prev = v
   396  	}
   397  	return out
