     1  // Package main は Dockerfile の `FROM` base image と docker compose の `image:` を
     2  // 不変の digest へ固定するツール。
     3  //
     4  //	resolve: docker/*/Dockerfile の FROM と docker-compose*.yaml の image: を走査し、
     5  //	         image:tag を registry の現在 digest へ解決して lockfile (docker/images-pin.toml) へ書き出す。
     6  //	apply:   lockfile を SSOT に各参照を `image:tag@sha256:...` へ固定する（FROM は AS stage を保持）。
     7  //	check:   apply と同じ判定を書き換えなしで行い、未固定/未登録/drift があれば非ゼロ終了する（CI / hook 用）。
     8  //
     9  // tag は版の SSOT として FROM 行に残し、digest を lock 側で管理する（tag 再割り当て攻撃の遮断）。
    10  //
    11  // supply-chain cooldown: --min-age-days 未満（＝公開直後）の digest は採用しない。
    12  // mutable tag は「N 日前に何を指していたか」を registry へ問えないため、step-back 先は
    13  // git tag のような版履歴ではなく自分の前回 lock。前回 lock が無い初回は退行先が無く、
    14  // 出来立ての未検証 digest を掴む危険があるため、resolve は tag のまま残さず非ゼロ終了する
    15  // （緊急ブートストラップのみ --min-age-days=0 で明示採用）。
    16  //
    17  // apply/check は fail-closed：管理対象の FROM は image:tag@sha256:... へ固定され、かつ lock(SSOT)
    18  // に登録済みでなければならない。lock 未登録の image は「未登録」、digest 無し（tag のみ）は
    19  // 「未固定」として非ゼロ終了する（tag のみ運用は一切許容しない）。

    48  	// FROM [--platform=...] <ref> [AS <stage>]
    49  	fromRe = regexp.MustCompile(`(?m)^(FROM[ \t]+)(?:--platform=\S+[ \t]+)?(\S+)([ \t]+(?i:AS)[ \t]+\S+)?[ \t]*$`)

   102  // run は、サブコマンドを解釈して固定処理へ振り分けます。
   103  // wd は走査の基点となるディレクトリの取得手段で、差し替えられるよう引数で受けます。
   104  func run(args []string, wd func() (string, error)) error {
   105  	if len(args) == 0 {
   106  		return errUsage
   107  	}
   108  
   109  	root, err := wd()
   110  	if err != nil {
   111  		return xerrors.Wrap(err, "getwd")
   112  	}
   113  
   114  	targets, err := targetFiles(root)
   115  	if err != nil {
   116  		return err
   117  	}
   118  
   119  	switch args[0] {
   120  	case "resolve":
   121  		fs := flag.NewFlagSet("resolve", flag.ContinueOnError)
   122  		minAge := fs.Int("min-age-days", 0, "N 日未満の新しすぎる digest は quarantine（0 で無効）")
   123  		if err := fs.Parse(args[1:]); err != nil {
   124  			// ヘルプ要求は失敗ではないので 0 で終える。usage は flag が既に出力している。
   125  			if xerrors.Is(err, flag.ErrHelp) {
   126  				return nil
   127  			}
   128  
   129  			return xerrors.Wrap(err, "failed to parse flags")
   130  		}
   131  
   132  		return resolve(root, targets, *minAge)
   133  	case "apply":
   134  		return applyOrCheck(root, targets, false)
   135  	case "check":
   136  		return applyOrCheck(root, targets, true)
   137  	default:
   138  		return errUsage
   139  	}
   140  }

   142  func targetFiles(root string) ([]target, error) {
   143  	dockerfiles, err := globFiles(root, "docker/*/Dockerfile")
   144  	if err != nil {
   145  		return nil, err
   146  	}
   147  	compose, err := globFiles(root, "docker-compose*.yaml")
   148  	if err != nil {
   149  		return nil, err
   150  	}
   151  	var targets []target
   152  	for _, f := range dockerfiles {
   153  		targets = append(targets, target{path: f, re: fromRe})
   154  	}
   155  	for _, f := range compose {
   156  		targets = append(targets, target{path: f, re: composeImageRe})
   157  	}
   158  	sort.Slice(targets, func(i, j int) bool { return targets[i].path < targets[j].path })
   159  	return targets, nil
   160  }

   173  // parseRef は FROM / compose image の ref を image:tag へ分解する。第2戻り値が false なら対象外
   174  // （tag 無し＝ビルドステージ参照 / scratch、あるいは registry port を tag と誤認する形）。
   175  func parseRef(ref string) (imageRef, bool) {
   176  	name, _, _ := strings.Cut(ref, "@") // 既存の @digest を捨てる
   177  	image, tag, ok := lastColonSplit(name)
   178  	if !ok || tag == "" || strings.Contains(tag, "/") {
   179  		return imageRef{}, false // tag が無い / 最後の ':' が registry:port だった
   180  	}
   181  	return imageRef{image: image, tag: tag}, true
   182  }

   184  // lastColonSplit は最後の ':' で分割する（tag に ':' は含まれない前提）。
   185  func lastColonSplit(s string) (string, string, bool) {
   186  	i := strings.LastIndex(s, ":")
   187  	if i < 0 {
   188  		return "", "", false
   189  	}
   190  	return s[:i], s[i+1:], true
   191  }

   193  func resolve(root string, targets []target, minAgeDays int) error {
   194  	keys, err := collectKeys(targets)
   195  	if err != nil {
   196  		return err
   197  	}
   198  	// lockfile 不在（初回）は空マップで続行するが、それ以外の読み込み失敗は握り潰さず fail-close する
   199  	// （既存ピンが lock から脱落して quarantine の退行先が消え、出来立ての未検証 digest を掴むか
   200  	// errNoStepBack で落ちるかの二択になるのを防ぐ。applyOrCheck と対称）。
   201  	existing, err := readLock(filepath.Join(root, lockFile))
   202  	if !isIgnorableLockErr(err) {
   203  		return xerrors.Wrap(err, "read lockfile")
   204  	}
   205  
   206  	ctx := context.Background()
   207  	lock := map[string]string{}
   208  	var notes, skipped []string
   209  	for k, r := range keys {
   210  		digest, err := resolveDigest(ctx, r.key())
   211  		if err != nil {
   212  			return xerrors.Wrap(err, "resolve "+k)
   213  		}
   214  		use, note, err := quarantine(ctx, r, k, digest, minAgeDays, existing)
   215  		if err != nil {
   216  			return xerrors.Wrap(err, "age "+k)
   217  		}
   218  		if note != "" {
   219  			notes = append(notes, note)
   220  		}
   221  		if use == "" {
   222  			skipped = append(skipped, k)
   223  			continue
   224  		}
   225  		lock[k] = use
   226  		log.Printf("  %s -> %s", k, use)
   227  	}
   228  	sort.Strings(notes)
   229  	for _, n := range notes {
   230  		log.Printf("  ⚠️ %s", n)
   231  	}
   232  
   233  	if err := writeLock(filepath.Join(root, lockFile), lock); err != nil {
   234  		return xerrors.Wrap(err, "write lockfile")
   235  	}
   236  	log.Printf("✅ %s に %d 件を書き出しました", lockFile, len(lock))
   237  
   238  	// 退行先の無い出来立て image（fresh + 既存ピン無し）は tag のまま残さず失敗させる。
   239  	// tag のみ運用を許すと未検証 digest を掴むため、check も同状態を弾く（fail-closed）。
   240  	if len(skipped) > 0 {
   241  		sort.Strings(skipped)
   242  
   243  		return xerrors.Wrap(errNoStepBack, fmt.Sprintf(
   244  			"%d 日未満・既存ピン無し。aged 後に再実行するか、緊急時のみ --min-age-days=0 で明示採用してください: %s",
   245  			minAgeDays, strings.Join(skipped, ", ")))
   246  	}
   247  
   248  	return nil
   249  }

   251  func collectKeys(targets []target) (map[string]imageRef, error) {
   252  	keys := map[string]imageRef{}
   253  	for _, t := range targets {
   254  		data, err := os.ReadFile(t.path)
   255  		if err != nil {
   256  			return nil, xerrors.Wrap(err, "read "+t.path)
   257  		}
   258  		for _, m := range t.re.FindAllStringSubmatch(string(data), -1) {
   259  			if r, ok := parseRef(m[2]); ok {
   260  				keys[r.key()] = r
   261  			}
   262  		}
   263  	}
   264  	return keys, nil
   265  }

   267  // quarantine は minAgeDays 未満の新しすぎる digest を採用しない。第1戻り値 "" は skip。
   268  // 経過日数の取得に失敗した場合は err で呼び出し元へ伝播する。
   269  func quarantine(ctx context.Context, r imageRef, key, candidate string, minAgeDays int, existing map[string]string) (string, string, error) {
   270  	if minAgeDays <= 0 {
   271  		return candidate, "", nil
   272  	}
   273  	age, err := digestAgeDays(ctx, r.key())
   274  	if err != nil {
   275  		return "", "", err
   276  	}
   277  	if age >= minAgeDays {
   278  		return candidate, "", nil
   279  	}
   280  	if prev, ok := existing[key]; ok {
   281  		return prev, fmt.Sprintf("%s: 現 digest が %d 日 (<%d) のため既存ピンを維持", key, age, minAgeDays), nil
   282  	}
   283  	return "", fmt.Sprintf("%s: 現 digest が %d 日 (<%d)・既存ピン無しのため tag のまま skip", key, age, minAgeDays), nil
   284  }

   286  // resolveDigest は image:tag の現在の index digest を registry から取得する。
   287  func resolveDigest(ctx context.Context, ref string) (string, error) {
   288  	out, err := inspect(ctx, ref)
   289  	if err != nil {
   290  		return "", err
   291  	}
   292  	m := digestRe.FindStringSubmatch(out)
   293  	if m == nil {
   294  		return "", xerrors.Wrap(errDigestUnparsable, ref)
   295  	}
   296  	return m[1], nil
   297  }

   299  // digestAgeDays は image:tag が指す digest の image config created から経過日数を返す。
   300  // 公式イメージのビルド日は publish 日にほぼ一致する。マルチアーキの場合は最も古い created を採る。
   301  func digestAgeDays(ctx context.Context, ref string) (int, error) {
   302  	created, err := earliestCreated(ctx, ref)
   303  	if err != nil {
   304  		return 0, err
   305  	}
   306  	return int(time.Since(created).Hours() / hoursPerDay), nil
   307  }

   309  func earliestCreated(ctx context.Context, ref string) (time.Time, error) {
   310  	// マルチアーキ (index) は .Image が map、単一アーキは struct。前者→後者でフォールバックする。
   311  	// inspect 自体のエラー（429 等）は握り潰さず伝播し、テンプレート不一致のときだけ次を試す。
   312  	var lastErr error
   313  	for _, tmpl := range []string{
   314  		`{{ range .Image }}{{ println .Created }}{{ end }}`,
   315  		`{{ println .Image.Created }}`,
   316  	} {
   317  		out, err := inspect(ctx, ref, "--format", tmpl)
   318  		if err != nil {
   319  			lastErr = err
   320  			continue
   321  		}
   322  		if t, ok := minCreated(out); ok {
   323  			return t, nil
   324  		}
   325  	}
   326  	if lastErr != nil {
   327  		return time.Time{}, lastErr
   328  	}
   329  	return time.Time{}, xerrors.Wrap(errCreatedUnparsable, ref)
   330  }

   332  // minCreated は inspect 出力の各行（"2006-01-02 15:04:05.9 +0000 UTC"）から最古の時刻を返す。
   333  func minCreated(out string) (time.Time, bool) {
   334  	var earliest time.Time
   335  	found := false
   336  	for line := range strings.SplitSeq(out, "\n") {
   337  		fields := strings.Fields(line)
   338  		if len(fields) < createdTimeCols {
   339  			continue
   340  		}
   341  		// "date time zone" の 3 フィールドだけを RFC 化して parse（末尾 "UTC" 等は捨てる）。
   342  		t, err := time.Parse("2006-01-02 15:04:05.999999999 -0700",
   343  			strings.Join(fields[:createdTimeCols], " "))
   344  		if err != nil {
   345  			continue
   346  		}
   347  		if !found || t.Before(earliest) {
   348  			earliest, found = t, true
   349  		}
   350  	}
   351  	return earliest, found
   352  }

   354  func inspect(ctx context.Context, ref string, extra ...string) (string, error) {
   355  	cctx, cancel := context.WithTimeout(ctx, inspectTimeout)
   356  	defer cancel()
   357  	args := append([]string{"buildx", "imagetools", "inspect", ref}, extra...)
   358  	cmd := exec.CommandContext(cctx, "docker", args...) //nolint:gosec // ref は Dockerfile 由来
   359  	var stderr strings.Builder
   360  	cmd.Stderr = &stderr
   361  	out, err := cmd.Output()
   362  	if err != nil {
   363  		return "", xerrors.Wrap(err, "docker buildx imagetools inspect "+ref+": "+strings.TrimSpace(stderr.String()))
   364  	}
   365  	return string(out), nil
   366  }

   368  // rewritePins は lock を元に FROM を digest 固定した内容と、lock 未登録の image キー一覧を返す。
   369  // lock に無い image は「未登録」として報告し、行は書き換えない（digest は剥がさない）。
   370  func rewritePins(data string, re *regexp.Regexp, lock map[string]string) (string, []string) {
   371  	var missing []string
   372  	out := re.ReplaceAllStringFunc(data, func(line string) string {
   373  		m := re.FindStringSubmatch(line)
   374  		r, ok := parseRef(m[2])
   375  		if !ok {
   376  			return line
   377  		}
   378  		digest, found := lock[r.key()]
   379  		if !found {
   380  			missing = append(missing, r.key())
   381  			return line
   382  		}
   383  		return m[1] + r.key() + "@" + digest + m[3]
   384  	})
   385  	return out, missing
   386  }

   388  // applyOrCheck は lockfile を SSOT に FROM を digest 固定する。dryRun=true は書き換えず
   389  // 未固定/未登録/drift を非ゼロ終了で報告する。tag のみへ戻す正規化はしない（fail-closed）。
   390  //
   391  // 全ファイルを読み切って未登録の有無を確定させてから書き込む。1 ファイルずつ書きながら進むと、
   392  // 後続ファイルの未登録参照で中断したときに「exit 1 なのに作業ツリーは書き換え済み」という
   393  // 中途半端な状態が残る。
   394  func applyOrCheck(root string, targets []target, dryRun bool) error {
   395  	lock, err := readLock(filepath.Join(root, lockFile))
   396  	if err != nil {
   397  		return xerrors.Wrap(err, "read lockfile（先に make pin-images-resolve を実行してください）")
   398  	}
   399  	var missing, drifted, pending []string
   400  	rewritten := map[string]string{}
   401  	for _, t := range targets {
   402  		data, err := os.ReadFile(t.path)
   403  		if err != nil {
   404  			return xerrors.Wrap(err, "read "+rel(root, t.path))
   405  		}
   406  		out, miss := rewritePins(string(data), t.re, lock)
   407  		missing = append(missing, miss...)
   408  		if out == string(data) {
   409  			continue
   410  		}
   411  		if dryRun {
   412  			drifted = append(drifted, rel(root, t.path))
   413  			continue
   414  		}
   415  		pending = append(pending, t.path)
   416  		rewritten[t.path] = out
   417  	}
   418  
   419  	if err := validateMissing(missing); err != nil {
   420  		return err
   421  	}
   422  
   423  	for _, path := range pending {
   424  		if err := os.WriteFile(path, []byte(rewritten[path]), filePerm); err != nil {
   425  			return xerrors.Wrap(err, "write "+rel(root, path))
   426  		}
   427  		log.Printf("  updated %s", rel(root, path))
   428  	}
   429  
   430  	return report(drifted, dryRun, len(pending))

   486  // readLock は lockfile を image:tag→digest として読む。空行とコメント行以外で代入として解釈
   487  // できない行、および既出キーの再定義はエラーにする。読み飛ばしや後勝ちの上書きは、そのエントリが
   488  // 「存在しない」あるいは「行順で決まる」状態を警告なく作り、lockfile が SSOT として機能しなくなる。
   489  func readLock(path string) (map[string]string, error) {
   490  	f, err := os.Open(path) //nolint:gosec // path is constructed from cwd + literal filename
   491  	if err != nil {
   492  		return nil, err
   493  	}
   494  	defer func() { _ = f.Close() }()
   495  	lock := map[string]string{}
   496  	sc := bufio.NewScanner(f)
   497  	for lineNo := 1; sc.Scan(); lineNo++ {
   498  		line := strings.TrimSpace(sc.Text())
   499  		if line == "" || strings.HasPrefix(line, "#") {
   500  			continue
   501  		}
   502  		m := lockRe.FindStringSubmatch(line)
   503  		if m == nil {
   504  			return nil, xerrors.Wrap(errLockInvalidLine,
   505  				fmt.Sprintf("%d 行目: %q（make pin-images-resolve を実行するか該当行を削除してください）", lineNo, line))
   506  		}
   507  		if _, dup := lock[m[1]]; dup {
   508  			return nil, xerrors.Wrap(errLockDuplicateKey,
   509  				fmt.Sprintf("%d 行目: %q（make pin-images-resolve を実行するか重複行を削除してください）", lineNo, m[1]))
   510  		}
   511  		lock[m[1]] = m[2]
   512  	}
   513  	return lock, sc.Err()
   514  }
