     1  // Package main は、フィーチャーブランチの分岐元となるリリースブランチを解決するツール。
     2  //
     3  //	base-branch    最新のリリースライン（`release/vX.Y.Z`）のブランチ名を標準出力へ 1 行で出す
     4  //
     5  // このリポジトリはリリースのたびにリリースラインが移る運用のため、「どこから切るか」の
     6  // 答えが参照先ごとに食い違う。ローカルの `refs/remotes/origin/HEAD` は clone 時に一度
     7  // 決まったきり `git fetch` では更新されず（更新には `git remote set-head` が要る）、
     8  // GitHub のデフォルトブランチは最新のリリースラインへ移る前の値であることがある。
     9  // どちらも警告を出さずに古い答えを返すので、判断の出所を 1 つに固定する。
    10  //
    11  // 出所は origin の実状態（`git ls-remote`）。ローカルの参照は一切読まない。上の 2 つが
    12  // 陳腐化していても答えは変わらない。
    13  //
    14  // 最新の定義はバージョン番号の数値比較であって、コミット日時ではない。リリースラインの
    15  // 新しさはバージョン番号そのものが表しており、日時は古いラインへの hotfix やベース merge で
    16  // 前後する（実際このリポジトリの release/v2.1.0 は、より新しい release/v2.2.0 を取り込んだ
    17  // merge を tip に持つ）。ブランチを切る scripts/release も同じ数値比較で次版を決めるので、
    18  // 作る側と解決する側の基準が揃う。文字列順を採らないのは v1.10.0 が v1.9.0 より前に並ぶため。
    19  //
    20  // git はホストの認証情報を使うため、ツールランナーではなくホストで実行する
    21  // （scripts/release と同じ扱い）。

    78  // run は、最新のリリースラインのブランチ名を out へ 1 行で書き出します。
    79  // list は origin の参照一覧の取得手段で、差し替えられるよう引数で受けます。
    80  //
    81  // 出力を装飾しないのは、呼び出し側がコマンド置換でそのまま変数へ受けるためです。
    82  // 進捗やエラーは log（標準エラー）側へ出るので、標準出力に混ざることはありません。
    83  func run(args []string, list func() (string, error), out io.Writer) error {
    84  	fs := flag.NewFlagSet("base-branch", flag.ContinueOnError)
    85  
    86  	if err := fs.Parse(args); err != nil {
    87  		// ヘルプ要求は失敗ではないので 0 で終える。usage は flag が既に出力している。
    88  		if xerrors.Is(err, flag.ErrHelp) {
    89  			return nil
    90  		}
    91  
    92  		return xerrors.Wrap(err, "failed to parse flags")
    93  	}
    94

    96  	// もっともらしいブランチ名が返る。

    98  		return xerrors.Wrap(errUnexpectedArgs, strings.Join(fs.Args(), " "))
    99  	}
   100  
   101  	refs, err := list()
   102  	if err != nil {
   103  		return err
   104  	}
   105  
   106  	latest, err := latestRelease(refs)
   107  	if err != nil {
   108  		return err

   110

   116  // lsRemoteReleases は、origin のリリースラインの参照一覧を取得します。
   117  func lsRemoteReleases() (string, error) {
   118  	ctx, cancel := context.WithTimeout(context.Background(), commandTimeout)
   119  	defer cancel()
   120  
   121  	cmd := exec.CommandContext(ctx, "git", "ls-remote", "--heads", remoteName, refPrefix+releasePrefix+"*")
   122  	// git の失敗理由（認証・名前解決）はそのまま利用者へ見せる。握り潰すと
   123  	// 「リリースラインが無い」との区別が付かなくなる。
   124  	cmd.Stderr = os.Stderr
   125  
   126  	out, err := cmd.Output()
   127  	if err != nil {
   128  		return "", xerrors.Wrap(err, "git ls-remote --heads "+remoteName)
   129  	}
   130  
   131  	return string(out), nil
   132  }

   134  // latestRelease は、`git ls-remote --heads` の出力から最新のリリースラインを選びます。
   135  //
   136  // 解釈できる行が 1 つも無い場合はエラーにします。取得自体は成功しうる（リモートに
   137  // リリースラインがまだ無い、参照の書式が変わった）ため、0 件を「最新は空文字」として
   138  // 返すと、呼び出し側は解決できなかったことに気付かないまま空のベースを使います。
   139  func latestRelease(lsRemoteOutput string) (releaseLine, error) {
   140  	var (
   141  		latest releaseLine
   142  		found  bool
   143  	)
   144  
   145  	for line := range strings.Lines(lsRemoteOutput) {
   146  		parsed, ok := parseLine(line)
   147  		if !ok {
   148  			continue
   149  		}
   150  
   151  		if !found || parsed.newerThan(latest) {
   152  			latest, found = parsed, true
   153  		}
   154  	}
   155  
   156  	if !found {
   157  		return releaseLine{}, errNoReleaseBranch
   158  	}
   159  
   160  	return latest, nil
   161  }

   163  // parseLine は、`git ls-remote --heads` の 1 行（`<sha>\t<ref>`）をリリースラインへ変換します。
   164  // リリースラインの書式に合わない行は対象外として false を返します。
   165  func parseLine(line string) (releaseLine, bool) {
   166  	_, ref, ok := strings.Cut(strings.TrimSpace(line), "\t")
   167  	if !ok {
   168  		return releaseLine{}, false
   169  	}
   170  
   171  	name, ok := strings.CutPrefix(strings.TrimSpace(ref), refPrefix)
   172  	if !ok {
   173  		return releaseLine{}, false
   174  	}
   175  
   176  	m := releasePattern.FindStringSubmatch(name)
   177  	if m == nil {
   178  		return releaseLine{}, false
   179  	}
   180  
   181  	major, minor, patch, ok := parseTriple(m[1], m[2], m[3])
   182  	if !ok {
   183  		return releaseLine{}, false
   184  	}
   185  
   186  	return releaseLine{name: name, major: major, minor: minor, patch: patch}, true
   187  }
   188  
   189  // parseTriple は、バージョンの 3 数値を整数へ変換します。桁が int に収まらない場合は
   190  // false を返します。0 として扱うと、あり得ない大きさの版が最古として紛れ込みます。
   191  func parseTriple(rawMajor, rawMinor, rawPatch string) (int, int, int, bool) {
   192  	major, err := strconv.Atoi(rawMajor)
   193  	if err != nil {
   194  		return 0, 0, 0, false
   195  	}
   196  
   197  	minor, err := strconv.Atoi(rawMinor)
   198  	if err != nil {
   199  		return 0, 0, 0, false
   200  	}
   201  
   202  	patch, err := strconv.Atoi(rawPatch)
   203  	if err != nil {
   204  		return 0, 0, 0, false
   205  	}
   206  
   207  	return major, minor, patch, true
   208  }

   210  // newerThan は、レシーバが other より新しいリリースラインかを返します。
   211  func (r releaseLine) newerThan(other releaseLine) bool {
   212  	if r.major != other.major {
   213  		return r.major > other.major
   214  	}
   215  
   216  	if r.minor != other.minor {
   217  		return r.minor > other.minor
   218  	}
   219  
   220  	return r.patch > other.patch
   221  }
