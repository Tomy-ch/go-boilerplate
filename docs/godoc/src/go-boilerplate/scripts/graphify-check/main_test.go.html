    23  // runGit は、dir で git を実行します。失敗は出力ごとテストの失敗にします。
    24  func runGit(t *testing.T, dir string, args ...string) {
    25  	t.Helper()
    26  
    27  	cmd := exec.CommandContext(t.Context(), "git", append([]string{"-C", dir}, args...)...) //nolint:gosec // 引数はテスト内で組み立てた固定値
    28  	cmd.Env = append(os.Environ(),
    29  		"GIT_AUTHOR_NAME=t", "GIT_AUTHOR_EMAIL=t@example.com",
    30  		"GIT_COMMITTER_NAME=t", "GIT_COMMITTER_EMAIL=t@example.com",
    31  	)
    32  
    33  	out, err := cmd.CombinedOutput()
    34  	require.NoError(t, err, string(out))
    35  }

    37  // repositoryWith は、与えたパスにファイルを持つ git リポジトリを作ってそのパスを返します。
    38  // staged は index に載せるパスで、それ以外は未追跡のまま残します。
    39  func repositoryWith(t *testing.T, files map[string]string, staged ...string) string {
    40  	t.Helper()
    41  
    42  	dir := filepath.Join(t.TempDir(), "repo")
    43  	require.NoError(t, os.MkdirAll(dir, 0o750))
    44  	runGit(t, dir, "init", "--quiet")
    45  
    46  	for name, body := range files {
    47  		path := filepath.Join(dir, name)
    48  		require.NoError(t, os.MkdirAll(filepath.Dir(path), 0o750))
    49  		require.NoError(t, os.WriteFile(path, []byte(body), 0o600))
    50  	}
    51  	for _, name := range staged {
    52  		runGit(t, dir, "add", name)
    53  	}
    54  
    55  	return dir
    56  }

    69  func captureLog(t *testing.T) *bytes.Buffer {
    70  	t.Helper()
    71  	buffer := &bytes.Buffer{}
    72  	previous := log.Writer()
    73  	flags := log.Flags()
    74  	log.SetOutput(buffer)
    75  	log.SetFlags(0)
    76  	t.Cleanup(func() {
    77  		log.SetOutput(previous)
    78  		log.SetFlags(flags)
    79  	})
    80  
    81  	return buffer

    92  	return func(name string) ([]byte, error) {
