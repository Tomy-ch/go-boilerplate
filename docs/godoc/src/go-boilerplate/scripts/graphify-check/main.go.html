     1  // Package main は、追跡対象の graphify 成果物にマシン固有の情報が残っていないかを検査するツール。
     2  //
     3  //	-dir:  検査する graphify の出力ディレクトリ（既定 graphify-out）
     4  //	-pin:  抽出プロンプトの pin ファイル（既定 .agents/graphify/spec-pin.toml）
     5  //	-spec: これから抽出に使う extraction-spec.md。与えると pin と一致するか照合する
     6  //	-base: この ref からの差分に出力ディレクトリが含まれていないか検査する
     7  //
     8  // graphify の出力ディレクトリには、生成したマシンでしか意味を持たないファイルが同居する。
     9  // 抽出キャッシュの AST 側はノード ID を渡されたパスから組み立てるため絶対パスを焼き込み、
    10  // `.graphify_root` / `.graphify_python` はホストのパスそのものを保持する。相対パスへ直す
    11  // 後処理は最終グラフにしか効かないので、キャッシュ側は絶対パスを抱えたまま残る。これらが
    12  // 追跡対象に入ると、他の checkout では意味を成さない差分が出続けるうえ、公開リポジトリに
    13  // マシンの利用者名が残る。編集で消せない公開でもあるため、混入は入る前に止める。
    14  //
    15  // 検査は 4 つ。
    16  //
    17  //  1. 追跡してよい成果物の一覧から外れたものが追跡されていないか。.gitignore のホワイトリスト
    18  //     が破れた、あるいは意図しない add が起きた場合に当たる。
    19  //  2. 追跡してよい成果物自身にホストの絶対パスが混ざっていないか。上流が新しい経路で漏らし
    20  //     始めた場合に当たる。1 だけでは回帰を捉えられず、2 だけでは絶対パスを含まないマシン
    21  //     ローカルなファイルを見逃すので、両方が要る。
    22  //  3. セマンティックキャッシュの名前空間が pin と一致するか。名前空間は抽出プロンプトの
    23  //     fingerprint で決まるが、そのプロンプトはリポジトリ管理外のファイルなので、キャッシュ
    24  //     だけをコミットすると素性が追えなくなる。-spec を与えれば、これから使うプロンプトが
    25  //     pin と同じものかも同時に照合する。
    26  //  4. -base を与えたとき、その ref からの差分に出力ディレクトリが含まれていないか。グラフは
    27  //     1 ファイル 45 万行の塊で、コードを 1 行触るだけで全面が書き換わるため、並行するブランチ
    28  //     が同時に更新すると必ず衝突する。しかも生成物の衝突を再生成で解くという通常の逃げ道は、
    29  //     意味論抽出にモデルが要る以上ここでは使えない。だから更新はデフォルトブランチ側に寄せ、
    30  //     feature ブランチからは持ち込ませない。発火させる条件は呼び出し側が決める。
    31  //     ただし base 側に出力がまだ無い場合だけは通す。寄せ先が存在しないので衝突しようが無く、
    32  //     ここで止めると出力を最初に置く経路そのものが無くなるためである。
    33  //
    34  // このツールは lefthook と CI のゲートから呼ばれる。壊れ方が「何も検査しなくなる」方向に
    35  // 出るため、判定ロジックはシェルの中ではなくテストの当たる Go 側に置く。

   144  // run は、追跡中の graphify 成果物を検査して違反を報告します。
   145  // list は追跡パスの列挙手段、read はファイルの読み出し手段、diff は base ref からの変更パスの
   146  // 列挙手段、atBase は base 時点の追跡内容の列挙手段で、いずれも差し替えられるよう引数で受けます。
   147  func run(
   148  	args []string,
   149  	list func(dir string) ([]string, error),
   150  	read func(name string) ([]byte, error),
   151  	diff pathLister,
   152  	atBase pathLister,
   153  ) error {
   154  	fs := flag.NewFlagSet("graphify-check", flag.ContinueOnError)
   155  	dir := fs.String("dir", defaultDir, "検査する graphify の出力ディレクトリ")
   156  	pin := fs.String("pin", defaultPin, "抽出プロンプトの pin ファイル")
   157  	spec := fs.String("spec", "", "これから抽出に使う extraction-spec.md（省略時は照合しない）")
   158  	base := fs.String("base", "", "この ref からの差分に出力が含まれていないか検査する（省略時は検査しない）")
   159  
   160  	if err := fs.Parse(args); err != nil {
   161  		// ヘルプ要求は失敗ではないので 0 で終える。usage は flag が既に出力している。
   162  		if xerrors.Is(err, flag.ErrHelp) {
   163  			return nil
   164  		}
   165  
   166  		return xerrors.Wrap(err, "failed to parse flags")
   167  	}
   168  
   169  	fingerprint, err := readPinnedFingerprint(*pin, read)
   170  	if err != nil {
   171  		return err
   172  	}
   173  
   174  	if *spec != "" {
   175  		if err := verifySpec(*spec, fingerprint, read); err != nil {
   176  			return err
   177  		}
   178  	}
   179  
   180  	if *base != "" {
   181  		if err := verifyBase(*base, *dir, diff, atBase); err != nil {
   182  			return err
   183  		}
   184  	}
   185  
   186  	tracked, err := list(*dir)
   187  	if err != nil {
   188  		return xerrors.Wrap(err, "❌ 追跡ファイルを列挙できません")
   189  	}
   190  	// 追跡ゼロは「何も見なかった」ことであって「違反が無かった」ことではないので、合格と
   191  	// 同じ文言では報告しない。ただし追跡が無ければコミットされる成果物も無く、守るべきものが
   192  	// 無いため失敗にはしない。graphify を回す前の checkout や、出力を丸ごと無視する運用が
   193  	// これに当たる。壊れた走査（git が答えない・成果物が読めない）は下で失敗になる。
   194  	if len(tracked) == 0 {
   195  		log.Printf("graphify: %s に追跡中の成果物がないため、検査対象はありません", *dir)
   196  
   197  		return nil
   198  	}
   199  
   200  	violations, err := inspect(*dir, fingerprint, tracked, read)
   201  	if err != nil {
   202  		return err
   203  	}
   204  	if len(violations) == 0 {
   205  		log.Printf("✅ graphify: %d ファイルを検査、マシン固有の情報はありません", len(tracked))
   206  
   207  		return nil
   208  	}
   209  
   210  	// 呼び出し側が分岐に使うのは違反の種類だけなので、戻り値には種類を、中身はログへ出す。
   211  	kinds := make([]error, 0, len(violations))
   212  	for _, found := range violations {
   213  		log.Printf("❌ %s: %s", found.path, found.detail)
   214  		if !slices.Contains(kinds, found.kind) {
   215  			kinds = append(kinds, found.kind)
   216  		}
   217  	}
   218  
   219  	return xerrors.Join(kinds...)
   220  }

   222  // verifySpec は、これから抽出に使うプロンプトが pin と同じものかを照合します。
   223  func verifySpec(path, pinned string, read func(name string) ([]byte, error)) error {
   224  	body, err := read(path)
   225  	if err != nil {
   226  		return xerrors.Wrap(err, "❌ "+path+msgUnreadable)
   227  	}
   228  
   229  	actual := promptFingerprint(body)

   231  		return xerrors.Wrap(errSpecPinMismatch, fmt.Sprintf(
   232  			"❌ %s の fingerprint は %s で、pin の %s と一致しません"+
   233  				"（別バージョンの graphify、または別エージェント向けの spec を掴んでいます）",
   234  			path, actual, pinned,
   235  		))
   236  	}
   237  
   238  	log.Printf("✅ graphify: 抽出プロンプトは pin と一致します (%s)", pinned)
   239  
   240  	return nil
   241  }

   243  // verifyBase は、base ref からの差分に graphify の出力が含まれていないかを見ます。
   244  // base 側に出力がまだ無いときは、その差分は導入そのものなので通します。
   245  func verifyBase(base, dir string, diff, atBase pathLister) error {
   246  	changed, err := diff(base, dir)
   247  	if err != nil {
   248  		return xerrors.Wrap(err, "❌ "+base+" からの差分を取得できません")
   249  	}
   250  	if len(changed) == 0 {
   251  		return nil
   252  	}
   253  
   254  	existing, err := atBase(base, dir)
   255  	if err != nil {
   256  		return xerrors.Wrap(err, "❌ "+base+" の "+dir+" を列挙できません")
   257  	}
   258  	if len(existing) == 0 {
   259  		log.Printf("graphify: %s にはまだ %s が無いため、この差分は導入として扱います（%d 件）",
   260  			base, dir, len(changed))
   261  
   262  		return nil
   263  	}
   264  
   265  	log.Printf("❌ %s からの差分に graphify の出力が %d 件含まれています:", base, len(changed))
   266  	for _, path := range changed[:min(len(changed), maxListedChanges)] {
   267  		log.Printf("     %s", path)
   268  	}
   269  	if len(changed) > maxListedChanges {
   270  		log.Printf("     ... 他 %d 件", len(changed)-maxListedChanges)
   271  	}
   272  
   273  	return xerrors.Wrap(errArtifactInDiff, fmt.Sprintf(
   274  		"❌ %s の更新はデフォルトブランチ側に寄せています。このブランチからは外してください"+
   275  			"（`git checkout %s -- %s` で戻せます）", dir, base, dir,
   276  	))
   277  }

   279  // diffAgainst は、base ref と作業ツリーの間で dir 配下に変更のあったパスを返します。
   280  // 三点ではなく二点の差分を使います。三点の分岐点は base が進むたびに動くので、同じブランチが
   281  // 触ってもいない出力を「変更した」と報告し得ます。
   282  func diffAgainst(base, dir string) ([]string, error) {
   283  	output, err := exec.CommandContext( //nolint:gosec // 引数は base と dir のみで、コマンド自体は固定
   284  		context.Background(), "git", "diff", "--name-only", "-z", base, "--", dir,
   285  	).Output()
   286  	if err != nil {
   287  		return nil, xerrors.Wrap(err, "git diff "+base+" -- "+dir)
   288  	}
   289  
   290  	var changed []string
   291  	for name := range strings.SplitSeq(string(output), "\x00") {
   292  		if name != "" {
   293  			changed = append(changed, name)
   294  		}
   295  	}
   296  	sort.Strings(changed)
   297  
   298  	return changed, nil
   299  }

   322  // readPinnedFingerprint は、pin ファイルから抽出プロンプトの fingerprint を読みます。
   323  // 読み飛ばしや後勝ちの上書きは、その宣言が「存在しない」あるいは「行順で決まる」状態を
   324  // 警告なく作るため、解釈できない行と重複キーはエラーにします。
   325  func readPinnedFingerprint(path string, read func(name string) ([]byte, error)) (string, error) {
   326  	body, err := read(path)
   327  	if err != nil {
   328  		return "", xerrors.Wrap(err, "❌ "+path+msgUnreadable)
   329  	}
   330  
   331  	values := map[string]string{}
   332  	scanner := bufio.NewScanner(bytes.NewReader(body))
   333  	for lineNo := 1; scanner.Scan(); lineNo++ {
   334  		line := strings.TrimSpace(scanner.Text())
   335  		if line == "" || strings.HasPrefix(line, "#") {
   336  			continue
   337  		}
   338  		matched := pinLine.FindStringSubmatch(line)
   339  		if matched == nil {
   340  			return "", xerrors.Wrap(errPinInvalidLine, fmt.Sprintf("%s の %d 行目: %q", path, lineNo, line))
   341  		}
   342  		if _, duplicate := values[matched[1]]; duplicate {
   343  			return "", xerrors.Wrap(errPinDuplicateKey, fmt.Sprintf("%s の %d 行目: %q", path, lineNo, matched[1]))
   344  		}
   345  		values[matched[1]] = matched[2]
   346  	}
   347  	if err := scanner.Err(); err != nil {
   348  		return "", xerrors.Wrap(err, "❌ "+path+" を読めません")
   349  	}
   350  
   351  	fingerprint := values["spec_fingerprint"]
   352  	if fingerprint == "" {
   353  		return "", xerrors.Wrap(errPinMissingKey, path+" に spec_fingerprint がありません")
   354  	}
   355  
   356  	return fingerprint, nil
   357  }

   359  // promptFingerprint は、抽出プロンプトの fingerprint を graphify.cache.prompt_fingerprint と
   360  // 同じ規則で求めます。
   361  func promptFingerprint(body []byte) string {
   362  	text := strings.ReplaceAll(string(body), "\r\n", "\n")
   363  	text = strings.ReplaceAll(text, "\r", "\n")
   364  
   365  	lines := strings.Split(text, "\n")
   366  	for i, line := range lines {
   367  		lines[i] = strings.TrimRightFunc(line, unicode.IsSpace)
   368  	}
   369  	sum := sha256.Sum256([]byte(strings.TrimSpace(strings.Join(lines, "\n"))))
   370  
   371  	return hex.EncodeToString(sum[:])[:fingerprintLength]
   372  }

   394  // inspect は、追跡パスをホワイトリスト・絶対パス混入・キャッシュ名前空間の 3 観点で検査します。
   395  func inspect(
   396  	dir, fingerprint string, tracked []string, read func(name string) ([]byte, error),
