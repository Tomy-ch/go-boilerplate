    35  // useOpenAPIValidationWithAuthFailure は、提示された資格情報の検証が必ず失敗する形で実 OpenAPI
    36  // ミドルウェアを登録します。任意認証の operation が、無効な資格情報を匿名として通さないことを
    37  // 実際のミドルウェア経路で確かめるために使います（ADR-0021 (optional-authentication-fail-closed)）。
    38  func useOpenAPIValidationWithAuthFailure(t *testing.T, e *echo.Echo) {
    39  	t.Helper()
    40  
    41  	spec, err := validator.GetValidator()
    42  	require.NoError(t, err)
    43  
    44  	skipper := func(*echo.Context) bool { return false }
    45  	authFunc := func(_ context.Context, input *openapi3filter.AuthenticationInput) error {
    46  		// スロットを持つのはリクエストの context だけで、バリデータが渡す context は空
    47  		// （internal/controller/httpstack/oapi/auth/auth.go の NewAuthenticator と同じ理由）。
    48  		req := input.RequestValidationInput.Request
    49  		//nolint:contextcheck // input が内包する request の context のスロットへ書き戻すため
    50  		ctxhelper.SetAuthnFailure(req.Context(), ctxhelper.ErrUnauthenticatedUser)
    51  		return ctxhelper.ErrUnauthenticatedUser
    52  	}
    53  	e.Use(oapi.Middleware(spec, skipper, authFunc))
    54  }
