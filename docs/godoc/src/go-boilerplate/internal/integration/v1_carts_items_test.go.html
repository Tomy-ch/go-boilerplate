   241  func readBody(t *testing.T, response *http.Response) string {
   242  	t.Helper()
   243  	defer func() { require.NoError(t, response.Body.Close()) }()
   244  
   245  	body, err := io.ReadAll(response.Body)
   246  	require.NoError(t, err)
   247  	return string(body)
   248  }
