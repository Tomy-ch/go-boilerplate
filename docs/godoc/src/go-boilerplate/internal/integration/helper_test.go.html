    79  func (s *Server) Do(
    80  	method, path string, reqBody io.Reader, contentType string, headers http.Header,
    81  ) *http.Response {
    82  	s.t.Helper()
    83  
    84  	req, err := http.NewRequestWithContext(s.t.Context(), method, s.baseURL+path, reqBody)
    85  	require.NoError(s.t, err)
    86  
    87  	for k, vals := range headers {
    88  		for _, v := range vals {
    89  			req.Header.Add(k, v)
    90  		}
    91  	}
    92  	if contentType != "" && req.Header.Get("Content-Type") == "" {
    93  		req.Header.Set("Content-Type", contentType)
    94  	}
    95  
    96  	res, err := s.client.Do(req)
    97  	require.NoError(s.t, err)
    98  	defer s.t.Cleanup(func() {
    99  		require.NoError(s.t, res.Body.Close())
   100  	})
   101  
   102  	return res
