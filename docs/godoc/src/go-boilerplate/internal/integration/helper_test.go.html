    73  // Do は、任意メソッド/パス/ボディでHTTPを実行する。
    74  func (s *Server) Do(
    75  	method, path string, reqBody io.Reader, contentType string, headers http.Header,
    76  ) *http.Response {
    77  	s.t.Helper()
    78  
    79  	req, err := http.NewRequestWithContext(s.t.Context(), method, s.baseURL+path, reqBody)
    80  	require.NoError(s.t, err)
    81  
    82  	for k, vals := range headers {
    83  		for _, v := range vals {
    84  			req.Header.Add(k, v)
    85  		}
    86  	}
    87  	if contentType != "" && req.Header.Get("Content-Type") == "" {
    88  		req.Header.Set("Content-Type", contentType)
    89  	}
    90  
    91  	res, err := s.client.Do(req)
    92  	require.NoError(s.t, err)
    93  	defer s.t.Cleanup(func() {
    94  		require.NoError(s.t, res.Body.Close())
    95  	})
    96  
    97  	return res
    98  }
