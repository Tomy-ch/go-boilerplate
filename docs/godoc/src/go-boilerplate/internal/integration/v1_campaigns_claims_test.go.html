    39  	// Idempotency-Key はこの操作では必須なので、ヘッダには必ず載せる。

    45  		headers.Set("Idempotency-Key", key)

    47  		return headers

    50  	newIdempotencyDeps := func(t *testing.T, claimCount, completes int) idempotency.Deps {
    51  		t.Helper()
    52  
    53  		ctrl := gomock.NewController(t)
    54  		store := mock_idempotency.NewMockStore(ctrl)
    55  		store.EXPECT().Claim(gomock.Any(), gomock.Any()).Return(true, nil).Times(claimCount)
    56  		store.EXPECT().Complete(gomock.Any(), gomock.Any()).Return(nil).Times(completes)
    57  		txm := mock_tx.NewMockManager(ctrl)
    58  		txm.EXPECT().Do(gomock.Any(), gomock.Any()).DoAndReturn(
    59  			func(ctx context.Context, fn func(context.Context) error) error { return fn(ctx) },
    60  		).Times(claimCount)
    61  
    62  		return idempotency.Deps{
    63  			Txm:   txm,
    64  			Store: store,
    65  			Clock: clocktest.NewMockClock(t, issuedAt),
    66  		}
    67  	}

    92  		t.Run("POST /v1/campaigns/claims が受け取ったクーポンの 201 を返す", func(t *testing.T) {
    93  			t.Parallel()
    94  
    95  			e := echo.New()
    96  			tf := observability.NewNoopTracerFactory(t)
    97  
    98  			var capturedCode string
    99  			mockUC := mock_campaign.NewMockUsecase(gomock.NewController(t))
   100  			mockUC.EXPECT().ClaimCoupon(gomock.Any(), gomock.Any(), gomock.Any()).DoAndReturn(
   101  				func(_ context.Context, authn *auth.Authn, code string) (campaignuc.ClaimedCouponView, error) {
   102  					require.NotNil(t, authn)
   103  					capturedCode = code
   104  
   105  					return newView(t), nil
   106  				},
   107  			)
   108  
   109  			claims.BindHandler(e, tf, mockUC, newIdempotencyDeps(t, 1, 1))
   110  
   111  			actual := StartServer(t, e).
   112  				DoJSON(http.MethodPost, campaignClaimsPath, newBody(), availableClaimer(t, e, "claim-ok"))
   113  
   114  			assert.Equal(t, http.StatusCreated, actual.StatusCode)
   115  			var body claimsgen.CouponResponse
   116  			require.NoError(t, json.NewDecoder(actual.Body).Decode(&body))
   117  			assert.Equal(t, claimsgen.CouponDiscountKind("rate"), body.Discount.Kind)
   118  			assert.Equal(t, "0.1", body.Discount.Value)
   119  			assert.Nil(t, body.UsedAt)
   120  			assert.Equal(t, expiresAt, body.ExpiresAt)
   121  			assert.Equal(t, issuedAt, body.IssuedAt)
   122  			// 正規化はドメインの責務なので、要求の値がそのまま届く。
   123  			assert.Equal(t, "welcome-2026", capturedCode)
   124  		})
   125  	})
