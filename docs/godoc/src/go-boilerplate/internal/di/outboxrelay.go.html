    44  // RunOutboxReplay は、dead 状態の outbox 行を pending へ戻すワンショット実行を行います。
    45  // poll ループを持たない（OutboxRelayModule を含まない）一時的な fx.App で ReplayUsecase を解決します。
    46  func RunOutboxReplay(ctx context.Context, messageID *uuid.UUID) (int64, error) {
    47  	var replay outboxuc.ReplayUsecase
    48  
    49  	opts := append(outboxRelayCommonOptions(), fx.Populate(&replay), fx.WithLogger(NewFxEventLogger))
    50  	app := fx.New(opts...)
    51  
    52  	if err := app.Start(ctx); err != nil {
    53  		return 0, err
    54  	}
    55  
    56  	result, runErr := replay.ReplayDead(ctx, messageID)
    57  
    58  	stopCtx, cancel := context.WithTimeout(context.Background(), outboxReplayStopTimeout)
    59  	defer cancel()
    60  	stopErr := app.Stop(stopCtx) //nolint:contextcheck // 停止用 context は意図的に ctx を継承しない
    61  
    62  	// replay 失敗と OnStop 失敗（DB プール Close 等）の双方をオペレータへ可視化する。
    63  	// errors.Join は両方 nil なら nil を返す。
    64  	return result, errors.Join(runErr, stopErr)
    65  }
