    48  // RunJob は、ジョブ実行用の開始関数・停止関数を生成して返します。context／タイムアウトの制御は返した関数の呼出側が担います。
    49  // grace（APP_SHUTDOWN_TIMEOUT）を fx.StopTimeout に設定し、停止時に fx 既定（15s）が
    50  // 停止猶予より先に teardown を打ち切らないようにします（停止軸を grace に一本化）。
    51  func RunJob(grace time.Duration) (StartFunc, StopFunc) {
    52  	var (
    53  		state  job.State
    54  		logger logging.Logger
    55  		osCfg  *config.OperatingSystemConfig
    56  	)
    57  
    58  	app := fx.New(
    59  		NewJobCore(),
    60  		fx.StopTimeout(grace),
    61  		fx.Populate(&state, &logger, &osCfg),
    62  		fx.WithLogger(NewFxEventLogger),
    63  	)
    64  
    65  	start := func(ctx context.Context, name string, args []string) <-chan error {
    66  		l := logger.Named("job.RunJob").CallerSkip(callSkip)
    67  		done := make(chan error, 1)
    68  		state.Set(name, args, done)
    69  
    70  		if err := app.Start(ctx); err != nil {
    71  			fields := append(jobEventFields(logging.EventTypeStart, osCfg.TimeZone()),
    72  				logging.String(logging.JobNameKey, name),
    73  				logging.Strings(logging.JobArgsKey, args),
    74  				logging.Error(logging.JobErrorKey, err),
    75  			)
    76  			l.Error("failed to start job application", fields...)
    77  			// 共有 done に触れると hook goroutine と二重 close／送信になり得るため、専用チャネルで返す。
    78  			failCh := make(chan error, 1)
    79  			failCh <- err
    80  			close(failCh)
    81  			return failCh
    82  		}
    83  
    84  		fields := append(jobEventFields(logging.EventTypeStart, osCfg.TimeZone()),
    85  			logging.String(logging.JobNameKey, name),
    86  			logging.Strings(logging.JobArgsKey, args),
    87  		)
    88  		l.Info("job started", fields...)
    89  
    90  		return done
    91  	}
    92  
    93  	stop := func(ctx context.Context) error {
    94  		l := logger.Named("job.RunJob").CallerSkip(callSkip)
    95  		if err := app.Stop(ctx); err != nil {
    96  			fields := append(jobEventFields(logging.EventTypeEnd, osCfg.TimeZone()),
    97  				logging.Error(logging.JobErrorKey, err),
    98  			)
    99  			l.Error("failed to stop job application", fields...)
   100  			return err
   101  		}
   102  
   103  		l.Info("job application finished", jobEventFields(logging.EventTypeEnd, osCfg.TimeZone())...)
   104  		return nil
   105  	}
   106  
   107  	return start, stop
   108  }
