    32  // RunWorker は、worker 実行用の開始関数・停止関数を生成して返します。
    33  // grace（APP_SHUTDOWN_TIMEOUT）を fx.StopTimeout に設定し、停止時に fx 既定（15s）が
    34  // DrainTimeout より先に drain を打ち切らないようにします。
    35  func RunWorker(grace time.Duration) (StartFunc, StopFunc) {
    36  	var (
    37  		state  workerboundary.State
    38  		logger logging.Logger
    39  	)
    40  
    41  	app := fx.New(
    42  		NewWorkerCore(),
    43  		fx.StopTimeout(grace),
    44  		fx.Populate(&state, &logger),
    45  		fx.WithLogger(NewFxEventLogger),
    46  	)
    47  
    48  	start := func(ctx context.Context, name string, args []string) <-chan error {
    49  		l := logger.Named("worker.RunWorker").CallerSkip(callSkip)
    50  		done := make(chan error, 1)
    51  		state.Set(name, args, done)
    52  
    53  		if err := app.Start(ctx); err != nil {
    54  			l.Error(
    55  				ctx,
    56  				"failed to start worker application",
    57  				logging.String(logging.WorkerNameKey, name),
    58  				logging.Error(logging.ErrorKey, err),
    59  			)
    60  			// done は hook goroutine が送信/close を所有するため、起動失敗は専用チャネルで返す。
    61  			failCh := make(chan error, 1)
    62  			failCh <- err
    63  			close(failCh)
    64  			return failCh
    65  		}
    66  
    67  		l.Info(ctx, "worker started", logging.String(logging.WorkerNameKey, name))
    68  		return done
    69  	}
    70  
    71  	stop := func(ctx context.Context) error {
    72  		l := logger.Named("worker.RunWorker").CallerSkip(callSkip)
    73  		if err := app.Stop(ctx); err != nil {
    74  			l.Error(ctx, "failed to stop worker application", logging.Error(logging.ErrorKey, err))
    75  			return err
    76  		}
    77  		l.Info(ctx, "worker application finished")
    78  		return nil
    79  	}
    80  
    81  	return start, stop
    82  }
