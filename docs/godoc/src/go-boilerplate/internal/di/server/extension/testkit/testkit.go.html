    12  // RequireProvidesOne は、与えたモジュール群が指定 group に要素をちょうど1件 provide することを検証する。
    13  // fx.NopLogger は構成時のログ出力を抑えるだけで、検証結果には影響しない。
    14  func RequireProvidesOne[T any](t *testing.T, group string, opts ...fx.Option) {
    15  	t.Helper()
    16  
    17  	var got []T
    18  	app := fx.New(append(opts,
    19  		fx.Populate(fx.Annotate(&got, fx.ParamTags(`group:"`+group+`"`))),
    20  		fx.NopLogger,
    21  	)...)
    22  
    23  	require.NoError(t, app.Start(context.Background()))
    24  	defer func() { require.NoError(t, app.Stop(context.Background())) }()
    25  
    26  	require.Len(t, got, 1)
    27  }
