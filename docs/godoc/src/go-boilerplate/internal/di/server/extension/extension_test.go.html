    18  func orderTagMiddleware(tag string) echo.MiddlewareFunc {
    19  	return func(next echo.HandlerFunc) echo.HandlerFunc {
    20  		return func(c echo.Context) error {
    21  			c.Response().Header().Add("X-Order", tag)
    22  			return next(c)
    23  		}
    24  	}
    25  }

   308  	t.Parallel()
   309  
   310  	t.Run("重複なしの場合は空スライスを返す", func(t *testing.T) {
   311  		t.Parallel()

   322  
   323  	t.Run("重複がある場合はそのpriorityのみ返す", func(t *testing.T) {
   324  		t.Parallel()
   325  
   326  		input := map[int][]string{
   327  			1: {"A", "B"}, // ★ 重複
   328  			2: {"C"},
   329  		}
   330  
   331  		got := extractPriorityConflicts(input)
   332  
   333  		require.Len(t, got, 1)
   334  		assert.Contains(t, got[0], "priority=1")
   335  		assert.Contains(t, got[0], "[A B]") // names の出力
   336  	})
   337  
   338  	t.Run("複数のpriorityが重複している場合は複数返す", func(t *testing.T) {
   339  		t.Parallel()
   340  
   341  		input := map[int][]string{
   342  			1: {"A", "B"}, // ★ 重複
   343  			2: {"C"},
   344  			3: {"X", "Y", "Z"}, // ★ 重複
   345  		}
   346  
   347  		got := extractPriorityConflicts(input)
   348  
   349  		require.Len(t, got, 2)
   350  		assert.Contains(t, got[0], "priority=")
   351  		assert.Contains(t, got[1], "priority=")
   352  
   353  		// priority=1 が含まれていること
   354  		assert.True(t, containsSubstring(got, "priority=1"))
   355  		// priority=3 が含まれていること
   356  		assert.True(t, containsSubstring(got, "priority=3"))
   357  	})
   358  
   359  	t.Run("names が3つ以上の場合でも正しくフォーマットされる", func(t *testing.T) {
   360  		t.Parallel()
   361  
   362  		input := map[int][]string{
   363  			5: {"A", "B", "C"},
   364  		}
   365

   367  		require.Len(t, got, 1)

   370  	})
