   118  // applyMiddlewares は、kind 種別のミドルウェアを優先度順に apply で適用します。
   119  func applyMiddlewares(logger logging.Logger, kind string, mws []middlewareEntry, apply func(...echo.MiddlewareFunc)) error {
   120  	if err := validatePriorityConflicts(kind, mws); err != nil {
   121  		return err
   122  	}
   123  
   124  	log := logger.Named("ApplyMiddlewares").CallerSkip(callerSkip)
   125  	log.Info(
   126  		fmt.Sprintf("Applying %s middleware", kind),
   127  		logging.Int("count", len(mws)),
   128  	)
   129  
   130  	sort.Slice(mws, func(i, j int) bool {
   131  		return mws[i].priority < mws[j].priority
   132  	})
   133  
   134  	for _, mw := range mws {
   135  		log.Info(
   136  			fmt.Sprintf("Applying %s middleware", kind),
   137  			logging.Int("priority", mw.priority),
   138  			logging.String("middleware", mw.name),
   139  		)
   140  		apply(mw.middleware)
   141  	}
   142  	return nil
   143  }

   145  // validatePriorityConflicts は、kind 種別のミドルウェアに Priority の重複がないか検証します。
   146  func validatePriorityConflicts(kind string, mws []middlewareEntry) error {
   147  	byPriority := make(map[int][]string)
   148  
   149  	for _, mw := range mws {
   150  		byPriority[mw.priority] = append(byPriority[mw.priority], mw.name)
   151  	}
   152  
   153  	conflicts := extractPriorityConflicts(byPriority)
   154  
   155  	if len(conflicts) > 0 {
   156  		return xerrors.New(fmt.Sprintf("duplicate %s middleware priorities: %s",
   157  			kind, strings.Join(conflicts, "; "),
   158  		))
   159  	}
   160  
   161  	return nil
   162  }
   163  
   164  // extractPriorityConflicts は、priority ごとに名前が重複しているものを抽出して返します。
   165  // 戻り値は priority 昇順の conflict 表現の文字列スライス。
   166  func extractPriorityConflicts(byPriority map[int][]string) []string {
   167  	priorities := make([]int, 0, len(byPriority))
   168  	for p := range byPriority {
   169  		priorities = append(priorities, p)
   170  	}
   171  	sort.Ints(priorities)
   172  
   173  	var conflicts []string
   174  	for _, p := range priorities {
   175  		names := byPriority[p]
   176  		if len(names) > 1 {
   177  			conflicts = append(conflicts,
   178  				fmt.Sprintf("priority=%d: %v", p, names),
   179  			)
   180  		}
   181  	}
   182  	return conflicts
   183  }
