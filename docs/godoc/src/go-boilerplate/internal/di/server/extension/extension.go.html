   110  // applyMiddlewares は、kind 種別のミドルウェアを優先度順に apply で適用します。
   111  func applyMiddlewares(logger logging.Logger, kind string, mws []middlewareEntry, apply func(...echo.MiddlewareFunc)) error {
   112  	if err := validatePriorityConflicts(kind, mws); err != nil {
   113  		return err
   114  	}
   115  
   116  	log := logger.Named("ApplyMiddlewares").CallerSkip(callerSkip)
   117  	log.Info(
   118  		fmt.Sprintf("Applying %s middleware", kind),
   119  		logging.Int("count", len(mws)),
   120  	)
   121  
   122  	sort.Slice(mws, func(i, j int) bool {
   123  		return mws[i].priority < mws[j].priority
   124  	})
   125  
   126  	for _, mw := range mws {
   127  		log.Info(
   128  			fmt.Sprintf("Applying %s middleware", kind),
   129  			logging.Int("priority", mw.priority),
   130  			logging.String("middleware", mw.name),
   131  		)
   132  		apply(mw.middleware)
   133  	}
   134  	return nil
   135  }

   137  // validatePriorityConflicts は、kind 種別のミドルウェアに Priority の重複がないか検証します。
   138  func validatePriorityConflicts(kind string, mws []middlewareEntry) error {
   139  	byPriority := make(map[int][]string)
   140  
   141  	for _, mw := range mws {
   142  		byPriority[mw.priority] = append(byPriority[mw.priority], mw.name)
   143  	}
   144  
   145  	conflicts := extractPriorityConflicts(byPriority)
   146  
   147  	if len(conflicts) > 0 {
   148  		return xerrors.New(fmt.Sprintf("duplicate %s middleware priorities: %s",
   149  			kind, strings.Join(conflicts, "; "),
   150  		))
   151  	}
   152  
   153  	return nil
   154  }

   156  // extractPriorityConflicts は、priority ごとに名前が重複しているものを抽出して返します。
   157  // 戻り値は priority 昇順の conflict 表現の文字列スライス。
   158  func extractPriorityConflicts(byPriority map[int][]string) []string {
   159  	priorities := make([]int, 0, len(byPriority))
   160  	for p := range byPriority {
   161  		priorities = append(priorities, p)
   162  	}
   163  	sort.Ints(priorities)
   164  
   165  	var conflicts []string
   166  	for _, p := range priorities {
   167  		names := byPriority[p]
   168  		if len(names) > 1 {
   169  			conflicts = append(conflicts,
   170  				fmt.Sprintf("priority=%d: %v", p, names),
   171  			)
   172  		}
   173  	}
   174  	return conflicts
   175  }
