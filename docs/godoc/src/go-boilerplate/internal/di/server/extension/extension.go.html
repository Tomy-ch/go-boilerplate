   124  // applyMiddlewares は、kind 種別のミドルウェアを優先度順に apply で適用します。
   125  func applyMiddlewares(logger logging.Logger, kind string, mws []middlewareEntry, apply func(...echo.MiddlewareFunc)) error {
   126  	if err := validatePriorityConflicts(kind, mws); err != nil {
   127  		return err
   128  	}
   129  
   130  	log := logger.Named("ApplyMiddlewares").CallerSkip(callerSkip)
   131  	log.Info(
   132  		context.Background(),
   133  		fmt.Sprintf("Applying %s middleware", kind),
   134  		logging.Int("count", len(mws)),
   135  	)
   136  
   137  	sort.Slice(mws, func(i, j int) bool {
   138  		return mws[i].priority < mws[j].priority
   139  	})
   140  
   141  	for _, mw := range mws {
   142  		log.Info(
   143  			context.Background(),
   144  			fmt.Sprintf("Applying %s middleware", kind),
   145  			logging.Int("priority", mw.priority),
   146  			logging.String("middleware", mw.name),
   147  		)
   148  		apply(mw.middleware)
   149  	}
   150  	return nil
   151  }

   153  // validatePriorityConflicts は、kind 種別のミドルウェアに Priority の重複がないか検証します。
   154  func validatePriorityConflicts(kind string, mws []middlewareEntry) error {
   155  	byPriority := make(map[int][]string)
   156  
   157  	for _, mw := range mws {
   158  		byPriority[mw.priority] = append(byPriority[mw.priority], mw.name)
   159  	}
   160  
   161  	conflicts := extractPriorityConflicts(byPriority)
   162  
   163  	if len(conflicts) > 0 {
   164  		return xerrors.New(fmt.Sprintf("duplicate %s middleware priorities: %s",
   165  			kind, strings.Join(conflicts, "; "),
   166  		))
   167  	}
   168  
   169  	return nil
   170  }

   172  // extractPriorityConflicts は、priority ごとに名前が重複しているものを抽出して返します。
   173  // 戻り値は priority 昇順の conflict 表現の文字列スライス。
   174  func extractPriorityConflicts(byPriority map[int][]string) []string {
   175  	priorities := make([]int, 0, len(byPriority))
   176  	for p := range byPriority {
   177  		priorities = append(priorities, p)
   178  	}
   179  	sort.Ints(priorities)
   180  
   181  	var conflicts []string
   182  	for _, p := range priorities {
   183  		names := byPriority[p]
   184  		if len(names) > 1 {
   185  			conflicts = append(conflicts,
   186  				fmt.Sprintf("priority=%d: %v", p, names),
   187  			)
   188  		}
   189  	}
   190  	return conflicts
   191  }
