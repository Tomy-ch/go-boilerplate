    16  // serveThroughMiddleware は mw 適用後に / を1回叩き応答コードを返す。ミドルウェアがリクエスト
    17  // チェーンを壊さず素通しさせることを検証するために用いる。
    18  func serveThroughMiddleware(t *testing.T, mw echo.MiddlewareFunc) int {
    19  	t.Helper()
    20  	e := echo.New()
    21  	e.Use(mw)
    22  	e.GET("/", func(c echo.Context) error { return c.NoContent(http.StatusNoContent) })
    23  	req := httptest.NewRequestWithContext(t.Context(), http.MethodGet, "/", nil)
    24  	rec := httptest.NewRecorder()
    25  	e.ServeHTTP(rec, req)
    26  	return rec.Code
    27  }
