   170  	t.Run("異常系", func(t *testing.T) {
   171  		t.Parallel()
   172  
   173  		t.Run("処理中接続が残りShutdownがタイムアウトするとErrorログを出しエラーを返す", func(t *testing.T) {
   174  			t.Parallel()
   175  
   176  			ctrl := gomock.NewController(t)
   177  			mockLogger := mock_logging.NewMockLogger(ctrl)
   178  			mockLogger.EXPECT().Named("server.Stop").Return(mockLogger).AnyTimes()
   179  			mockLogger.EXPECT().CallerSkip(serverCallerSkip).Return(mockLogger).AnyTimes()
   180  			mockLogger.EXPECT().Info("http stopping", gomock.Any(), gomock.Any(), gomock.Any()).Times(1)
   181  			mockLogger.EXPECT().Error("failed to shutdown http server", gomock.Any()).Times(1)
   182  
   183  			cfg := config.MockConfigForTest(t)
   184  			srvCfg := config.NewServerConfig(cfg)
   185  			osCfg := config.NewOperatingSystemConfig(cfg)
   186  
   187  			e := server.NewAppServer(srvCfg)
   188  
   189  			// ハンドラを処理中にして接続を active に保ち、Shutdown を idle 完了させない
   190  			entered := make(chan struct{})
   191  			release := make(chan struct{})
   192  			e.GET("/block", func(c echo.Context) error {
   193  				close(entered)
   194  				<-release
   195  				return c.NoContent(http.StatusOK)
   196  			})
   197  
   198  			lc := &net.ListenConfig{}
   199  			ln, err := lc.Listen(context.Background(), "tcp", "127.0.0.1:0")
   200  			require.NoError(t, err)
   201  			e.Listener = ln
   202  			go func() { _ = e.Start("") }()
   203  			t.Cleanup(func() { close(release) })
   204  
   205  			go func() {
   206  				req, rerr := http.NewRequestWithContext(context.Background(), http.MethodGet, "http://"+ln.Addr().String()+"/block", nil)
   207  				if rerr != nil {
   208  					return
   209  				}
   210  				resp, gerr := http.DefaultClient.Do(req)
   211  				if gerr == nil {
   212  					_ = resp.Body.Close()
   213  				}
   214  			}()
   215  			<-entered // リクエストが処理中＝接続が active になったことを保証
   216  
   217  			// 既に期限の切れた context で Shutdown → 処理中接続が残り context error を返す
   218  			ctx, cancel := context.WithTimeout(context.Background(), 20*time.Millisecond)
   219  			defer cancel()
   220  			fn := newStopServerFunc(e, mockLogger, osCfg)
   221  			require.Error(t, fn(ctx))
   222  		})
   223  	})
