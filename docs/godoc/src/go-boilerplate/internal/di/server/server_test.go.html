    90  func TestServerShutdownValidationWiring(t *testing.T) {
    91  	t.Run("正常系", func(t *testing.T) {
    92  		t.Run("shutdown>=request の設定では fx.Invoke 経由でも起動できる", func(t *testing.T) {
    93  			cfg := config.MockConfigForTest(t)

    95  				fx.NopLogger,
    96  				fx.Supply(config.NewApplicationConfig(cfg), config.NewServerConfig(cfg)),
    97  				fx.Invoke(config.ValidateServerShutdown),
    98  			)
    99  			require.NoError(t, app.Start(context.Background()))
   100  			require.NoError(t, app.Stop(context.Background()))
   101  		})
   102  	})
   103  
   104  	t.Run("異常系", func(t *testing.T) {
   105  		t.Run("shutdown<request の設定では fx.Invoke が起動を失敗させる", func(t *testing.T) {
   106  			config.EnsureRepoRootAndEnv(t, config.TestingEnvValue)
   107  			t.Setenv("SERVER_REQUEST_TIMEOUT", "60s")
   108  			t.Setenv("APP_SHUTDOWN_TIMEOUT", "30s")
   109  
   110  			cfg, err := config.New()
   111  			require.NoError(t, err) // New() は交差検証しないため、不正設定でも構築は成功する
   112  
   113  			app := fx.New(
   114  				fx.NopLogger,
   115  				fx.Supply(config.NewApplicationConfig(cfg), config.NewServerConfig(cfg)),
   116  				fx.Invoke(config.ValidateServerShutdown),
