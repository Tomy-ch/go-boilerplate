    29  		f.record(e.Err,

    31  			[]*logging.Field{logging.String("callee", e.FunctionName), logging.String("caller", e.CallerName)},

    33  			logging.String("callee", e.FunctionName), logging.String("caller", e.CallerName),
    34  			logging.DurationMs("runtime_ms", e.Runtime))
    35  	case *fxevent.Supplied:

    37  			"Supply failed",
    38  			[]*logging.Field{logging.String("type", e.TypeName), logging.String("module", e.ModuleName)},
    39  			f.logger.Debug, "Supplied",
    40  			logging.String("type", e.TypeName), logging.String("module", e.ModuleName))
    41  	case *fxevent.Provided:

    43  			"Provide failed",
    44  			[]*logging.Field{logging.String("constructor", e.ConstructorName), logging.String("module", e.ModuleName)},
    45  			f.logger.Debug, "Provided",
    46  			logging.String("constructor", e.ConstructorName), logging.String("module", e.ModuleName),
    47  			logging.Strings("output_types", e.OutputTypeNames))
    48  	case *fxevent.Invoked:

    70  			f.logger.Debug, "Custom logger initialized",
    71  			logging.String("constructor", e.ConstructorName))
    72  	case *fxevent.Started:
