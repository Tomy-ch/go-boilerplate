    20  // WorkerModule は、worker engine 関連の依存関係を提供する fx.Module です。
    21  // 既定では worker を 1 つも登録しません。
    22  func WorkerModule() fx.Option {
    23  	return fx.Module("worker",
    24  		provideWorkers(
    25  		// ここに worker のコンストラクタを追加します（例: <pkg>.New）。
    26  		// 各 worker は usecase/boundary/worker.Worker を実装し、Consumer(broker adapter) を内包します。
    27  		),
    28  		provideQueueStatsTargets(
    29  		// ここに QueueStatsTarget のコンストラクタを追加します。
    30  		// SQS など QueueStatsProvider を実装する adapter を使う worker のときだけ任意登録します。
    31  		// コンストラクタは queuemetrics.Target を返し、sqs.NewQueueStatsProvider で Provider を組み立てます。
    32  		),
    33  		fx.Provide(
    34  			observability.NewWorkerMetrics,
    35  			diworker.ProvideEngine,
    36  			workercontroller.NewState,
    37  			provideQueueStatsCollector,
    38  		),
    39  		// 起動時に DrainTimeout < grace を検証する。違反時は app.Start が失敗する。
    40  		fx.Invoke(diworker.ValidateShutdownGrace),
    41  		fx.Invoke(hook.RegisterWorkerHooks),
    42  		// queue depth / DLQ 収集器を Prometheus レジストリへ登録する（対象が無ければ何も出さない）。
    43  		fx.Invoke(queuemetrics.RegisterStatsCollector),
    44  	)
    45  }
