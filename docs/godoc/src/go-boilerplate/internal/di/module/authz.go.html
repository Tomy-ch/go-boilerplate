    31  // provideAuthorizer は、環境（EnvLocal / EnvCI / EnvTest）に対応した Authorizer を返します。
    32  // 現状の実装は全許可（allowall）の割り切りであるため、本番相当の環境では誤って全許可を
    33  // 配線しないようエラーを返し、RBAC / 外部ポリシーエンジン実装への差し替えを強制します。
    34  func provideAuthorizer(appCfg *config.ApplicationConfig, logger logging.Logger) (authzbd.Authorizer, error) {
    35  	switch appCfg.Env() {
    36  	case config.EnvLocal, config.EnvCI, config.EnvTest:
    37  		logger.Named("authz").CallerSkip(callerSkipCount).Warn(
    38  			context.Background(),
    39  			"Allow-all authorizer wired: every request is permitted (non-production only)",
    40  			logging.String("env", appCfg.Env()),
    41  		)
    42  
    43  		return allowall.New(), nil
    44  	default:
    45  		logger.Named("authz").CallerSkip(callerSkipCount).Error(
    46  			context.Background(),
    47  			"No authorizer configured for the current environment",
    48  			logging.String("env", appCfg.Env()),
    49  		)
    50  
    51  		return nil, xerrors.New("no authorizer configured for environment: " + appCfg.Env())
    52  	}
    53  }
