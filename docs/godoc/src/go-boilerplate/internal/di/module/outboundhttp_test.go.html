    48  func Test_provideOutboundHTTPClient(t *testing.T) {
    49  	t.Parallel()
    50  
    51  	t.Run("正常系", func(t *testing.T) {
    52  		t.Parallel()
    53  
    54  		t.Run("env に応じて private 網宛ての可否が変わる", func(t *testing.T) {
    55  			t.Parallel()
    56  			// httptest は loopback（private）で待つ。計装なしの実 transport を使うのは、
    57  			// テスト用の noop transport が全接続を許可する dial control を持ち、
    58  			// 方針の違いが結果に出ないため。
    59  			srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
    60  				w.WriteHeader(http.StatusNoContent)
    61  			}))
    62  			defer srv.Close()
    63  
    64  			// transport を共有すると接続プールも共有され、許可側が張った接続を拒否側が再利用して
    65  			// dial 時のガードが発火しない。方針ごとに transport を分けて判定する。
    66  			localErr := getWith(
    67  				t,
    68  				provideOutboundHTTPClient(observability.NewGuardedHTTPClientTransport(t), newAppCfgForEnv(t, config.EnvLocal)),
    69  				srv.URL,
    70  			)
    71  			prdErr := getWith(
    72  				t,
    73  				provideOutboundHTTPClient(observability.NewGuardedHTTPClientTransport(t), newAppCfgForEnv(t, config.EnvProduction)),
    74  				srv.URL,
    75  			)
    76  
    77  			require.NoError(t, localErr, "local で private 網宛てが拒否されている")
    78  			require.Error(t, prdErr, "prd で private 網宛てが許可されており、env の方針が下流へ効いていない")
    79  		})
    80  	})
    81  }

    84  func getWith(t *testing.T, client *observability.OutboundHTTPClient, url string) error {
    85  	t.Helper()
