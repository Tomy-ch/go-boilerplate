    23  func TestObservabilityModule_ProvidesTracerFactory(t *testing.T) {
    24  	t.Parallel()
    25  
    26  	t.Run("正常系", func(t *testing.T) {
    27  		t.Parallel()
    28  
    29  		t.Run("fx アプリで TracerFactory が提供される", func(t *testing.T) {
    30  			t.Parallel()
    31  
    32  			ctrl := gomock.NewController(t)
    33  
    34  			mockReg := mock_lifecycle.NewMockRegistrar(ctrl)
    35  			mockLog := mock_logging.NewMockLogger(ctrl)
    36  			mockLF := mock_logging.NewMockLogFieldBuilder(ctrl)
    37  
    38  			// ProviderShutdowner の Shutdown が単一の Stop フックとして登録される。
    39  			mockReg.EXPECT().RegisterStop(gomock.Any()).Times(1)
    40  
    41  			// buildinfo.Register はデフォルトレジストリへ登録するため、テスト間の汚染を避ける。
    42  			origReg := prometheus.DefaultRegisterer
    43  			prometheus.DefaultRegisterer = prometheus.NewRegistry()
    44  			t.Cleanup(func() { prometheus.DefaultRegisterer = origReg })
    45  
    46  			var tf observability.TracerFactory
    47  
    48  			app := fx.New(
    49  				ObservabilityModule(),
    50  				fx.Provide(func() testing.TB { return t }),
    51  				fx.Provide(func() lifecycle.Registrar { return mockReg }),
    52  				fx.Provide(func() logging.Logger { return mockLog }),
    53  				fx.Provide(func() logging.LogFieldBuilder { return mockLF }),
    54  				fx.Provide(func() *config.ApplicationConfig {
    55  					return config.NewApplicationConfig(config.MockConfigForTest(t))
    56  				}),
    57  				fx.Provide(func() *config.ObservabilityConfig {
    58  					oc := config.NewObservabilityConfig(config.MockConfigForTest(t))
    59  					oc.SetObservabilityTracesExporter(t, "")
    60  					oc.SetObservabilityMetricsExporter(t, "")
    61  					oc.SetObservabilityLogsExporter(t, "")
    62  					return oc
    63  				}),
    64  				fx.Provide(system.NewBuildInfo),
    65  				fx.Populate(&tf),
    66  				fx.NopLogger,
    67  			)
    68  
    69  			require.NoError(t, app.Start(context.Background()))
    70  			t.Cleanup(func() { require.NoError(t, app.Stop(context.Background())) })
    71  			assert.NotNil(t, tf)
    72  		})
    73  	})

    76  func TestObservabilityModule(t *testing.T) {
    77  	t.Parallel()
    78  
    79  	// ObservabilityModule は config / logging / BuildInfo を外から受け取るため、
    80  	// 依存だけを供給した最小構成に対象モジュールを重ねて検証する。
    81  	obsDeps := func(t *testing.T) []fx.Option {
    82  		t.Helper()
    83  
    84  		ctrl := gomock.NewController(t)
    85  		mockReg := mock_lifecycle.NewMockRegistrar(ctrl)
    86  		mockLog := mock_logging.NewMockLogger(ctrl)
    87  		mockLF := mock_logging.NewMockLogFieldBuilder(ctrl)
    88  
    89  		return []fx.Option{
    90  			fx.Provide(func() testing.TB { return t }),
    91  			fx.Provide(func() lifecycle.Registrar { return mockReg }),
    92  			fx.Provide(func() logging.Logger { return mockLog }),
    93  			fx.Provide(func() logging.LogFieldBuilder { return mockLF }),
    94  			fx.Provide(func() *config.ApplicationConfig {
    95  				return config.NewApplicationConfig(config.MockConfigForTest(t))
    96  			}),
    97  			fx.Provide(func() *config.ObservabilityConfig {
    98  				return config.NewObservabilityConfig(config.MockConfigForTest(t))
    99  			}),
   100  			fx.Provide(system.NewBuildInfo),
   101  			fx.NopLogger,
   102  		}
   103  	}
   104  
   105  	t.Run("正常系", func(t *testing.T) {
   106  		t.Parallel()
   107  
   108  		t.Run("logging が要求する LogCore と TraceExtractor の seam を提供する", func(t *testing.T) {
   109  			t.Parallel()
   110  
   111  			var (
   112  				core    logging.LogCore
   113  				extract logging.TraceExtractor
   114  			)
   115  
   116  			opts := append(obsDeps(t), ObservabilityModule(), fx.Populate(&core, &extract))
   117  			require.NoError(t, fx.ValidateApp(opts...))
   118  		})
   119  
   120  		t.Run("DB / HTTP client 計装が要求する pgx tracer と transport を提供する", func(t *testing.T) {
   121  			t.Parallel()
   122  
   123  			var (
   124  				pgxTracer *otelpgx.Tracer
   125  				transport *observability.HTTPClientTransport
   126  			)
   127  
   128  			opts := append(obsDeps(t), ObservabilityModule(), fx.Populate(&pgxTracer, &transport))
   129  			require.NoError(t, fx.ValidateApp(opts...))
   130  		})
   131  	})
   132  
   133  	t.Run("異常系", func(t *testing.T) {
   134  		t.Parallel()
   135  
   136  		t.Run("未配線では TracerFactory が解決できずグラフ検証に失敗する", func(t *testing.T) {
   137  			t.Parallel()
   138  
   139  			var tf observability.TracerFactory
   140  
   141  			opts := append(obsDeps(t), fx.Populate(&tf))
   142  			require.Error(t, fx.ValidateApp(opts...))
   143  		})
   144  	})
   145  }
