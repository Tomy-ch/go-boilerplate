    39  			// buildinfo.Register はデフォルトレジストリへ登録するため、テスト間の汚染を避ける。
    40  			origReg := prometheus.DefaultRegisterer
    41  			prometheus.DefaultRegisterer = prometheus.NewRegistry()
    42  			t.Cleanup(func() { prometheus.DefaultRegisterer = origReg })
    43  
    44  			var tf observability.TracerFactory
    45  
    46  			app := fx.New(
    47  				ObservabilityModule(),
    48  				fx.Provide(func() testing.TB { return t }),
    49  				fx.Provide(func() lifecycle.Registrar { return mockReg }),
    50  				fx.Provide(func() logging.Logger { return mockLog }),
    51  				fx.Provide(func() logging.LogFieldBuilder { return mockLF }),
    52  				fx.Provide(func() *config.ApplicationConfig {
    53  					return config.NewApplicationConfig(config.MockConfigForTest(t))
    54  				}),
    55  				fx.Provide(func() *config.ObservabilityConfig {
    56  					oc := config.NewObservabilityConfig(config.MockConfigForTest(t))
    57  					oc.SetObservabilityTracesExporter(t, "")
    58  					oc.SetObservabilityMetricsExporter(t, "")
    59  					oc.SetObservabilityLogsExporter(t, "")
    60  					return oc
    61  				}),
    62  				fx.Provide(system.NewBuildInfo),
    63  				fx.Populate(&tf),
    64  				fx.NopLogger,
    65  			)
    66  
    67  			require.NoError(t, app.Start(context.Background()))
    68  			require.NotNil(t, tf)
    69  			require.NoError(t, app.Stop(context.Background()))
    70  		})
    71  	})
