    63  					return config.NewApplicationConfig(config.MockConfigForTest(t))
    64  				}))
    65  			}
    66  			profiles := collectGroup[httpclient.DownstreamProfile](t, `group:"httpclient_profiles"`, deps())
    67  			required := collectGroup[httpclient.Downstream](t, `group:"required_downstreams"`, deps())
    68  
    69  			assert.Empty(t, httpclient.MissingDownstreams(profiles, required))
    70  		})
