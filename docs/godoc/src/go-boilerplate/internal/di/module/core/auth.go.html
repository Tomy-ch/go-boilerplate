    62  // provideAuthenticator は、環境に対応した Authenticator を選んで返します。
    63  // 対応する case が無い環境（staging / production 等）は誤った Authenticator を配線しないよう
    64  // default で起動エラーにします（fail-closed）。
    65  func provideAuthenticator(p authenticatorParams) (authbd.Authenticator, error) {
    66  	logger := p.Logger.Named("core.authn").CallerSkip(callerSkipCount)
    67  
    68  	switch p.AppCfg.Env() {
    69  	case config.EnvCI, config.EnvTest:
    70  		logger.Warn(
    71  			context.Background(),
    72  			"Local authenticator wired: authentication is stubbed (non-production only)",
    73  			logging.String("env", p.AppCfg.Env()),
    74  		)
    75  
    76  		return local.New(), nil
    77  	case config.EnvLocal, config.EnvDevelopment:
    78  		return provideJWKSAuthenticator(p, logger)
    79  	default:
    80  		logger.Error(
    81  			context.Background(),
    82  			"No authenticator configured for the current environment",
    83  			logging.String("env", p.AppCfg.Env()),
    84  		)
    85  
    86  		return nil, xerrors.New("no authenticator configured for environment: " + p.AppCfg.Env())
    87  	}
    88  }

    90  // provideJWKSAuthenticator は、AUTH_* 設定から JWKS backed の JWT authenticator を構築します。
    91  // JWKS の取得は httpclient substrate に委ねるため、goroutine / lifecycle の管理は不要です（遅延取得）。
    92  func provideJWKSAuthenticator(p authenticatorParams, logger logging.Logger) (authbd.Authenticator, error) {
    93  	authenticator, err := jwt.NewJWKS(jwt.JWKSParams{
    94  		Params: jwt.Params{
    95  			Issuer:       p.AuthCfg.Issuer(),
    96  			Audience:     p.AuthCfg.Audience(),
    97  			AllowedAlgs:  p.AuthCfg.AllowedAlgorithms(),
    98  			Leeway:       p.AuthCfg.ClockSkew(),
    99  			ExpectedType: accessTokenType,
   100  			Clock:        p.Clock,
   101  		},
   102  		JWKSURL:            p.AuthCfg.JWKSURL(),
   103  		CacheTTL:           p.AuthCfg.JWKSCacheTTL(),
   104  		DiscoveryTTL:       p.AuthCfg.DiscoveryTTL(),
   105  		UnknownKidCooldown: p.AuthCfg.UnknownKidCooldown(),
   106  		// local は疑似 provider（http）へ接続するため非 https を許容。それ以外の環境では https を強制する。
   107  		AllowInsecureURL: p.AppCfg.Env() == config.EnvLocal,
   108  	}, p.HTTPClient)
   109  	if err != nil {
   110  		logger.Error(
   111  			context.Background(),
   112  			"Failed to wire JWKS JWT authenticator",
   113  			logging.String("env", p.AppCfg.Env()),
   114  			logging.Error("error", err),
   115  		)
   116  
   117  		return nil, err
   118  	}
   119  
   120  	logger.Info(
   121  		context.Background(),
   122  		"JWKS JWT authenticator wired",
   123  		logging.String("env", p.AppCfg.Env()),
   124  		logging.String("issuer", p.AuthCfg.Issuer()),
   125  	)
   126  
   127  	return authenticator, nil
   128  }
