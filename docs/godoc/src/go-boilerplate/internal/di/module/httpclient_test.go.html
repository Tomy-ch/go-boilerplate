    25  //nolint:paralleltest // 上記の理由により fx アプリ起動ケースは非並列
    26  func Test_httpClientModule_ProvidesClient(t *testing.T) {
    27  	t.Run("正常系", func(t *testing.T) {
    28  		t.Run("実モジュール配線(clock + httpclient)から Client が構築される", func(t *testing.T) {
    29  			var client httpclient.Client
    30  			app := newHTTPClientTestApp(t, fx.Populate(&client))
    31  
    32  			require.NoError(t, app.Start(context.Background()))
    33  			t.Cleanup(func() { require.NoError(t, app.Stop(context.Background())) })
    34  			assert.NotNil(t, client)
    35  		})
    36  
    37  		t.Run("required と対応 profile が揃っていれば起動する", func(t *testing.T) {
    38  			var client httpclient.Client
    39  			app := newHTTPClientTestApp(t,
    40  				provideHTTPClientProfiles(func() httpclient.DownstreamProfile {
    41  					return httpclient.DownstreamProfile{Name: "svc", Profile: httpclient.DefaultProfile()}
    42  				}),
    43  				provideRequiredDownstreams(func() httpclient.Downstream { return "svc" }),
    44  				fx.Populate(&client),
    45  			)
    46  
    47  			require.NoError(t, app.Start(context.Background()))
    48  			t.Cleanup(func() { require.NoError(t, app.Stop(context.Background())) })
    49  			assert.NotNil(t, client)
    50  		})
    51  	})
    52  
    53  	t.Run("異常系", func(t *testing.T) {
    54  		t.Run("required に対応する profile が未登録の場合、起動に失敗する", func(t *testing.T) {
    55  			// profile を登録せず required だけ宣言すると、silent fallback ではなく起動失敗になる。
    56  			var client httpclient.Client
    57  			app := newHTTPClientTestApp(t,
    58  				provideRequiredDownstreams(func() httpclient.Downstream { return "orphan" }),
    59  				fx.Populate(&client),
    60  			)
    61  
    62  			require.Error(t, app.Start(context.Background()))
    63  		})
    64  	})
