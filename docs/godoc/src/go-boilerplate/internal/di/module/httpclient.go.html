    21  // provideHTTPClientRegistry は、グループに集まった Profile から Registry を生成します。
    22  // 各 gateway が宣言した required Downstream に対応する Profile が欠けている場合は起動時に失敗させ、
    23  // 登録漏れが silent な DefaultProfile fallback へ流れるのを防ぎます。
    24  func provideHTTPClientRegistry(in HTTPClientProfilesIn) (httpclient.Registry, error) {
    25  	if missing := httpclient.MissingDownstreams(in.Profiles, in.Required); len(missing) > 0 {
    26  		return nil, xerrors.New(fmt.Sprintf("httpclient profile missing for required downstreams: %v", missing))
    27  	}
    28  	return httpclient.NewRegistryFromProfiles(in.Profiles)
    29  }

    45  	opts := make([]fx.Option, len(constructors))

    53  	}
    54  	return fx.Options(opts...)
    55  }
