    24  // OutboxRelayModule は、outbox relay engine とそのライフサイクルフックを提供するfx.Moduleです。
    25  // relay 専用プロセス（cmd outbox-relay）でのみ使用します。
    26  // outboxPublisherModule は非標準の httpclient profile（MaxAttempts=1 等）を value group へ寄与するため、
    27  // relay 以外のプロセスへ漏れないよう共有 InfrastructureModule ではなくここに閉じ込めます。
    28  func OutboxRelayModule() fx.Option {
    29  	return fx.Module("outbox-relay",
    30  		outboxPublisherModule(),
    31  		fx.Provide(
    32  			// 具象 OutboxMetrics を usecase 境界の Metrics interface として供給する。
    33  			fx.Annotate(
    34  				observability.NewOutboxMetrics,
    35  				fx.As(new(outboxuc.Metrics)),
    36  			),
    37  			outboxuc.NewRelay,
    38  			provideRelaySettings,
    39  			outboxengine.NewEngine,
    40  		),
    41  		fx.Invoke(relayhook.RegisterRelayHooks),
    42  	)
    43  }

    45  // provideRelaySettings は、OutboxConfig から relay engine の設定を生成します。
    46  // BatchSize / PollInterval / ErrorBackoff は 0 以下だとホットループ（スピン / Sleep 即 return）を
    47  // 招くため、それぞれ既定値へ clamp します。
    48  // BatchSize が engine の int32 に収まらないときは、既定値へ丸めて誤設定を隠すのではなく起動を失敗させます。
    49  func provideRelaySettings(cfg *config.OutboxConfig) (outboxengine.Settings, error) {
    50  	batchSize := outboxuc.DefaultBatchSize
    51  	if cfg.BatchSize() > 0 {
    52  		v, err := safecast.IntToInt32(cfg.BatchSize())
    53  		if err != nil {
    54  			return outboxengine.Settings{}, xerrors.Wrap(err, "outbox relay batch size is out of range")
    55  		}
    56  		batchSize = v
    57  	}
    58  
    59  	pollInterval := defaultRelayPollInterval
    60  	if cfg.PollInterval() > 0 {
    61  		pollInterval = cfg.PollInterval()
    62  	}
    63  
    64  	errorBackoff := defaultRelayErrorBackoff
    65  	if cfg.ErrorBackoff() > 0 {
    66  		errorBackoff = cfg.ErrorBackoff()
    67  	}
    68  
    69  	return outboxengine.Settings{
    70  		BatchSize:    batchSize,
    71  		PollInterval: pollInterval,
    72  		ErrorBackoff: errorBackoff,
    73  	}, nil
    74  }
