    15  // RegisterWorkerHooks は、worker engine と health listener のライフサイクルフックを登録します。
    16  //   - OnStart: health listener を起動し、選択された worker を detached goroutine で実行する。
    17  //   - OnStop:  engine の context をキャンセルして drain 完了を（stopCtx の範囲で）待ち、health listener を停止する。
    18  func RegisterWorkerHooks(
    19  	reg lifecycle.Registrar,
    20  	engine *workerengine.Engine,
    21  	state workerboundary.State,
    22  	wc *config.WorkerConfig,
    23  	logger logging.Logger,
    24  ) {
    25  	startHealth, stopHealth := cliworker.NewHealthServer(wc.HealthListenAddr(), engine.Healthy, logger)
    26  
    27  	lifecycle.SupervisedRunner{
    28  		OnStartAux: startHealth,
    29  		Body: func(ctx context.Context) {
    30  			name, _, done := state.Snapshot()
    31  			if done == nil {
    32  				logger.Named("worker.Hooks").Info("No worker to run", logging.String(logging.WorkerNameKey, name))
    33  				return
    34  			}
    35  			defer close(done)
    36  			done <- engine.Run(ctx, name) // 猶予超過時の未完は Ack されず再配送へ
    37  		},
    38  		OnStopAux: stopHealth,
    39  	}.Register(reg)
