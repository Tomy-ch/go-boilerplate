    58  func TestRegisterWorkerHooks(t *testing.T) {
    59  	t.Parallel()
    60  
    61  	t.Run("正常系", func(t *testing.T) {
    62  		t.Parallel()
    63  
    64  		t.Run("起動対象がある場合_engineを実行しdoneへ結果を送って閉じる", func(t *testing.T) {
    65  			t.Parallel()
    66  
    67  			log := logging.NewTestLogger(t)
    68  			engine := newTestEngine(t, log)
    69  			state := workerengine.NewState()
    70  			done := make(chan error, 1)

    72  
    73  			wc := config.NewWorkerConfig(config.MockConfigForTest(t))
    74  			// 並列サブテスト間で health listener の既定ポート(:8081)が衝突するため空きポートを使う。
    75  			wc.SetHealthListenAddr(t, "127.0.0.1:0")
    76  			start, stop := captureWorkerHooks(t, engine, state, wc, log)
    77  
    78  			require.NoError(t, start(context.Background()))
    79  
    80  			// Body goroutine が engine.Run の結果を done へ送って close する。
    81  			err := <-done
    82  			require.ErrorIs(t, err, workerengine.ErrUnknownWorker)
    83  			_, ok := <-done
    84  			assert.False(t, ok, "done は close される")
    85  
    86  			require.NoError(t, stop(context.Background()))
    87  		})
    88  
    89  		t.Run("起動対象がない場合_engineを起動せずNoWorkerToRunをログ出力する", func(t *testing.T) {
