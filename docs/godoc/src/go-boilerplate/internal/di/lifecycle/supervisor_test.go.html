    20  func TestSupervisedRunner_Register(t *testing.T) {
    21  	t.Parallel()
    22  
    23  	t.Run("正常系", func(t *testing.T) {
    24  		t.Parallel()
    25  
    26  		t.Run("OnStartでBodyを起動しOnStopで実行contextをキャンセルして完了を待つ", func(t *testing.T) {
    27  			t.Parallel()
    28  
    29  			var auxStarted, auxStopped bool
    30  			bodyCtxDone := make(chan struct{})
    31  			var bodyCtxErr error
    32  
    33  			reg := &fakeRegistrar{}
    34  			SupervisedRunner{
    35  				OnStartAux: func() { auxStarted = true },
    36  				Body: func(ctx context.Context) {
    37  					<-ctx.Done() // OnStop の cancel で解放される
    38  					bodyCtxErr = ctx.Err()
    39  					close(bodyCtxDone)
    40  				},
    41  				OnStopAux: func(context.Context) { auxStopped = true },
    42  			}.Register(reg)
    43  
    44  			require.NotNil(t, reg.start)
    45  			require.NotNil(t, reg.stop)
    46  
    47  			// OnStart: aux 起動 → Body goroutine 起動。
    48  			require.NoError(t, reg.start(context.Background()))
    49  			assert.True(t, auxStarted)
    50  
    51  			// OnStop: cancel → Body 完了待ち → aux 停止。
    52  			require.NoError(t, reg.stop(context.Background()))
    53  			<-bodyCtxDone
    54  			require.ErrorIs(t, bodyCtxErr, context.Canceled)
    55  			assert.True(t, auxStopped)
    56  		})
    57  
    58  		t.Run("起動contextのキャンセルはBodyのcontextへ伝播しない", func(t *testing.T) {
    59  			t.Parallel()
    60  
    61  			bodyRan := make(chan struct{})
    62  			var bodyCtxErr error
    63  
    64  			reg := &fakeRegistrar{}
    65  			SupervisedRunner{
    66  				Body: func(ctx context.Context) {
    67  					// Body 起動時点で起動 ctx のキャンセルに巻き込まれていないこと（Background 由来）。
    68  					bodyCtxErr = ctx.Err()
    69  					close(bodyRan)
    70  				},
    71  			}.Register(reg)
    72  
    73  			// 既にキャンセル済みの起動 ctx を渡しても Body の ctx は無傷であること。
    74  			startCtx, cancel := context.WithCancel(context.Background())
    75  			cancel()
    76  			require.NoError(t, reg.start(startCtx))
    77  
    78  			<-bodyRan
    79  			require.NoError(t, bodyCtxErr)
    80  
    81  			// 後始末（goroutine リーク防止のため OnStop を呼ぶ）。
    82  			require.NoError(t, reg.stop(context.Background()))
    83  		})
    84  
    85  		t.Run("Bodyが既に完了していればOnStopは即座に返る", func(t *testing.T) {
    86  			t.Parallel()
    87  
    88  			bodyDone := make(chan struct{})
    89  			reg := &fakeRegistrar{}
    90  			SupervisedRunner{
    91  				Body: func(context.Context) { close(bodyDone) },
    92  			}.Register(reg)
    93  
    94  			require.NoError(t, reg.start(context.Background()))
    95  			<-bodyDone // Body は即時完了
    96  
    97  			require.NoError(t, reg.stop(context.Background()))
    98  		})
    99  
   100  		t.Run("Body完了前でも猶予切れ_stopCtx満了_でOnStopは返る", func(t *testing.T) {
   101  			t.Parallel()
   102  
   103  			block := make(chan struct{})

   106  				// ctx を無視してブロックし続ける Body（drain が完了しないケース）。

   108  			}.Register(reg)
   109  
   110  			require.NoError(t, reg.start(context.Background()))
   111  
   112  			// 既に満了した stopCtx を渡すと drain を待たず即座に返る。
   113  			stopCtx, cancel := context.WithCancel(context.Background())
   114  			cancel()
   115  			require.NoError(t, reg.stop(stopCtx))
   116  
   117  			close(block) // goroutine の後始末
   118  		})
