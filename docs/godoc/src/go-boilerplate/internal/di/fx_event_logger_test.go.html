    27  func TestFxEventLogger_LogEvent(t *testing.T) {
    28  	t.Parallel()
    29  
    30  	boom := errors.New("boom")
    31  
    32  	cases := []struct {
    33  		name      string
    34  		event     fxevent.Event
    35  		wantMsg   string // "" のときはログが出ないことを期待
    36  		wantLevel string
    37  	}{
    38  		{"OnStart失敗はErrorで記録される", &fxevent.OnStartExecuted{FunctionName: "f", CallerName: "c", Err: boom}, "fx OnStart hook failed", "error"},
    39  		{"OnStart成功は無視される", &fxevent.OnStartExecuted{FunctionName: "f"}, "", ""},
    40  		{"OnStop失敗はErrorで記録される", &fxevent.OnStopExecuted{FunctionName: "f", Err: boom}, "fx OnStop hook failed", "error"},
    41  		{"Supply失敗はErrorで記録される", &fxevent.Supplied{TypeName: "T", Err: boom}, "fx supply failed", "error"},
    42  		{"Provide失敗はErrorで記録される", &fxevent.Provided{ConstructorName: "ctor", Err: boom}, "fx provide failed", "error"},
    43  		{"Invoke失敗はErrorで記録される", &fxevent.Invoked{FunctionName: "f", Err: boom}, "fx invoke failed", "error"},
    44  		{"Replace失敗はErrorで記録される", &fxevent.Replaced{Err: boom}, "fx replace failed", "error"},
    45  		{"Decorate失敗はErrorで記録される", &fxevent.Decorated{DecoratorName: "dec", Err: boom}, "fx decorate failed", "error"},
    46  		{"LoggerInitialized失敗はErrorで記録される", &fxevent.LoggerInitialized{Err: boom}, "fx logger initialization failed", "error"},
    47  		{"Started失敗はErrorで記録される", &fxevent.Started{Err: boom}, "fx application failed to start", "error"},
    48  		{"Started成功はInfoで記録される", &fxevent.Started{}, "fx application started", "info"},
    49  		{"Stopped失敗はErrorで記録される", &fxevent.Stopped{Err: boom}, "fx application failed to stop", "error"},
    50  		{"Stopped成功はInfoで記録される", &fxevent.Stopped{}, "fx application stopped", "info"},
    51  		{"RollingBackは起動失敗をErrorで記録する", &fxevent.RollingBack{StartErr: boom}, "fx start failed, rolling back", "error"},
    52  		{"RolledBack失敗はErrorで記録される", &fxevent.RolledBack{Err: boom}, "fx rollback failed", "error"},
    53  		{"RolledBack成功(Err nil)は無視される", &fxevent.RolledBack{}, "", ""},
    54  		{"対象外イベントは無視される", &fxevent.Invoking{FunctionName: "f"}, "", ""},
    55  	}
    56  
    57  	for _, tc := range cases {
    58  		t.Run(tc.name, func(t *testing.T) {
    59  			t.Parallel()
    60  
    61  			obs, logs := logging.NewObservedTestLogger(t)
    62  			NewFxEventLogger(obs).LogEvent(tc.event)
    63  
    64  			if tc.wantMsg == "" {
    65  				assert.Equal(t, 0, logs.Len())
    66  				return
    67  			}
    68  
    69  			entries := logs.FilterMessage(tc.wantMsg).All()
    70  			require.Len(t, entries, 1)
    71  			assert.Equal(t, tc.wantLevel, entries[0].Level.String())
    72  		})
    73  	}
