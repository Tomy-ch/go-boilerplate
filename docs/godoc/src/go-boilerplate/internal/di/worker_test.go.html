    27  	// worker 未選択(state 未設定)でも起動・停止でき、health listener の起動/停止まで通ることを確認する。
    28  	config.EnsureRepoRootAndEnv(t, config.TestingEnvValue)
    29  	// health listener のポート衝突を避けるため OS 割り当ての空きポートを使う。

    31  
    32  	ctrl := gomock.NewController(t)
    33  	mockDB := mock_driver.NewMockDatabaseDriver(ctrl)
    34  	mockDB.EXPECT().Close().Return(nil).AnyTimes()
    35  	mockDB.EXPECT().Stats().Return(&pgxpool.Stat{}).AnyTimes()
    36  	mockDB.EXPECT().Ping(gomock.Any()).Return(nil).AnyTimes()
    37  
    38  	app := fx.New(
    39  		NewWorkerCore(),
    40  		fx.Replace(fx.Annotate(mockDB, fx.As(new(driver.DatabaseDriver)))),
    41  		fx.NopLogger,
    42  	)
    43  
    44  	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
    45  	defer cancel()
    46  	require.NoError(t, app.Start(ctx))
    47  
    48  	stopCtx, stopCancel := context.WithTimeout(context.Background(), 10*time.Second)
    49  	defer stopCancel()
    50  	require.NoError(t, app.Stop(stopCtx))
