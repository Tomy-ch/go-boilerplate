    24  func TestNewWorkerCore_BootsWithMockedDB(t *testing.T) {
    25  	// worker 未選択(state 未設定)でも起動・停止でき、health listener の起動/停止まで通ることを確認する。
    26  	config.EnsureRepoRootAndEnv(t, config.TestingEnvValue)
    27  	// health listener のポート衝突を避けるため OS 割り当ての空きポートを使う。
    28  	t.Setenv("WORKER_HEALTH_LISTEN_ADDR", "127.0.0.1:0")
    29  
    30  	ctrl := gomock.NewController(t)
    31  	mockDB := mock_driver.NewMockDatabaseDriver(ctrl)
    32  	mockDB.EXPECT().Close().Return(nil).AnyTimes()
    33  	mockDB.EXPECT().Stats().Return(&pgxpool.Stat{}).AnyTimes()
    34  	mockDB.EXPECT().Ping(gomock.Any()).Return(nil).AnyTimes()
    35  
    36  	app := fx.New(
    37  		NewWorkerCore(),
    38  		fx.Replace(fx.Annotate(mockDB, fx.As(new(driver.DatabaseDriver)))),
    39  		fx.NopLogger,
    40  	)
    41  
    42  	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
    43  	defer cancel()
    44  	require.NoError(t, app.Start(ctx))
    45  
    46  	stopCtx, stopCancel := context.WithTimeout(context.Background(), 10*time.Second)
    47  	defer stopCancel()
    48  	require.NoError(t, app.Stop(stopCtx))
