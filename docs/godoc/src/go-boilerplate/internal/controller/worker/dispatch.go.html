    10  // keyedDispatcher は、PartitionKey ごとにメッセージを直列化しつつ全体並列で処理します（B3）。
    11  //   - PartitionKey が空: そのまま並列に proc を起動。
    12  //   - PartitionKey が非空: key ごとの単一 goroutine が FIFO で proc を順次実行。
    13  //
    14  // in-flight トークンは呼び出し側（poll loop）が dispatch 前に取得済みで、proc が解放します。

    36  // dispatch は、メッセージを並列 or key 直列で処理へ回します。
    37  func (kd *keyedDispatcher) dispatch(ctx context.Context, m worker.Message) {
    38  	if m.PartitionKey == "" {
    39  		go kd.proc(ctx, m)
    40  		return
    41  	}
    42  
    43  	kd.mu.Lock()
    44  	r, ok := kd.runners[m.PartitionKey]
    45  	if !ok {
    46  		r = &keyRunner{ch: make(chan worker.Message, kd.buffer)}
    47  		kd.runners[m.PartitionKey] = r
    48  		go kd.runKey(ctx, m.PartitionKey, r)
    49  	}
    50  	r.pending++
    51  	kd.mu.Unlock()
    52  
    53  	r.ch <- m
    54  }

    56  // runKey は、特定 key のメッセージを FIFO で順次処理し、滞留が無くなれば自身を破棄します。
    57  func (kd *keyedDispatcher) runKey(ctx context.Context, key string, r *keyRunner) {
    58  	for {
    59  		m := <-r.ch
    60  		kd.proc(ctx, m)
    61  
    62  		kd.mu.Lock()
    63  		r.pending--
    64  		if r.pending == 0 {
    65  			delete(kd.runners, key)
    66  			kd.mu.Unlock()
    67  			return
    68  		}
    69  		kd.mu.Unlock()
    70  	}
    71  }
