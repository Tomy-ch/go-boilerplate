    15  // Engine は、選択された worker を pull-ack で実行する driving adapter です。
    16  // seam（worker.Consumer / Handler / FailureHandler）のみに依存し、broker 実装には依存しません。

    28  // New は、Engine を生成します。worker 名の重複は ErrDuplicateWorker を返します。
    29  func New(
    30  	workers []worker.Worker,
    31  	set Settings,
    32  	tf observability.TracerFactory,
    33  	met *observability.WorkerMetrics,
    34  	log logging.Logger,
    35  ) (*Engine, error) {
    36  	m := make(map[string]worker.Worker, len(workers))
    37  	for _, w := range workers {
    38  		name := w.Name()
    39  		if _, exists := m[name]; exists {
    40  			return nil, xerrors.Wrap(ErrDuplicateWorker, name)
    41  		}
    42  		m[name] = w
    43  	}
    44  	set.normalize()
    45  
    46  	return &Engine{
    47  		workers: m,
    48  		set:     set,
    49  		tracer:  tf.Controller(),
    50  		log:     log,
    51  		met:     met,
    52  	}, nil
    53  }

    65  // Run は、指定された worker を実行します。ctx 完了 or Fatal で drain して返ります。
    66  // 通常（Retryable/Permanent のみ）の場合は ctx 完了まで常駐します。
    67  func (e *Engine) Run(ctx context.Context, name string) error {
    68  	w, ok := e.workers[name]
    69  	if !ok {
    70  		return xerrors.Wrap(ErrUnknownWorker, name)
    71  	}
    72  
    73  	e.active.Store(true)
    74  	e.markProgress()
    75  	defer e.active.Store(false)
    76  
    77  	return newRun(e, w).loop(ctx)
    78  }

    80  // Healthy は、readiness 判定を返します（C2）。Run 実行中かつ poll 進捗が閾値内のとき true。
    81  func (e *Engine) Healthy() bool {
    82  	if !e.active.Load() {
    83  		return false
    84  	}
    85  	last := time.Unix(0, e.progress.Load())
    86  	return time.Since(last) < e.set.ProgressStaleAfter
    87  }
