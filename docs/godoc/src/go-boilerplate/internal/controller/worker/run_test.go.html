   106  func Test_run_nackBackoff(t *testing.T) {
   107  	t.Parallel()
   108  
   109  	newTestRun := func(t *testing.T) *run {
   110  		t.Helper()
   111  		f := testkit.NewFake()
   112  		w := testWorker{name: "w", cons: f, handler: handlerFunc(func(context.Context, bw.Message) error { return nil })}
   113  		set := baseSettings()
   114  		set.NackBackoffInitial = 10 * time.Millisecond
   115  		set.NackBackoffMax = time.Second
   116  		return newRun(newTestEngine(t, set, w), w)
   117  	}
   118  
   119  	// full jitter で散らされるため、値そのものではなく指数 backoff の算出値を上限として判定する。
   120  	t.Run("正常系", func(t *testing.T) {
   121  		t.Parallel()
   122  
   123  		t.Run("初回配送（ReceiveCount=1）は初期 backoff を上限とする", func(t *testing.T) {
   124  			t.Parallel()
   125  
   126  			got := newTestRun(t).nackBackoff(1)
   127  
   128  			assert.GreaterOrEqual(t, got, time.Duration(0))
   129  			assert.LessOrEqual(t, got, 10*time.Millisecond)
   130  		})
   131  
   132  		t.Run("再配送回数に応じて上限が指数的に伸びる", func(t *testing.T) {
   133  			t.Parallel()
   134  
   135  			r := newTestRun(t)
   136  
   137  			// ReceiveCount=3 は attempt=2 なので上限は 10ms * 2^2 = 40ms。full jitter は [0, 上限] の
   138  			// 一様乱数なので、上限が初期値 10ms のままなら 200 回引いて 10ms を超えない確率は事実上 0。
   139  			var maxSeen time.Duration
   140  			for range 200 {
   141  				got := r.nackBackoff(3)
   142  				require.LessOrEqual(t, got, 40*time.Millisecond)
   143  				maxSeen = max(maxSeen, got)
   144  			}
   145  
   146  			assert.Greater(t, maxSeen, 10*time.Millisecond)
   147  		})
   148  
   149  		t.Run("ReceiveCount が 0 以下でも attempt は 0 に丸められる", func(t *testing.T) {
   150  			t.Parallel()
   151  
   152  			// 0 起算へ変換すると負になる入力でも backoff.Duration へ負の attempt を渡さない。
   153  			assert.LessOrEqual(t, newTestRun(t).nackBackoff(0), 10*time.Millisecond)
   154  			assert.LessOrEqual(t, newTestRun(t).nackBackoff(-5), 10*time.Millisecond)
   155  		})
   156  	})
