    23  // 信号は handler の Retryable 連続失敗と poll エラー（onFailure）。成功と Permanent は onSuccess。

    55  // toHalfOpen は、Open から Half-open へ遷移します（cooldown 経過後に poll loop が呼ぶ）。
    56  // 新しい Half-open エピソードなので probing は未投入（false）へ戻します。
    57  func (c *circuit) toHalfOpen() {
    58  	c.mu.Lock()
    59  	defer c.mu.Unlock()
    60  	if c.phase == phaseOpen {
    61  		c.phase = phaseHalfOpen
    62  		c.probing = false
    63  	}
    64  }

    66  // abortProbe は、Half-open の probe バッチが空振り（0 件受信）だったとき probing を解除します。
    67  // probe が in-flight にならないと結果（onSuccess/onFailure）が出ず slotFreed も鳴らないため、
    68  // これを解除しないと poll loop が次周の acquire で永久停止する。解除後は次周で再 probe できる。
    69  func (c *circuit) abortProbe() {
    70  	c.mu.Lock()
    71  	defer c.mu.Unlock()
    72  	if c.phase == phaseHalfOpen {
    73  		c.probing = false
    74  	}
    75  }
    76  
    77  // tryBeginProbe は、Half-open で probe バッチをまだ投入していなければ probing を確定して true を返します。
    78  // 既に probing 中（結果待ち）または Half-open でない場合は false を返し、呼び出し側は新規 Receive を止めます。
    79  // これにより 1 Half-open エピソードで投入する probe は 1 バッチ（総試行数 ≤ CircuitHalfOpenProbe）に限定されます。
    80  func (c *circuit) tryBeginProbe() bool {
    81  	c.mu.Lock()
    82  	defer c.mu.Unlock()
    83  	if c.phase == phaseHalfOpen && !c.probing {
    84  		c.probing = true
    85  		return true
    86  	}
    87  	return false
    88  }

    90  // onSuccess は、成功（または Permanent 退避完了）を記録します。Half-open なら Closed へ復帰します。
    91  func (c *circuit) onSuccess() {
    92  	c.mu.Lock()
    93  	defer c.mu.Unlock()
    94  	c.failures = 0
    95  	if c.phase == phaseHalfOpen {
    96  		c.phase = phaseClosed
    97  		c.probing = false
    98  		c.openCount = 0
    99  	}
   100  }

   102  // onFailure は、Retryable 失敗または poll エラーを記録します。閾値超過 or Half-open 失敗で Open へ遷移します。
   103  func (c *circuit) onFailure() {
   104  	c.mu.Lock()
   105  	defer c.mu.Unlock()
   106  	if c.threshold <= 0 {
   107  		return
   108  	}
   109  	c.failures++
   110  	switch c.phase {
   111  	case phaseHalfOpen:
   112  		c.trip()
   113  	case phaseClosed:
   114  		if c.failures >= c.threshold {
   115  			c.trip()
   116  		}
   117  	case phaseOpen:
   118  		// Open 中は Receive しないため通常は到達しない。
   119  	}
   120  }

   122  // trip は、Open へ遷移し cooldown を確定します（mu ロック下で呼ぶこと）。
   123  func (c *circuit) trip() {
   124  	c.phase = phaseOpen
   125  	c.probing = false
   126  	c.curCooldown = c.backoff.Duration(c.openCount)
   127  	c.openCount++
   128  }
