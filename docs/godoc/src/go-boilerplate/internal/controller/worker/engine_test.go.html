    52  // startEngine は、worker を別 goroutine で起動し、cancel と完了 channel を返します。
    53  func startEngine(t *testing.T, set Settings, log logging.Logger, w bw.Worker) (context.CancelFunc, <-chan error) {
    54  	t.Helper()
    55  	eng, err := New([]bw.Worker{w}, set, observability.NewNoopTracerFactory(t), observability.NewNoopWorkerMetrics(t), log)
    56  	require.NoError(t, err)
    57  
    58  	ctx, cancel := context.WithCancel(context.Background())
    59  	done := make(chan error, 1)
    60  	go func() { done <- eng.Run(ctx, w.Name()) }()
    61  	return cancel, done
    62  }
