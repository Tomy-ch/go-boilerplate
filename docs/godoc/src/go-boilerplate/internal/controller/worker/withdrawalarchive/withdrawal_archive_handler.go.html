    37  // Handle は、user.withdrawn.v1 の payload を退会証跡として保存します。
    38  //
    39  // キューには outbox が publish する全種別が流れるため、種別が一致しないメッセージは保存せず成功を返します
    40  // （engine が Ack して取り除きます）。再配送しても種別は変わらず、DLQ へ送っても運用者の対処が無いためです。
    41  // 種別属性を持たないメッセージも同じ扱いにします。属性を載せる前に publish された残留メッセージが
    42  // DLQ を埋めないようにするためです。
    43  //
    44  // 復元できない payload はリトライで直らないため Permanent として分類し、engine の退避経路へ渡します。
    45  // 保存の失敗は分類せずそのまま返し、engine 既定の Retryable として再配送させます。
    46  func (h *handler) Handle(ctx context.Context, m workerbd.Message) error {
    47  	ctx, endSpan := h.tracer.Start(ctx)
    48  	defer endSpan()
    49  
    50  	if m.Attributes[workerbd.AttrEventType] != event.TypeWithdrawn {
    51  		h.logging.Named(Name).Debug(ctx, skippedMessage,
    52  			logging.String(logging.MessageIDKey, m.ID),
    53  			logging.String(logging.EventTypeKey, m.Attributes[workerbd.AttrEventType]),
    54  		)
    55  		return nil
    56  	}
    57  
    58  	withdrawn, err := event.ParseWithdrawn(m.Body)
    59  	if err != nil {
    60  		return xerrors.Wrap(apperror.ErrPermanent, err.Error())
    61  	}
    62  
    63  	if _, err := h.archive.ArchiveWithdrawal(ctx, user.ArchiveWithdrawalParams{
    64  		UserID:  withdrawn.UserID,
    65  		Payload: m.Body,
    66  	}); err != nil {
    67  		// 入力が不正なメッセージは、何度配送されても同じ結果になるため退避側へ回す。
    68  		if xerrors.Is(err, apperror.ErrValidation) {
    69  			return xerrors.Wrap(apperror.ErrPermanent, err.Error())
    70  		}
    71  		return err
    72  	}
    73  
    74  	return nil
    75  }
