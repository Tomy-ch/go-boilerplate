    37  // 種別が一致しないメッセージと種別属性を持たないメッセージは保存せず成功を返します。
    38  // 復元できない payload は Permanent、保存の失敗は分類せずそのまま返します。
    39  // これらの選別・分類の根拠は README（Message selection / Error classification）を参照。
    40  func (h *handler) Handle(ctx context.Context, m workerbd.Message) error {
    41  	ctx, endSpan := h.tracer.Start(ctx)
    42  	defer endSpan()
    43  
    44  	if m.Attributes[workerbd.AttrEventType] != event.TypeWithdrawn {
    45  		h.logging.Named(Name).Debug(ctx, skippedMessage,
    46  			logging.String(logging.MessageIDKey, m.ID),
    47  			logging.String(logging.EventTypeKey, m.Attributes[workerbd.AttrEventType]),
    48  		)
    49  		return nil
    50  	}
    51  
    52  	withdrawn, err := event.ParseWithdrawn(m.Body)
    53  	if err != nil {
    54  		return xerrors.Wrap(apperror.ErrPermanent, err.Error())
    55  	}
    56  
    57  	if _, err := h.archive.ArchiveWithdrawal(ctx, user.ArchiveWithdrawalParams{
    58  		UserID:  withdrawn.UserID,
    59  		Payload: m.Body,
    60  	}); err != nil {
    61  		// 入力が不正なメッセージは、何度配送されても同じ結果になるため退避側へ回す。
    62  		if xerrors.Is(err, apperror.ErrValidation) {
    63  			return xerrors.Wrap(apperror.ErrPermanent, err.Error())
    64  		}
    65  		return err
    66  	}
    67  
    68  	return nil
