    40  func newRun(e *Engine, w worker.Worker) *run {
    41  	set := e.set
    42  	r := &run{
    43  		e:         e,
    44  		name:      w.Name(),
    45  		consumer:  w.Consumer(),
    46  		handler:   w.Handler(),
    47  		failure:   w.FailureHandler(),
    48  		cb:        newCircuit(set.CircuitFailureThreshold, set.circuitBackoff()),
    49  		inflight:  make(chan struct{}, set.MaxInFlight),

    57  // loop は、poll loop 本体です。ctx 完了 or Fatal で抜け、drain して返ります。
    58  func (r *run) loop(parent context.Context) error {
    59  	ctx, cancel := context.WithCancel(parent)
    60  	r.cancel = cancel
    61  	defer cancel()
    62  	defer r.drain()
    63  
    64  	for {
    65  		r.e.markProgress() // stuck 検出用に進捗時刻を更新
    66  		n, ok := r.acquire(ctx)
    67  		if !ok {
    68  			return r.fatalErr()
    69  		}
    70  		msgs, err := r.consumer.Receive(ctx, n)
    71  		if err != nil {
    72  			if ctx.Err() != nil {
    73  				return r.fatalErr()
    74  			}
    75  			r.onPollError(ctx, err)
    76  			continue
    77  		}
    78  		if len(msgs) == 0 {
    79  			// Half-open の probe が空振りしたので probing を解除し、次周で再 probe させる。
    80  			r.cb.abortProbe()
    81  			continue
    82  		}
    83  		r.dispatchAll(ctx, msgs)
    84  	}
    85  }

    87  // acquire は、Receive を許可できる状態になるまで待ち、取得件数を返します。
    88  // サーキットが Open なら cooldown を待ち、in-flight に空きが出るまで待つ二段ゲートです。
    89  // スロット待ちの間に Open へ遷移しうるため、待機後に状態を再確認します。
    90  func (r *run) acquire(ctx context.Context) (int, bool) {
    91  	for {
    92  		if ctx.Err() != nil {
    93  			return 0, false
    94  		}
    95  		if r.cb.phaseNow() == phaseOpen {
    96  			if !r.waitCooldown(ctx) {
    97  				return 0, false
    98  			}
    99  			continue
   100  		}
   101  		free, ok := r.waitForSlot(ctx)
   102  		if !ok {
   103  			return 0, false
   104  		}
   105  		// phase は 1 度だけ snapshot して判定する（複数回 phaseNow() すると
   106  		// その間の trip() 割り込みで Open なのに BatchSize 件 Receive しうる TOCTOU を防ぐ）。
   107  		switch r.cb.phaseNow() {
   108  		case phaseOpen:
   109  			continue // スロット待ちの間に Open へ遷移したので Receive せず再評価
   110  		case phaseHalfOpen:
   111  			if r.cb.tryBeginProbe() {
   112  				return min(r.e.set.CircuitHalfOpenProbe, free), true
   113  			}
   114  			// 既に probe 投入済み（probing 中）。結果が確定して Closed/Open へ遷移するまで新規 Receive を
   115  			// 止め、in-flight 解放（probe 結果の到来で slotFreed が鳴る）で起床して再評価する。
   116  			select {
   117  			case <-ctx.Done():
   118  				return 0, false
   119  			case <-r.slotFreed:
   120  				r.e.markProgress() // probing 待機からの起床は進捗とみなし、probe 待ちの間の readiness stale を避ける
   121  			}
   122  			continue
   123  		default:
   124  			return min(r.e.set.BatchSize, free), true
   125  		}
   126  	}
   127  }

   129  // waitForSlot は、in-flight に空きが出るまで待ち、空き数を返します（B2 のゲート）。
   130  func (r *run) waitForSlot(ctx context.Context) (int, bool) {
   131  	for {
   132  		free := cap(r.inflight) - len(r.inflight)
   133  		if free > 0 {
   134  			return free, true
   135  		}
   136  		select {
   137  		case <-ctx.Done():
   138  			return 0, false
   139  		case <-r.slotFreed:
   140  		}
   141  	}
   142  }

   144  // waitCooldown は、Open の cooldown を待って Half-open へ遷移します。
   145  func (r *run) waitCooldown(ctx context.Context) bool {
   146  	select {
   147  	case <-ctx.Done():
   148  		return false
   149  	case <-time.After(r.cb.cooldown()):
   150  		r.cb.toHalfOpen()
   151  		return true
   152  	}
   153  }

   155  // dispatchAll は、受信メッセージを in-flight 計上のうえ dispatch へ回します。
   156  func (r *run) dispatchAll(ctx context.Context, msgs []worker.Message) {
   157  	if len(msgs) == 0 {
   158  		return
   159  	}
   160  	r.e.met.Received(ctx, int64(len(msgs)))
   161  	for _, m := range msgs {
   162  		r.inflight <- struct{}{}
   163  		r.e.met.InFlightAdd(ctx, 1)
   164  		r.wg.Add(1)
   165  		r.keyed.dispatch(ctx, m)
   166  	}
   167  }

   169  // process は、1 メッセージの処理単位です（B1 の同時数制御・span・分類処理）。
   170  func (r *run) process(ctx context.Context, m worker.Message) {
   171  	defer r.finishMessage(ctx)
   172  
   173  	ctx = r.withTrace(ctx, m) // D1: traceparent から trace 継続
   174  	ctx, endSpan := r.e.tracer.Start(ctx)
   175  	defer endSpan()
   176  
   177  	select {
   178  	case r.conc <- struct{}{}:
   179  	case <-ctx.Done():
   180  		return // 停止中は未処理のまま離脱（Ack/Nack せず再配送へ）。in-flight は finishMessage が解放
   181  	}
   182  	defer func() { <-r.conc }()
   183  
   184  	r.warnIfPoison(ctx, m)
   185  	start := time.Now()
   186  	err := r.safeHandle(ctx, m)
   187  	r.e.met.RecordLatencyMs(ctx, msSince(start))
   188  	r.handleResult(ctx, m, err)
   189  }

   191  // finishMessage は、in-flight トークンを解放し poll loop を起床させます。
   192  func (r *run) finishMessage(ctx context.Context) {
   193  	<-r.inflight
   194  	r.e.met.InFlightAdd(ctx, -1)
   195  	select {
   196  	case r.slotFreed <- struct{}{}:
   197  	default:
   198  	}
   199  	r.wg.Done()
   200  }

   207  // safeHandle は、per-message recover（A6）と Extend ハートビート（A3）付きで Handle を実行します。
   208  func (r *run) safeHandle(ctx context.Context, m worker.Message) (err error) {
   209  	defer func() {
   210  		if rec := recover(); rec != nil {
   211  			// panic 値はログにのみ残し、伝播するエラーには含めない（秘密情報の漏洩防止）。
   212  			r.e.log.Named("worker.recover").Error(
   213  				ctx,
   214  				"panic recovered in handler",
   215  				append(msgFields(r.name, m), logging.String(logging.PanicKey, fmt.Sprintf("%v", rec)))...,
   216  			)
   217  			err = xerrors.Wrap(apperror.ErrRetryable, "panic recovered in handler")
   218  		}
   219  	}()
   220  
   221  	stop := r.startHeartbeat(ctx, m)
   222  	defer stop()
   223  
   224  	return r.handler.Handle(ctx, m)
   225  }

   227  // startHeartbeat は、ExtendInterval ごとに Extend を呼ぶハートビートを開始し、停止関数を返します。
   228  func (r *run) startHeartbeat(ctx context.Context, m worker.Message) func() {
   229  	interval := r.e.set.ExtendInterval
   230  	if interval <= 0 {
   231  		return func() {}
   232  	}
   233  
   234  	ticker := time.NewTicker(interval)
   235  	done := make(chan struct{})
   236  	// stopped は goroutine の終了を通知する。停止クロージャがこれを待つことで、worker 停止後に
   237  	// heartbeat goroutine が残らないこと（graceful-stop 契約）を保証する。
   238  	stopped := make(chan struct{})
   239  	go func() {
   240  		defer close(stopped)
   241  		for {
   242  			select {
   243  			case <-done:
   244  				return
   245  			case <-ctx.Done():
   246  				return
   247  			case <-ticker.C:
   248  				if err := r.consumer.Extend(ctx, m, interval*extendVisibilityFactor); err != nil {
   249  					if ctx.Err() != nil {
   250  						return // 停止中の Extend 失敗は握り潰してよい（未 Ack なので再配送される）
   251  					}
   252  					// 握り潰さず可視化する。lease 延長失敗は早期再配送→重複処理の予兆。
   253  					r.e.met.ExtendError(ctx)
   254  					r.e.log.Named("worker.extend").Warn(
   255  						ctx,
   256  						"extend failed",
   257  						append(msgFields(r.name, m), logging.Error(logging.ErrorKey, err))...,
   258  					)
   259  				}
   260  			}
   261  		}
   262  	}()
   263  
   264  	return func() {
   265  		ticker.Stop()
   266  		close(done)
   267  		<-stopped // goroutine の完全終了を待つ（停止後に heartbeat が残らないことを保証）
   268  	}
   269  }

   271  // handleResult は、Handle の結果を分類して Ack / Nack / FailureHandler / engine 停止に振り分けます。
   272  // 併せて engine 所有 metric の更新（D2）と構造化ログ（D3）を行います。
   273  func (r *run) handleResult(ctx context.Context, m worker.Message, err error) {
   274  	if err == nil {
   275  		r.ack(ctx, m) // A1: 成功時のみ Ack
   276  		r.cb.onSuccess()
   277  		r.e.met.Processed(ctx)
   278  		r.e.log.Named("worker.process").Debug(ctx, "message processed", msgFields(r.name, m)...)
   279  		return
   280  	}
   281  
   282  	r.e.met.Failed(ctx)
   283  	fields := append(msgFields(r.name, m), logging.Error(logging.ErrorKey, err))
   284  
   285  	switch classify(err) {
   286  	case catRetryable:
   287  		r.nack(ctx, m) // A2
   288  		r.cb.onFailure()
   289  		r.e.met.Retried(ctx)
   290  		r.e.log.Named("worker.process").Warn(ctx, "retryable failure, nacked", fields...)
   291  	case catPermanent:
   292  		if ferr := r.routePermanent(ctx, m, err); ferr != nil { // A5
   293  			// dead-letter 退避に失敗した。実際には退避できていないため DLQ 計上・成功ログは出さず、
   294  			// Ack もしない（暗黙の再配送へ委ねる）。退避先障害は下流失敗として circuit へ計上する。
   295  			r.cb.onFailure()
   296  			r.e.log.Named("worker.process").Error(ctx, "permanent failure, dead-letter routing failed", fields...)
   297  			return
   298  		}
   299  		r.cb.onSuccess()
   300  		r.e.met.DLQ(ctx)
   301  		r.e.log.Named("worker.process").Warn(ctx, "permanent failure, routed to dead-letter", fields...)
   302  	case catFatal:
   303  		r.e.log.Named("worker.process").Error(ctx, "fatal failure, stopping engine", fields...)
   304  		r.triggerFatal(err) // A5（Fatal）
   305  	}
   306  }

   308  // routePermanent は、Permanent を FailureHandler へ退避してから Ack します。
   309  // 退避に失敗した場合は Ack せず、その error を返します（呼び出し元が circuit / ログを分岐する）。
   310  func (r *run) routePermanent(ctx context.Context, m worker.Message, cause error) error {
   311  	if r.failure != nil {
   312  		if err := r.failure.Fail(ctx, m, cause); err != nil {
   313  			r.logErr(ctx, "worker.failure", "failure handler error", err)
   314  			return err
   315  		}
   316  	}
   317  	r.ack(ctx, m)
   318  	return nil
   319  }

   327  // nack は、retryable 失敗時に per-message 再配送 backoff（指数 + full jitter）を計算し、
   328  // その遅延つきで再配送します。policy は engine が持ち、adapter が native 機構で honor します。
   329  func (r *run) nack(ctx context.Context, m worker.Message) {
   330  	d := r.nackBackoff(m.ReceiveCount)
   331  	if err := r.consumer.NackWithBackoff(ctx, m, d); err != nil {
   332  		r.logErr(ctx, "worker.nack", "nack error", err)
   333  	}
   334  }
   335  
   336  // nackBackoff は、ReceiveCount（1 起算）に対する再配送遅延を返します。
   337  // 指数 backoff を full jitter で散らし、同時失敗の thundering herd を避けます。
   338  func (r *run) nackBackoff(receiveCount int) time.Duration {
   339  	// ReceiveCount は 1 起算、backoff.Duration は 0 起算。
   340  	attempt := max(0, receiveCount-1)
   341  	return retry.Full(r.e.set.nackBackoff().Duration(attempt))
   342  }
