   114  // TestMiddleware_overWire は、実 HTTP 経路（commit 済みレスポンス）でも Content-Type が
   115  // 上書きされることを検証する。recorder の生ヘッダマップでは commit 後の挙動を検出できないため
   116  // httptest.NewServer を用いてワイヤ上の最終ヘッダを確認する。
   117  func TestMiddleware_overWire(t *testing.T) {
   118  	t.Parallel()
   119  
   120  	exec := func(t *testing.T, handler echo.HandlerFunc) string {
   121  		t.Helper()
   122  		e := echo.New()
   123  		e.Use(Middleware())
   124  		e.GET("/t", handler)
   125  
   126  		srv := httptest.NewServer(e)
   127  		t.Cleanup(srv.Close)
   128  
   129  		req, err := http.NewRequestWithContext(context.Background(), http.MethodGet, srv.URL+"/t", nil)
   130  		require.NoError(t, err)
   131  		resp, err := http.DefaultClient.Do(req)
   132  		require.NoError(t, err)
   133  		t.Cleanup(func() { _ = resp.Body.Close() })
   134  		_, err = io.ReadAll(resp.Body)
   135  		require.NoError(t, err)
   136  		return resp.Header.Get(echo.HeaderContentType)
   137  	}
   138  
   139  	t.Run("正常系", func(t *testing.T) {
   140  		t.Parallel()
   141  
   142  		t.Run("HTMLボディはapplication/jsonへ上書きされる", func(t *testing.T) {
   143  			t.Parallel()
   144  			got := exec(t, func(c echo.Context) error { return c.HTML(http.StatusOK, "<p>hi</p>") })
   145  			assert.Equal(t, echo.MIMEApplicationJSON, got)
   146  		})
   147  
   148  		t.Run("Content-Type未設定のボディもapplication/jsonへ上書きされる", func(t *testing.T) {
   149  			t.Parallel()
   150  			got := exec(t, func(c echo.Context) error { return c.Blob(http.StatusOK, "", []byte("x")) })
   151  			assert.Equal(t, echo.MIMEApplicationJSON, got)
   152  		})
   153  
   154  		t.Run("application/jsonは変更されない", func(t *testing.T) {
   155  			t.Parallel()
   156  			got := exec(t, func(c echo.Context) error { return c.JSON(http.StatusOK, map[string]string{"k": "v"}) })
   157  			assert.Equal(t, echo.MIMEApplicationJSON, got)
   158  		})
   159  
   160  		t.Run("text/plainは変更されない", func(t *testing.T) {
   161  			t.Parallel()
   162  			got := exec(t, func(c echo.Context) error { return c.String(http.StatusOK, "plain") })
   163  			assert.Equal(t, echo.MIMETextPlainCharsetUTF8, got)
   164  		})
   165  	})
