    44  		t.Parallel()

    46  		t.Run("本番モードでは信頼プロキシ経由のXFFヘッダからIPを抽出する", func(t *testing.T) {
    47  			t.Parallel()
    48  
    49  			cfg := config.MockConfigForTest(t)
    50  			appCfg := config.NewApplicationConfig(cfg)
    51  			appCfg.SetApplicationMode(t, config.ProductionMode)
    52  
    53  			secCfg := config.NewSecurityConfig(cfg)
    54  			secCfg.SetCIDR(t, parsedCIDR)
    55  
    56  			actual := NewIPExtractor(appCfg, secCfg)
    57  			require.NotNil(t, actual)
    58  
    59  			// 接続元(RemoteAddr)を信頼 CIDR(192.168.0.0/16)内に置くと、XFF のクライアント IP が採用される。
    60  			req := httptest.NewRequestWithContext(context.Background(), http.MethodGet, "/", nil)
    61  			req.RemoteAddr = "192.168.1.1:1234"
    62  			req.Header.Set(echo.HeaderXForwardedFor, "203.0.113.5")
    63  			assert.Equal(t, "203.0.113.5", actual(req))
    64  		})
    65  
    66  		t.Run("開発モードでは接続元IPを直接抽出しXFFを無視する", func(t *testing.T) {
    67  			t.Parallel()

    72  
    73  			secCfg := config.NewSecurityConfig(cfg)
    74  			secCfg.SetCIDR(t, parsedCIDR)
    75  
    76  			actual := NewIPExtractor(appCfg, secCfg)
    77  			require.NotNil(t, actual)
    78  
    79  			// direct 抽出は RemoteAddr をそのまま採用し、XFF は無視する。

    81  			req.RemoteAddr = "192.168.1.1:1234"

    84  		})
