    42  // strictHandlerFunc は、oapi-codegen 生成の gen.StrictHandlerFunc と同型のテスト用型です。
    43  type strictHandlerFunc func(ec echo.Context, request any) (any, error)

    45  // newEcho は、テスト用の echo.Context（POST testPath）を生成します。key 非空ならヘッダを付与し、
    46  // withAuthn なら subject を持つ Authn を ctx に仕込みます。subject が UUID として解釈できる場合は
    47  // 内部 UserID も解決済みにします（冪等性スコープは内部 UserID を使うため）。
    48  func newEcho(key string, withAuthn bool, subject string) echo.Context {
    49  	ctx := context.Background()
    50  	if withAuthn {
    51  		ctx = ctxhelper.WithAuthn(ctx)
    52  		a, _ := auth.New(subject, "test", nil, nil)
    53  		if id, err := uuid.Parse(subject); err == nil {
    54  			a = a.WithUserID(id)
    55  		}
    56  		ctxhelper.SetAuthn(ctx, *a)
    57  	}
    58  	req := httptest.NewRequestWithContext(ctx, http.MethodPost, testPath, nil)
    59  	if key != "" {
    60  		req.Header.Set(headerName, key)
    61  	}
    62  	return echo.New().NewContext(req, httptest.NewRecorder())
    63  }
