    42  // newEcho は、テスト用の echo.Context（POST testPath）を生成します。key 非空ならヘッダを付与し、
    43  // withAuthn なら subject を持つ Authn を ctx に仕込みます。
    44  func newEcho(key string, withAuthn bool, subject string) echo.Context {
    45  	ctx := context.Background()
    46  	if withAuthn {
    47  		ctx = ctxhelper.WithAuthn(ctx)
    48  		a, _ := auth.New(subject, "test", nil, nil)
    49  		ctxhelper.SetAuthn(ctx, *a)
    50  	}
    51  	req := httptest.NewRequestWithContext(ctx, http.MethodPost, testPath, nil)
    52  	if key != "" {
    53  		req.Header.Set(headerName, key)
    54  	}
    55  	return echo.New().NewContext(req, httptest.NewRecorder())
    56  }
