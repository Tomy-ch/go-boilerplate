    30  // PrometheusRecorder は、Recorder を Prometheus メトリクスとして公開する実装です。
    31  // prometheus.Collector も満たし、pool_stats_collector と同じく registry へ登録して使用します。
    32  type PrometheusRecorder struct {
    33  	requests *prometheus.CounterVec
    34  	duration *prometheus.HistogramVec

    37  // NewPrometheusRecorder は、PrometheusRecorder を初期化して返します。
    38  // 副作用は持たず、registry への登録は RegisterRecorder で別途行います。
    39  func NewPrometheusRecorder() *PrometheusRecorder {
    40  	return &PrometheusRecorder{
    41  		requests: prometheus.NewCounterVec(prometheus.CounterOpts{
    42  			Namespace: namespace,
    43  			Subsystem: subsystem,
    44  			Name:      "requests_total",
    45  			Help:      "Total number of HTTP server requests.",
    46  		}, labelNames),
    47  		duration: prometheus.NewHistogramVec(prometheus.HistogramOpts{
    48  			Namespace: namespace,
    49  			Subsystem: subsystem,
    50  			Name:      "request_duration_seconds",
    51  			Help:      "HTTP server request latencies in seconds.",
    52  			Buckets:   prometheus.DefBuckets,
    53  		}, labelNames),
    54  	}
    55  }

    57  // Observe は、request count を increment し、duration を histogram に記録します。
    58  func (r *PrometheusRecorder) Observe(method, route string, statusCode int, statusClass string, duration time.Duration) {
    59  	labels := prometheus.Labels{
    60  		"method":       method,
    61  		"route":        route,
    62  		"status_code":  statusCodeLabel(statusCode),
    63  		"status_class": statusClass,
    64  	}
    65  	r.requests.With(labels).Inc()
    66  	r.duration.With(labels).Observe(duration.Seconds())
    67  }

    87  // ignoreAlreadyRegistered は、AlreadyRegisteredError を無視し、それ以外のエラーはそのまま返します。
    88  func ignoreAlreadyRegistered(err error) error {
    89  	if err == nil {
    90  		return nil
    91  	}
    92  
    93  	var alreadyRegisteredErr prometheus.AlreadyRegisteredError
    94  	if xerrors.As(err, &alreadyRegisteredErr) {
    95  		return nil
    96  	}

    98  }
