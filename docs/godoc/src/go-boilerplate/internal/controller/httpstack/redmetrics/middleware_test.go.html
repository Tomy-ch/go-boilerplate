    15  // serveCfg は、ミドルウェア経由のリクエスト実行設定です。
    16  type serveCfg struct {
    17  	registerPath string
    18  	requestPath  string
    19  	handler      echo.HandlerFunc
    20  }

    22  // serve は、redmetrics ミドルウェアを適用した Echo に rec を計測先として 1 リクエスト流します。
    23  // 計測の検証は rec（生成 mock）に設定した EXPECT で行います。
    24  func serve(t *testing.T, rec Recorder, cfg serveCfg) {
    25  	t.Helper()
    26  
    27  	e := echo.New()
    28  	e.Use(Middleware(rec))
    29  	if cfg.registerPath != "" {
    30  		e.GET(cfg.registerPath, cfg.handler)
    31  	}
    32  
    33  	req := httptest.NewRequestWithContext(context.Background(), http.MethodGet, cfg.requestPath, nil)
    34  	res := httptest.NewRecorder()
    35  	e.ServeHTTP(res, req)
    36  }

    38  // okHandler は、After フックを発火させるためボディ付きで応答するハンドラを返します。
    39  func okHandler() echo.HandlerFunc {
    40  	return func(c echo.Context) error {
    41  		return c.String(http.StatusOK, "ok")
    42  	}
    43  }
