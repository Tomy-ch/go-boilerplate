    12  // Middleware は、HTTP リクエストの count / duration / status を計測する Echo ミドルウェアを返します。
    13  // status は c.Response().After フックで確定後に取得し、error handler / recovery による
    14  // 最終 status を安全に観測します。/metrics などの運用系パスは計測対象から除外します。
    15  //
    16  // After フックはレスポンスの Write ごとに発火するため、ストリーミング応答（チャンク / SSE 等）
    17  // では複数回呼ばれうる。これによる重複計上を防ぐため、Observe は 1 リクエストにつき
    18  // 厳密に 1 回だけ呼び出されることを保証する。
    19  //
    20  // 計測の既知の限界（After フック方式に起因する仕様）:
    21  //   - ボディ無し応答（204 No Content / 304 Not Modified など）は Write が呼ばれず
    22  //     After フックが発火しないため、計測されない。
    23  //
    24  // これは status を確定後に安全に観測するための After フック採用とのトレードオフであり、
    25  // 現時点では仕様として許容する。
    26  func Middleware(rec Recorder) echo.MiddlewareFunc {
    27  	return func(next echo.HandlerFunc) echo.HandlerFunc {
    28  		return func(c echo.Context) error {
    29  			if ops.IsOpsPath(c.Request().URL.Path) {
    30  				return next(c)
    31  			}
    32  
    33  			start := time.Now()
    34  			method := c.Request().Method
    35  
    36  			var once sync.Once
    37  			c.Response().After(func() {
    38  				once.Do(func() {
    39  					status := normalizeStatus(c.Response().Status)
    40  					rec.Observe(method, routeOf(c), status, statusClass(status), time.Since(start))
    41  				})
    42  			})
    43  
    44  			return next(c)
    45  		}
    46  	}
    47  }
