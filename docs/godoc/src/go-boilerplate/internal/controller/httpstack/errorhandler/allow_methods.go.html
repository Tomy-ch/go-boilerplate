    15  // allowProbeMethods は、リクエストのパステンプレートを逆引きするために試行するメソッドです。
    16  // 405 のリクエスト自身のメソッドは定義上どのルートにも解決しないため、
    17  // 同じパスに定義された別メソッドで解決させてテンプレートを得ます。

    19  // OpenAPI が operation を定義できるメソッドを網羅します。1 つでも欠けると、
    20  // 欠けたメソッドだけを定義したパスでテンプレートを引けず Allow が欠落します。

    47  // NewOpenAPIAllowPolicy は、spec からパスごとの許可メソッド一覧を構築します。
    48  // 解決は Host に依存しないため、router は [newHostAgnosticRouter] で構築します。
    49  func NewOpenAPIAllowPolicy(spec *openapi3.T) (AllowPolicy, error) {
    50  	router, err := newHostAgnosticRouter(spec)
    51  	if err != nil {
    52  		return nil, err
    53  	}
    54  
    55  	return &openAPIAllowPolicy{
    56  		router: router,
    57  		allow:  buildAllowMap(spec),
    58  	}, nil
    59  }

    61  // Allow は [AllowPolicy] を実装します。
    62  // 405 のリクエストはそのメソッドでは解決しないため、他メソッドで探索してパステンプレートを特定します。
    63  func (p *openAPIAllowPolicy) Allow(req *http.Request) string {
    64  	for _, method := range allowProbeMethods {
    65  		probe := req.Clone(req.Context())
    66  		probe.Method = method
    67  
    68  		route, _, err := p.router.FindRoute(probe)
    69  		if err != nil || route == nil {
    70  			continue
    71  		}
    72  		return p.allow[route.Path]
    73  	}
    74  	return ""
    75  }
    76  
    77  // buildAllowMap は、パステンプレートごとの Allow ヘッダー値を組み立てます。
    78  // Echo のルータが返す値と揃えるため、spec に定義された全メソッドへ OPTIONS を加えます
    79  // (OPTIONS は Echo が自動で応答するため、spec の定義有無によらず常に許可されます)。
    80  func buildAllowMap(spec *openapi3.T) map[string]string {
    81  	allow := make(map[string]string)
    82  	if spec.Paths == nil {
    83  		return allow
    84  	}
    85  
    86  	for path, pathItem := range spec.Paths.Map() {
    87  		methods := make([]string, 0, len(pathItem.Operations())+1)
    88  		for method := range pathItem.Operations() {
    89  			if method != http.MethodOptions {
    90  				methods = append(methods, method)
    91  			}
    92  		}
    93  		sort.Strings(methods)
    94  
    95  		allow[path] = strings.Join(append([]string{http.MethodOptions}, methods...), allowHeaderSeparator)
    96  	}
    97  	return allow
    98  }
