    67  // operationsReferencingSchema は、いずれかの error レスポンス(4xx/5xx)の JSON スキーマが
    68  // target と同一(kin-openapi は $ref を共有ポインタへ解決)である operation の集合を返します。
    69  // policy の details プロパティ判定とは独立した契約導出です。
    70  func operationsReferencingSchema(spec *openapi3.T, target *openapi3.Schema) map[string]bool {
    71  	found := make(map[string]bool)
    72  	for _, item := range spec.Paths.Map() {
    73  		for _, op := range item.Operations() {
    74  			if op.OperationID != "" && op.Responses != nil && operationReferencesSchema(op, target) {
    75  				found[op.OperationID] = true
    76  			}
    77  		}
    78  	}
    79  	return found
    80  }

    82  // operationReferencesSchema は、operation の error レスポンスのいずれかが target スキーマを参照するかを返します。
    83  func operationReferencesSchema(op *openapi3.Operation, target *openapi3.Schema) bool {
    84  	for status, respRef := range op.Responses.Map() {
    85  		if isErrorStatusCode(status) && responseReferencesSchema(respRef, target) {
    86  			return true
    87  		}
    88  	}
    89  	return false
    90  }
    91  
    92  // responseReferencesSchema は、レスポンスの application/json スキーマが target と同一かを返します。
    93  func responseReferencesSchema(respRef *openapi3.ResponseRef, target *openapi3.Schema) bool {
    94  	if respRef == nil || respRef.Value == nil {
    95  		return false
    96  	}
    97  	mediaType := respRef.Value.Content.Get("application/json")
    98  	if mediaType == nil || mediaType.Schema == nil {
    99  		return false
   100  	}
   101  	return mediaType.Schema.Value == target
   102  }
