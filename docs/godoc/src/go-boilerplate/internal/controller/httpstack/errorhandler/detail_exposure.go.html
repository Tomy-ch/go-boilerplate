    30  // NewOpenAPIDetailPolicy は、spec からエンドポイント opt-in ポリシーを構築します。
    31  // details を返せるのは、error レスポンス(4xx/5xx)の JSON スキーマに details プロパティを持つ operation のみです。
    32  // 判定は Host に依存しないため、router は [newHostAgnosticRouter] で構築します。
    33  func NewOpenAPIDetailPolicy(spec *openapi3.T) (DetailPolicy, error) {
    34  	router, err := newHostAgnosticRouter(spec)
    35  	if err != nil {
    36  		return nil, err
    37  	}
    38  
    39  	return &openAPIDetailPolicy{
    40  		router:  router,
    41  		allowed: buildDetailExposureMap(spec),
    42  	}, nil
    43  }

    45  // newHostAgnosticRouter は、Host 制約を持たない router を spec から構築します。
    46  // servers を残すと gorillamux が Host マッチを行い、proxy やテストの Host 不一致で
    47  // 全てのルート解決が失敗するため、servers を除去した複製から構築します。
    48  func newHostAgnosticRouter(spec *openapi3.T) (routers.Router, error) {
    49  	hostAgnostic := *spec
    50  	hostAgnostic.Servers = nil
    51  
    52  	return gorillamux.NewRouter(&hostAgnostic)

    55  // Allows は [DetailPolicy] を実装します。ルート解決に失敗した場合は false(fail-closed)を返します。
    56  func (p *openAPIDetailPolicy) Allows(req *http.Request) bool {
    57  	route, _, err := p.router.FindRoute(req)
    58  	if err != nil || route == nil || route.Operation == nil {
    59  		return false
    60  	}
    61  	return p.allowed[route.Operation.OperationID]
    62  }

    64  // buildDetailExposureMap は、error レスポンスの JSON スキーマに details プロパティを持つ operation を集めます。
    65  func buildDetailExposureMap(spec *openapi3.T) map[string]bool {
    66  	allowed := make(map[string]bool)
    67  	if spec.Paths == nil {
    68  		return allowed
    69  	}
    70  	for _, pathItem := range spec.Paths.Map() {
    71  		for _, op := range pathItem.Operations() {
    72  			if op.OperationID == "" || op.Responses == nil {
    73  				continue
    74  			}
    75  			if operationExposesDetails(op) {
    76  				allowed[op.OperationID] = true
    77  			}
    78  		}
    79  	}
    80  	return allowed
    81  }

    84  func operationExposesDetails(op *openapi3.Operation) bool {
    85  	for status, respRef := range op.Responses.Map() {
    86  		if !isErrorStatusCode(status) {
    87  			continue
    88  		}
    89  		if responseHasDetailsProperty(respRef) {
    90  			return true
    91  		}
    92  	}
    93  	return false
    94  }

    96  // responseHasDetailsProperty は、レスポンスの application/json スキーマが details プロパティを持つかを返します。
    97  func responseHasDetailsProperty(respRef *openapi3.ResponseRef) bool {
    98  	if respRef == nil || respRef.Value == nil {
    99  		return false
   100  	}
   101  	mediaType := respRef.Value.Content.Get("application/json")
   102  	if mediaType == nil || mediaType.Schema == nil {
   103  		return false
   104  	}
   105  	return schemaHasDetailsProperty(mediaType.Schema.Value)
   106  }

   108  // schemaHasDetailsProperty は、スキーマ(または allOf 合成の各要素)が details プロパティを持つかを返します。
   109  // ErrorResponseWithDetails は現状トップレベルに details を持つが、allOf 合成へ変えても壊れないよう allOf も走査します。
   110  func schemaHasDetailsProperty(schema *openapi3.Schema) bool {
   111  	if schema == nil {
   112  		return false
   113  	}
   114  	if _, ok := schema.Properties[detailsPropertyName]; ok {
   115  		return true
   116  	}
   117  	for _, sub := range schema.AllOf {
   118  		if sub != nil && schemaHasDetailsProperty(sub.Value) {
   119  			return true
   120  		}
   121  	}
   122  	return false
   123  }
