    36  // handleHTTPError は、HTTPエラーを処理し、適切なレスポンスをクライアントに返します。
    37  func handleHTTPError(c echo.Context, logger logging.Logger, lf logging.LogFieldBuilder, obsCfg *config.ObservabilityConfig, err error) {
    38  	if handled, _ := ctxhelper.GetErrorHandledFromEcho(c); handled {
    39  		return
    40  	}
    41  	ctxhelper.SetErrorHandledToEcho(c, true)
    42  
    43  	resp := normalizeHTTPError(err, requestid.GetRequestIDFromResponse(c))
    44  
    45  	if !c.Response().Committed {
    46  		if writeErr := writeErrorResponse(c, resp); writeErr != nil {
    47  			reqIn := server.BuildHTTPRequestLogInput(c, logging.EventTypeError)
    48  			writeErrFields := []*logging.Field{logging.String(logging.InternalErrorKey, writeErr.Error())}
    49  			fields := append(lf.BuildHTTPRequestFields(reqIn), writeErrFields...)
    50  			logger.Named("errorhandler.handleHTTPError").Error(c.Request().Context(), "failed to write error response", fields...)
    51  			// ヘッダ送出途中の失敗ではレスポンスが commit 済みになり得るため、未 commit 時のみ 500 を書く。
    52  			if !c.Response().Committed {
    53  				c.Response().WriteHeader(http.StatusInternalServerError)
    54  			}
    55  			return
    56  		}
    57  	}
    58  
    59  	// リカバリ済みのパニックは middleware.recover が既にログ済みのため、二重ログを抑止する（500 応答は返す）。
    60  	if recovered, _ := ctxhelper.GetRecoveredFromEcho(c); !recovered {
    61  		logHTTPError(c, logger, lf, obsCfg, resp)
    62  	}
    63  }

    72  func normalizeHTTPError(
    73  	err error,
    74  	requestID string,

    79  		if !isErrorStatus(he.HTTPStatus) {

    81  			if he.Details != nil {
    82  				res.Details = he.Details
    83  			}
    84  			res.RequestId = requestID
    85  			return res
    86  		}
    87  		he.RequestId = requestID
    88  		return he
    89  	}
    90  
    91  	var ehe *echo.HTTPError
    92  	if xerrors.As(err, &ehe) {
    93  		if res := normalizeOpenAPIError(ehe); res != nil {
    94  			res.RequestId = requestID
    95  			return res
    96  		}
    97  		if res := normalizeEchoHTTPError(ehe); res != nil {
    98  			res.RequestId = requestID
    99  			return res
   100  		}
   101  	}
   102  
   103  	res := response.NewHTTPErrorFromAppError(err)
   104  	res.RequestId = requestID
   105  	return res

   109  func httpErrorField(
   110  	c echo.Context,
   111  	lf logging.LogFieldBuilder,
   112  	he *response.HTTPErrorResponse,
   113  ) []*logging.Field {
   114  	fields := []*logging.Field{
   115  		logging.Int(logging.StatusKey, he.HTTPStatus),
   116  		logging.String(logging.ErrorCodeKey, he.Code),
   117  		logging.String(logging.ErrorMessageKey, he.Message),
   118  		logging.String(logging.RequestIDKey, he.RequestId),
   119  	}
   120  	fields = append(fields, lf.BuildHTTPRequestFields(server.BuildHTTPRequestLogInput(c, logging.EventTypeError))...)
   121  	if he.Details != nil {
   122  		fields = append(fields, logging.Strings(logging.ErrorDetailsKey, *he.Details))
   123  	}
   124  	if he.Internal != nil {
   125  		additionalFields := []*logging.Field{
   126  			logging.String(logging.InternalErrorKey, he.Internal.Error()),
   127  			logging.Stacktrace(logging.InternalStackTraceKey, he.Internal),
   128  		}
   129  		fields = append(fields, additionalFields...)
   130  	}
   131  	return fields
   132  }

   134  // logHTTPError は、HTTPエラーをログに記録します。
   135  func logHTTPError(
   136  	c echo.Context,
   137  	logger logging.Logger,
   138  	lf logging.LogFieldBuilder,
   139  	obsCfg *config.ObservabilityConfig,
   140  	he *response.HTTPErrorResponse,
   141  ) {
   142  	if !obsCfg.TargetStatusCodeSet()[he.HTTPStatus] {
   143  		return
   144  	}
   145  	fields := httpErrorField(c, lf, he)
   146  	ctx := c.Request().Context()
   147  	switch {
   148  	case he.HTTPStatus >= errorLevelBoundHTTPStatus:
   149  		logger.Error(ctx, "errorhandler.server_error", fields...)
   150  	default:
   151  		logger.Warn(ctx, "errorhandler.client_error", fields...)
   152  	}
   153  }
