    70  	t.Helper()
    71  
    72  	ctrl := gomock.NewController(t)

    74  	resolver := mock_auth.NewMockIdentityResolver(ctrl)
    75  
    76  	authn, err := authbd.New("user-john-doe", authbd.IssuerMock, nil, nil)
    77  	require.NoError(t, err)
    78  
    79  	switch {
    80  	case stub.authenticateErr != nil:
    81  		authenticator.EXPECT().Authenticate(gomock.Any(), gomock.Any()).
    82  			Return(nil, stub.authenticateErr).AnyTimes()
    83  	case stub.resolveErr != nil:
    84  		authenticator.EXPECT().Authenticate(gomock.Any(), gomock.Any()).Return(authn, nil).AnyTimes()
    85  		resolver.EXPECT().Resolve(gomock.Any(), gomock.Any()).Return(nil, stub.resolveErr).AnyTimes()
    86  	default:
    87  		authenticator.EXPECT().Authenticate(gomock.Any(), gomock.Any()).Return(authn, nil).AnyTimes()
    88  		resolver.EXPECT().Resolve(gomock.Any(), gomock.Any()).Return(authn, nil).AnyTimes()
    89  	}
    90  
    91  	return auth.NewAuthenticator(authenticator, resolver)

    94  // serveTestSpec は、合成 spec のミドルウェアへ 1 リクエスト通し、ステータスとハンドラ到達の有無を返す。
    95  func serveTestSpec(t *testing.T, stub authnStub, path, authorization string) (int, bool) {
    96  	t.Helper()
    97  
    98  	loader := openapi3.NewLoader()
    99  	spec, err := loader.LoadFromData([]byte(testSpec))
   100  	require.NoError(t, err)
   101  	require.NoError(t, spec.Validate(context.Background()))
   102  
   103  	e := echo.New()
   104  	mw := Middleware(spec, nil, newTestAuthFunc(t, stub))
   105  
   106  	reached := false
   107  	handler := mw(func(c *echo.Context) error {
   108  		reached = true
   109  		return c.NoContent(http.StatusOK)
   110  	})
   111  
   112  	req := httptest.NewRequestWithContext(context.Background(), http.MethodGet, path, nil)
   113  	if authorization != "" {
   114  		req.Header.Set(echo.HeaderAuthorization, authorization)
   115  	}
   116  	rec := httptest.NewRecorder()
   117  	c := e.NewContext(req, rec)
   118  	c.SetPath(path)
   119  
   120  	if herr := handler(c); herr != nil {
   121  		var he *echo.HTTPError
   122  		if xerrors.As(herr, &he) {
   123  			return he.Code, reached
   124  		}
   125  		return http.StatusInternalServerError, reached
   126  	}
   127  	return rec.Code, reached
   128  }
