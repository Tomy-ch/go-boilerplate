   129  // decodeSpec returns the embedded OpenAPI spec as raw JSON bytes,
   130  // after base64-decoding and flate-decompressing the embedded blob.
   131  func decodeSpec() ([]byte, error) {
   132  	encoded := strings.Join(swaggerSpec, "")
   133  	compressed, err := base64.StdEncoding.DecodeString(encoded)
   134  	if err != nil {
   135  		return nil, fmt.Errorf("error base64 decoding spec: %w", err)
   136  	}
   137  	zr := flate.NewReader(bytes.NewReader(compressed))
   138  	var buf bytes.Buffer
   139  	if _, err := buf.ReadFrom(zr); err != nil {
   140  		return nil, fmt.Errorf("read flate: %w", err)
   141  	}
   142  	if err := zr.Close(); err != nil {
   143  		return nil, fmt.Errorf("close flate reader: %w", err)
   144  	}
   145  
   146  	return buf.Bytes(), nil
   147  }
