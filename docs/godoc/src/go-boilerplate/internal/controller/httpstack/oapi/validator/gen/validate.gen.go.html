   119  	if err != nil {
   120  		return nil, fmt.Errorf("error base64 decoding spec: %w", err)
   121  	}
   122  	zr := flate.NewReader(bytes.NewReader(compressed))
   123  	var buf bytes.Buffer
   124  	if _, err := buf.ReadFrom(zr); err != nil {
   125  		return nil, fmt.Errorf("read flate: %w", err)
   126  	}
   127  	if err := zr.Close(); err != nil {
   128  		return nil, fmt.Errorf("close flate reader: %w", err)
   129  	}
   130  
   131  	return buf.Bytes(), nil
