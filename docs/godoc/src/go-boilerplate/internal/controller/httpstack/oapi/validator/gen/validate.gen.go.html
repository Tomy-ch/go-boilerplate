   109  	}
   110  	zr := flate.NewReader(bytes.NewReader(compressed))

   112  	if _, err := buf.ReadFrom(zr); err != nil {
   113  		return nil, fmt.Errorf("read flate: %w", err)
   114  	}
   115  	if err := zr.Close(); err != nil {
   116  		return nil, fmt.Errorf("close flate reader: %w", err)
   117  	}
   118  
   119  	return buf.Bytes(), nil
