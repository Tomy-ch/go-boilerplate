   183  // after base64-decoding and flate-decompressing the embedded blob.
   184  func decodeSpec() ([]byte, error) {
   185  	encoded := strings.Join(swaggerSpec, "")
   186  	compressed, err := base64.StdEncoding.DecodeString(encoded)
   187  	if err != nil {
   188  		return nil, fmt.Errorf("error base64 decoding spec: %w", err)
   189  	}
   190  	zr := flate.NewReader(bytes.NewReader(compressed))
   191  	var buf bytes.Buffer
   192  	if _, err := buf.ReadFrom(zr); err != nil {
   193  		return nil, fmt.Errorf("read flate: %w", err)
   194  	}
   195  	if err := zr.Close(); err != nil {
   196  		return nil, fmt.Errorf("close flate reader: %w", err)
   197  	}
   198  
   199  	return buf.Bytes(), nil
   200  }

   224  // PathToRawSpec; externally-referenced files must be embedded in their

   233  		pathToFile := url.String()
   234  		pathToFile = path.Clean(pathToFile)
   235  		getSpec, ok := resolvePath[pathToFile]
   236  		if !ok {
   237  			err1 := fmt.Errorf("path not found: %s", pathToFile)
   238  			return nil, err1
   239  		}
   240  		return getSpec()
   241  	}

   243  	specData, err = rawSpec()
   244  	if err != nil {
   245  		return
   246  	}
   247  	swagger, err = loader.LoadFromData(specData)
   248  	if err != nil {
   249  		return
   250  	}
   251  	return
   252  }
