    20  // NewAuthenticator は、認証用のOpenAPIオプションを返します。
    21  func NewAuthenticator(
    22  	authenticator authbd.Authenticator,
    23  	resolver authbd.IdentityResolver,
    24  ) openapi3filter.AuthenticationFunc {
    25  	return func(ctx context.Context, input *openapi3filter.AuthenticationInput) error {
    26  		req := input.RequestValidationInput.Request
    27  
    28  		// エラー変換は authExtractor に一本化。
    29  		authn, err := authExtractor(ctx, req, authenticator, resolver)
    30  		if err != nil {
    31  			return withHTTPStatus(err)
    32  		}
    33  		if authn == nil {
    34  			return withHTTPStatus(ErrUnauthorizedTokenNotProvided)
    35  		}
    36  
    37  		//nolint:contextcheck // input が内包する request の context のスロットへ書き戻すため
    38  		if !ctxhelper.SetAuthn(req.Context(), *authn) {
    39  			return withHTTPStatus(ErrAuthnSlotNotFound)
    40  		}
    41  		return nil
    42  	}
    43  }

    45  // withHTTPStatus は、認証フェーズのエラーへ 401 のステータスを持たせます（元のエラーは保持）。
    46  // OpenAPI バリデータは認証エラーのうち HTTP ステータスを解決できないものを 403 へ丸めるため、
    47  // 401 として返すにはこの段でステータスを持たせる必要があります。
    48  // infra 障害は infraErrorToHTTP が先にステータスを付与するため、ここで 401 に潰されません。
    49  func withHTTPStatus(err error) error {
    50  	var he *echo.HTTPError
    51  	if xerrors.As(err, &he) {
    52  		return err
    53  	}
    54  	return echo.NewHTTPError(http.StatusUnauthorized, "").Wrap(err)
    55  }

    57  // authExtractor は、Bearer トークンを検証して内部ユーザーを解決した Authn を返します。
    58  // トークンが無い場合は nil, nil を返します。
    59  func authExtractor(
    60  	ctx context.Context,
    61  	req *http.Request,
    62  	authenticator authbd.Authenticator,
    63  	resolver authbd.IdentityResolver,
    64  ) (*authbd.Authn, error) {
    65  	scheme, token := extractBearerToken(req)
    66  	if token == "" {
    67  		//nolint:nilnil // トークン未提供を表す。呼び出し側で authn==nil を判定し未提供エラーへ変換するため意図的にnil,nilを返す
    68  		return nil, nil
    69  	}
    70  
    71  	cred, err := authbd.NewCredential(scheme, token)
    72  	if err != nil {
    73  		return nil, err
    74  	}
    75  
    76  	authn, err := authenticator.Authenticate(ctx, cred)
    77  	if err != nil {
    78  		return nil, ErrUnauthorizedInvalidToken
    79  	}
    80  	if authn == nil {
    81  		//nolint:nilnil // Authenticate が nil,nil を返した場合は未提供として扱い、呼び出し側で ErrUnauthorizedTokenNotProvided へ変換する
    82  		return nil, nil
    83  	}
    84  
    85  	// 認証済みの外部アイデンティティ（issuer + subject）を内部ユーザーへ解決する。
    86  	// 未登録 / 削除済み等の認証失敗（ErrUnauthenticated）は 401となる。
    87  	// DB 等の infra 障害は認証フェーズで 401 に潰さず、apperror の分類に応じた 5xx として返す。
    88  	resolved, err := resolver.Resolve(ctx, authn)
    89  	if err != nil {
    90  		if xerrors.Is(err, apperror.ErrUnauthenticated) {
    91  			return nil, err
    92  		}
    93  		return nil, infraErrorToHTTP(err)
    94  	}
    95  	if resolved == nil {
    96  		// resolver は成功時に非 nil の Authn を返す契約。防御的に未認証として扱う。
    97  		return nil, authbd.ErrIdentityNotFound
    98  	}
    99  
   100  	return resolved, nil
   101  }

   103  // infraErrorToHTTP は、 IdentityResolver が返した非認証エラーを、apperror の分類に応じたエラーへ変換します。
   104  func infraErrorToHTTP(err error) error {
   105  	status := http.StatusInternalServerError
   106  	if xerrors.Is(err, apperror.ErrUnavailable) {
   107  		status = http.StatusServiceUnavailable
   108  	}
   109  	return echo.NewHTTPError(status, "").Wrap(err)
   110  }

   112  // extractBearerToken は、Authorization ヘッダから Bearer トークンを抽出します。
   113  // Bearer トークンは RFC 6750 で Authorization ヘッダに固定されるため、ヘッダ名・スキームは可変にしない。
   114  // Authorization: Bearer <token> 形式のときだけ scheme=SchemeBearer と token を返し、
   115  // それ以外は scheme/token とも空を返します。
   116  func extractBearerToken(r *http.Request) (string, string) {
   117  	raw := strings.TrimSpace(r.Header.Get(echo.HeaderAuthorization))
   118  	if raw == "" {
   119  		return "", ""
   120  	}
   121  	after, ok := strings.CutPrefix(raw, prefixBearer)
   122  	if !ok {
   123  		return "", ""
   124  	}
   125  	return authbd.SchemeBearer, strings.TrimSpace(after)
   126  }
