    45  func authExtractor(
    46  	ctx context.Context,
    47  	req *http.Request,
    48  	authCfg *config.AuthConfig,
    49  	authenticator authbd.Authenticator,

    52  	if token == "" {

    54  	}
    55  
    56  	cred, err := authbd.NewCredential(token)
    57  	if err != nil {
    58  		return nil, err
    59  	}
    60  
    61  	authn, err := authenticator.Authenticate(ctx, cred)
    62  	if err != nil {
    63  		return nil, ErrUnauthorizedInvalidToken
    64  	}
    65  
    66  	return authn, nil
    67  }

    69  // extractToken は、認証トークンを抽出します。
    70  func extractToken(r *http.Request, authCfg *config.AuthConfig) string {
    71  	// extract from Cookie
    72  	if authCfg.CookieName() != "" {
    73  		if ck, err := r.Cookie(authCfg.CookieName()); err == nil && ck != nil {
    74  			if v := strings.TrimSpace(ck.Value); v != "" {
    75  				return v
    76  			}
    77  		}
    78  	}
    79  
    80  	// extract from Header
    81  	if authCfg.HeaderName() == "" {
    82  		return ""
    83  	}
    84  
    85  	raw := strings.TrimSpace(r.Header.Get(authCfg.HeaderName()))
    86  	if raw == "" {
    87  		return ""
    88  	}
    89  	if authCfg.AllowedHeaderBearer() && strings.EqualFold(authCfg.HeaderName(), echo.HeaderAuthorization) {
    90  		if strings.HasPrefix(raw, prefixBearer) {
    91  			return strings.TrimSpace(strings.TrimPrefix(raw, prefixBearer))
    92  		}
    93  		return ""
    94  	}
    95  	return raw
    96  }
