    21  // NewAuthenticator は、認証用のOpenAPIオプションを返します。
    22  func NewAuthenticator(
    23  	authenticator authbd.Authenticator,
    24  	resolver authbd.IdentityResolver,
    25  ) openapi3filter.AuthenticationFunc {
    26  	return func(_ context.Context, input *openapi3filter.AuthenticationInput) error {
    27  		req := input.RequestValidationInput.Request
    28  
    29  		// OpenAPI バリデータが渡す context は context.Background() から組み立てられており、
    30  		// スパン・deadline・キャンセルのいずれも持たない。認証は request の予算の内側で行う。
    31  		//nolint:contextcheck // 引数の context ではなく input が内包する request の context を用いるため
    32  		authn, err := authExtractor(req.Context(), req, authenticator, resolver)
    33  		if err != nil {
    34  			failure := withHTTPStatus(err)
    35  			// 記録するのは資格情報が提示されたうえでの失敗だけ。未提示は失敗ではない。
    36  			//nolint:contextcheck // input が内包する request の context のスロットへ書き戻すため
    37  			ctxhelper.SetAuthnFailure(req.Context(), failure)
    38  			return failure
    39  		}
    40  		if authn == nil {
    41  			return withHTTPStatus(ErrUnauthorizedTokenNotProvided)
    42  		}
    43  
    44  		//nolint:contextcheck // input が内包する request の context のスロットへ書き戻すため
    45  		if !ctxhelper.SetAuthn(req.Context(), *authn) {
    46  			failure := withHTTPStatus(ErrAuthnSlotNotFound)
    47  			// 認証は成立したが結果を運べていない。記録しなければ、認証済みの主体が匿名として通る。
    48  			//nolint:contextcheck // 同上
    49  			ctxhelper.SetAuthnFailure(req.Context(), failure)
    50  			return failure
    51  		}
    52  		return nil
    53  	}
    54  }

    56  // withHTTPStatus は、認証フェーズのエラーへ apperror の分類に対応するステータスを持たせます（元のエラーは保持）。
    57  // OpenAPI バリデータはステータスを解決できないエラーを 403 へ丸めるため、この段で持たせないと、
    58  // 認可の判定を行っていないものが認可の結論として外へ出ます。
    59  // メッセージは空のまま返します（本文は errorhandler が組み立てる）。
    60  func withHTTPStatus(err error) error {
    61  	var he *echo.HTTPError
    62  	if xerrors.As(err, &he) {
    63  		return err
    64  	}
    65  	return echo.NewHTTPError(response.NewHTTPErrorFromAppError(err).HTTPStatus, "").Wrap(err)
    66  }

    68  // authExtractor は、Bearer トークンを検証して内部ユーザーを解決した Authn を返します。
    69  // トークンが無い場合は nil, nil を返します。
    70  func authExtractor(
    71  	ctx context.Context,
    72  	req *http.Request,
    73  	authenticator authbd.Authenticator,
    74  	resolver authbd.IdentityResolver,
    75  ) (*authbd.Authn, error) {
    76  	scheme, token := extractBearerToken(req)
    77  	if token == "" {
    78  		//nolint:nilnil // トークン未提供を表す。呼び出し側で authn==nil を判定し未提供エラーへ変換するため意図的にnil,nilを返す
    79  		return nil, nil
    80  	}
    81  
    82  	cred, err := authbd.NewCredential(scheme, token)
    83  	if err != nil {
    84  		return nil, err
    85  	}
    86  
    87  	// 資格情報についての結論だけを認証エラーへ寄せ、原因は外へ出さない。
    88  	// 検証を遂行できなかった場合（鍵が引けない / 予算切れ）は結論ではないため、分類を保ったまま返す。
    89  	authn, err := authenticator.Authenticate(ctx, cred)
    90  	if err != nil {
    91  		if xerrors.Is(err, apperror.ErrUnauthenticated) {
    92  			return nil, ErrUnauthorizedInvalidToken
    93  		}
    94  		return nil, err
    95  	}
    96  	if authn == nil {
    97  		//nolint:nilnil // Authenticate が nil,nil を返した場合は未提供として扱い、呼び出し側で ErrUnauthorizedTokenNotProvided へ変換する
    98  		return nil, nil
    99  	}
   100  
   101  	// 認証済みの外部アイデンティティ（issuer + subject）を内部ユーザーへ解決する。エラーは分類を変えずに返す。
   102  	resolved, err := resolver.Resolve(ctx, authn)
   103  	if err != nil {
   104  		return nil, err
   105  	}
   106  	if resolved == nil {
   107  		// resolver は成功時に非 nil の Authn を返す契約。防御的に未認証として扱う。
   108  		return nil, authbd.ErrIdentityNotFound
   109  	}
   110  
   111  	return resolved, nil
   112  }

   114  // extractBearerToken は、Authorization ヘッダから Bearer トークンを抽出します（ヘッダ名・スキームは RFC 6750 により固定）。
   115  // Authorization: Bearer <token> 形式のときだけ scheme=SchemeBearer と token を返し、
   116  // それ以外は scheme/token とも空を返します。
   117  func extractBearerToken(r *http.Request) (string, string) {
   118  	raw := strings.TrimSpace(r.Header.Get(echo.HeaderAuthorization))
   119  	if raw == "" {
   120  		return "", ""
   121  	}
   122  	after, ok := strings.CutPrefix(raw, prefixBearer)
   123  	if !ok {
   124  		return "", ""
   125  	}
   126  	return authbd.SchemeBearer, strings.TrimSpace(after)
   127  }
