    13  // headerCrossOriginResourcePolicy は、Cross-Origin-Resource-Policy ヘッダーの名前です。
    14  // echo の Secure ミドルウェアはこのヘッダーを扱わないため、名前を自前で持ちます。
    15  const headerCrossOriginResourcePolicy = "Cross-Origin-Resource-Policy"

    17  // Middleware は、Content-Type-Options / Referrer-Policy / X-Frame-Options / HSTS /
    18  // Cross-Origin-Resource-Policy を設定する Echo セキュリティミドルウェアを返します。
    19  // 各値は SecurityConfig から取得します。
    20  func Middleware(secCfg *config.SecurityConfig) echo.MiddlewareFunc {
    21  	secure := middleware.SecureWithConfig(buildSecureConfig(secCfg))
    22  	corp := secCfg.CrossOriginResourcePolicy()
    23  
    24  	return func(next echo.HandlerFunc) echo.HandlerFunc {
    25  		return secure(func(c *echo.Context) error {
    26  			applyCORP(c.Response().Header(), corp)
    27  			return next(c)
    28  		})
    29  	}
    30  }
    31  
    32  // applyCORP は、Cross-Origin-Resource-Policy ヘッダーを設定します。
    33  // policy が空ならヘッダーを設定しません。空文字は「このヘッダーを出さない」という設定値であり、
    34  // 値の無いヘッダーを送ることとは区別されます。
    35  func applyCORP(h http.Header, policy string) {
    36  	if policy == "" {
    37  		return
    38  	}
    39  	h.Set(headerCrossOriginResourcePolicy, policy)
    40  }

    42  func buildSecureConfig(secCfg *config.SecurityConfig) middleware.SecureConfig {
    43  	return middleware.SecureConfig{

    47  		HSTSMaxAge:            int(secCfg.HSTSMaxAge().Seconds()),
    48  		HSTSExcludeSubdomains: secCfg.HSTSExcludeSubdomains(),
    49  		HSTSPreloadEnabled:    secCfg.HSTSPreloadEnabled(),
    50  	}
