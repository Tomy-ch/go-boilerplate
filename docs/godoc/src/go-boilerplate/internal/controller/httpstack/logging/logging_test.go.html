   147  func Test_requestLog_buildResponseLogFields(t *testing.T) {
   148  	t.Parallel()
   149  
   150  	lf := logging.NewTestLogFieldBuilder(t)
   151  
   152  	// 各サブテストは並列実行されるため、共有の echo.Context を使うと Response() の
   153  	// ステータス/ヘッダ書き込みが競合する。サブテストごとに専用の Context を生成する。
   154  	newContext := func() echo.Context {
   155  		e := echo.New()
   156  		ctx := context.Background()
   157  		req := httptest.NewRequestWithContext(ctx, http.MethodGet, "/resp", nil)
   158  		req.RemoteAddr = "5.6.7.8:9000"
   159  		rec := httptest.NewRecorder()
   160  		return e.NewContext(req, rec)
   161  	}
   162  
   163  	t.Run("正常系", func(t *testing.T) {
   164  		t.Parallel()
   165  
   166  		t.Run("基本フィールドを返す（trace_id/span_idはLoggerがctxから注入する）", func(t *testing.T) {
   167  			t.Parallel()
   168  
   169  			c := newContext()
   170  			expectedStatus := http.StatusCreated
   171  			expectedRequestID := "req-123"
   172  
   173  			c.Response().Status = expectedStatus
   174  			c.Response().Header().Set("X-Request-Id", expectedRequestID)
   175  
   176  			l := requestLog{c: c, lf: lf}
   177  			fields := l.buildResponseLogFields(150 * time.Millisecond)
   178  
   179  			assert.Contains(t, fields, logging.Int(logging.StatusKey, expectedStatus))
   180  			assert.Contains(t, fields, logging.String(logging.RequestIDKey, expectedRequestID))
   181  		})
   182  	})
