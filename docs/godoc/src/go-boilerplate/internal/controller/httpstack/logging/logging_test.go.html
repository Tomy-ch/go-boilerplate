   138  func Test_requestLog_buildResponseLogFields(t *testing.T) {
   139  	t.Parallel()
   140  
   141  	lf := logging.NewTestLogFieldBuilder(t)
   142  
   143  	// 各サブテストは並列実行されるため、共有の echo.Context を使うと Response() の
   144  	// ステータス/ヘッダ書き込みが競合する。サブテストごとに専用の Context を生成する。
   145  	newContext := func() echo.Context {
   146  		e := echo.New()
   147  		ctx := context.Background()
   148  		req := httptest.NewRequestWithContext(ctx, http.MethodGet, "/resp", nil)
   149  		req.RemoteAddr = "5.6.7.8:9000"
   150  		rec := httptest.NewRecorder()
   151  		return e.NewContext(req, rec)
   152  	}
   153  
   154  	t.Run("正常系", func(t *testing.T) {
   155  		t.Parallel()
   156  
   157  		t.Run("スパンなしはtrace/spanは含まれないレスポンスログフィールドセットを返す", func(t *testing.T) {
   158  			t.Parallel()
   159  
   160  			c := newContext()
   161  			expectedStatus := http.StatusCreated
   162  			expectedRequestID := "req-123"
   163  
   164  			c.Response().Status = expectedStatus
   165  			c.Response().Header().Set("X-Request-Id", expectedRequestID)
   166  
   167  			l := requestLog{c: c, lf: lf, traceCtx: &observability.TraceContext{}}
   168  			fields := l.buildResponseLogFields(150 * time.Millisecond)
   169  
   170  			assert.Contains(t, fields, logging.Int(logging.StatusKey, expectedStatus))
   171  			assert.Contains(t, fields, logging.String(logging.RequestIDKey, expectedRequestID))
   172  		})
   173  
   174  		t.Run("スパンありはtrace/spanが含まれるレスポンスログフィールドセットを返す", func(t *testing.T) {
   175  			t.Parallel()
   176  
   177  			c := newContext()
   178  			expectedStatus := http.StatusAccepted
   179  			expectedRequestID := "req-accepted"
   180  
   181  			cWithSpan, end := testspan.StartTestSpanForEcho(t, c)
   182  			defer end()
   183  
   184  			cWithSpan.Response().Status = expectedStatus
   185  			cWithSpan.Response().Header().Set("X-Request-Id", expectedRequestID)
   186  
   187  			tc := observability.ExtractTraceContext(cWithSpan.Request().Context())
   188  			l := requestLog{c: cWithSpan, lf: lf, traceCtx: tc}
   189  			fields := l.buildResponseLogFields(20 * time.Millisecond)
   190  
   191  			assert.Contains(t, fields, logging.Int(logging.StatusKey, expectedStatus))
   192  			assert.Contains(t, fields, logging.String(logging.RequestIDKey, expectedRequestID))
   193  		})
   194  	})
