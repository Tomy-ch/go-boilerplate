   172  	t.Parallel()
   173  
   174  	t.Run("正常系", func(t *testing.T) {
   175  		t.Parallel()
   176  
   177  		t.Run("origがHijackerを実装している場合は書き換えて透過する", func(t *testing.T) {
   178  			t.Parallel()
   179  			orig := newFakeOrig()
   180  			b := true
   181  			cfg := &SecurityCookie{applyToAll: true, forceHTTPOnly: &b}
   182  			w := newCookieRewriteWriter(orig, cfg)
   183  
   184  			w.hdr.Add("Set-Cookie", "id=1")
   185  			conn, rw, err := w.Hijack()
   186  			require.NoError(t, err)
   187  			require.NotNil(t, conn)
   188  			require.NotNil(t, rw)
   189  			assert.True(t, w.wroteHdr)
   190  			sc := w.hdr.Values("Set-Cookie")
   191  			require.Len(t, sc, 1)
   192  			assert.Contains(t, sc[0], "HttpOnly")
   193  			require.NoError(t, conn.Close())
   194  		})
   195  
   196  		t.Run("既にwroteHdrがtrueの場合はrewriteをスキップする", func(t *testing.T) {
   197  			t.Parallel()
   198  			orig := newFakeOrig()
   199  			b := true
   200  			cfg := &SecurityCookie{applyToAll: true, forceHTTPOnly: &b}
   201  			w := newCookieRewriteWriter(orig, cfg)
   202  			w.hdr.Add("Set-Cookie", "id=1")
   203  			w.wroteHdr = true
   204  			conn, rw, err := w.Hijack()
   205  			require.NoError(t, err)
   206  			require.NotNil(t, conn)
   207  			require.NotNil(t, rw)
   208  			assert.Equal(t, []string{"id=1"}, w.hdr.Values("Set-Cookie"))
   209  			require.NoError(t, conn.Close())
   210  		})
   211  
   212  		t.Run("Rewriteが失敗したら元のrawをhdrに残す", func(t *testing.T) {
   213  			t.Parallel()
   214  			orig := newFakeOrig()
   215  			cfg := &SecurityCookie{applyToAll: true}
   216  			w := newCookieRewriteWriter(orig, cfg)
   217  			// 不正な Cookie（'=' 無し）なので RewriteSetCookie は "" を返す
   218  			w.hdr.Add("Set-Cookie", "NoEquals")
   219  
   220  			conn, rw, err := w.Hijack()
   221  			require.NoError(t, err)
   222  			require.NotNil(t, conn)
   223  			require.NotNil(t, rw)
   224  			assert.Equal(t, []string{"NoEquals"}, w.hdr.Values("Set-Cookie"))
   225  			require.NoError(t, conn.Close())
   226  		})
   227  	})
   228  
   229  	t.Run("異常系", func(t *testing.T) {
   230  		t.Parallel()
   231  
   232  		t.Run("origがHijackerを実装していない場合はErrNotSupportedを返す", func(t *testing.T) {
   233  			t.Parallel()
