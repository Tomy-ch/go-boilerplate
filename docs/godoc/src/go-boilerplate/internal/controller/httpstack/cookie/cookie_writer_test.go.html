   147  func Test_cookieRewriteWriter_Hijack(t *testing.T) {
   148  	t.Parallel()
   149  
   150  	t.Run("正常系", func(t *testing.T) {
   151  		t.Parallel()
   152  
   153  		t.Run("origがHijackerを実装している場合は書き換えて透過する", func(t *testing.T) {
   154  			t.Parallel()
   155  			orig := newFakeOrig()
   156  			b := true
   157  			cfg := &SecurityCookie{applyToAll: true, forceHTTPOnly: &b}
   158  			w := newCookieRewriteWriter(orig, cfg)
   159  
   160  			w.hdr.Add("Set-Cookie", "id=1")
   161  			conn, rw, err := w.Hijack()
   162  			require.NoError(t, err)
   163  			require.NotNil(t, conn)
   164  			require.NotNil(t, rw)
   165  			assert.True(t, w.wroteHdr)
   166  			sc := w.hdr.Values("Set-Cookie")
   167  			require.Len(t, sc, 1)
   168  			assert.Contains(t, sc[0], "HttpOnly")
   169  			require.NoError(t, conn.Close())
   170  		})
   171  
   172  		t.Run("既にwroteHdrがtrueの場合はrewriteをスキップする", func(t *testing.T) {
   173  			t.Parallel()
   174  			orig := newFakeOrig()
   175  			b := true
   176  			cfg := &SecurityCookie{applyToAll: true, forceHTTPOnly: &b}
   177  			w := newCookieRewriteWriter(orig, cfg)
   178  			w.hdr.Add("Set-Cookie", "id=1")
   179  			w.wroteHdr = true
   180  			conn, rw, err := w.Hijack()
   181  			require.NoError(t, err)
   182  			require.NotNil(t, conn)
   183  			require.NotNil(t, rw)
   184  			// rewrite がスキップされ HttpOnly は付与されない
   185  			assert.Equal(t, []string{"id=1"}, w.hdr.Values("Set-Cookie"))
   186  			require.NoError(t, conn.Close())
   187  		})
   188  
   189  		t.Run("Rewriteが失敗したら元のrawをhdrに残す", func(t *testing.T) {
   190  			t.Parallel()
   191  			orig := newFakeOrig()
   192  			cfg := &SecurityCookie{applyToAll: true}
   193  			w := newCookieRewriteWriter(orig, cfg)
   194  			// 不正な Cookie（'=' 無し）なので RewriteSetCookie は "" を返す
   195  			w.hdr.Add("Set-Cookie", "NoEquals")
   196  
   197  			conn, rw, err := w.Hijack()
   198  			require.NoError(t, err)
   199  			require.NotNil(t, conn)
   200  			require.NotNil(t, rw)
   201  			assert.Equal(t, []string{"NoEquals"}, w.hdr.Values("Set-Cookie"))
   202  			require.NoError(t, conn.Close())
   203  		})
   204  	})
   205  
   206  	t.Run("異常系", func(t *testing.T) {
   207  		t.Parallel()
   208
