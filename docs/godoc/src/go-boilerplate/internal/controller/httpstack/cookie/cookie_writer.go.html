    66  // Hijack はコネクションをハイジャックします。
    67  func (w *cookieRewriteWriter) Hijack() (net.Conn, *bufio.ReadWriter, error) {
    68  	// WebSocket 等は WriteHeader を経由せずに Hijack されることがあるため、w.hdr 上で
    69  	// Set-Cookie rewrite を確定させる。これは w.Header() を参照するタイプの Upgrade 実装にのみ
    70  	// 効く防御であり、生バッファへ直書きする通常の hijack 経路では配線に出ない。
    71  	if !w.wroteHdr {
    72  		w.wroteHdr = true
    73  
    74  		rawCookies := w.hdr.Values(headerSetCookie)
    75  		if len(rawCookies) > 0 {
    76  			w.hdr.Del(headerSetCookie)
    77  			for _, raw := range rawCookies {
    78  				w.hdr.Add(headerSetCookie, w.rewriteOrKeep(raw))
    79  			}
    80  		}
    81  	}
    82  
    83  	// WebSocket などで http.Hijacker を透過する
    84  	h, ok := w.orig.(http.Hijacker)
    85  	if !ok {
    86  		return nil, nil, http.ErrNotSupported
    87  	}
    88  	return h.Hijack()
    89  }

   101  // wrapper 側でヘッダ書き換えを確定させてから orig に委譲します。
   102  func (w *cookieRewriteWriter) ReadFrom(r io.Reader) (int64, error) {
   103  	if !w.wroteHdr {
   104  		w.WriteHeader(http.StatusOK)
   105  	}
   106  
   107  	if rf, ok := w.orig.(io.ReaderFrom); ok {
   108  		return rf.ReadFrom(r)

   120  // rewriteOrKeep は Set-Cookie を書き換え、失敗（空文字）時は元の raw を残します。
   121  // 失敗時に消すと Cookie 消失になるため、この不変条件を1箇所に集約します。
   122  func (w *cookieRewriteWriter) rewriteOrKeep(raw string) string {
   123  	if r := w.cfg.RewriteSetCookie(raw); r != "" {
   124  		return r
   125  	}
   126  	return raw
   127  }

   129  // flushHeadersWithRewrite はヘッダを書き込みます（Set-Cookie を書き換え）。
   130  func (w *cookieRewriteWriter) flushHeadersWithRewrite() {
   131  	for k, vv := range w.hdr {
   132  		if strings.EqualFold(k, headerSetCookie) {
   133  			continue
   134  		}
   135  		for _, v := range vv {
   136  			w.orig.Header().Add(k, v)
   137  		}
   138  	}
   139  
   140  	// Set-Cookie を rewrite してから追加
   141  	for _, raw := range w.hdr.Values(headerSetCookie) {
   142  		w.orig.Header().Add(headerSetCookie, w.rewriteOrKeep(raw))
   143  	}
   144  }
