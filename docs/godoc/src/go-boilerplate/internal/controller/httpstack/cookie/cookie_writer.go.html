    65  func (w *cookieRewriteWriter) Hijack() (net.Conn, *bufio.ReadWriter, error) {
    66  	// WebSocket 等は WriteHeader を経由せずに Hijack されることがあるため、ここで
    67  	// Set-Cookie rewrite を確定させる。これは w.Header() を参照するタイプの Upgrade 実装にのみ
    68  	// 効く防御であり、生バッファへ直書きする通常の hijack 経路では配線に出ない。
    69  	if !w.wroteHdr {
    70  		w.wroteHdr = true
    71  
    72  		w.rewriteSetCookies()
    73  	}
    74  
    75  	// WebSocket などで http.Hijacker を透過する
    76  	h, ok := w.orig.(http.Hijacker)
    77  	if !ok {
    78  		return nil, nil, http.ErrNotSupported
    79  	}
    80  	return h.Hijack()
    81  }

   122  func (w *cookieRewriteWriter) rewriteSetCookies() {

   126  	if len(raws) == 0 {
