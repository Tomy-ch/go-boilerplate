    16  func TestMiddleware(t *testing.T) {
    17  	t.Parallel()
    18  
    19  	// exec は、limitMB 上限のミドルウェア配下で body を POST し、(handler 実行有無, 返り error) を返す。
    20  	exec := func(t *testing.T, limitMB int, body []byte) (bool, error) {
    21  		t.Helper()
    22  		e := echo.New()
    23  		req := httptest.NewRequestWithContext(context.Background(), http.MethodPost, "/", bytes.NewReader(body))
    24  		rec := httptest.NewRecorder()
    25  		c := e.NewContext(req, rec)
    26  
    27  		handled := false
    28  		handler := func(c *echo.Context) error {
    29  			_, _ = io.ReadAll(c.Request().Body) // BodyLimit は reader をラップするので読み切る
    30  			handled = true
    31  			return c.NoContent(http.StatusOK)
    32  		}
    33  		return handled, Middleware(limitMB)(handler)(c)
    34  	}
    35  
    36  	t.Run("正常系", func(t *testing.T) {
    37  		t.Parallel()
    38  
    39  		t.Run("上限内のボディは通過しハンドラが実行される", func(t *testing.T) {
    40  			t.Parallel()
    41  
    42  			handled, err := exec(t, 1, bytes.Repeat([]byte("a"), 1000)) // 1KB < 1MB
    43  			require.NoError(t, err)
    44  			assert.True(t, handled)
    45  		})
    46  
    47  		t.Run("ちょうど上限値のボディは通過しハンドラが実行される", func(t *testing.T) {
    48  			t.Parallel()
    49  
    50  			// limit 1MB=1,000,000 byte ちょうどは上限以下として通過する（境界の通過側）。
    51  			handled, err := exec(t, 1, bytes.Repeat([]byte("a"), 1_000_000))
    52  			require.NoError(t, err)
    53  			assert.True(t, handled)
    54  		})
    55  	})
    56  
    57  	t.Run("異常系", func(t *testing.T) {
    58  		t.Parallel()
    59  
    60  		t.Run("上限超過のボディは413で弾かれハンドラは実行されない", func(t *testing.T) {
    61  			t.Parallel()
    62  
    63  			// limit 1MB=1,000,000 byte に対し 2,000,000 byte を送る。
    64  			handled, err := exec(t, 1, bytes.Repeat([]byte("a"), 2_000_000))
    65  
    66  			require.Error(t, err)
    67  			assert.Equal(t, http.StatusRequestEntityTooLarge, echo.StatusCode(err))
    68  			assert.False(t, handled)
    69  		})
    70  
    71  		t.Run("上限を1バイト超えるボディは413で弾かれハンドラは実行されない", func(t *testing.T) {
    72  			t.Parallel()
    73  
    74  			// limit 1MB=1,000,000 byte に対し 1,000,001 byte（境界の超過側）を送る。
    75  			handled, err := exec(t, 1, bytes.Repeat([]byte("a"), 1_000_001))
    76  
    77  			require.Error(t, err)
    78  			assert.Equal(t, http.StatusRequestEntityTooLarge, echo.StatusCode(err))
    79  			assert.False(t, handled)
    80  		})
    81  
    82  		t.Run("Content-Length不明でも実読み取りが上限超過なら413になる", func(t *testing.T) {
    83  			t.Parallel()
    84  
    85  			// io.NopCloser で包むと Content-Length が確定せず、BodyLimit は ContentLength の
    86  			// 早期判定を通過し、実読み取り中の上限チェックで 413 を返す。
    87  			e := echo.New()
    88  			body := io.NopCloser(bytes.NewReader(bytes.Repeat([]byte("a"), 2_000_000)))
    89  			req := httptest.NewRequestWithContext(context.Background(), http.MethodPost, "/", body)
    90  			rec := httptest.NewRecorder()
    91  			c := e.NewContext(req, rec)
    92  
    93  			handled := false
    94  			handler := func(c *echo.Context) error {
    95  				_, err := io.ReadAll(c.Request().Body) // 実読み取りで上限超過し 413 を返す
    96  				handled = true
    97  				return err
    98  			}
    99  			err := Middleware(1)(handler)(c)
   100  
   101  			require.Error(t, err)
   102  			assert.Equal(t, http.StatusRequestEntityTooLarge, echo.StatusCode(err))
   103  			assert.True(t, handled, "実読み取り経路を通すためハンドラ自体は実行される")
   104  		})
   105  
   106  		t.Run("limitMBが0以下の場合はパニックする", func(t *testing.T) {
   107  			t.Parallel()
   108  
   109  			assert.Panics(t, func() { Middleware(0) })
   110  			assert.Panics(t, func() { Middleware(-1) })
   111  		})
   112  	})
