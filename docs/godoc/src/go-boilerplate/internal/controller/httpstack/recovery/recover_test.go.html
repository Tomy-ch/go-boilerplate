    94  			cfg := config.MockConfigForTest(t)
    95  			appCfg := config.NewApplicationConfig(cfg)
    96  			lf := logging.NewTestLogFieldBuilder(t)
    97  			obsLogger, observed := logging.NewObservedTestLogger(t)
    98  
    99  			e := echo.New()
   100  			e.Use(Middleware(obsLogger, lf, appCfg))
   101  			e.GET("/panic", func(_ echo.Context) error { panic("boom-panic") })
   102  
   103  			req := httptest.NewRequestWithContext(context.Background(), http.MethodGet, "/panic", nil)
   104  			e.ServeHTTP(httptest.NewRecorder(), req)
   105

   107  			require.Len(t, entries, 1)
   108  			cm := entries[0].ContextMap()
   109  
   110  			assert.Equal(t, logging.EventTypePanic, cm[logging.EventTypeKey])
   111  
   112  			errStr, ok := cm[logging.InternalErrorKey].(string)
   113  			require.True(t, ok)
   114  			assert.Contains(t, errStr, "boom-panic")
   115  
   116  			// 実ランタイムスタックが行配列で出力されること（Grafana 表示用に []string 化済み）。
   117  			// zap observer は []any として保持するため、要素ごとに string にアサートする。
   118  			stackLines, ok := cm[logging.InternalStackTraceKey].([]any)
   119  			require.True(t, ok)
   120  			require.NotEmpty(t, stackLines)
   121  			first, ok := stackLines[0].(string)
