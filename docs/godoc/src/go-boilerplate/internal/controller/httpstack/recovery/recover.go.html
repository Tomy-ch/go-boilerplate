    31  // newRecoverConfig は、環境設定に基づいてリカバリミドルウェアの設定を生成します。
    32  func newRecoverConfig(logger logging.Logger, appCfg *config.ApplicationConfig) middleware.RecoverConfig {
    33  	switch {
    34  	case appCfg.IsDevelopmentMode():
    35  		return developmentConfig()
    36  	case appCfg.IsProductionMode():
    37  		return productionConfig()
    38  	default:
    39  		logger.Named("middleware.recover").Warn(
    40  			context.Background(),
    41  			"Unknown environment, using production config for recover middleware",
    42  			logging.String("env", appCfg.Mode()),
    43  		)
    44  		return productionConfig()
    45  	}
    46  }

    48  // newRecoverLogErrorFunc は、リカバリミドルウェアのログ出力関数を生成します。
    49  func newRecoverLogErrorFunc(logger logging.Logger, lf logging.LogFieldBuilder) func(c echo.Context, err error, stack []byte) error {
    50  	return func(c echo.Context, err error, stack []byte) error {
    51  		reqIn := server.BuildHTTPRequestLogInput(c, logging.EventTypePanic)
    52  		recoverFields := []*logging.Field{
    53  			logging.String(logging.InternalErrorKey, err.Error()),
    54  			logging.Strings(logging.InternalStackTraceKey, logging.SplitStackLines(string(stack))),
    55  		}
    56  		fields := append(lf.BuildHTTPRequestFields(reqIn), recoverFields...)
    57  		logger.Named("middleware.recover").Error(c.Request().Context(), "panic recovered", fields...)
    58  		// ログ済みを記録し err を返す（echo が c.Error で 500 を返す。二重ログは ctxhelper.GetRecoveredFromEcho で抑止）。
    59  		ctxhelper.SetRecoveredToEcho(c, true)
    60  		return err
    61  	}
    62  }
