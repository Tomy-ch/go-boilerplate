   116  func (s *server) PostPurchases(ctx context.Context, request gen.PostPurchasesRequestObject) (gen.PostPurchasesResponseObject, error) {
   117  	ctx, endSpan := s.tracer.Start(ctx)
   118  	defer endSpan()
   119  
   120  	userID, err := ctxhelper.RequireUserID(ctx)
   121  	if err != nil {

   123  	}
   124  
   125  	details := make([]purchaseuc.DetailParam, len(request.Body.Details))
   126  	for i, d := range request.Body.Details {
   127  		details[i] = purchaseuc.DetailParam{
   128  			ProductID: conv.UUID(d.ProductId),
   129  			Quantity:  int(d.Quantity),

   131  	}
   132  
   133  	var displayCurrency *string

   135  		s := string(*request.Params.DisplayCurrency)
   136  		displayCurrency = &s
   137  	}
   138  
   139  	view, _, err := idempotency.Run(ctx, s.idem, http.StatusCreated, func(ctx context.Context) (checkoutuc.PurchaseView, error) {
   140  		return s.checkout.CreatePurchase(ctx, checkoutuc.CreatePurchaseParams{
   141  			UserID:          userID,
   142  			Details:         details,
   143  			DisplayCurrency: displayCurrency,
   144  		})
   145  	})
   146  	if err != nil {
   147  		return nil, err
   148  	}
   149  
   150  	res, err := toPurchaseResponse(view.Purchase, view.ReferenceAmount)
   151  	if err != nil {
   152  		return nil, err
   153  	}
   154  	return gen.PostPurchases201JSONResponse(res), nil

   157  // toPurchaseResponse は、ユースケースの DTO を HTTP レスポンスへ変換します。
   158  // 数量が int32 に収まらない場合はエラーを返します。
   159  func toPurchaseResponse(v purchaseuc.PurchaseView, ref *checkoutuc.ReferenceAmountView) (gen.PurchaseResponse, error) {
   160  	details := make([]gen.PurchaseDetailResponse, len(v.Details))
   161  	for i, d := range v.Details {
   162  		quantity, err := safecast.IntToInt32(d.Quantity)
   163  		if err != nil {
   164  			return gen.PurchaseResponse{}, xerrors.Wrap(err, "invalid purchase detail quantity")
   165  		}
   166  		details[i] = gen.PurchaseDetailResponse{
   167  			ProductId: d.ProductID.ToPrimitive(),
   168  			Quantity:  quantity,
   169  			UnitPrice: d.UnitPrice.String(),
   170  		}
   171  	}
   172  
   173  	return gen.PurchaseResponse{
   174  		Code:            v.Code,
   175  		UserId:          v.UserID.ToPrimitive(),
   176  		StatusId:        v.StatusID.ToPrimitive(),
   177  		SubtotalAmount:  int64(v.SubtotalAmount),
   178  		TaxAmount:       int64(v.TaxAmount),
   179  		ShippingFee:     int64(v.ShippingFee),
   180  		TotalAmount:     int64(v.TotalAmount),
   181  		Details:         details,
   182  		OrderedAt:       v.OrderedAt,
   183  		ReferenceAmount: toReferenceAmount(ref),
   184  	}, nil
