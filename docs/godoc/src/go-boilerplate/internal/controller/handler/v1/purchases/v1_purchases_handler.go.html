   139  func toPurchaseResponse(v purchaseuc.PurchaseView, ref *checkoutuc.ReferenceAmountView) (gen.PurchaseResponse, error) {
   140  	details := make([]gen.PurchaseDetailResponse, len(v.Details))
   141  	for i, d := range v.Details {
   142  		quantity, err := safecast.IntToInt32(d.Quantity)
   143  		if err != nil {
   144  			return gen.PurchaseResponse{}, xerrors.Wrap(err, "invalid purchase detail quantity")
   145  		}
   146  		details[i] = gen.PurchaseDetailResponse{
   147  			ProductId: d.ProductID.ToPrimitive(),
   148  			Quantity:  quantity,
   149  			UnitPrice: d.UnitPrice.String(),
   150  		}
   151  	}
   152  
   153  	return gen.PurchaseResponse{
   154  		Id:              v.ID.ToPrimitive(),
   155  		Code:            v.Code,
   156  		UserId:          v.UserID.ToPrimitive(),
   157  		StatusId:        v.StatusID.ToPrimitive(),
   158  		SubtotalAmount:  int64(v.SubtotalAmount),
   159  		TaxAmount:       int64(v.TaxAmount),
   160  		ShippingFee:     int64(v.ShippingFee),
   161  		TotalAmount:     int64(v.TotalAmount),
   162  		Details:         details,
   163  		OrderedAt:       v.OrderedAt,
   164  		ReferenceAmount: toReferenceAmount(ref),

   166  }
