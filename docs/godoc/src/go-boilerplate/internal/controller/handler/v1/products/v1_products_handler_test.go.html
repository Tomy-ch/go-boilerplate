    57  // wantProductResponse は、本番 toProductResponse とは独立な検証用オラクル（フィールド取り違え検出）。
    58  func wantProductResponse(dto productuc.ProductView) gen.ProductResponse {
    59  	//nolint:gosec // G115: テストデータは int32 範囲内の固定値です
    60  	quantity := int32(dto.Quantity)
    61  	var threshold *int32
    62  	if dto.StockWarningThreshold != nil {
    63  		//nolint:gosec // G115: テストデータは int32 範囲内の固定値です
    64  		v := int32(*dto.StockWarningThreshold)
    65  		threshold = &v
    66  	}
    67  	return gen.ProductResponse{
    68  		Id:                    dto.ID.ToPrimitive(),
    69  		Name:                  dto.Name,
    70  		Description:           dto.Description,
    71  		Price:                 dto.Price.String(),
    72  		Quantity:              quantity,
    73  		StockWarningThreshold: threshold,
    74  		Status: gen.ProductStatusRef{
    75  			Id:   dto.StatusID.ToPrimitive(),
    76  			Name: dto.StatusName,
    77  		},
    78  		Category: gen.ProductCategoryRef{
    79  			Id:   dto.CategoryID.ToPrimitive(),
    80  			Name: dto.CategoryName,
    81  		},
    82  		PublishedAt: dto.PublishedAt,
    83  		Images:      expectedImageItems(dto.Images),
    84  		//nolint:gosec // G115: テストデータは int32 範囲内の固定値です
    85  		Version: int32(dto.Version),
    86  	}
    87  }

    89  // expectedImageItems は、期待レスポンスの商品画像を組み立てます。
    90  // production の写像と同じ順序・同じ値になることを固定するため、要素は素直に書き下します。
    91  func expectedImageItems(dtos []productuc.ProductImageItemView) []gen.ProductImageItem {
    92  	items := make([]gen.ProductImageItem, len(dtos))
    93  	for i, dto := range dtos {
    94  		//nolint:gosec // G115: テストデータは int32 範囲内の固定値です
    95  		items[i] = gen.ProductImageItem{ImagePath: dto.Path, DisplaySort: int32(dto.DisplaySort)}
    96  	}
    97  	return items
    98  }
