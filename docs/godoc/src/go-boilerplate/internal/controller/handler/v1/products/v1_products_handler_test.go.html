    48  func wantProductResponse(dto productuc.ProductView) gen.ProductResponse {
    49  	//nolint:gosec // G115: テストデータは int32 範囲内の固定値です
    50  	price, quantity := int32(dto.Price), int32(dto.Quantity)
    51  	var threshold *int32
    52  	if dto.StockWarningThreshold != nil {
    53  		//nolint:gosec // G115: テストデータは int32 範囲内の固定値です
    54  		v := int32(*dto.StockWarningThreshold)
    55  		threshold = &v
    56  	}
    57  	return gen.ProductResponse{
    58  		Id:                    dto.ID.ToPrimitive(),
    59  		Name:                  dto.Name,
    60  		Description:           dto.Description,
    61  		Price:                 price,

    64  		StatusId:              dto.StatusID.ToPrimitive(),
