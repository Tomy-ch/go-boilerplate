   117  // toCartItemResponses は、明細の DTO を HTTP レスポンスへ変換します。
   118  func toCartItemResponses(views []cartuc.CartItemView) ([]gen.CartItemResponse, error) {
   119  	items := make([]gen.CartItemResponse, len(views))
   120  	for i, v := range views {
   121  		quantity, err := safecast.IntToInt32(v.Quantity)
   122  		if err != nil {
   123  			return nil, xerrors.Wrap(err, "invalid cart item quantity")
   124  		}
   125  
   126  		available, err := toAvailableQuantity(v.AvailableQuantity)
   127  		if err != nil {
   128  			return nil, err
   129  		}
   130  
   131  		items[i] = gen.CartItemResponse{
   132  			ProductId:         v.ProductID.ToPrimitive(),
   133  			ProductName:       v.ProductName,
   134  			ImagePath:         v.ImagePath,
   135  			Quantity:          quantity,
   136  			UnitPrice:         toUnitPrice(v.UnitPrice),
   137  			Issues:            toCartItemIssues(v.Issues),
   138  			AvailableQuantity: available,
   139  		}
   140  	}
   141  	return items, nil
   142  }
