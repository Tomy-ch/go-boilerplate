   109  // toCartItemResponses は、明細の DTO を HTTP レスポンスへ変換します。
   110  func toCartItemResponses(views []cartuc.CartItemView) ([]gen.CartItemResponse, error) {
   111  	items := make([]gen.CartItemResponse, len(views))
   112  	for i, v := range views {
   113  		quantity, err := safecast.IntToInt32(v.Quantity)
   114  		if err != nil {
   115  			return nil, xerrors.Wrap(err, "invalid cart item quantity")
   116  		}
   117  
   118  		available, err := toAvailableQuantity(v.AvailableQuantity)
   119  		if err != nil {
   120  			return nil, err
   121  		}
   122  
   123  		items[i] = gen.CartItemResponse{
   124  			ProductId:         v.ProductID.ToPrimitive(),
   125  			ProductName:       v.ProductName,
   126  			ImagePath:         v.ImagePath,
   127  			Quantity:          quantity,
   128  			UnitPrice:         toUnitPrice(v.UnitPrice),
   129  			Issues:            toCartItemIssues(v.Issues),
   130  			AvailableQuantity: available,
   131  		}
   132  	}
   133  	return items, nil
   134  }

   139  func toAvailableQuantity(quantity *int) (*int32, error) {
