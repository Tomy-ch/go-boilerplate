    66  // wantProductResponse は、本番 toProductResponse とは独立な検証用オラクル（フィールド取り違え検出）。
    67  func wantProductResponse(dto productuc.ProductView) gen.ProductResponse {
    68  	//nolint:gosec // G115: テストデータは int32 範囲内の固定値です
    69  	quantity := int32(dto.Quantity)
    70  	var threshold *int32
    71  	if dto.StockWarningThreshold != nil {
    72  		//nolint:gosec // G115: テストデータは int32 範囲内の固定値です
    73  		v := int32(*dto.StockWarningThreshold)
    74  		threshold = &v
    75  	}
    76  	return gen.ProductResponse{
    77  		Id:                    dto.ID.ToPrimitive(),
    78  		Name:                  dto.Name,
    79  		Description:           dto.Description,
    80  		Price:                 dto.Price.String(),
    81  		Quantity:              quantity,
    82  		StockWarningThreshold: threshold,
    83  		Status: gen.ProductStatusRef{
    84  			Id:   dto.StatusID.ToPrimitive(),
    85  			Name: dto.StatusName,
    86  		},
    87  		Category: gen.ProductCategoryRef{
    88  			Id:   dto.CategoryID.ToPrimitive(),
    89  			Name: dto.CategoryName,
    90  		},
    91  		PublishedAt: dto.PublishedAt,
    92  		Images:      expectedImageItems(dto.Images),
    93  		//nolint:gosec // G115: テストデータは int32 範囲内の固定値です
    94  		Version: int32(dto.Version),
    95  	}
    96  }

   100  func expectedImageItems(dtos []productuc.ProductImageItemView) []gen.ProductImageItem {

   105  	}
   106  	return items
