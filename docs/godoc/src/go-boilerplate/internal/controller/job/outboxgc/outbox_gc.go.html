    61  // Execute は、retention を超えた published エントリをバッチ削除します。--batch-size=N で 1 バッチの件数を指定できます。
    62  // 途中で失敗した場合も、確定した削除件数を出力してからエラーを返します。
    63  func (j *jobImpl) Execute(ctx context.Context, args []string) error {
    64  	ctx, endSpan := j.tracer.Start(ctx)
    65  	defer endSpan()
    66  
    67  	batchSize, err := parseBatchSize(args)
    68  	if err != nil {
    69  		return err
    70  	}
    71  
    72  	deleted, err := j.gc.SweepPublished(ctx, batchSize)
    73  	if err != nil {
    74  		// 中断までにコミットされた削除は取り消せない。エラーだけを返すと消えた件数が運用者に届かないため、
    75  		// 確定した件数を記録してから伝播する。
    76  		j.logging.Named(jobName).Warn(ctx, abortedMessage, logging.Int64(logging.JobResultKey, deleted))
    77  		return err
    78  	}
    79  
    80  	j.logging.Named(jobName).Info(ctx, resultMessage, logging.Int64(logging.JobResultKey, deleted))
    81  	return nil
    82  }

    84  // parseBatchSize は、--batch-size=N フラグを解釈します。未指定なら 0（usecase 側で既定値）。
    85  func parseBatchSize(args []string) (int32, error) {
    86  	var batchSize int32
    87  	seen := false
    88  	for _, a := range args {
    89  		if !strings.HasPrefix(a, batchSizeFlagPrefix) {
    90  			return 0, xerrors.Wrap(errUnknownFlag, a)
    91  		}
    92  		if seen {
    93  			return 0, xerrors.Wrap(errDuplicateFlag, batchSizeFlagPrefix)
    94  		}
    95  		seen = true
    96  
    97  		n, err := strconv.ParseInt(strings.TrimPrefix(a, batchSizeFlagPrefix), 10, 32)
    98  		if err != nil || n <= 0 {
    99  			return 0, xerrors.Wrap(errInvalidBatchSize, a)
   100  		}
   101  		batchSize = int32(n)
   102  	}
   103  	return batchSize, nil
   104  }
