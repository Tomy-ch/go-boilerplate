    25  // NewRunner は、Runnerを生成します。
    26  func NewRunner(jobs []job.Job) (job.Runner, error) {
    27  	m := make(map[string]job.Job, len(jobs))
    28  	for _, j := range jobs {
    29  		name := j.Name()
    30  		if _, exists := m[name]; exists {
    31  			return nil, xerrors.Wrap(ErrDuplicateJob, name)
    32  		}
    33  		m[name] = j
    34  	}
    35  	return &runner{registry: m}, nil
    36  }

    38  // Run は、指定されたジョブを実行します。
    39  func (r *runner) Run(ctx context.Context, jobName string, args []string) error {
    40  	j, ok := r.registry[jobName]
    41  	if !ok {
    42  		return xerrors.Wrap(ErrUnknownJob, fmt.Sprintf("%s (available: %v)", jobName, r.Names()))
    43  	}
    44  	return j.Execute(ctx, args)
    45  }
