   190  			ctrl := gomock.NewController(t)

   192  			purge.EXPECT().PurgeDeleted(gomock.Any(), time.Duration(0), int32(0), false).
   193  				Return(user.PurgeResult{}, nil)
   194  
   195  			job := &jobImpl{logging: logging.NewTestLogger(t), tracer: tf.Controller(), purge: purge}
   196  			require.NoError(t, job.Execute(t.Context(), nil))
   197  		})
   198  
   199  		t.Run("3フラグの併用をそのまま usecase へ渡す", func(t *testing.T) {
   200  			t.Parallel()
   201  			ctrl := gomock.NewController(t)
   202  			purge := mock_user.NewMockPurgeUsecase(ctrl)
   203  			purge.EXPECT().PurgeDeleted(gomock.Any(), 720*time.Hour, int32(100), true).
   204  				Return(user.PurgeResult{}, nil)
   205  
   206  			job := &jobImpl{logging: logging.NewTestLogger(t), tracer: tf.Controller(), purge: purge}
   207  			require.NoError(t, job.Execute(t.Context(), []string{"--older-than=720h", "--batch-size=100", "--dry-run"}))
   208  		})
   209  
   210  		t.Run("削除件数とスキップ件数を結果ログのキーへ出力する", func(t *testing.T) {
   211  			t.Parallel()
   212  			ctrl := gomock.NewController(t)
   213  			purge := mock_user.NewMockPurgeUsecase(ctrl)
   214  			purge.EXPECT().PurgeDeleted(gomock.Any(), gomock.Any(), gomock.Any(), gomock.Any()).
   215  				Return(user.PurgeResult{Purged: 3, SkippedWithPurchases: 2}, nil)
   216  
   217  			log, logs := logging.NewObservedTestLogger(t)
   218  			job := &jobImpl{logging: log, tracer: tf.Controller(), purge: purge}
   219  			require.NoError(t, job.Execute(t.Context(), nil))
   220  
   221  			entries := logs.All()
   222  			require.Len(t, entries, 1)
   223  			assert.Equal(t, int64(3), entries[0].ContextMap()[logging.JobResultKey])
   224  			assert.Equal(t, int64(2), entries[0].ContextMap()[logging.JobSkippedKey])
   225  		})
   226  	})
