    69  // Execute は、退会から retention を過ぎたユーザーをバッチ物理削除し、件数をログに出力します。
    70  // --older-than=<duration> で保持期間を、--batch-size=N で 1 バッチの件数を指定でき、
    71  // --dry-run では削除せず対象件数だけを出力します。
    72  // 途中で失敗した場合も、確定した件数を出力してからエラーを返します。
    73  func (j *jobImpl) Execute(ctx context.Context, args []string) error {
    74  	ctx, endSpan := j.tracer.Start(ctx)
    75  	defer endSpan()
    76  
    77  	opts, err := parseArgs(args)
    78  	if err != nil {
    79  		return err
    80  	}
    81  
    82  	result, err := j.purge.PurgeDeleted(ctx, opts.retention, opts.batchSize, opts.dryRun)
    83  	fields := []*logging.Field{
    84  		logging.Int64(logging.JobResultKey, result.Purged),
    85  		logging.Int64(logging.JobSkippedKey, result.SkippedWithPurchases),
    86  	}
    87  	if err != nil {
    88  		// 中断までにコミットされた物理削除は取り消せない。エラーだけを返すと消えた件数が運用者に届かないため、
    89  		// 確定した件数を記録してから伝播する。
    90  		j.logging.Named(jobName).Warn(ctx, abortedMessage(opts.dryRun), fields...)
    91  		return err
    92  	}
    93  
    94  	j.logging.Named(jobName).Info(ctx, resultMessage(opts.dryRun), fields...)
    95  	return nil
    96  }

    98  // parseArgs は、--older-than=<duration> / --batch-size=N / --dry-run を解釈します。
    99  // 未指定の retention / batchSize は 0（usecase 側で既定値）です。
   100  // 同一フラグの重複・未知フラグは、後勝ちで黙殺せずエラーにします。
   101  func parseArgs(args []string) (options, error) {
   102  	var (
   103  		opts          options
   104  		seenOlderThan bool
   105  		seenBatchSize bool
   106  	)
   107  	for _, a := range args {
   108  		switch {
   109  		case strings.HasPrefix(a, olderThanFlagPrefix):
   110  			if seenOlderThan {
   111  				return options{}, xerrors.Wrap(errDuplicateFlag, olderThanFlagPrefix)
   112  			}
   113  			seenOlderThan = true
   114  
   115  			d, err := time.ParseDuration(strings.TrimPrefix(a, olderThanFlagPrefix))
   116  			if err != nil || d <= 0 {
   117  				return options{}, xerrors.Wrap(errInvalidOlderThan, a)
   118  			}
   119  			opts.retention = d
   120  		case strings.HasPrefix(a, batchSizeFlagPrefix):
   121  			if seenBatchSize {
   122  				return options{}, xerrors.Wrap(errDuplicateFlag, batchSizeFlagPrefix)
   123  			}
   124  			seenBatchSize = true
   125  
   126  			n, err := strconv.ParseInt(strings.TrimPrefix(a, batchSizeFlagPrefix), 10, 32)
   127  			if err != nil || n <= 0 {
   128  				return options{}, xerrors.Wrap(errInvalidBatchSize, a)
   129  			}
   130  			opts.batchSize = int32(n)
   131  		case a == dryRunFlag:
   132  			if opts.dryRun {
   133  				return options{}, xerrors.Wrap(errDuplicateFlag, dryRunFlag)
   134  			}
   135  			opts.dryRun = true
   136  		default:
   137  			return options{}, xerrors.Wrap(errUnknownFlag, a)
   138  		}
   139  	}
   140  	return opts, nil
   141  }
