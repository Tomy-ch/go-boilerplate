   136  // lookupErrorMetaByAppError は、アプリケーションエラーに対応するエラーメタデータを取得します。
   137  // 既知のエラー型に一致しない場合は 500 Internal Server Error のメタを返します。
   138  func lookupErrorMetaByAppError(err error) httpErrorMeta {
   139  	switch {
   140  	case xerrors.Is(err, apperror.ErrInvalidArgument): // 400
   141  		return lookupErrorMetaByHTTPStatus(http.StatusBadRequest)
   142  	case xerrors.Is(err, apperror.ErrValidation): // 422
   143  		return lookupErrorMetaByHTTPStatus(http.StatusUnprocessableEntity)
   144  	case xerrors.Is(err, apperror.ErrUnauthenticated): // 401
   145  		return lookupErrorMetaByHTTPStatus(http.StatusUnauthorized)
   146  	case xerrors.Is(err, apperror.ErrPermissionDenied): // 403
   147  		return lookupErrorMetaByHTTPStatus(http.StatusForbidden)
   148  	case xerrors.Is(err, apperror.ErrNotFound): // 404
   149  		return lookupErrorMetaByHTTPStatus(http.StatusNotFound)
   150  	case xerrors.Is(err, apperror.ErrConflict): // 409
   151  		return lookupErrorMetaByHTTPStatus(http.StatusConflict)
   152  	case xerrors.Is(err, apperror.ErrCanceled): // 499
   153  		return lookupErrorMetaByHTTPStatus(statusClientClosedRequest)
   154  	case xerrors.Is(err, apperror.ErrUnavailable): // 503
   155  		return lookupErrorMetaByHTTPStatus(http.StatusServiceUnavailable)
   156  	case xerrors.Is(err, apperror.ErrUnimplemented): // 501
   157  		return lookupErrorMetaByHTTPStatus(http.StatusNotImplemented)
   158  	case xerrors.Is(err, apperror.ErrTooManyRequests): // 429
   159  		return lookupErrorMetaByHTTPStatus(http.StatusTooManyRequests)
   160  	case xerrors.Is(err, apperror.ErrInternal): // 500
   161  		return lookupErrorMetaByHTTPStatus(http.StatusInternalServerError)
   162  	default:
   163  		return lookupErrorMetaByHTTPStatus(http.StatusInternalServerError)
