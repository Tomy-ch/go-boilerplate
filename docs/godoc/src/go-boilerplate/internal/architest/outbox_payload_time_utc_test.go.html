    78  // collectOutboxPayloadTimeViolations は、root 配下の event パッケージを走査して、時刻を自分で整形して
    79  // いる行を集めます。並びはパッケージ・ファイル・行番号の順で、同じツリーからは常に同じ結果になります。
    80  func collectOutboxPayloadTimeViolations(root string) ([]string, error) {
    81  	dirs, err := collectEventPackages(root)
    82  	if err != nil {
    83  		return nil, err
    84  	}
    85  
    86  	violations := []string{}
    87  	for _, dir := range dirs {
    88  		paths, perr := plainGoFiles(dir)
    89  		if perr != nil {
    90  			return nil, perr
    91  		}
    92  
    93  		for _, path := range paths {
    94  			found, ferr := collectFileTimeViolations(root, path)
    95  			if ferr != nil {
    96  				return nil, ferr
    97  			}
    98  			violations = append(violations, found...)
    99  		}
   100  	}
   101  
   102  	return violations, nil
   103  }

   105  // collectFileTimeViolations は、ファイル 1 つ分の違反行を `<相対パス>:<行番号>: <行>` の形で返します。
   106  func collectFileTimeViolations(root, path string) ([]string, error) {
   107  	lines, err := readLines(path)

   115  // timeViolationsInLines は、時刻を自分で整形している行を `<rel>:<行番号>: <行>` の形で返します。
   116  func timeViolationsInLines(rel string, lines []string) []string {
   117  	violations := []string{}
   118  	for i, line := range lines {
   119  		if !strings.Contains(line, timeFormatCall) {
   120  			continue
   121  		}
   122  		violations = append(violations, fmt.Sprintf("%s:%d: %s", rel, i+1, strings.TrimSpace(line)))
   123  	}
   124  
   125  	return violations
   126  }
