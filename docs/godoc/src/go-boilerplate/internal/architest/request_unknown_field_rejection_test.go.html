    34  // TestRequestBodyRejectsUnknownFields は、リクエストボディから到達できる全ての object スキーマが
    35  // 未知フィールドを拒否することを機械検証する。この規則は
    36  // openapi/components/requests/README.md と openapi/components/schemas/README.md の Rules が定める。
    37  //
    38  // 突き合わせる集合は「リクエストボディから到達可能な object スキーマ」と「未知フィールドを実効的に
    39  // 拒否するスキーマ」で、前者 ⊆ 後者 を表明する。宣言が無いスキーマは契約に書かれていないフィールドを
    40  // 黙って受け取るため、クライアント側のタイポや古いフィールド名が無視される。
    41  //
    42  // 検査は requestBody 直下で止めず properties / items / allOf を再帰する。ネストした要素スキーマ
    43  // （購入明細・商品画像）はトップレベルの宣言に守られず、そこだけ素通りが残るためである。
    44  //
    45  // 読み出すのは生成物ではなく手書き正本の openapi.yaml で、理由は TestRouteSpecParity と同じ。
    46  //
    47  // 対象が 0 件でも失敗させない。サンプル API を撤去すると requestBody を持つ operation が
    48  // 残らないためで、0 件と壊れた走査を区別するのは Test_walkRequestSchema と Test_evaluateClosure
    49  // が担う（internal/architest/README.md の Notes）。
    50  func TestRequestBodyRejectsUnknownFields(t *testing.T) {
    51  	t.Parallel()
    52  
    53  	subjects := collectRequestBodySchemas(t)
    54  	t.Logf("検査した object スキーマ: %d 件", len(subjects))
    55  
    56  	missing := make([]string, 0, len(subjects))
    57  	blind := make([]string, 0, len(subjects))
    58  	for _, subject := range subjects {
    59  		verdict, fields := evaluateClosure(subject.schema)
    60  		switch verdict {
    61  		case closureMissing:
    62  			missing = append(missing, fmt.Sprintf("%s (properties: %v)", subject.location, fields))
    63  		case closureBlind:
    64  			blind = append(blind, fmt.Sprintf("%s (見えていない properties: %v)", subject.location, fields))
    65  		case closureNotSubject, closureEffective:
    66  		}
    67  	}
    68  	sort.Strings(missing)
    69  	sort.Strings(blind)
    70  
    71  	assert.Empty(t, missing,
    72  		"リクエストボディから到達する object スキーマに additionalProperties: false が無い。"+
    73  			"契約に無いフィールドを送っても拒否されないため、宣言を追加すること")
    74  	assert.Empty(t, blind,
    75  		"additionalProperties: false が、同じスキーマオブジェクトに無い properties を見ていない。"+
    76  			"allOf の兄弟が持つプロパティは宣言から見えず正当な入力まで拒否されるため、"+
    77  			"宣言は properties を持つスキーマオブジェクト自身に置くこと")
    78  }

    80  // evaluateClosure は、allOf を展開した composition 全体として未知フィールドを拒否できているかを判定します。
    81  // 第 2 戻り値は、closureMissing では守られていない properties、closureBlind では宣言から見えていない
    82  // properties で、いずれも辞書順です。
    83  //
    84  // additionalProperties は JSON Schema の意味論で「同じスキーマオブジェクトの properties に一致しない
    85  // フィールド」だけを見るため、宣言と properties が別のスキーマオブジェクトに分かれていると、
    86  // 拒否しすぎるか（closureBlind）拒否できないかのどちらかになります。
    87  func evaluateClosure(schema *openapi3.Schema) (closureVerdict, []string) {
    88  	members := flattenAllOf(schema, nil, map[*openapi3.Schema]struct{}{})
    89  
    90  	properties := map[string]struct{}{}
    91  	closures := make([]*openapi3.Schema, 0, len(members))
    92  	for _, member := range members {
    93  		for name := range member.Properties {
    94  			properties[name] = struct{}{}
    95  		}
    96  		if member.AdditionalProperties.Has != nil && !*member.AdditionalProperties.Has {
    97  			closures = append(closures, member)
    98  		}
    99  	}
   100  
   101  	if len(properties) == 0 {
   102  		return closureNotSubject, nil
   103  	}
   104  	if len(closures) == 0 {
   105  		return closureMissing, sortedNames(properties)
   106  	}
   107  
   108  	unseen := map[string]struct{}{}
   109  	for _, closure := range closures {
   110  		for name := range properties {
   111  			if _, ok := closure.Properties[name]; !ok {
   112  				unseen[name] = struct{}{}
   113  			}
   114  		}
   115  	}
   116  	if len(unseen) > 0 {
   117  		return closureBlind, sortedNames(unseen)
   118  	}
   119  	return closureEffective, nil
   120  }

   122  // flattenAllOf は、スキーマ自身と allOf の枝を再帰的に平坦化して返します。
   123  // composition は allOf の全ての枝を同時に満たす必要があるため、宣言と properties の所在は
   124  // 枝をまたいで数えます。visited は allOf の循環に対する保険です。
   125  func flattenAllOf(
   126  	schema *openapi3.Schema, acc []*openapi3.Schema, visited map[*openapi3.Schema]struct{},
   127  ) []*openapi3.Schema {
   128  	if schema == nil {
   129  		return acc
   130  	}
   131  	if _, seen := visited[schema]; seen {
   132  		return acc
   133  	}
   134  	visited[schema] = struct{}{}
   135  
   136  	acc = append(acc, schema)
   137  	for _, branch := range schema.AllOf {
   138  		if branch != nil {
   139  			acc = flattenAllOf(branch.Value, acc, visited)
   140  		}
   141  	}
   142  	return acc
   143  }

   145  // collectRequestBodySchemas は、手書き正本の spec を読み、全 operation のリクエストボディから
   146  // 到達できる object スキーマを報告用の位置つきで返します。
   147  func collectRequestBodySchemas(t *testing.T) []requestSchemaSubject {
   148  	t.Helper()
   149  
   150  	loader := &openapi3.Loader{IsExternalRefsAllowed: true}
   151  	spec, err := loader.LoadFromFile(filepath.Join(moduleRoot(t), filepath.FromSlash(specSourceFile)))
   152  	require.NoError(t, err)
   153  	require.NotNil(t, spec.Paths)
   154  
   155  	subjects := make([]requestSchemaSubject, 0)
   156  	visited := map[*openapi3.Schema]struct{}{}
   157  	for _, path := range sortedNames(pathKeys(spec.Paths)) {
   158  		item := spec.Paths.Value(path)
   159  		operations := item.Operations()
   160  		for _, method := range sortedNames(operationKeys(operations)) {
   161  			body := operations[method].RequestBody
   162  			if body == nil || body.Value == nil {
   163  				continue
   164  			}
   165  			for _, mediaType := range sortedNames(contentKeys(body.Value.Content)) {
   166  				subjects = walkRequestSchema(
   167  					fmt.Sprintf("%s %s [%s]", method, path, mediaType),
   168  					body.Value.Content[mediaType].Schema, visited, subjects,
   169  				)
   170  			}
   171  		}
   172  	}
   173  	return subjects
   174  }

   176  // walkRequestSchema は、スキーマを再帰的に辿り object スキーマを収集します。
   177  // visited は $ref の循環による無限再帰を防ぎます。この spec の $ref はファイルパス参照で、
   178  // kin-openapi は参照ごとに別のスキーマを確保するため、同じ定義を複数の operation から
   179  // 参照しても重複排除はされません（同じ内容が別の位置で 2 度検査されるだけで、判定は変わりません）。
   180  //
   181  // not は辿りません。not が述べるのは拒否したい形であって受理できるフィールドの集合ではなく、
   182  // そこへ additionalProperties: false を要求すると、排他制約を書いただけの spec を違反と判定します。
   183  func walkRequestSchema(
   184  	location string, ref *openapi3.SchemaRef,
   185  	visited map[*openapi3.Schema]struct{}, acc []requestSchemaSubject,
   186  ) []requestSchemaSubject {
   187  	if ref == nil || ref.Value == nil {
   188  		return acc
   189  	}
   190  	schema := ref.Value
   191  	if _, seen := visited[schema]; seen {
   192  		return acc
   193  	}
   194  	visited[schema] = struct{}{}
   195  
   196  	if len(schema.Properties) > 0 || len(schema.AllOf) > 0 {
   197  		acc = append(acc, requestSchemaSubject{location: location, schema: schema})
   198  	}
   199  
   200  	for _, name := range sortedNames(schema.Properties) {
   201  		acc = walkRequestSchema(location+".properties."+name, schema.Properties[name], visited, acc)
   202  	}
   203  	acc = walkRequestSchema(location+".items", schema.Items, visited, acc)
   204  	for i, branch := range schema.AllOf {
   205  		acc = walkRequestSchema(fmt.Sprintf("%s.allOf[%d]", location, i), branch, visited, acc)
   206  	}
   207  	for i, branch := range schema.OneOf {
   208  		acc = walkRequestSchema(fmt.Sprintf("%s.oneOf[%d]", location, i), branch, visited, acc)
   209  	}
   210  	for i, branch := range schema.AnyOf {
   211  		acc = walkRequestSchema(fmt.Sprintf("%s.anyOf[%d]", location, i), branch, visited, acc)
   212  	}
   213  	return walkRequestSchema(location+".additionalProperties", schema.AdditionalProperties.Schema, visited, acc)
   214  }
