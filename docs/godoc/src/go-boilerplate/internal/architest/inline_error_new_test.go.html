    78  // collectInlineXerrorsNew は、gofmt 済みソースの行列から、package-level の var 宣言以外の位置に
    79  // 現れる xerrors.New 呼び出しを `file:line: 該当行` 形式で列挙します。
    80  //
    81  // gofmt は package-level の var ブロックを行頭 `var (` … 行頭 `)` に整形するため、その区間だけを
    82  // 許可領域とみなします。関数本体内の var ブロックは行頭がタブになるので許可領域に入りません。
    83  // 許可領域の中でも、宣言行そのものの形に一致しない位置（関数リテラルの内側など）は違反として扱います。
    84  // 行コメントは対象外です。
    85  func collectInlineXerrorsNew(lines []string, file string) []string {
    86  	var violations []string
    87  	inVarBlock := false
    88  
    89  	for i, line := range lines {
    90  		switch {
    91  		case strings.HasPrefix(line, "var ("):
    92  			inVarBlock = true
    93  			continue
    94  		case inVarBlock && strings.HasPrefix(line, ")"):
    95  			inVarBlock = false
    96  			continue
    97  		}
    98  		// 行コメントは規約の解説そのものが `xerrors.New(` の字面を含むため、実呼び出しと区別する。
    99  		// 行途中のコメント・文字列リテラル内の字面までは見分けないが、誤検出の現実的な発生源は行コメント。
   100  		if !strings.Contains(line, xerrorsNewCall) || strings.HasPrefix(strings.TrimSpace(line), "//") {
   101  			continue
   102  		}
   103  		// ブロック内でも宣言行の形（1 段インデントの直接代入）に一致するものだけを許可する。
   104  		// ブロック全体を無条件に許可すると、関数リテラルへ包んだ動的生成を見逃す。
   105  		if inVarBlock && blockSentinelDeclRe.MatchString(line) {
   106  			continue
   107  		}
   108  		if !inVarBlock && sentinelDeclRe.MatchString(line) {
   109  			continue
   110  		}
   111  		violations = append(violations, fmt.Sprintf("%s:%d: %s", file, i+1, strings.TrimSpace(line)))
   112  	}
   113  	return violations
   114  }
