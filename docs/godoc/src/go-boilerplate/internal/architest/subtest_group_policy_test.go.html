    18  // 判定ロジックのテストは違反そのものを合成入力として書くため、自己走査すると自分のテストデータを違反として拾います。

    21  var (
    22  	// testFuncDeclRe は、レシーバを持たないトップレベル関数の宣言を検出する。
    23  	// テスト関数以外にもマッチするのは意図的で、ヘルパー関数へ入った時点で走査を止めるために名前を捕捉する。
    24  	testFuncDeclRe = regexp.MustCompile(`^func (\w+)\(`)
    25  	// wrappedRunRe は、引数が同一行に無い t.Run 呼び出しを検出する。
    26  	// ケース名を次行へ折り返すとインデント段数による構造判定が成立しないため、解析不能として違反にする（黙って読み飛ばすと検出漏れになる）。
    27  	wrappedRunRe = regexp.MustCompile(`^\t+t\.Run\($`)
    28  	// topLevelRunRe は、TestXxx 直下（インデント 1 段）の t.Run を検出する。
    29  	topLevelRunRe = regexp.MustCompile(`^\tt\.Run\(`)
    30  	// literalGroupRe は、最外周グループとして許容される唯一の形にマッチする。
    31  	literalGroupRe = regexp.MustCompile(`^\tt\.Run\("(正常系|異常系)", func\(`)
    32  	// nestedGroupRe は、グループの内側（インデント 2 段以上）でサブケース名が 正常系 / 異常系 で始まるものを検出する。
    33  	// 規約はグループ内でのさらなるネストを許すため、深さを 2 段に固定すると 3 段目以降のプレフィックス形式を取りこぼす。
    34  	nestedGroupRe = regexp.MustCompile(`^\t{2,}t\.Run\("(正常系|異常系)`)

    37  // TestSubtestGroupPolicy は、テストのサブテスト構造が規約どおりであることを機械検証する。
    38  // t.Run サブケースを持つ TestXxx の最外周はリテラル 正常系 / 異常系 に限り、
    39  // 挙動文の直置き・正常系_xxx のプレフィックス形式・境界ケースのような第 3 のグループはいずれも違反になる。
    40  // 単一シナリオの TestXxx はグループを省略してよいため、t.Run を 1 つも持たない関数は対象外。
    41  //
    42  // 検出は depguard が go/ast を禁じるためテキスト走査で行う（既存 architest と同方針）。
    43  // gofmt 済みであることを前提に、インデント段数でネスト深さを判定する。
    44  // どちら側のグループへ属すべきかは意味の判断であり機械検証できないため、構造のみを対象とし、
    45  // 帰属は testing-conventions.md §10 とレビューが担う。構造違反を許す allowlist は持たない。
    46  func TestSubtestGroupPolicy(t *testing.T) {
    47  	t.Parallel()
    48  
    49  	var violations []string
    50  	scanned := 0
    51  
    52  	for _, root := range moduleSubdirs(t, "internal", "pkg") {
    53  		err := filepath.WalkDir(root, func(path string, d fs.DirEntry, werr error) error {
    54  			if werr != nil {
    55  				return werr
    56  			}
    57  			if d.IsDir() || !strings.HasSuffix(path, "_test.go") || filepath.Base(path) == subtestGroupSelfFile {
    58  				return nil
    59  			}
    60  			src, rerr := pkgfs.OS{}.ReadFile(path)
    61  			if rerr != nil {
    62  				return rerr
    63  			}
    64  			scanned++
    65  			violations = append(violations, collectSubtestGroupViolations(path, strings.Split(string(src), "\n"))...)
    66  			return nil
    67  		})
    68  		require.NoError(t, err)
    69  	}
    70  
    71  	// 走査対象が 0 件だと検証が常に成功してしまうため、ルート解決の破綻を空振りとして検出する。
    72  	require.NotZero(t, scanned, "走査対象のテストファイルが 0 件（moduleSubdirs のルート解決を疑う）")
    73  
    74  	sort.Strings(violations)
    75  	for _, v := range violations {
    76  		t.Log("サブテスト構造の違反: " + v)
    77  	}
    78  
    79  	require.Empty(t, violations,
    80  		"サブテストのグループ構造が規約に反している。t.Run を使う TestXxx の最外周はリテラル 正常系 / 異常系 に限り、"+
    81  			"分割の余地がない単一シナリオはグループごと省略すること。")
    82  }

    84  // collectSubtestGroupViolations は、規約に反するサブテストのグループ構造を違反として列挙する。
    85  func collectSubtestGroupViolations(file string, lines []string) []string {
    86  	var violations []string
    87  	inTestFunc := false
    88  
    89  	for i, line := range lines {
    90  		if m := testFuncDeclRe.FindStringSubmatch(line); m != nil {
    91  			inTestFunc = strings.HasPrefix(m[1], "Test")
    92  			continue
    93  		}
    94  		if !inTestFunc {
    95  			continue
    96  		}
    97  
    98  		switch {
    99  		case wrappedRunRe.MatchString(line):
   100  			violations = append(violations, describe(file, i, line,
   101  				"t.Run の引数が同一行に無く構造を解析できない"))
   102  		case topLevelRunRe.MatchString(line) && !literalGroupRe.MatchString(line):
   103  			violations = append(violations, describe(file, i, line,
   104  				"TestXxx 直下の t.Run はリテラル 正常系 / 異常系 のみ"))
   105  		case nestedGroupRe.MatchString(line):
   106  			violations = append(violations, describe(file, i, line,
   107  				"グループ内のサブケース名に 正常系 / 異常系 を含めない"))
   108  		}
   109  	}
   110  	return violations
   111  }
