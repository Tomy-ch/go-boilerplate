    24  // TestIdempotencyCompleteness は、OpenAPI で Idempotency-Key ヘッダを宣言した操作
    25  // （= 生成コードの <Op>Params 型が IdempotencyKey フィールドを持つ操作）のハンドラが、
    26  // 必ず idempotency.Run 経由で処理していることを機械検証する完全性テストです。
    27  //
    28  // 冪等性は「変更系すべて」ではなく「Idempotency-Key を宣言した操作」に閉じる（PUT/DELETE は
    29  // HTTP 的に冪等で Run を要さない）ため、OpenAPI のヘッダ宣言を唯一のマーカー（source of truth）と
    30  // して扱う。middleware 登録済みで Run 呼び忘れ、のような silent な dedup 欠落を loud な失敗に変える。
    31  //
    32  // 本リポジトリは depguard で go/ast 等のツールチェーンパッケージを禁止するため、AST ではなく
    33  // gofmt 済みソースのテキスト走査で検出する（`type ...Params struct {` / `func (recv) Name(` は
    34  // gofmt により行頭固定なので、この走査は安定する）。
    35  func TestIdempotencyCompleteness(t *testing.T) {
    36  	t.Parallel()
    37  
    38  	root := handlerRoot(t)
    39  
    40  	marked := map[string]string{}    // operationID -> 宣言元ファイル（Idempotency-Key を宣言する操作）
    41  	wrapped := map[string]struct{}{} // idempotency.Run を呼ぶハンドラメソッド名
    42  
    43  	err := filepath.WalkDir(root, func(path string, d fs.DirEntry, walkErr error) error {
    44  		if walkErr != nil {
    45  			return walkErr
    46  		}
    47  		if d.IsDir() || !strings.HasSuffix(path, ".go") || strings.HasSuffix(path, "_test.go") {
    48  			return nil
    49  		}
    50  
    51  		src, readErr := pkgfs.OS{}.ReadFile(path)
    52  		if readErr != nil {
    53  			return readErr
    54  		}
    55  		lines := strings.Split(string(src), "\n")
    56  
    57  		if strings.HasSuffix(path, ".gen.go") {
    58  			collectMarkedParams(lines, path, marked)
    59  		} else {
    60  			collectRunWrapped(lines, wrapped)
    61  		}
    62  		return nil
    63  	})
    64  	require.NoError(t, err)
    65  
    66  	// マーカー検出ロジックが生成コードの命名変更等で陳腐化して空振りすると、完全性検証が
    67  	// 常に成功してしまう。少なくとも1件は検出できることを保証する。
    68  	require.NotEmpty(t, marked, "Idempotency-Key を宣言する操作が生成コードから1件も検出できない（マーカー検出の陳腐化を疑う）")
    69  
    70  	for op, file := range marked {
    71  		_, ok := wrapped[op]
    72  		assert.Truef(t, ok,
    73  			"操作 %s は Idempotency-Key を宣言している（%s）が、ハンドラが idempotency.Run を呼んでいない", op, file)
    74  	}
    75  }

    86  // collectMarkedParams は、`type <Op>Params struct {` ブロックが IdempotencyKey フィールドを
    87  // 含むとき Op を marked へ加えます。gofmt 済みソースでは閉じ括弧が行頭 `}` に現れる前提です。
    88  func collectMarkedParams(lines []string, file string, marked map[string]string) {
    89  	for i, line := range lines {
    90  		m := paramsStructRe.FindStringSubmatch(line)
    91  		if m == nil {
    92  			continue
    93  		}
    94  		op := m[1]
    95  		for _, body := range lines[i+1:] {
    96  			if body == "}" {
    97  				break
    98  			}
    99  			if strings.Contains(body, "IdempotencyKey") {
   100  				marked[op] = file
   101  				break
   102  			}
   103  		}
   104  	}
   105  }

   107  // collectRunWrapped は、メソッド宣言ブロック内に idempotency.Run 呼び出しがあるとき、その
   108  // メソッド名を wrapped へ加えます。ブロックは行頭 `func ` から次の行頭 `func ` までとします。
   109  //
   110  // 検出は import 別名なしの直接呼び出し `idempotency.Run(` に限る。別名 import（`idem.Run(` 等）や
   111  // 変数へ束ねてからの間接呼び出しは検出できない。現行ハンドラは直接呼び出しのみのため問題ないが、
   112  // この規約から外れると完全性検証が偽陰性になりうる点に注意。
   113  func collectRunWrapped(lines []string, wrapped map[string]struct{}) {
   114  	currentMethod := ""
   115  	for _, line := range lines {
   116  		if m := handlerMethodRe.FindStringSubmatch(line); m != nil {
   117  			currentMethod = m[1]
   118  			continue
   119  		}
   120  		if strings.HasPrefix(line, "func ") {
   121  			currentMethod = "" // メソッド以外の関数宣言に入ったらリセット
   122  			continue
   123  		}
   124  		if currentMethod != "" && strings.Contains(line, "idempotency.Run(") {
   125  			wrapped[currentMethod] = struct{}{}
   126  		}
   127  	}
   128  }
