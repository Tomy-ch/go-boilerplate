    24  // TestIdempotencyCompleteness は、OpenAPI で Idempotency-Key を宣言した操作
    25  // （= 生成コードの <Op>Params 型が IdempotencyKey フィールドを持つ操作）のハンドラが、
    26  // 必ず idempotency.Run 経由で処理していることを機械検証する完全性テストです。
    27  // OpenAPI のヘッダ宣言を source of truth とし、Run 呼び忘れのような silent な dedup 欠落を loud な失敗に変える。
    28  //
    29  // 対象操作は 0 件でも許容する（サンプル API 削除後は該当が無くなり得る）が、検出ロジックの空振りを
    30  // 防ぐため、param 検出用の正規表現が既知の生成形にマッチすることを別途 assert する。depguard が
    31  // go/ast を禁止するため、AST ではなく gofmt 済みソースのテキスト走査で検出する。
    32  func TestIdempotencyCompleteness(t *testing.T) {
    33  	t.Parallel()
    34  
    35  	root := handlerRoot(t)
    36  
    37  	marked := map[string]string{}    // operationID -> 宣言元ファイル（Idempotency-Key を宣言する操作）
    38  	wrapped := map[string]struct{}{} // idempotency.Run を呼ぶハンドラメソッド名
    39  	totalParamsSeen := 0             // 走査中に見た `<Op>Params struct` の総数（検出健全性の指標）
    40  
    41  	err := filepath.WalkDir(root, func(path string, d fs.DirEntry, walkErr error) error {
    42  		if walkErr != nil {
    43  			return walkErr
    44  		}
    45  		if d.IsDir() || !strings.HasSuffix(path, ".go") || strings.HasSuffix(path, "_test.go") {
    46  			return nil
    47  		}
    48  
    49  		src, readErr := pkgfs.OS{}.ReadFile(path)
    50  		if readErr != nil {
    51  			return readErr
    52  		}
    53  		lines := strings.Split(string(src), "\n")
    54  
    55  		if strings.HasSuffix(path, ".gen.go") {
    56  			totalParamsSeen += collectMarkedParams(lines, path, marked)
    57  		} else {
    58  			collectRunWrapped(lines, wrapped)
    59  		}
    60  		return nil
    61  	})
    62  	require.NoError(t, err)
    63  
    64  	// マーカー検出ロジックが生成コードの命名変更等で陳腐化して空振りすると、完全性検証が
    65  	// 常に成功してしまう。これを防ぐため、param 検出の正規表現が既知の生成形にマッチすることを
    66  	// 直接検証する。サンプル API を全削除すると param を持つ操作は 0 件になり得る（totalParamsSeen==0）
    67  	// ため、リポジトリ走査の件数ではなく正規表現の自己検査で陳腐化を検出する。
    68  	t.Logf("param 構造体の検出数: %d（サンプル全削除後は 0 になり得る）", totalParamsSeen)
    69  	require.NotEmpty(t, paramsStructRe.FindStringSubmatch("type XxxParams struct {"),
    70  		"paramsStructRe が `<Op>Params struct {` 形にマッチしない（正規表現の陳腐化を疑う）")
    71  
    72  	for op, file := range marked {
    73  		_, ok := wrapped[op]
    74  		assert.Truef(t, ok,
    75  			"操作 %s は Idempotency-Key を宣言している（%s）が、ハンドラが idempotency.Run を呼んでいない", op, file)
    76  	}
    77  }

    88  // collectMarkedParams は、`type <Op>Params struct {` ブロックが IdempotencyKey フィールドを
    89  // 含むとき Op を marked へ加えます。gofmt 済みソースでは閉じ括弧が行頭 `}` に現れる前提です。
    90  // 戻り値は、IdempotencyKey の有無に依らず走査中に検出した `<Op>Params struct` の件数で、
    91  // 呼び出し側の検出健全性チェック（正規表現の陳腐化検出）に用います。
    92  func collectMarkedParams(lines []string, file string, marked map[string]string) int {
    93  	seen := 0
    94  	for i, line := range lines {
    95  		m := paramsStructRe.FindStringSubmatch(line)
    96  		if m == nil {
    97  			continue
    98  		}
    99  		seen++
   100  		op := m[1]
   101  		for _, body := range lines[i+1:] {
   102  			if body == "}" {
   103  				break
   104  			}
   105  			if strings.Contains(body, "IdempotencyKey") {
   106  				marked[op] = file
   107  				break
   108  			}
   109  		}
   110  	}
   111  	return seen
   112  }

   114  // collectRunWrapped は、メソッド宣言ブロック内に idempotency.Run 呼び出しがあるとき、その
   115  // メソッド名を wrapped へ加えます。ブロックは行頭 `func ` から次の行頭 `func ` までとします。
   116  //
   117  // 検出は import 別名なしの直接呼び出し `idempotency.Run(` に限る。別名 import（`idem.Run(` 等）や
   118  // 変数へ束ねてからの間接呼び出しは検出できない。現行ハンドラは直接呼び出しのみのため問題ないが、
   119  // この規約から外れると完全性検証が偽陰性になりうる点に注意。
   120  func collectRunWrapped(lines []string, wrapped map[string]struct{}) {
   121  	currentMethod := ""
   122  	for _, line := range lines {
   123  		if m := handlerMethodRe.FindStringSubmatch(line); m != nil {
   124  			currentMethod = m[1]
   125  			continue
   126  		}
   127  		if strings.HasPrefix(line, "func ") {
   128  			currentMethod = "" // メソッド以外の関数宣言に入ったらリセット
   129  			continue
   130  		}
   131  		if currentMethod != "" && strings.Contains(line, "idempotency.Run(") {
   132  			wrapped[currentMethod] = struct{}{}
   133  		}
   134  	}
   135  }
