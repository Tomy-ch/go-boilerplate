    17  // 命名規約: メソッドは `Test<Type>_<Method>`（Type が非公開なら `Test_<type>_<Method>`）、
    18  // 自由関数は `Test<Func>`（非公開なら `Test_<func>`）。depguard が go/ast を禁じるため、

    24  	branchRe     = regexp.MustCompile(`^\t+(if|for|switch|select)[ ({]`)

    34  // TestUnitTestMappingCompleteness は、分岐（if/for/switch/select）を持つ production の
    35  // func / method が、命名規約に一致する TestXxx を同一パッケージに持つことを機械検証する。
    36  // テスト対象と TestXxx の 1:1 を強制し、レビュー頼みの見落とし（例: keyFunc の防御分岐が
    37  // 呼び出し側テストに埋もれて専用テストが無い状態）を loud な失敗に変える。
    38  //
    39  // 「1:1 を採らない」ことは許さない。真にユニットテスト不要 / 現時点で書けない対象も、
    40  // 命名規約どおりの TestXxx を宣言し中で t.Skip("理由") することで意図を明示する
    41  // （allowlist は持たない。理由がコード上に残る）。
    42  func TestUnitTestMappingCompleteness(t *testing.T) {
    43  	t.Parallel()
    44  
    45  	prodByDir, testsByDir := scanPackages(t)
    46  	violations := collectViolations(prodByDir, testsByDir)
    47  	sort.Strings(violations)
    48  	for _, v := range violations {
    49  		t.Log("1:1 未対応: " + v)
    50  	}
    51  	t.Logf("total 1:1 violations: %d", len(violations))
    52  
    53  	require.Empty(t, violations,
    54  		"分岐を持つ production 関数/メソッドに対応する TestXxx が無い（1:1 違反）。"+
    55  			"専用テストを追加するか、真に不要なら命名規約どおりの TestXxx を宣言し t.Skip(理由) で明示すること。")
    56  }

    58  // scanPackages は internal / pkg 配下を走査し、ディレクトリ単位で
    59  // production subject と TestXxx 名を収集して返す。
    60  func scanPackages(t *testing.T) (map[string][]prodSubject, map[string]map[string]struct{}) {
    61  	t.Helper()
    62  	prodByDir := map[string][]prodSubject{}
    63  	testsByDir := map[string]map[string]struct{}{}
    64  
    65  	for _, root := range moduleSubdirs(t, "internal", "pkg") {
    66  		err := filepath.WalkDir(root, func(path string, d fs.DirEntry, werr error) error {
    67  			if werr != nil {
    68  				return werr
    69  			}
    70  			if d.IsDir() || !strings.HasSuffix(path, ".go") {
    71  				return nil
    72  			}
    73  			src, rerr := pkgfs.OS{}.ReadFile(path)
    74  			if rerr != nil {
    75  				return rerr
    76  			}
    77  			indexGoFile(path, strings.Split(string(src), "\n"), prodByDir, testsByDir)
    78  			return nil
    79  		})
    80  		require.NoError(t, err)
    81  	}
    82  	return prodByDir, testsByDir
    83  }

    85  // indexGoFile は 1 ファイルを分類する。_test.go は TestXxx 名を、production は分岐持ち subject を収集する。
    86  func indexGoFile(path string, lines []string, prodByDir map[string][]prodSubject, testsByDir map[string]map[string]struct{}) {
    87  	dir := filepath.Dir(path)
    88  	if strings.HasSuffix(path, "_test.go") {
    89  		if testsByDir[dir] == nil {
    90  			testsByDir[dir] = map[string]struct{}{}
    91  		}
    92  		for _, l := range lines {
    93  			if m := testFuncRe.FindStringSubmatch(l); m != nil {
    94  				testsByDir[dir][m[1]] = struct{}{}
    95  			}
    96  		}
    97  		return
    98  	}
    99  	if isGeneratedGo(path, lines) {
   100  		return
   101  	}
   102  	prodByDir[dir] = append(prodByDir[dir], collectBranchingSubjects(lines, path)...)
   103  }

   105  // collectViolations は、対応する TestXxx を持たない subject を違反として列挙する。
   106  func collectViolations(prodByDir map[string][]prodSubject, testsByDir map[string]map[string]struct{}) []string {
   107  	var violations []string
   108  	for dir, subs := range prodByDir {
   109  		tests := testsByDir[dir]
   110  		for _, s := range subs {
   111  			if !hasAnyTest(tests, s.candidates) {
   112  				violations = append(violations, s.file+": "+s.name+" → 期待テスト名: "+strings.Join(s.candidates, " / "))
   113  			}
   114  		}
   115  	}
   116  	return violations
   117  }

   119  // collectBranchingSubjects は、分岐を持つ func / method を subject 化して返す。
   120  // 分岐が無い trivial（getter / 単純ラッパ等）は 1:1 の対象外として除外する。
   121  func collectBranchingSubjects(lines []string, file string) []prodSubject {
   122  	var subs []prodSubject
   123  	for i, line := range lines {
   124  		var name string
   125  		var candidates []string
   126  
   127  		if m := methodDeclRe.FindStringSubmatch(line); m != nil {
   128  			typ, method := m[1], m[2]
   129  			name = typ + "." + method
   130  			candidates = []string{"Test" + typ + "_" + method, "Test_" + typ + "_" + method}
   131  		} else if m := freeFuncRe.FindStringSubmatch(line); m != nil {
   132  			fn := m[1]
   133  			if fn == "main" || fn == "init" {
   134  				continue
   135  			}
   136  			name = fn
   137  			candidates = []string{"Test" + fn, "Test_" + fn}
   138  		} else {
   139  			continue
   140  		}
   141  
   142  		if funcBodyHasBranch(lines, i) {
   143  			subs = append(subs, prodSubject{name: name, file: file, candidates: candidates})
   144  		}
   145  	}
   146  	return subs
   147  }

   149  // funcBodyHasBranch は、宣言行 declIdx から次の行頭 `}`（トップレベルの閉じ括弧）までを本文とみなし、
   150  // 文頭に分岐キーワードが現れるかを判定する。コメント行（`//`）は除外される（branchRe が行頭タブ+キーワード限定のため）。
   151  //
   152  // gofmt は短い body を 1 行に保つ（例: getter `func (u *User) City() string { return u.city }`）。
   153  // 1 行関数は trivial とみなし分岐なし扱いにする（複数行本文の走査に入ると後続関数の分岐を誤検出するため）。
   154  func funcBodyHasBranch(lines []string, declIdx int) bool {
   155  	if strings.HasSuffix(strings.TrimRight(lines[declIdx], " \t"), "}") {
   156  		return false // 1 行関数 = trivial
   157  	}
   158  	for j := declIdx + 1; j < len(lines); j++ {
   159  		if lines[j] == "}" { // gofmt: トップレベル関数の閉じ括弧は行頭 `}`
   160  			return false
   161  		}
   162  		if branchRe.MatchString(lines[j]) {
   163  			return true
   164  		}
   165  	}
   166  	return false
   167  }

   169  func hasAnyTest(tests map[string]struct{}, candidates []string) bool {
   170  	for _, c := range candidates {
   171  		if _, ok := tests[c]; ok {
   172  			return true
   173  		}
   174  	}
   175  	return false
   176  }

   178  // isGeneratedGo は、生成物（ファイル名 or `// Code generated ... DO NOT EDIT.` ヘッダ）を判定する。
   179  func isGeneratedGo(path string, lines []string) bool {
   180  	base := filepath.Base(path)
   181  	if strings.HasSuffix(base, ".gen.go") || strings.HasSuffix(base, ".sql.go") ||
   182  		strings.HasSuffix(base, "_mock.go") || strings.HasSuffix(base, ".pb.go") {
   183  		return true
   184  	}
   185  	for k := 0; k < len(lines) && k < 5; k++ {
   186  		if genHeaderRe.MatchString(lines[k]) {
   187  			return true
   188  		}
   189  	}
   190  	return false
   191  }

   194  // 指定サブディレクトリの絶対パス群を返す。
   195  func moduleSubdirs(t *testing.T, subs ...string) []string {
   196  	t.Helper()
   197  	_, thisFile, _, ok := runtime.Caller(0)
   198  	require.True(t, ok)
   199  	root := filepath.Join(filepath.Dir(thisFile), "..", "..") // internal/architest -> module root
   200  	out := make([]string, 0, len(subs))
   201  	for _, s := range subs {
   202  		out = append(out, filepath.Join(root, s))
   203  	}
   204  	return out
   205  }
