    17  // 命名規約: メソッドは `Test<Type>_<Method>`（Type が非公開なら `Test_<type>_<Method>`）、
    18  // 自由関数は `Test<Func>`（非公開なら `Test_<func>`）。depguard が go/ast を禁じるため、
    19  // gofmt 済みソースのテキスト走査で検出する（既存 architest と同方針）。
    20  var (
    21  	// 型パラメータ節（`[T any]` 等）を許容する。レシーバ型・自由関数名の直後に挟まりうるため、
    22  	// 省略可能な `(?:\[.*\])?` を挟んでからメソッド引数の `(` を要求する。
    23  	// レシーバ変数名は省略可能（`func (*T) M()` は Go として合法で、名前を省くだけで検証を逃れられないようにする）。
    24  	methodDeclRe = regexp.MustCompile(`^func \(\s*(?:\w+\s+)?\*?(\w+)(?:\[.*\])?\) (\w+)\(`)
    25  	freeFuncRe   = regexp.MustCompile(`^func (\w+)(?:\[.*\])?\(`)
    26  	testFuncRe   = regexp.MustCompile(`^func (Test\w+)\(`)
    27  	genHeaderRe  = regexp.MustCompile(`^// Code generated .* DO NOT EDIT\.$`)
    28  )

    36  // TestUnitTestMappingCompleteness は、production code の全 func / method が、命名規約に一致する
    37  // TestXxx を同一パッケージに持つことを機械検証する。テスト対象と TestXxx の 1:1 を強制し、
    38  // レビュー頼みの見落とし（例: keyFunc の防御分岐が呼び出し側テストに埋もれて専用テストが
    39  // 無い状態、ルート登録だけを行う分岐なしの BindHandler が未テストのまま残る状態）を
    40  // loud な失敗に変える。
    41  //
    42  // 「1:1 を採らない」ことは許さない。対象が検証不可能であるために到達できない場合に限り、
    43  // 命名規約どおりの TestXxx を宣言し中で t.Skip("なぜ検証不可能か") することで意図を明示する
    44  // （allowlist は持たない。理由がコード上に残る）。「他のテストでカバー済み」は skip の
    45  // 理由にならない。テスト可能な対象は実テストを書く。
    46  //
    47  // 候補名が複数あるのは非公開シンボルの命名揺れを許容するためであって、両方を並べてよい
    48  // という意味ではないため、候補名の TestXxx を 2 つ以上持つ subject も違反とする。
    49  // 逆方向の検査はここまでで、候補名に一致しない TestXxx が同じ subject を対象にしている
    50  // 状態（1 シンボルが規約外の名前のテストと分業している状態）は検知できない。テスト名から
    51  // subject を逆引きする手段が無く、production code に対応シンボルを持たない TestXxx は
    52  // docs/testing-conventions.md §1 が by construction 正当と認めているため、そこまで踏み込むと
    53  // 検出のほとんどが誤検知になる。その形は test-review のレビューが受け持つ。
    54  func TestUnitTestMappingCompleteness(t *testing.T) {
    55  	t.Parallel()
    56  
    57  	prodByDir, testsByDir := scanPackages(t)
    58  	violations := collectViolations(prodByDir, testsByDir)
    59  	sort.Strings(violations)
    60  	for _, v := range violations {
    61  		t.Log("1:1 違反: " + v)
    62  	}
    63  	t.Logf("total 1:1 violations: %d", len(violations))
    64  
    65  	require.Empty(t, violations,
    66  		"production code の関数/メソッドと TestXxx の 1:1 が崩れている。"+
    67  			"対応する TestXxx が無い場合は専用テストを追加すること。検証不可能であるために到達できない"+
    68  			"対象に限り、命名規約どおりの TestXxx を宣言し、なぜ検証不可能かを t.Skip の理由文に書いて"+
    69  			"明示できる。候補名の TestXxx が重複している場合は、どちらか 1 本へケースを統合すること。")

    73  // production code の subject と TestXxx 名を収集して返す。
    74  //
    75  // scripts を含めるのは、そこに置かれた Go ツールがゲート（供給網ピン・連番検査・カバレッジ
    76  // 閾値・負荷帯の解決）であり、壊れ方が「何も検査しなくなる」方向に出るためである。
    77  // カバレッジ母数からは GO_TEST_EXCLUDE で外れているので、1:1 を欠くと対応の有無すら誰も見ない。
    78  func scanPackages(t *testing.T) (map[string][]prodSubject, map[string]map[string]struct{}) {
    79  	t.Helper()
    80  	prodByDir := map[string][]prodSubject{}
    81  	testsByDir := map[string]map[string]struct{}{}
    82  
    83  	for _, root := range moduleSubdirs(t, "internal", "pkg", "scripts") {
    84  		err := filepath.WalkDir(root, func(path string, d fs.DirEntry, werr error) error {
    85  			if werr != nil {
    86  				return werr
    87  			}
    88  			if d.IsDir() || !strings.HasSuffix(path, ".go") {
    89  				return nil
    90  			}
    91  			src, rerr := pkgfs.OS{}.ReadFile(path)
    92  			if rerr != nil {
    93  				return rerr
    94  			}
    95  			indexGoFile(path, strings.Split(string(src), "\n"), prodByDir, testsByDir)
    96  			return nil
    97  		})
    98  		require.NoError(t, err)
    99  	}
   100  	return prodByDir, testsByDir
   101  }

   103  // indexGoFile は 1 ファイルを分類する。_test.go は TestXxx 名を、production code は分岐持ち subject を収集する。
   104  func indexGoFile(path string, lines []string, prodByDir map[string][]prodSubject, testsByDir map[string]map[string]struct{}) {
   105  	dir := filepath.Dir(path)
   106  	if strings.HasSuffix(path, "_test.go") {
   107  		if testsByDir[dir] == nil {
   108  			testsByDir[dir] = map[string]struct{}{}
   109  		}
   110  		for _, l := range lines {
   111  			if m := testFuncRe.FindStringSubmatch(l); m != nil {
   112  				testsByDir[dir][m[1]] = struct{}{}
   113  			}
   114  		}
   115  		return
   116  	}
   117  	if isGeneratedGo(path, lines) {
   118  		return
   119  	}
   120  	prodByDir[dir] = append(prodByDir[dir], collectSubjects(lines, path)...)
   121  }

   123  // collectViolations は、対応する TestXxx を持たない subject と、候補名の TestXxx を
   124  // 2 つ以上持つ subject を違反として列挙する。
   125  func collectViolations(prodByDir map[string][]prodSubject, testsByDir map[string]map[string]struct{}) []string {
   126  	var violations []string
   127  	for dir, subs := range prodByDir {
   128  		tests := testsByDir[dir]
   129  		for _, s := range subs {
   130  			switch matched := matchingTests(tests, s.candidates); len(matched) {
   131  			case 0:
   132  				violations = append(violations, s.file+": "+s.name+" → 期待テスト名: "+strings.Join(s.candidates, " / "))
   133  			case 1:
   134  			default:
   135  				violations = append(violations, s.file+": "+s.name+" → 候補名の TestXxx が重複: "+strings.Join(matched, " / "))
   136  			}
   137  		}
   138  	}
   139  	return violations
   140  }

   142  // collectSubjects は、宣言されている全ての func / method を subject 化して返す。
   143  // 分岐の有無・body の行数では絞らない（getter / 単純ラッパ / ルート登録のみの関数も
   144  // 契約を持ちうるため）。除外するのは main / init のみ。
   145  func collectSubjects(lines []string, file string) []prodSubject {
   146  	var subs []prodSubject
   147  	for _, line := range lines {
   148  		var name string
   149  		var candidates []string
   150  
   151  		if m := methodDeclRe.FindStringSubmatch(line); m != nil {
   152  			typ, method := m[1], m[2]
   153  			name = typ + "." + method
   154  			candidates = []string{"Test" + typ + "_" + method, "Test_" + typ + "_" + method}
   155  		} else if m := freeFuncRe.FindStringSubmatch(line); m != nil {
   156  			fn := m[1]
   157  			if fn == "main" || fn == "init" {
   158  				continue
   159  			}
   160  			name = fn
   161  			candidates = []string{"Test" + fn, "Test_" + fn}
   162  		} else {
   163  			continue
   164  		}
   165  
   166  		subs = append(subs, prodSubject{name: name, file: file, candidates: candidates})
   167  	}
   168  	return subs
   169  }

   171  // matchingTests は、candidates のうち実在する TestXxx 名を宣言順で返す。
   172  func matchingTests(tests map[string]struct{}, candidates []string) []string {
   173  	var matched []string
   174  	for _, c := range candidates {
   175  		if _, ok := tests[c]; ok {
   176  			matched = append(matched, c)
   177  		}
   178  	}
   179  	return matched
   180  }

   182  // isGeneratedGo は、生成物（ファイル名 or `// Code generated ... DO NOT EDIT.` ヘッダ）を判定する。
   183  func isGeneratedGo(path string, lines []string) bool {
   184  	base := filepath.Base(path)
   185  	if strings.HasSuffix(base, ".gen.go") || strings.HasSuffix(base, ".sql.go") ||
   186  		strings.HasSuffix(base, "_mock.go") || strings.HasSuffix(base, ".pb.go") {
   187  		return true
   188  	}
   189  	for k := 0; k < len(lines) && k < 5; k++ {
   190  		if genHeaderRe.MatchString(lines[k]) {
   191  			return true
   192  		}
   193  	}
   194  	return false
   195  }
