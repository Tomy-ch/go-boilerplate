    25  var (
    26  	// echoImportRe は、Echo v5 を import するファイルを判定します。ルート登録形の走査を
    27  	// このファイル群に限ることで、同名メソッド（logging.Any 等）の誤検出を避けます。
    28  	echoImportRe = regexp.MustCompile(`"github\.com/labstack/echo/v5"`)
    29  	// routeCallRe は、HTTP メソッド名によるルート登録呼び出しを捕捉します。レシーバ名は問いません
    30  	// （手書きは `e`、生成コードは `router` を使う）。
    31  	routeCallRe = regexp.MustCompile(`\.(GET|POST|PUT|PATCH|DELETE|HEAD|OPTIONS|CONNECT|TRACE|QUERY)\(`)
    32  	// literalPathRe は、ルート登録の第 1 引数がパス文字列リテラルであることを要求し、その値を捕捉します。
    33  	// 生成コードは options.BaseURL を前置するため、その形も許容します。
    34  	literalPathRe = regexp.MustCompile(`^\s*(?:options\.BaseURL\s*\+\s*)?"(/[^"]*)"`)
    35  	// opaqueCallRe は、メソッドとパスの組を静的に確定できない Echo の登録形を捕捉します。
    36  	// 引数の形は問わない。パスがリテラルかどうかで見逃しが生まれるより、Echo 依存ファイルで
    37  	// 同名メソッドを呼んだときに loud に落ちるほうが安全な倒れ方だからです。
    38  	opaqueCallRe = regexp.MustCompile(`\.(Any|Match|Static|StaticFS|File|FileFS|RouteNotFound|Group|AddRoute)\(`)
    39  	// opaqueAddCallRe は、Add 系のルート登録を捕捉します。Add は http.Header・url.Values にも
    40  	// あるため、第 1 引数が HTTP メソッドか echo.Route リテラルであることを要求して区別します。
    41  	opaqueAddCallRe = regexp.MustCompile(
    42  		`\.Add\(\s*(?:"(?:GET|POST|PUT|PATCH|DELETE|HEAD|OPTIONS|CONNECT|TRACE|QUERY)"` +
    43  			`|http\.Method\w+|(?:echo\.)?Route\{)`)
    44  	// specParamRe は、OpenAPI のパスパラメータ表記 {name} を捕捉します。
    45  	specParamRe = regexp.MustCompile(`\{(\w+)\}`)

    70  // TestRouteSpecParity は、Echo に登録されるルートと OpenAPI spec の operation が 1:1 で対応することを
    71  // 機械検証する。OpenAPI リクエストバリデーション・details の opt-in ゲート・404/405 の判定はいずれも
    72  // 「登録されたルート = spec の operation」を暗黙の前提にしており、spec に無いルートを足すとテストは
    73  // 緑のまま実行時の挙動だけが変わるため、その前提をここで明示的な契約に変える。
    74  //
    75  // 意図的な spec 外ルートのための許可リストは持たない。許可リスト自体がドリフト源になるため、
    76  // spec 外のルートは常に失敗させる。
    77  //
    78  // ルートの列挙は実 DI グラフではなくソースの静的走査で行う（depguard が go/ast を禁じるため、
    79  // 既存 architest と同じく gofmt 済みソースのテキスト走査を採る）。ソースを見る以上、
    80  // di へ BindHandler を登録し忘れて実行時にルートが生えない配線漏れは本テストでは見えない。
    81  // その配線漏れは TestBindHandlerDIParity が「BindHandler の宣言 = fx.Invoke の列挙」および
    82  // 「RegisterHandlers を持つ生成物あり ⇒ BindHandler の実装あり」として補完する。
    83  //
    84  // spec 側も手書き正本を読むため、サンプル API を削除すると両者は同時に縮み、この契約は保たれる。
    85  func TestRouteSpecParity(t *testing.T) {
    86  	t.Parallel()
    87  
    88  	index := scanRepositoryRoutes(t)
    89  	require.NotEmpty(t, index.scanned, "Echo を import するファイルが 1 件も見つからない（走査ルートの誤りを疑う）")
    90  	t.Logf("走査した Echo 依存ファイル: %d 件 / 検出したルート登録: %d 件", index.scanned, len(index.registered))
    91  
    92  	assert.Empty(t, index.opaque,
    93  		"メソッドとパスを静的に確定できない登録形がある。spec との 1:1 を検証できないため、"+
    94  			"生成された RegisterHandlers 経由の登録に置き換えること")
    95  
    96  	spec := specRouteSet(t)
    97  
    98  	assert.Empty(t, index.routesMissingFrom(spec),
    99  		"spec に無いルートが Echo に登録されている。OpenAPI リクエストバリデーションと "+
   100  			"details の opt-in ゲートはいずれも spec 上の operation を前提とするため、"+
   101  			"まず OpenAPI に operation を定義すること")
   102  	assert.Empty(t, index.unregisteredIn(spec),
   103  		"spec の operation に対応するルート登録がソース上に無い。ハンドラパッケージの生成漏れを疑うこと")
   104  }
