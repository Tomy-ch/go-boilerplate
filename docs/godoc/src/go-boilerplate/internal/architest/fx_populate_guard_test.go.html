   103  // collectUnguardedFxPopulate は、gofmt 済みソースの行列から fx.Populate の呼び出し数・そのうち
   104  // 解析できた数・populate 対象を構築エラーの検査より前に参照している箇所
   105  // （`file:line: 該当行` 形式）を返します。
   106  //
   107  // 参照ごとに、それを囲む最も内側の関数リテラル（無ければ fx.Populate の直後）から参照までの間に
   108  // ガードがあるかを見ます。start / stop のように独立して呼ばれうるクロージャが複数ある場合、
   109  // 一方のガードは他方をカバーしないため、クロージャ単位で判定します。
   110  // 同じクロージャ内の複数参照は同一の欠落なので、最初の 1 件だけを報告します。
   111  func collectUnguardedFxPopulate(lines []string, file string) (int, int, []string) {
   112  	code := stripCommentsAndRawStrings(lines)
   113  
   114  	var violations []string
   115  	sites, analyzed := 0, 0
   116  
   117  	for i, line := range code {
   118  		if !strings.Contains(line, fxPopulateCall) {
   119  			continue
   120  		}
   121  		sites++
   122  
   123  		end := funcEndLine(code, i)
   124  
   125  		call, callEnd := joinCall(code, i, fxPopulateCall)
   126  		names := populatedNames(call)
   127  		appVar, newAt := fxNewSite(code, funcStartLine(code, i), end)
   128  		if len(names) == 0 || appVar == "" {
   129  			continue
   130  		}
   131  		analyzed++
   132  
   133  		// fx.Populate は fx.New の引数なので、対象名は構築式そのものにも現れる。
   134  		// 走査は構築式が閉じた後から始め、引数の記述を参照と取り違えないようにする。
   135  		_, newEnd := joinCall(code, newAt, fxNewCall)
   136  		scanFrom := max(callEnd, newEnd)
   137  
   138  		violations = append(violations, unguardedUses(code, lines, file, scanFrom, end, appVar, names)...)
   139  	}
   140  
   141  	return sites, analyzed, violations
   142  }

   144  // unguardedUses は、fx.Populate（populateAt 行）から関数終端 end までの間で、
   145  // ガード（`<appVar>.Err()`）に先行されていない populate 対象の参照を列挙します。
   146  func unguardedUses(code, lines []string, file string, populateAt, end int, appVar string, names []string) []string {
   147  	var violations []string
   148  	guard := appVar + ".Err()"
   149  	reported := make(map[int]bool)
   150  
   151  	for use := populateAt + 1; use < end && use < len(code); use++ {
   152  		if !referencesAny(code[use], names) {
   153  			continue
   154  		}
   155  		scope := enclosingClosure(code, populateAt, use)
   156  		if reported[scope] {
   157  			continue
   158  		}
   159  		if hasGuardBefore(code, populateAt, use, scope, guard) {
   160  			continue
   161  		}
   162  		reported[scope] = true
   163  		violations = append(violations,
   164  			fmt.Sprintf("%s:%d: %s", file, use+1, strings.TrimSpace(lines[use])))
   165  	}
   166  
   167  	return violations
   168  }

   170  // hasGuardBefore は、use 行より前にあり、かつ use を必ず先行実行するガードが存在するかを返します。
   171  //
   172  // ガードが関数直下（囲む関数リテラルなし）にあれば、後続のどのクロージャの参照も先行実行されます。
   173  // クロージャの中にあるガードは、同じクロージャ内の参照しかカバーしません（別のクロージャは独立に
   174  // 呼ばれうるため）。入れ子は最も内側のクロージャだけで判定するので、多重入れ子では
   175  // カバーしていても違反とみなす側（保守的）に倒れます。
   176  func hasGuardBefore(code []string, populateAt, use, useScope int, guard string) bool {
   177  	for g := populateAt + 1; g < use; g++ {
   178  		if !strings.Contains(code[g], guard) {
   179  			continue
   180  		}
   181  		guardScope := enclosingClosure(code, populateAt, g)
   182  		if guardScope == -1 || guardScope == useScope {
   183  			return true
   184  		}
   185  	}
   186  	return false
   187  }

   189  // enclosingClosure は、line を囲む最も内側の関数リテラルの開始行を返します。
   190  // bound より外側は見ず、関数直下（囲む関数リテラルが無い）の場合は -1 を返します。
   191  //
   192  // ブロックが外側へ出るほどインデントは浅くなるので、line から遡って「それまでの最小インデント
   193  // より浅い行」だけを外側のブロック開始とみなします。空行はインデントを持たないため飛ばします。
   194  func enclosingClosure(code []string, bound, line int) int {
   195  	if line >= len(code) {
   196  		return -1
   197  	}
   198  	minIndent := indentWidth(code[line])
   199  
   200  	for j := line - 1; j > bound; j-- {
   201  		if strings.TrimSpace(code[j]) == "" {
   202  			continue
   203  		}
   204  		indent := indentWidth(code[j])
   205  		if indent >= minIndent {
   206  			continue
   207  		}
   208  		minIndent = indent
   209  		if closureStartRe.MatchString(strings.TrimRight(code[j], " \t")) {
   210  			return j
   211  		}
   212  	}
   213  	return -1
   214  }

   216  // indentWidth は、行頭のタブ数を返します（gofmt はインデントにタブのみを使います）。
   217  func indentWidth(line string) int {
   218  	return len(line) - len(strings.TrimLeft(line, "\t"))
   219  }

   221  // fxNewSite は、[from, to] の範囲で fx.New の戻り値を受ける変数名とその行を返します。
   222  // 変数で受けていない（戻り値を直接返している等）場合は空文字と -1 を返します。
   223  // fx.Populate は fx.New の引数なので、呼び出し行は fx.New より前にも後にも来ます。
   224  // そのため範囲は fx.Populate の位置ではなく関数全体を渡します。
   225  func fxNewSite(code []string, from, to int) (string, int) {
   226  	for i := from; i <= to && i < len(code); i++ {
   227  		if !strings.Contains(code[i], fxNewCall) {
   228  			continue
   229  		}
   230  		if m := fxNewAssignRe.FindStringSubmatch(code[i]); m != nil {
   231  			return m[1], i
   232  		}
   233  	}
   234  	return "", -1
   235  }

   237  // funcStartLine は、line を含む関数宣言の開始行（行頭 `func `）を返します。
   238  // 見つからない場合は 0 を返します。
   239  func funcStartLine(code []string, line int) int {
   240  	for i := line; i >= 0 && i < len(code); i-- {
   241  		if strings.HasPrefix(code[i], "func ") {
   242  			return i
   243  		}
   244  	}
   245  	return 0
   246  }

   248  // joinCall は、from 行に現れる call の開き括弧から対応する閉じ括弧までを 1 つの文字列に連結し、
   249  // 閉じ括弧のある行を返します。gofmt は引数を複数行に分けた形をそのまま保つため、
   250  // 単一行前提だと取りこぼします。呼び出しが見つからない / 括弧が閉じない場合は空文字と from を返します。
   251  func joinCall(code []string, from int, call string) (string, int) {
   252  	if from < 0 || from >= len(code) {
   253  		return "", from
   254  	}
   255  	idx := strings.Index(code[from], call)
   256  	if idx < 0 {
   257  		return "", from
   258  	}
   259  
   260  	var b strings.Builder
   261  	depth := 0
   262  	for i := from; i < len(code); i++ {
   263  		line := code[i]
   264  		start := 0
   265  		if i == from {
   266  			start = idx + len(call) - 1 // 開き括弧から読み始める
   267  		}
   268  		for _, r := range line[start:] {
   269  			switch r {
   270  			case '(':
   271  				depth++
   272  			case ')':
   273  				depth--
   274  			}
   275  			b.WriteRune(r)
   276  			if depth == 0 {
   277  				return call[:len(call)-1] + b.String(), i
   278  			}
   279  		}
   280  	}
   281  	return "", from
   282  }

   284  // populatedNames は、`fx.Populate(&a, &b)` の呼び出し文字列から populate 対象の識別子を取り出します。
   285  // すべての引数が `&識別子` の直接形である場合だけ名前を返し、`fx.Annotate(...)` などで
   286  // 包まれていて対象を一意に特定できない場合は nil を返します（解析対象から外す）。
   287  func populatedNames(call string) []string {
   288  	open := strings.Index(call, "(")
   289  	if open < 0 || !strings.HasSuffix(call, ")") {
   290  		return nil
   291  	}
   292  	args := strings.TrimSpace(call[open+1 : len(call)-1])
   293  	if args == "" {
   294  		return nil
   295  	}
   296  
   297  	parts := strings.Split(args, ",")
   298  	names := make([]string, 0, len(parts))
   299  	for _, part := range parts {
   300  		// 複数行で書いた呼び出しは末尾にカンマが残るため、空の要素は区切りの余りとして読み飛ばす。
   301  		trimmed := strings.TrimSpace(part)
   302  		if trimmed == "" {
   303  			continue
   304  		}
   305  		name, ok := strings.CutPrefix(trimmed, "&")
   306  		if !ok || !identifierRe.MatchString(name) {
   307  			return nil
   308  		}
   309  		names = append(names, name)
   310  	}
   311  	if len(names) == 0 {
   312  		return nil
   313  	}
   314  	return names
   315  }

   317  // stripCommentsAndRawStrings は、行数を保ったままコメントと raw string の中身を空白へ置き換えた
   318  // ソースを返します。対象名を字面で含むコメントや、`}` を含む埋め込みテンプレートが走査を
   319  // 誤らせるため、判定はこの view に対して行います。
   320  func stripCommentsAndRawStrings(lines []string) []string {
   321  	out := make([]string, len(lines))
   322  	var s stripState
   323  
   324  	for i, line := range lines {
   325  		var b strings.Builder
   326  		s.inQuote = false
   327  		for j := 0; j < len(line); {
   328  			j = s.step(&b, line, j)
   329  		}
   330  		out[i] = b.String()
   331  	}
   332  	return out
   333  }

   335  // step は line の j バイト目を処理し、出力へ書き込んだうえで次に読む位置を返します。
   336  func (s *stripState) step(b *strings.Builder, line string, j int) int {
   337  	switch {
   338  	case s.inRaw:
   339  		return s.stepRaw(b, line, j)
   340  	case s.inBlock:
   341  		return s.stepBlock(b, line, j)
   342  	case s.inQuote:
   343  		return s.stepQuote(b, line, j)
   344  	default:
   345  		return s.stepCode(b, line, j)
   346  	}
   347  }

   349  // stepRaw は、raw string の内部を空白へ置き換えつつ、閉じるバッククォートを検出します。
   350  func (s *stripState) stepRaw(b *strings.Builder, line string, j int) int {
   351  	if line[j] == '`' {
   352  		s.inRaw = false
   353  		b.WriteByte('`')
   354  		return j + 1
   355  	}
   356  	b.WriteByte(' ')
   357  	return j + 1
   358  }

   360  // stepBlock は、ブロックコメントの内部を空白へ置き換えつつ、閉じる `*/` を検出します。
   361  func (s *stripState) stepBlock(b *strings.Builder, line string, j int) int {
   362  	if line[j] == '*' && byteAt(line, j+1) == '/' {
   363  		s.inBlock = false
   364  		b.WriteString("  ")
   365  		return j + 2
   366  	}
   367  	b.WriteByte(' ')
   368  	return j + 1
   369  }

   371  // stepQuote は、文字列リテラルの内部をそのまま写し、エスケープを飛ばして閉じ `"` を検出します。
   372  func (s *stripState) stepQuote(b *strings.Builder, line string, j int) int {
   373  	c := line[j]
   374  	b.WriteByte(c)
   375  	if c == '\\' && j+1 < len(line) {
   376  		b.WriteByte(line[j+1])
   377  		return j + 2
   378  	}
   379  	if c == '"' {
   380  		s.inQuote = false
   381  	}
   382  	return j + 1
   383  }

   385  // stepCode は、コード領域を写しつつリテラル・コメントの開始を検出します。
   386  // 行コメントは以降を丸ごと落とすため、行末位置を返します。
   387  func (s *stripState) stepCode(b *strings.Builder, line string, j int) int {
   388  	c, next := line[j], byteAt(line, j+1)
   389  	switch {
   390  	case c == '/' && next == '/':
   391  		return len(line)
   392  	case c == '/' && next == '*':
   393  		s.inBlock = true
   394  		b.WriteString("  ")
   395  		return j + 2
   396  	case c == '"':
   397  		s.inQuote = true
   398  	case c == '`':
   399  		s.inRaw = true
   400  	}
   401  	b.WriteByte(c)
   402  	return j + 1
   403  }

   405  // funcEndLine は、from 行を含む関数の終端（行頭 `}`）の行インデックスを返します。
   406  // gofmt は関数の閉じ波括弧を行頭に置くため、次の行頭 `}` までを関数本体とみなせます。
   407  // 見つからない場合はファイル末尾を返します。
   408  func funcEndLine(code []string, from int) int {
   409  	for i := from; i < len(code); i++ {
   410  		if strings.HasPrefix(code[i], "}") {
   411  			return i
   412  		}
   413  	}
   414  	return len(code) - 1
   415  }

   417  // referencesAny は、行が names のいずれかを識別子として含むかを返します。
   418  // 部分一致で別の識別子（`logger` に対する `loggerFactory` など）を拾わないよう前後の文字を見ます。
   419  func referencesAny(line string, names []string) bool {
   420  	for _, name := range names {
   421  		if containsIdentifier(line, name) {
   422  			return true
   423  		}
   424  	}
   425  	return false
   426  }
   427  
   428  // containsIdentifier は、line 中に name が識別子として現れるかを返します。
   429  // 前後が Go の識別子構成文字（英数字 / `_`）でない出現だけを識別子とみなします。
   430  func containsIdentifier(line, name string) bool {
   431  	if name == "" {
   432  		return false
   433  	}
   434  	for offset := 0; ; {
   435  		idx := strings.Index(line[offset:], name)
   436  		if idx < 0 {
   437  			return false
   438  		}
   439  		at := offset + idx
   440  		if !isIdentifierChar(byteAt(line, at-1)) && !isIdentifierChar(byteAt(line, at+len(name))) {
   441  			return true
   442  		}
   443  		offset = at + len(name)
   444  	}
   445  }
   446  
   447  // byteAt は、範囲外を 0 として line の i バイト目を返します。
   448  func byteAt(line string, i int) byte {
   449  	if i < 0 || i >= len(line) {
   450  		return 0
   451  	}
   452  	return line[i]
   453  }
   454  
   455  // isIdentifierChar は、Go の識別子を構成しうるバイトかを返します。
   456  func isIdentifierChar(b byte) bool {
   457  	return b == '_' ||
   458  		(b >= '0' && b <= '9') ||
   459  		(b >= 'a' && b <= 'z') ||
   460  		(b >= 'A' && b <= 'Z')
   461  }
