   101  // collectUnguardedFxPopulate は、gofmt 済みソースの行列から fx.Populate の呼び出し数・そのうち
   102  // 解析できた数・populate 対象を構築エラーの検査より前に参照している箇所
   103  // （`file:line: 該当行` 形式）を返します。
   104  //
   105  // 参照ごとに、それを囲む最も内側の関数リテラル（無ければ fx.Populate の直後）から参照までの間に
   106  // ガードがあるかを見ます。start / stop のように独立して呼ばれうるクロージャが複数ある場合、
   107  // 一方のガードは他方をカバーしないため、クロージャ単位で判定します。
   108  // 同じクロージャ内の複数参照は同一の欠落なので、最初の 1 件だけを報告します。
   109  func collectUnguardedFxPopulate(lines []string, file string) (int, int, []string) {
   110  	code := stripCommentsAndRawStrings(lines)
   111  
   112  	var violations []string
   113  	sites, analyzed := 0, 0
   114  
   115  	for i, line := range code {
   116  		if !strings.Contains(line, fxPopulateCall) {
   117  			continue
   118  		}
   119  		sites++
   120  
   121  		end := funcEndLine(code, i)
   122  
   123  		call, callEnd := joinCall(code, i, fxPopulateCall)
   124  		names := populatedNames(call)
   125  		appVar, newAt := fxNewSite(code, funcStartLine(code, i), end)
   126  		if len(names) == 0 || appVar == "" {
   127  			continue
   128  		}
   129  		analyzed++
   130  
   131  		// fx.Populate は fx.New の引数なので、対象名は構築式そのものにも現れる。
   132  		// 走査は構築式が閉じた後から始め、引数の記述を参照と取り違えないようにする。
   133  		_, newEnd := joinCall(code, newAt, fxNewCall)
   134  		scanFrom := max(callEnd, newEnd)
   135  
   136  		violations = append(violations, unguardedUses(code, lines, file, scanFrom, end, appVar, names)...)
   137  	}
   138  
   139  	return sites, analyzed, violations
   140  }

   142  // unguardedUses は、fx.Populate（populateAt 行）から関数終端 end までの間で、
   143  // ガード（`<appVar>.Err()`）に先行されていない populate 対象の参照を列挙します。
   144  func unguardedUses(code, lines []string, file string, populateAt, end int, appVar string, names []string) []string {
   145  	var violations []string
   146  	guard := appVar + ".Err()"
   147  	reported := make(map[int]bool)
   148  
   149  	for use := populateAt + 1; use < end && use < len(code); use++ {
   150  		if !referencesAny(code[use], names) {
   151  			continue
   152  		}
   153  		scope := enclosingClosure(code, populateAt, use)
   154  		if reported[scope] {
   155  			continue
   156  		}
   157  		if hasGuardBefore(code, populateAt, use, scope, guard) {
   158  			continue
   159  		}
   160  		reported[scope] = true
   161  		violations = append(violations,
   162  			fmt.Sprintf("%s:%d: %s", file, use+1, strings.TrimSpace(lines[use])))
   163  	}
   164  
   165  	return violations
   166  }

   168  // hasGuardBefore は、use 行より前にあり、かつ use を必ず先行実行するガードが存在するかを返します。
   169  //
   170  // ガードが関数直下（囲む関数リテラルなし）にあれば、後続のどのクロージャの参照も先行実行されます。
   171  // クロージャの中にあるガードは、同じクロージャ内の参照しかカバーしません（別のクロージャは独立に
   172  // 呼ばれうるため）。入れ子は最も内側のクロージャだけで判定するので、多重入れ子では
   173  // カバーしていても違反とみなす側（保守的）に倒れます。
   174  func hasGuardBefore(code []string, populateAt, use, useScope int, guard string) bool {
   175  	for g := populateAt + 1; g < use; g++ {
   176  		if !strings.Contains(code[g], guard) {
   177  			continue
   178  		}
   179  		guardScope := enclosingClosure(code, populateAt, g)
   180  		if guardScope == -1 || guardScope == useScope {
   181  			return true
   182  		}
   183  	}
   184  	return false
   185  }

   187  // enclosingClosure は、line を囲む最も内側の関数リテラルの開始行を返します。
   188  // bound より外側は見ず、関数直下（囲む関数リテラルが無い）の場合は -1 を返します。
   189  //
   190  // ブロックが外側へ出るほどインデントは浅くなるので、line から遡って「それまでの最小インデント
   191  // より浅い行」だけを外側のブロック開始とみなします。空行はインデントを持たないため飛ばします。
   192  func enclosingClosure(code []string, bound, line int) int {
   193  	if line >= len(code) {
   194  		return -1
   195  	}
   196  	minIndent := indentWidth(code[line])
   197  
   198  	for j := line - 1; j > bound; j-- {
   199  		if strings.TrimSpace(code[j]) == "" {
   200  			continue
   201  		}
   202  		indent := indentWidth(code[j])
   203  		if indent >= minIndent {
   204  			continue
   205  		}
   206  		minIndent = indent
   207  		if closureStartRe.MatchString(strings.TrimRight(code[j], " \t")) {
   208  			return j
   209  		}
   210  	}
   211  	return -1
   212  }

   214  // indentWidth は、行頭のタブ数を返します（gofmt はインデントにタブのみを使います）。
   215  func indentWidth(line string) int {
   216  	return len(line) - len(strings.TrimLeft(line, "\t"))
   217  }

   219  // fxNewSite は、[from, to] の範囲で fx.New の戻り値を受ける変数名とその行を返します。
   220  // 変数で受けていない（戻り値を直接返している等）場合は空文字と -1 を返します。
   221  // fx.Populate は fx.New の引数なので、呼び出し行は fx.New より前にも後にも来ます。
   222  // そのため範囲は fx.Populate の位置ではなく関数全体を渡します。
   223  func fxNewSite(code []string, from, to int) (string, int) {
   224  	for i := from; i <= to && i < len(code); i++ {
   225  		if !strings.Contains(code[i], fxNewCall) {
   226  			continue
   227  		}
   228  		if m := fxNewAssignRe.FindStringSubmatch(code[i]); m != nil {
   229  			return m[1], i
   230  		}
   231  	}
   232  	return "", -1
   233  }

   241  		}
   242  	}
   243  	return 0

   246  // joinCall は、from 行に現れる call の開き括弧から対応する閉じ括弧までを 1 つの文字列に連結し、
   247  // 閉じ括弧のある行を返します。gofmt は引数を複数行に分けた形をそのまま保つため、
   248  // 単一行前提だと取りこぼします。呼び出しが見つからない / 括弧が閉じない場合は空文字と from を返します。
   249  func joinCall(code []string, from int, call string) (string, int) {
   250  	if from < 0 || from >= len(code) {
   251  		return "", from
   252  	}
   253  	idx := strings.Index(code[from], call)
   254  	if idx < 0 {
   255  		return "", from
   256  	}
   257  
   258  	var b strings.Builder
   259  	depth := 0
   260  	for i := from; i < len(code); i++ {
   261  		line := code[i]
   262  		start := 0
   263  		if i == from {
   264  			start = idx + len(call) - 1 // 開き括弧から読み始める
   265  		}
   266  		for _, r := range line[start:] {
   267  			switch r {
   268  			case '(':
   269  				depth++
   270  			case ')':
   271  				depth--
   272  			}
   273  			b.WriteRune(r)
   274  			if depth == 0 {
   275  				return call[:len(call)-1] + b.String(), i
   276  			}
   277  		}
   278  	}
   279  	return "", from
   280  }

   282  // populatedNames は、`fx.Populate(&a, &b)` の呼び出し文字列から populate 対象の識別子を取り出します。
   283  // すべての引数が `&識別子` の直接形である場合だけ名前を返し、`fx.Annotate(...)` などで
   284  // 包まれていて対象を一意に特定できない場合は nil を返します（解析対象から外す）。
   285  func populatedNames(call string) []string {
   286  	open := strings.Index(call, "(")
   287  	if open < 0 || !strings.HasSuffix(call, ")") {
   288  		return nil
   289  	}
   290  	args := strings.TrimSpace(call[open+1 : len(call)-1])
   291  	if args == "" {
   292  		return nil
   293  	}
   294  
   295  	parts := strings.Split(args, ",")
   296  	names := make([]string, 0, len(parts))
   297  	for _, part := range parts {
   298  		// 複数行で書いた呼び出しは末尾にカンマが残るため、空の要素は区切りの余りとして読み飛ばす。
   299  		trimmed := strings.TrimSpace(part)
   300  		if trimmed == "" {
   301  			continue
   302  		}
   303  		name, ok := strings.CutPrefix(trimmed, "&")
   304  		if !ok || !identifierRe.MatchString(name) {
   305  			return nil
   306  		}
   307  		names = append(names, name)
   308  	}
   309  	if len(names) == 0 {
   310  		return nil
   311  	}
   312  	return names
   313  }

   315  // stripCommentsAndRawStrings は、行数を保ったままコメントと raw string の中身を空白へ置き換えた
   316  // ソースを返します。対象名を字面で含むコメントや、`}` を含む埋め込みテンプレートが走査を
   317  // 誤らせるため、判定はこの view に対して行います。
   318  func stripCommentsAndRawStrings(lines []string) []string {
   319  	out := make([]string, len(lines))
   320  	var s stripState
   321  
   322  	for i, line := range lines {
   323  		var b strings.Builder
   324  		s.inQuote = false
   325  		for j := 0; j < len(line); {
   326  			j = s.step(&b, line, j)
   327  		}
   328  		out[i] = b.String()
   329  	}
   330  	return out
   331  }

   333  // step は line の j バイト目を処理し、出力へ書き込んだうえで次に読む位置を返します。
   334  func (s *stripState) step(b *strings.Builder, line string, j int) int {
   335  	switch {
   336  	case s.inRaw:
   337  		return s.stepRaw(b, line, j)
   338  	case s.inBlock:
   339  		return s.stepBlock(b, line, j)
   340  	case s.inQuote:
   341  		return s.stepQuote(b, line, j)
   342  	default:
   343  		return s.stepCode(b, line, j)
   344  	}
   345  }

   347  // stepRaw は、raw string の内部を空白へ置き換えつつ、閉じるバッククォートを検出します。
   348  func (s *stripState) stepRaw(b *strings.Builder, line string, j int) int {
   349  	if line[j] == '`' {
   350  		s.inRaw = false
   351  		b.WriteByte('`')
   352  		return j + 1
   353  	}
   354  	b.WriteByte(' ')
   355  	return j + 1
   356  }

   358  // stepBlock は、ブロックコメントの内部を空白へ置き換えつつ、閉じる `*/` を検出します。
   359  func (s *stripState) stepBlock(b *strings.Builder, line string, j int) int {
   360  	if line[j] == '*' && byteAt(line, j+1) == '/' {
   361  		s.inBlock = false
   362  		b.WriteString("  ")
   363  		return j + 2
   364  	}
   365  	b.WriteByte(' ')
   366  	return j + 1
   367  }

   369  // stepQuote は、文字列リテラルの内部をそのまま写し、エスケープを飛ばして閉じ `"` を検出します。
   370  func (s *stripState) stepQuote(b *strings.Builder, line string, j int) int {
   371  	c := line[j]
   372  	b.WriteByte(c)
   373  	if c == '\\' && j+1 < len(line) {
   374  		b.WriteByte(line[j+1])
   375  		return j + 2
   376  	}
   377  	if c == '"' {
   378  		s.inQuote = false
   379  	}
   380  	return j + 1
   381  }

   383  // stepCode は、コード領域を写しつつリテラル・コメントの開始を検出します。
   384  // 行コメントは以降を丸ごと落とすため、行末位置を返します。
   385  func (s *stripState) stepCode(b *strings.Builder, line string, j int) int {
   386  	c, next := line[j], byteAt(line, j+1)
   387  	switch {
   388  	case c == '/' && next == '/':
   389  		return len(line)
   390  	case c == '/' && next == '*':
   391  		s.inBlock = true
   392  		b.WriteString("  ")
   393  		return j + 2
   394  	case c == '"':
   395  		s.inQuote = true
   396  	case c == '`':
   397  		s.inRaw = true
   398  	}
   399  	b.WriteByte(c)
   400  	return j + 1
   401  }

   403  // funcEndLine は、from 行を含む関数の終端（行頭 `}`）の行インデックスを返します。
   404  // gofmt は関数の閉じ波括弧を行頭に置くため、次の行頭 `}` までを関数本体とみなせます。
   405  // 見つからない場合はファイル末尾を返します。
   406  func funcEndLine(code []string, from int) int {
   407  	for i := from; i < len(code); i++ {
   408  		if strings.HasPrefix(code[i], "}") {
   409  			return i
   410  		}
   411  	}
   412  	return len(code) - 1
   413  }

   415  // referencesAny は、行が names のいずれかを識別子として含むかを返します。
   416  // 部分一致で別の識別子（`logger` に対する `loggerFactory` など）を拾わないよう前後の文字を見ます。
   417  func referencesAny(line string, names []string) bool {
   418  	for _, name := range names {
   419  		if containsIdentifier(line, name) {
   420  			return true
   421  		}
   422  	}
   423  	return false
   424  }
   425  
   426  // containsIdentifier は、line 中に name が識別子として現れるかを返します。
   427  // 前後が Go の識別子構成文字（英数字 / `_`）でない出現だけを識別子とみなします。
   428  func containsIdentifier(line, name string) bool {
   429  	if name == "" {
   430  		return false
   431  	}
   432  	for offset := 0; ; {
   433  		idx := strings.Index(line[offset:], name)
   434  		if idx < 0 {
   435  			return false
   436  		}
   437  		at := offset + idx
   438  		if !isIdentifierChar(byteAt(line, at-1)) && !isIdentifierChar(byteAt(line, at+len(name))) {
   439  			return true
   440  		}
   441  		offset = at + len(name)
   442  	}
   443  }

   445  // byteAt は、範囲外を 0 として line の i バイト目を返します。
   446  func byteAt(line string, i int) byte {
   447  	if i < 0 || i >= len(line) {
   448  		return 0
   449  	}
   450  	return line[i]
   451  }
