    34  	coveringTestRe = regexp.MustCompile(`Test[A-Z_]`)

    37  // TestSkipReasonDoesNotNameCoveringTest は、Skip の理由が他のテストを名指ししていないことを機械検証する。
    38  // Skip が許されるのは対象が検証不可能であるために到達できない場合のみで、理由には「なぜ検証不可能か」を書く。
    39  // 「他のテストでカバー済み」を理由にした skip はテストを別テストの実装へ依存させ、カバー元が縮小・削除されても
    40  // 無言で green のまま残るため、1:1 マッピングを名前だけの殻にする。
    41  //
    42  // 理由は文字列リテラル直書きに限る。変数越しに渡されると本文をテキストで追えず、
    43  // 一段挟むだけで規約を骨抜きにできてしまうため、リテラル以外は違反として扱う。
    44  func TestSkipReasonDoesNotNameCoveringTest(t *testing.T) {
    45  	t.Parallel()
    46  
    47  	var violations []string
    48  	for _, root := range moduleSubdirs(t, "internal", "pkg") {
    49  		err := filepath.WalkDir(root, func(path string, d fs.DirEntry, werr error) error {
    50  			if werr != nil {
    51  				return werr
    52  			}
    53  			if d.IsDir() || !strings.HasSuffix(path, "_test.go") || filepath.Base(path) == selfFile {
    54  				return nil
    55  			}

    57  			if rerr != nil {
    58  				return rerr
    59  			}

    69  	}

    72  		"skip の理由が規約に反している。テスト可能な対象は実テストを書くこと。"+

    76  // collectSkipReasonViolations は、規約に反する Skip 呼び出しを違反として列挙する。
    77  func collectSkipReasonViolations(file string, lines []string) []string {
    78  	var violations []string
    79  	for i, line := range lines {
    80  		loc := skipCallRe.FindStringIndex(line)
    81  		if loc == nil || inComment(line, loc[0]) {
    82  			continue
    83  		}
    84  
    85  		if skipNowRe.MatchString(line[loc[0]:loc[1]]) {
    86  			violations = append(violations, describe(file, i, lines[i], "理由を持たない SkipNow は使えない"))
    87  			continue
    88  		}
    89  
    90  		reason, ok := skipCallText(lines, i, loc[1])
    91  		// 折り返された呼び出しはリテラルが次行以降に来るため、連結後の先頭空白を落としてから判定する。
    92  		literal := strings.TrimLeft(reason, " \t")
    93  		switch {
    94  		case !ok:
    95  			violations = append(violations, describe(file, i, lines[i], "理由を解析できない（複数行が長すぎる）"))
    96  		case !strings.HasPrefix(literal, `"`) && !strings.HasPrefix(literal, "`"):
    97  			violations = append(violations, describe(file, i, lines[i], "理由が文字列リテラルではない"))
    98  		case coveringTestRe.MatchString(reason):
    99  			violations = append(violations, describe(file, i, lines[i], "理由が他のテストを名指ししている"))
   100  		}
   101  	}
   102  	return violations
   103  }

   110  // inComment は、位置 at がその行のコメント部分にあるかを返す。
   111  // 行頭コメントだけでなく、コード末尾に付いた行内コメント中の言及も走査対象から外すために使う。
   112  func inComment(line string, at int) bool {
   113  	c := commentStart(line)
   114  	return c >= 0 && c < at
   115  }
   116  
   117  // commentStart は、行コメントの開始位置を返す（無ければ -1）。
   118  // 文字列リテラル中の `//`（URL など）をコメント開始と誤認すると、同じ行にある実在の Skip 呼び出しを
   119  // 走査対象から外してしまうため、クオートの開閉を追いながらリテラルの外だけを探す。
   120  func commentStart(line string) int {
   121  	var inQuote, inRaw, escaped bool
   122  	for i := range len(line) {
   123  		switch {
   124  		case escaped:
   125  			escaped = false
   126  		case inRaw:
   127  			inRaw = line[i] != '`'
   128  		case inQuote && line[i] == '\\':
   129  			escaped = true
   130  		case inQuote:
   131  			inQuote = line[i] != '"'
   132  		case line[i] == '`':
   133  			inRaw = true
   134  		case line[i] == '"':
   135  			inQuote = true
   136  		case line[i] == '/' && i+1 < len(line) && line[i+1] == '/':
   137  			return i
   138  		}
   139  	}
   140  	return -1
   141  }

   143  // skipCallText は、Skip 呼び出しの引数テキストを返す。
   144  // gofmt は長い引数を複数行へ折り返すため、閉じ括弧で終わる行までを連結する。
   145  // 上限行数内に閉じ括弧が見つからない場合は ok=false を返し、呼び出し元が違反として扱う。
   146  func skipCallText(lines []string, declIdx, from int) (string, bool) {
   147  	var text strings.Builder
   148  	text.WriteString(lines[declIdx][from:])
   149  	if strings.HasSuffix(strings.TrimSpace(lines[declIdx]), ")") {
   150  		return text.String(), true
   151  	}
   152  	for j := declIdx + 1; j < len(lines) && j-declIdx <= skipReasonMaxLines; j++ {
   153  		text.WriteString(lines[j])
   154  		if strings.HasSuffix(strings.TrimSpace(lines[j]), ")") {
   155  			return text.String(), true
   156  		}
   157  	}
   158  	return text.String(), false
   159  }
