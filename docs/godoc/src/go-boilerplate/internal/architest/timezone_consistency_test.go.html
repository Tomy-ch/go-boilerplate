   141  // TestPostgresProvisionersDeclareTimeZone は、PostgreSQL を起動する compose / ワークフローが
   142  // TZ と PGTZ を宣言していることを機械検証します。
   143  //
   144  // 値の一致を見る TestTimezoneMechanismValuesMatch は、宣言が丸ごと消えた欠落を検出できません
   145  // （他のファイルに同じキーが残る限り、比較対象が減るだけで一致は保たれます）。宣言の存在を
   146  // 要求するのはこのテストだけで、compose とワークフローの両方を対象にするのはそのためです。
   147  //
   148  // GitHub Actions はサービス定義をワークフロー間で共有できないため、同じ宣言が全ファイルに写経
   149  // されます。写経の漏れは変数名での grep には映らず（宣言済みのファイルしか引っ掛からない）、
   150  // 漏れたファイルだけが UTC のクラスタ既定で走ります。判定を image 名側から行うことで、
   151  // 「PostgreSQL を使うのに TZ を宣言していない」ファイルを取りこぼしません。
   152  //
   153  // 宣言の有無はファイル全体で見ており、当該サービスのブロックへは紐付けていません。1 ファイルに
   154  // PostgreSQL 以外のサービスが増え、そちらだけが TZ を宣言する状態は素通りします。YAML の階層を
   155  // テキスト走査で切り出す脆さを負うより、1 ファイル 1 PostgreSQL という現状に合わせた割り切りです。

   279  // parseDockerfileStages は、Dockerfile の内容を FROM 単位のステージへ分け、ローカルタイムに
   280  // 関わる宣言を取り出して返します。継続行は前の行へ畳んでから解釈します。
   281  // depguard が go/ast を禁じるためテキスト走査で行います（既存 architest と同方針）。
   282  func parseDockerfileStages(content string) []dockerfileStage {
   283  	var stages []dockerfileStage
   284  	for _, line := range joinDockerfileContinuations(content) {
   285  		if from := dockerfileFromRe.FindStringSubmatch(line); from != nil {
   286  			name := from[1]
   287  			if name == "" {
   288  				name = unnamedDockerfileStage
   289  			}
   290  			stages = append(stages, dockerfileStage{name: name})
   291  			continue
   292  		}
   293  		if len(stages) == 0 {
   294  			continue
   295  		}
   296  
   297  		current := &stages[len(stages)-1]
   298  		if env := dockerfileEnvRe.FindStringSubmatch(line); env != nil {
   299  			if tz := dockerfileEnvTZRe.FindStringSubmatch(env[1]); tz != nil {
   300  				current.timeZone = tz[1]
   301  				current.hasTimeZone = true
   302  			}
   303  		}
   304  		if dockerfileTzdataRe.MatchString(line) {
   305  			current.installsTzdata = true
   306  		}
   307  	}
   308  
   309  	return stages
   310  }

   312  // joinDockerfileContinuations は、末尾を \ で継続する行を後続行と 1 行へ畳み、前後の空白を
   313  // 落とした行の並びを返します。複数行に散った ENV や RUN の記述を 1 行として解釈するためです。
   314  func joinDockerfileContinuations(content string) []string {
   315  	var (
   316  		lines   []string
   317  		pending string
   318  	)
   319  	for raw := range strings.SplitSeq(content, "\n") {
   320  		line := strings.TrimSpace(raw)
   321  		if dockerfileContinuationRe.MatchString(line) {
   322  			pending += strings.TrimSpace(dockerfileContinuationRe.ReplaceAllString(line, "")) + " "
   323  			continue
   324  		}
   325  		lines = append(lines, strings.TrimSpace(pending+line))
   326  		pending = ""
   327  	}
   328  	if pending != "" {
   329  		lines = append(lines, strings.TrimSpace(pending))
   330  	}
   331  
   332  	return lines
   333  }

   335  // globRepoFiles は、リポジトリルートからの相対 glob に一致するファイルを、同じく相対パスで返します。
   336  func globRepoFiles(t *testing.T, root string, patterns ...string) []string {
   337  	t.Helper()
   338  
   339  	var files []string
   340  	for _, pattern := range patterns {
   341  		matched, err := pkgfs.OS{}.Glob(filepath.Join(root, pattern))
   342  		require.NoErrorf(t, err, "%s に一致するファイルを列挙できない", pattern)
   343  
   344  		for _, path := range matched {
   345  			rel, err := filepath.Rel(root, path)
   346  			require.NoErrorf(t, err, "%s の相対パスを求められない", path)
   347  			files = append(files, filepath.ToSlash(rel))
   348  		}
   349  	}
   350  
   351  	slices.Sort(files)
   352  	return files
   353  }
