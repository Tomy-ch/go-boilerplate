    39  // Run は、ctx に Idempotency-Key が無ければ businessFn を素通し実行し、ある場合は業務 tx 内で
    40  // claim → businessFn → complete を 1 tx で行います。再送は replay、並行 claim は 409、指紋不一致は 422。
    41  // successStatus は completed 行へ保存する HTTP ステータス、戻り値 replayed は再生したことを表します。
    42  //
    43  // 前提: 本関数は「成功時のステータスが常に successStatus 単一」の操作向けです（例: PostUsers は 201 のみ）。
    44  // replay 時は保存済みの応答ボディ（T）のみを復元し、保存済み response_status は呼び出し側へ伝播しません。
    45  func Run[T any](
    46  	ctx context.Context,
    47  	deps Deps,
    48  	successStatus int,
    49  	businessFn func(ctx context.Context) (T, error),
    50  ) (T, bool, error) {
    51  	req, ok := requestFromContext(ctx)
    52  	if !ok || req.Scope == "" {
    53  		res, err := businessFn(ctx)
    54  		return res, false, err
    55  	}
    56  
    57  	var result T
    58  	var replayed bool
    59  	err := deps.Txm.Do(ctx, func(ctx context.Context) error {
    60  		claimed, err := deps.Store.Claim(ctx, idempotencybndry.ClaimParams{
    61  			Scope:       req.Scope,
    62  			Key:         req.Key,
    63  			Method:      req.Method,
    64  			Path:        req.Path,
    65  			Fingerprint: req.Fingerprint,
    66  			ExpiresAt:   deps.Clock.Now().Add(ttl),
    67  		})
    68  		if err != nil {
    69  			if errors.Is(err, idempotencybndry.ErrLockTimeout) {
    70  				deps.metrics().IncConflict(req.OperationID)
    71  				return xerrors.Wrap(apperror.ErrConflict, "idempotency key is being processed, retry later")
    72  			}
    73  			return err
    74  		}
    75  
    76  		if !claimed {
    77  			res, replay, derr := decideExisting[T](ctx, deps, req)
    78  			if derr != nil {
    79  				return derr
    80  			}
    81  			result, replayed = res, replay
    82  			return nil
    83  		}
    84  
    85  		// 新規 claim 成立。業務処理を同一 tx で実行する（失敗は tx ロールバックで claim ごと解放）。
    86  		res, err := businessFn(ctx)
    87  		if err != nil {
    88  			return err
    89  		}
    90  		payload, err := json.Marshal(res)
    91  		if err != nil {
    92  			return xerrors.Wrap(apperror.ErrInternal, "failed to encode idempotent response: "+err.Error())
    93  		}
    94  		if err := deps.Store.Complete(ctx, idempotencybndry.CompleteParams{
    95  			Scope:           req.Scope,
    96  			Key:             req.Key,
    97  			ResponseStatus:  int32(successStatus), //nolint:gosec // HTTP ステータスコードは int32 に収まる
    98  			ResponsePayload: payload,
    99  		}); err != nil {
   100  			return err
   101  		}
   102  		result = res
   103  		return nil
   104  	})
   105  	if err != nil {
   106  		var zero T
   107  		return zero, false, err
   108  	}
   109  	return result, replayed, nil
   110  }

   112  // decideExisting は、既存キー（claim 衝突）に対して replay / 409 / 422 を判定します。
   113  func decideExisting[T any](
   114  	ctx context.Context, deps Deps, req Request,
   115  ) (T, bool, error) {
   116  	var zero T
   117  	rec, err := deps.Store.Get(ctx, req.Scope, req.Key)
   118  	if err != nil {
   119  		return zero, false, err
   120  	}
   121  	if rec == nil {
   122  		// claim 衝突直後に行が消えた稀なレース。後で再試行させる。
   123  		return zero, false, xerrors.Wrap(apperror.ErrConflict, "idempotency key state unavailable, retry later")
   124  	}
   125  
   126  	if !bytes.Equal(rec.Fingerprint, req.Fingerprint) {
   127  		deps.metrics().IncFingerprintMismatch(req.OperationID)
   128  		return zero, false, xerrors.Wrap(apperror.ErrValidation, "idempotency key reused with a different request")
   129  	}
   130  
   131  	if rec.Status != idempotencybndry.StatusCompleted {
   132  		deps.metrics().IncConflict(req.OperationID)
   133  		return zero, false, xerrors.Wrap(apperror.ErrConflict, "idempotency key is being processed, retry later")
   134  	}
   135  
   136  	// completed → 保存済み DTO を復元して replay。
   137  	var result T
   138  	if err := json.Unmarshal(rec.ResponsePayload, &result); err != nil {
   139  		return zero, false, xerrors.Wrap(apperror.ErrInternal, "failed to decode stored idempotent response: "+err.Error())
   140  	}
   141  	deps.metrics().IncReplay(req.OperationID)
   142  	return result, true, nil
   143  }
