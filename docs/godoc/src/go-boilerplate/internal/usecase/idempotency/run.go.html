    49  // Run は、ctx に Idempotency-Key が無ければ businessFn を素通し実行し、ある場合は業務 tx 内で
    50  // claim → businessFn → complete を 1 tx で行います。再送は replay、並行 claim は 409、指紋不一致は 422。
    51  // successStatus は completed 行へ保存する HTTP ステータス、戻り値 replayed は再生したことを表します。
    52  //
    53  // 前提: 本関数は「成功時のステータスが常に successStatus 単一」の操作向けです（例: PostUsers は 201 のみ）。
    54  // replay 時は保存済みの応答ボディ（T）のみを復元し、保存済み response_status は呼び出し側へ伝播しません。
    55  func Run[T any](
    56  	ctx context.Context,
    57  	deps Deps,
    58  	successStatus int,
    59  	businessFn func(ctx context.Context) (T, error),
    60  ) (T, bool, error) {
    61  	req, ok := requestFromContext(ctx)
    62  	if !ok || req.Scope == "" {
    63  		res, err := businessFn(ctx)
    64  		return res, false, err
    65  	}
    66  
    67  	var result T
    68  	var replayed bool
    69  	err := deps.Txm.Do(ctx, func(ctx context.Context) error {
    70  		claimed, err := deps.Store.Claim(ctx, idempotencybndry.ClaimParams{
    71  			Scope:       req.Scope,
    72  			Key:         req.Key,
    73  			Method:      req.Method,
    74  			Path:        req.Path,
    75  			Fingerprint: req.Fingerprint,
    76  			ExpiresAt:   deps.Clock.Now().Add(ttl),
    77  		})
    78  		if err != nil {
    79  			if errors.Is(err, idempotencybndry.ErrLockTimeout) {
    80  				deps.metrics().IncConflict(ctx, req.OperationID)
    81  				return xerrors.Wrap(apperror.ErrConflict, "idempotency key is being processed, retry later")
    82  			}
    83  			deps.metrics().IncClaimFailure(ctx, req.OperationID)
    84  			return err
    85  		}
    86  
    87  		if !claimed {
    88  			res, replay, derr := decideExisting[T](ctx, deps, req)
    89  			if derr != nil {
    90  				return derr
    91  			}
    92  			result, replayed = res, replay
    93  			return nil
    94  		}
    95  
    96  		// 新規 claim 成立。業務処理を同一 tx で実行する（失敗は tx ロールバックで claim ごと解放）。
    97  		deps.metrics().IncMiss(ctx, req.OperationID)
    98  		res, err := businessFn(ctx)
    99  		if err != nil {
   100  			return err
   101  		}
   102  		payload, err := json.Marshal(res)
   103  		if err != nil {
   104  			return xerrors.Join(apperror.ErrInternal, xerrors.Wrap(err, "failed to encode idempotent response"))
   105  		}
   106  		if err := deps.Store.Complete(ctx, idempotencybndry.CompleteParams{
   107  			Scope:           req.Scope,
   108  			Key:             req.Key,
   109  			ResponseStatus:  int32(successStatus), //nolint:gosec // HTTP ステータスコードは int32 に収まる
   110  			ResponsePayload: payload,
   111  		}); err != nil {
   112  			deps.metrics().IncCompleteFailure(ctx, req.OperationID)
   113  			return err
   114  		}
   115  		result = res
   116  		return nil
   117  	})
   118  	if err != nil {
   119  		var zero T
   120  		return zero, false, err
   121  	}
   122  	return result, replayed, nil
   123  }

   125  // decideExisting は、既存キー（claim 衝突）に対して replay / 409 / 422 を判定します。
   126  func decideExisting[T any](
   127  	ctx context.Context, deps Deps, req Request,
   128  ) (T, bool, error) {
   129  	var zero T
   130  	rec, err := deps.Store.Get(ctx, req.Scope, req.Key)
   131  	if err != nil {
   132  		return zero, false, err
   133  	}
   134  	if rec == nil {
   135  		// claim 衝突直後に行が消えた稀なレース。後で再試行させる。
   136  		deps.metrics().IncConflict(ctx, req.OperationID)
   137  		return zero, false, xerrors.Wrap(apperror.ErrConflict, "idempotency key state unavailable, retry later")
   138  	}
   139  
   140  	if !bytes.Equal(rec.Fingerprint, req.Fingerprint) {
   141  		deps.metrics().IncFingerprintMismatch(ctx, req.OperationID)
   142  		return zero, false, xerrors.Wrap(apperror.ErrValidation, "idempotency key reused with a different request")
   143  	}
   144  
   145  	if rec.Status != idempotencybndry.StatusCompleted {
   146  		deps.metrics().IncConflict(ctx, req.OperationID)
   147  		return zero, false, xerrors.Wrap(apperror.ErrConflict, "idempotency key is being processed, retry later")
   148  	}
   149  
   150  	// completed → 保存済み DTO を復元して replay。
   151  	var result T
   152  	if err := json.Unmarshal(rec.ResponsePayload, &result); err != nil {
   153  		return zero, false, xerrors.Join(apperror.ErrInternal, xerrors.Wrap(err, "failed to decode stored idempotent response"))
   154  	}
   155  	deps.metrics().IncHit(ctx, req.OperationID)
   156  	return result, true, nil
   157  }
