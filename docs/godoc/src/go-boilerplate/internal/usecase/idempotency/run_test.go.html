    36  // newDeps は、業務 tx を 1 回だけ実行する想定の Deps を返します（Txm.Do は Times(1)）。
    37  // 冪等処理に入る（Idempotency-Key あり）経路専用で、素通し経路では newDepsNoTx を使います。
    38  func newDeps(t *testing.T, store idempotencybndry.Store) idempotency.Deps {
    39  	t.Helper()
    40  	txm := mock_tx.NewMockManager(gomock.NewController(t))
    41  	txm.EXPECT().Do(gomock.Any(), gomock.Any()).DoAndReturn(
    42  		func(ctx context.Context, fn func(context.Context) error) error { return fn(ctx) }).Times(1)
    43  
    44  	return idempotency.Deps{
    45  		Txm:   txm,
    46  		Store: store,
    47  		Clock: testkit.NewMockClock(t, fixedNow),
    48  	}
    49  }
    50  
    51  // newDepsNoTx は、Txm.Do が一度も呼ばれないこと（Times(0)）を検証するための Deps を返します。
    52  // 素通し経路（Idempotency-Key 無し / Scope 空）で tx が使われないことの確認に用います。
    53  func newDepsNoTx(t *testing.T, store idempotencybndry.Store) idempotency.Deps {
    54  	t.Helper()
    55  	txm := mock_tx.NewMockManager(gomock.NewController(t))
    56  	txm.EXPECT().Do(gomock.Any(), gomock.Any()).Times(0)
    57  
    58  	return idempotency.Deps{
    59  		Txm:   txm,
    60  		Store: store,
    61  		Clock: testkit.NewMockClock(t, fixedNow),
    62  	}
    63  }

    65  // newDepsWithMetrics は、newDeps に Metrics mock を差した Deps を返します。
    66  func newDepsWithMetrics(
    67  	t *testing.T, store idempotencybndry.Store, m idempotency.Metrics,
    68  ) idempotency.Deps {

    70  	deps := newDeps(t, store)
    71  	deps.Metrics = m
    72  	return deps
