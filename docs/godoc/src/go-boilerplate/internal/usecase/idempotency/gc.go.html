    44  // SweepExpired は、失効行を batchSize 件ずつ削除し、合計削除件数を返します。
    45  func (g *gcUsecase) SweepExpired(ctx context.Context, batchSize int32) (int64, error) {
    46  	if batchSize <= 0 {
    47  		batchSize = DefaultGCBatchSize
    48  	}
    49  	cutoff := g.clock.Now()
    50  
    51  	var total int64
    52  	for {
    53  		deleted, err := g.store.DeleteExpired(ctx, cutoff, batchSize)
    54  		if err != nil {
    55  			g.metrics().IncExpiredCleanupFailure(ctx)
    56  			return total, err
    57  		}
    58  		if deleted > 0 {
    59  			g.metrics().IncExpiredCleanup(ctx, deleted)
    60  		}
    61  		total += deleted
    62  		// バッチが満たなかった = もう失効行は無い。
    63  		if deleted < int64(batchSize) {
    64  			return total, nil
    65  		}
    66  	}
    67  }
