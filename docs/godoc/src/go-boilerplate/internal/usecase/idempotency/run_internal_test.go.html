   127  func Test_decideExisting(t *testing.T) {
   128  	t.Parallel()
   129  
   130  	type payloadT struct {
   131  		Value string
   132  	}
   133  	fp := []byte("fingerprint")
   134  	req := Request{Scope: "user-1", Key: "key-1", Fingerprint: fp}
   135  
   136  	newDeps := func(t *testing.T, rec *idempotencybndry.Record, getErr error) Deps {
   137  		t.Helper()
   138  		store := mock_idempotency.NewMockStore(gomock.NewController(t))
   139  		store.EXPECT().Get(gomock.Any(), req.Scope, req.Key).Return(rec, getErr)
   140  		return Deps{Store: store}
   141  	}
   142  
   143  	t.Run("正常系", func(t *testing.T) {
   144  		t.Parallel()
   145  
   146  		t.Run("completed かつ fingerprint 一致なら保存済み DTO を復元して replay する", func(t *testing.T) {
   147  			t.Parallel()
   148  
   149  			deps := newDeps(t, &idempotencybndry.Record{
   150  				Status:          idempotencybndry.StatusCompleted,
   151  				Fingerprint:     fp,
   152  				ResponsePayload: []byte(`{"Value":"stored"}`),
   153  			}, nil)
   154  
   155  			result, replayed, err := decideExisting[payloadT](context.Background(), deps, req)
   156  
   157  			require.NoError(t, err)
   158  			assert.True(t, replayed)
   159  			assert.Equal(t, payloadT{Value: "stored"}, result)
   160  		})
   161  	})
   162  
   163  	t.Run("異常系", func(t *testing.T) {
   164  		t.Parallel()
   165  
   166  		t.Run("Store.Get がエラーならそのまま伝播する", func(t *testing.T) {
   167  			t.Parallel()
   168  
   169  			wantErr := xerrors.New("store down")
   170  			deps := newDeps(t, nil, wantErr)
   171  
   172  			_, replayed, err := decideExisting[payloadT](context.Background(), deps, req)
   173  
   174  			require.ErrorIs(t, err, wantErr)
   175  			assert.False(t, replayed)
   176  		})
   177  
   178  		t.Run("claim 直後にエントリが消えたレースは ErrConflict を返す", func(t *testing.T) {
   179  			t.Parallel()
   180  
   181  			deps := newDeps(t, nil, nil)
   182  
   183  			_, replayed, err := decideExisting[payloadT](context.Background(), deps, req)
   184  
   185  			require.ErrorIs(t, err, apperror.ErrConflict)
   186  			assert.False(t, replayed)
   187  		})
   188  
   189  		t.Run("fingerprint 不一致は ErrValidation を返す", func(t *testing.T) {

   192  			deps := newDeps(t, &idempotencybndry.Record{
