    81  func (u *purgeUsecase) PurgeDeleted(
    82  	ctx context.Context, retention time.Duration, batchSize int32, dryRun bool,
    83  ) (PurgeResult, error) {
    84  	ctx, endSpan := u.tracer.Start(ctx)
    85  	defer endSpan()
    86  
    87  	if retention <= 0 {
    88  		retention = DefaultPurgeRetention
    89  	}
    90  	if batchSize <= 0 {
    91  		batchSize = DefaultPurgeBatchSize
    92  	}
    93  	cutoff := u.clock.Now().Add(-retention)
    94  
    95  	var (
    96  		result  PurgeResult
    97  		afterID *uuid.UUID
    98  	)
    99  	for {
   100  		batch, err := u.purgeBatch(ctx, cutoff, afterID, batchSize, dryRun)
   101  		if err != nil {
   102  			// 失敗したバッチはロールバックされるが、それ以前のバッチのコミット済み削除は戻らない。
   103  			// 物理削除の実績を呼び手が失わないよう、累計を捨てずに返す。
   104  			return result, err
   105  		}
   106  		result.Purged += batch.purged
   107  		result.SkippedWithPurchases += batch.skipped
   108  
   109  		// 候補がバッチを満たさなかった = これ以上対象は無い。
   110  		if batch.candidates < int(batchSize) {
   111  			return result, nil
   112  		}
   113  		// 削除しなかった候補を跨いで前進するため、境界は削除可否によらず候補の末尾まで進める。
   114  		// ここを進めないと、先頭バッチが全件スキップ対象のとき同じ候補を取り直し続けてしまう。
   115  		afterID = batch.nextAfter
   116  	}
   117  }

   119  // purgeBatch は、1 バッチ分の候補列挙・購入照会・物理削除を 1 トランザクションで実行します。
   120  func (u *purgeUsecase) purgeBatch(
   121  	ctx context.Context, cutoff time.Time, afterID *uuid.UUID, batchSize int32, dryRun bool,
   122  ) (purgeBatchResult, error) {
   123  	return tx.DoWithResult(ctx, u.txm, func(ctx context.Context) (purgeBatchResult, error) {
   124  		ids, err := u.userRepo.FindDeletedBefore(ctx, cutoff, afterID, batchSize)
   125  		if err != nil {
   126  			return purgeBatchResult{}, err
   127  		}
   128  		if len(ids) == 0 {
   129  			return purgeBatchResult{}, nil
   130  		}
   131  
   132  		withPurchases, err := u.purchaseRepo.FindUserIDsWithPurchases(ctx, ids)
   133  		if err != nil {
   134  			return purgeBatchResult{}, err
   135  		}
   136  
   137  		batch := purgeBatchResult{
   138  			candidates: len(ids),
   139  			skipped:    int64(len(withPurchases)),
   140  			nextAfter:  &ids[len(ids)-1],
   141  		}
   142  		targets := excludeIDs(ids, withPurchases)
   143  		if dryRun {
   144  			batch.purged = int64(len(targets))
   145  			return batch, nil
   146  		}
   147  
   148  		purged, err := u.userRepo.PurgeByIDs(ctx, targets)
   149  		if err != nil {
   150  			return purgeBatchResult{}, err
   151  		}
   152  		batch.purged = purged
   153  		return batch, nil
   154  	})
   155  }

   157  // excludeIDs は、ids から excluded に含まれる ID を取り除いた並びを返します。
   158  func excludeIDs(ids, excluded []uuid.UUID) []uuid.UUID {
   159  	if len(excluded) == 0 {
   160  		return ids
   161  	}
   162  	excludedSet := make(map[uuid.UUID]struct{}, len(excluded))
   163  	for _, id := range excluded {
   164  		excludedSet[id] = struct{}{}
   165  	}
   166  	remained := make([]uuid.UUID, 0, len(ids))
   167  	for _, id := range ids {
   168  		if _, ok := excludedSet[id]; !ok {
   169  			remained = append(remained, id)
   170  		}
   171  	}
   172  	return remained
   173  }
