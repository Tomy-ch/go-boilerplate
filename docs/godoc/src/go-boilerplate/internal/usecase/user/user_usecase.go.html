   161  	defer endSpan()
   162  
   163  	now := u.clock.Now()
   164  	rawPassword, err := user.NewRawPassword(dto.RawPassword)

   171  		return UserView{}, err

   174  	var (
   175  		userEntity *user.User
   176  		pftName    string
   177  	)
   178  	err = u.txm.Do(ctx, func(ctx context.Context) error {
   179  		pftDomain, err := u.pftRepo.FindByName(ctx, dto.PrefectureName)
   180  		if err != nil {
   181  			return err
   182  		}
   183  		pftName = pftDomain.Name()
   184  
   185  		userEntity, err = user.New(
   186  			dto.UserID,
   187  			dto.FirstName,
   188  			dto.LastName,
   189  			passwordHash,
   190  			dto.Email,
   191  			dto.Phone,
   192  			pftDomain.ID(),
   193  			dto.City,
   194  			dto.Street,
   195  			dto.Building,
   196  			dto.PostalCode,
   197  			now,
   198  			now,
   199  			nil,
   200  		)
   201  		if err != nil {
   202  			return err
   203  		}
   204  
   205  		return u.userRepo.Create(ctx, userEntity)
   206  	})
   207  	if err != nil {
   208  		return UserView{}, err
   209  	}
   210  
   211  	return toUserView(userEntity, pftName), nil
   212  }

   237  // ListUsersFeed は、未削除ユーザーを作成日時の降順（cursor ページネーション）で取得するユースケースです。
   238  func (u *usecase) ListUsersFeed(ctx context.Context, cursor *paging.Cursor) (*UserFeedView, error) {
   239  	if cursor == nil {
   240  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "cursor must not be nil")
   241  	}
   242  
   243  	ctx, endSpan := u.tracer.Start(ctx)
   244  	defer endSpan()
   245  
   246  	after, err := decodeFeedCursor(cursor)
   247  	if err != nil {
   248  		return nil, err
   249  	}
   250  
   251  	us, err := u.userRepo.FindFeed(ctx, after, cursor.Limit32()+1)
   252  	if err != nil {
   253  		return nil, err
   254  	}
   255  
   256  	limit := cursor.Limit()
   257  	hasNext := len(us) > limit
   258  	if hasNext {
   259  		us = us[:limit]
   260  	}
   261  
   262  	items, err := u.toUserViews(ctx, us)
   263  	if err != nil {
   264  		return nil, err
   265  	}
   266  
   267  	var nextCursor *string
   268  	if hasNext && len(us) > 0 {
   269  		encoded := encodeFeedCursor(us[len(us)-1])
   270  		nextCursor = &encoded
   271  	}
   272  
   273  	return &UserFeedView{Items: items, NextCursor: nextCursor}, nil
   274  }

   345  // ChangePassword は、現在のパスワードを照合したうえでユーザーのパスワードを変更するユースケースです。
   346  func (u *usecase) ChangePassword(ctx context.Context, id uuid.UUID, currentPassword, newPassword string) error {
   347  	ctx, endSpan := u.tracer.Start(ctx)
   348  	defer endSpan()
   349  
   350  	now := u.clock.Now()
   351  	// 現パスワードも newPassword と同じ長さ制約で検証する（bcrypt の 72 バイト切り詰めを避け、入力の対称性を保つ）。
   352  	if _, err := user.NewRawPassword(currentPassword); err != nil {
   353  		return err
   354  	}
   355  	rawNew, err := user.NewRawPassword(newPassword)
   356  	if err != nil {
   357  		return err
   358  	}
   359  
   360  	return u.txm.Do(ctx, func(ctx context.Context) error {
   361  		userEntity, err := u.userRepo.FindByID(ctx, id)
   362  		if err != nil {
   363  			return err
   364  		}
   365  
   366  		matched, err := u.encrypter.Compare(userEntity.PasswordHash(), currentPassword)
   367  		if err != nil {
   368  			return err
   369  		}
   370  		if !matched {
   371  			return user.ErrCurrentPasswordMismatch
   372  		}
   373  
   374  		newHash, err := u.encrypter.Hash(rawNew.Value())
   375  		if err != nil {
   376  			return err
   377  		}
   378  
   379  		if err = userEntity.ChangePassword(newHash, now); err != nil {
   380  			return err
   381  		}
   382  
   383  		return u.userRepo.Update(ctx, userEntity)
   384  	})
   385  }

   387  // UpdateUserPartially は、ユーザーを部分更新するユースケースです（PATCH）。
   388  // 指定フィールドのみ更新し、未指定/null は据え置く（クリアは非対応。クリアは PUT を使う）。password は更新しない。
   389  func (u *usecase) UpdateUserPartially(ctx context.Context, id uuid.UUID, dto *PatchParamsDTO) (UserView, error) {
   390  	ctx, endSpan := u.tracer.Start(ctx)
   391  	defer endSpan()
   392  
   393  	now := u.clock.Now()
   394  
   395  	var (
   396  		userEntity *user.User
   397  		pftName    string
   398  	)
   399  	err := u.txm.Do(ctx, func(ctx context.Context) error {
   400  		var err error
   401  		userEntity, err = u.userRepo.FindByID(ctx, id)
   402  		if err != nil {
   403  			return err
   404  		}
   405  
   406  		prefectureID := userEntity.PrefectureID()
   407  		pftDomain, err := u.resolvePatchPrefecture(ctx, dto.PrefectureName, prefectureID)
   408  		if err != nil {
   409  			return err
   410  		}
   411  		prefectureID = pftDomain.ID()
   412  		pftName = pftDomain.Name()
   413  
   414  		// provided なフィールドのみ現在値に上書きしたフルセットを構築
   415  		building := userEntity.Building()
   416  		if dto.Building != nil {
   417  			building = dto.Building
   418  		}
   419  
   420  		if err = userEntity.UpdateProfile(
   421  			ptr.Deref(dto.FirstName, userEntity.FirstName()),
   422  			ptr.Deref(dto.LastName, userEntity.LastName()),
   423  			ptr.Deref(dto.Email, userEntity.Email()),
   424  			ptr.Deref(dto.Phone, userEntity.Phone()),
   425  			prefectureID,
   426  			ptr.Deref(dto.City, userEntity.City()),
   427  			ptr.Deref(dto.Street, userEntity.Street()),
   428  			building,
   429  			ptr.Deref(dto.PostalCode, userEntity.PostalCode()),
   430  			now,
   431  		); err != nil {
   432  			return err
   433  		}
   434  		return u.userRepo.Update(ctx, userEntity)
   435  	})
   436  	if err != nil {
   437  		return UserView{}, err
   438  	}
   439  
   440  	return toUserView(userEntity, pftName), nil
   441  }

   461  // toUserViews は、ユーザーエンティティ列を、都道府県名を一括解決した DTO 列へ変換します。
   462  // いずれかのユーザーが参照する都道府県を解決できない場合は参照整合性破れ（errOrphanPrefecture）を返します。
   463  func (u *usecase) toUserViews(ctx context.Context, us user.Users) ([]UserView, error) {
   464  	_, prefectureMap, err := observability.RunWithSpan(
   465  		ctx, u.tracer, "usecase", "user", "prefectureMap", func(ctx context.Context) (map[uuid.UUID]*prefecture.Prefecture, error) {
   466  			pids := make([]uuid.UUID, len(us))
   467  			for i, ue := range us {
   468  				pids[i] = ue.PrefectureID()
   469  			}
   470  
   471  			ps, pftErr := u.pftRepo.FindByIDs(ctx, pids)
   472  			if pftErr != nil {
   473  				return nil, pftErr
   474  			}
   475  
   476  			prefectureMap := make(map[uuid.UUID]*prefecture.Prefecture, len(ps))
   477  			for _, p := range ps {
   478  				prefectureMap[p.ID()] = p
   479  			}
   480  
   481  			return prefectureMap, nil
   482  		})
   483  	if err != nil {
   484  		return nil, err
   485  	}
   486  
   487  	dtos := make([]UserView, len(us))
   488  	for i, ue := range us {
   489  		p, ok := prefectureMap[ue.PrefectureID()]
   490  		if !ok {
   491  			return nil, errOrphanPrefecture
   492  		}
   493  		dtos[i] = toUserView(ue, p.Name())
   494  	}
   495  
   496  	return dtos, nil
   497  }

   499  // resolvePatchPrefecture は、PATCH の都道府県を解決します。
   500  // 名前指定があれば名前で解決（入力エラーは伝播）、なければ既存 ID で解決します（未解決は参照整合性破れ）。
   501  func (u *usecase) resolvePatchPrefecture(ctx context.Context, name *string, currentID uuid.UUID) (*prefecture.Prefecture, error) {
   502  	if name != nil {
   503  		return u.pftRepo.FindByName(ctx, *name)
   504  	}
   505  	pftDomain, err := u.pftRepo.FindByID(ctx, currentID)
   506  	if errors.Is(err, apperror.ErrNotFound) {
   507  		return nil, errOrphanPrefecture
   508  	}
   509  	if err != nil {
   510  		return nil, err
   511  	}
   512  	return pftDomain, nil
   513  }

   515  // toUserView は、ユーザーエンティティと都道府県名から DTO を構築します。
   516  func toUserView(u *user.User, prefectureName string) UserView {
   517  	return UserView{
   518  		FirstName:      u.FirstName(),
   519  		LastName:       u.LastName(),
   520  		Email:          u.Email(),
   521  		Phone:          u.Phone(),
   522  		PostalCode:     u.PostalCode(),
   523  		PrefectureName: prefectureName,
   524  		City:           u.City(),
