   132  func (u *usecase) ListUsers(ctx context.Context, active *bool, page *paging.Paging) ([]UserView, error) {
   133  	if page == nil {
   134  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "page must not be nil")
   135  	}
   136  
   137  	ctx, endSpan := u.tracer.Start(ctx)
   138  	defer endSpan()
   139  
   140  	us, err := u.userRepo.FindByActive(ctx, active, page.Limit32(), page.Offset32())
   141  	if err != nil {
   142  		return nil, err
   143  	}
   144  
   145  	_, prefectureMap, err := observability.RunWithSpan(
   146  		ctx, u.tracer, "usecase", "user", "prefectureMap", func(ctx context.Context) (map[uuid.UUID]*prefecture.Prefecture, error) {
   147  			pids := make([]uuid.UUID, len(us))
   148  			for i, ue := range us {
   149  				pids[i] = ue.PrefectureID()
   150  			}
   151  
   152  			ps, pftErr := u.pftRepo.FindByIDs(ctx, pids)
   153  			if pftErr != nil {
   154  				return nil, pftErr
   155  			}
   156  
   157  			prefectureMap := make(map[uuid.UUID]*prefecture.Prefecture, len(ps))
   158  			for _, p := range ps {
   159  				prefectureMap[p.ID()] = p
   160  			}
   161  
   162  			return prefectureMap, nil
   163  		})
   164  	if err != nil {
   165  		return nil, err
   166  	}
   167  
   168  	dtos := make([]UserView, len(us))
   169  	for i, ue := range us {
   170  		p, ok := prefectureMap[ue.PrefectureID()]
   171  		if !ok {
   172  			return nil, errOrphanPrefecture
   173  		}
   174  		dtos[i] = toUserView(ue, p.Name())
   175  	}
   176  
   177  	return dtos, nil
   178  }

   207  		userEntity, err = user.New(

   328  // ChangePassword は、現在のパスワードを照合したうえでユーザーのパスワードを変更するユースケースです。
   329  func (u *usecase) ChangePassword(ctx context.Context, id uuid.UUID, currentPassword, newPassword string) error {
   330  	ctx, endSpan := u.tracer.Start(ctx)
   331  	defer endSpan()
   332  
   333  	now := u.clock.Now()
   334  	// 現パスワードも newPassword と同じ長さ制約で検証する（bcrypt の 72 バイト切り詰めを避け、入力の対称性を保つ）。
   335  	if _, err := user.NewRawPassword(currentPassword); err != nil {
   336  		return err
   337  	}
   338  	rawNew, err := user.NewRawPassword(newPassword)
   339  	if err != nil {
   340  		return err
   341  	}
   342  
   343  	return u.txm.Do(ctx, func(ctx context.Context) error {
   344  		userEntity, err := u.userRepo.FindByID(ctx, id)
   345  		if err != nil {
   346  			return err
   347  		}
   348  
   349  		matched, err := u.encrypter.Compare(userEntity.PasswordHash(), currentPassword)
   350  		if err != nil {
   351  			return err
   352  		}
   353  		if !matched {
   354  			return user.ErrCurrentPasswordMismatch
   355  		}
   356  
   357  		newHash, err := u.encrypter.Hash(rawNew.Value())
   358  		if err != nil {
   359  			return err
   360  		}
   361  
   362  		if err = userEntity.ChangePassword(newHash, now); err != nil {
   363  			return err
   364  		}
   365  
   366  		return u.userRepo.Update(ctx, userEntity)
   367  	})
   368  }
