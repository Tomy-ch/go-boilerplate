    85  func (u *usecase) GetDashboardSummary(
    86  	ctx context.Context, authn *auth.Authn, params GetSummaryParams,
    87  ) (SummaryView, error) {
    88  	ctx, endSpan := u.tracer.Start(ctx)
    89  	defer endSpan()
    90  
    91  	if authn == nil {
    92  		return SummaryView{}, xerrors.Wrap(apperror.ErrUnauthenticated, "requires authenticated user")
    93  	}
    94  	// ダッシュボードは運用（admin）の参照で所有者概念を持たないため、所有者なしリソースとして認可する。
    95  	// 所有者を渡さないことで Authorizer の所有者フォールバックが働かず、admin だけが通る。
    96  	if err := u.authorizer.Authorize(
    97  		ctx, authn, authz.ActionDashboardRead, authz.NewResource(resourceKindDashboard, nil),
    98  	); err != nil {
    99  		return SummaryView{}, err
   100  	}
   101  
   102  	period, err := normalizePeriod(params)
   103  	if err != nil {
   104  		return SummaryView{}, err
   105  	}
   106  
   107  	sales, err := u.qs.SummarizeSales(ctx, period)
   108  	if err != nil {
   109  		return SummaryView{}, err
   110  	}
   111  	statusCounts, err := u.qs.CountPurchasesByStatus(ctx, period)
   112  	if err != nil {
   113  		return SummaryView{}, err
   114  	}
   115  	// 商品数は商品集約の属性による単純な件数であり、期間集計の投影ではないため Repository から取る。
   116  	productCounts, err := u.productRepo.Count(ctx)
   117  	if err != nil {
   118  		return SummaryView{}, err
   119  	}
   120  
   121  	return SummaryView{
   122  		SalesAmount:           sales.Amount,
   123  		SalesCount:            sales.Count,
   124  		PurchaseStatusCounts:  toStatusCountViews(statusCounts),
   125  		TotalProductCount:     productCounts.Total,
   126  		PublishedProductCount: productCounts.Published,
   127  	}, nil

   130  // normalizePeriod は、入力期間を集計区分へ正規化します。"month" / "range" のみ該当区分とし、それ以外は today として扱います。
   131  // range のときだけ開始日・終了日の相関を検証し、QueryService へは検証済みの指定だけが渡ります。
   132  func normalizePeriod(params GetSummaryParams) (query.Period, error) {
   133  	switch params.Period {
   134  	case string(query.PeriodMonth):
   135  		return query.Period{Kind: query.PeriodMonth}, nil
   136  	case string(query.PeriodRange):
   137  		if params.From == nil || params.To == nil {
   138  			return query.Period{}, xerrors.Wrap(apperror.ErrInvalidArgument, "period=range requires both from and to")
   139  		}
   140  		// 開始日・終了日は暦日のみが意味を持つため、比較も QueryService へ渡す値も暦日へ揃える。
   141  		from, to := dateOnly(*params.From), dateOnly(*params.To)
   142  		if to.Before(from) {
   143  			return query.Period{}, xerrors.Wrap(apperror.ErrInvalidArgument, "to must not be before from")
   144  		}
   145  		return query.Period{Kind: query.PeriodRange, From: from, To: to}, nil
   146  	default:
   147  		return query.Period{Kind: query.PeriodToday}, nil
   148  	}
   149  }

   151  // dateOnly は、時刻成分を落として t の暦日だけを表す時刻を返します。ロケーションは t のものを保ちます。
   152  func dateOnly(t time.Time) time.Time {
   153  	y, m, d := t.Date()
   154  	return time.Date(y, m, d, 0, 0, 0, 0, t.Location())
   155  }
