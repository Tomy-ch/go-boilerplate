    11  // resolveTemplate は、キャンペーンのテンプレートが保つ種別の名前を、クーポン集約の語彙へ解決します。
    12  // 名前の解決と値の検証はドメインが行い、ここは橋渡しだけです
    13  // （設計は docs/spec/usecase/campaign.md の Overview を参照）。
    14  func resolveTemplate(t campaign.Template) (coupon.Discount, coupon.Scope, error) {
    15  	discountKind, err := coupon.NewDiscountKindByName(t.DiscountKindName())
    16  	if err != nil {
    17  		return coupon.Discount{}, coupon.Scope{}, err
    18  	}
    19  	discount, err := coupon.NewDiscount(discountKind, t.DiscountValue())
    20  	if err != nil {
    21  		return coupon.Discount{}, coupon.Scope{}, err
    22  	}
    23  	if maxAmount := t.DiscountMaxAmount(); maxAmount != nil {
    24  		discount, err = discount.WithMaxAmount(*maxAmount)
    25  		if err != nil {
    26  			return coupon.Discount{}, coupon.Scope{}, err
    27  		}
    28  	}
    29  
    30  	scopeKind, err := coupon.NewScopeKindByName(t.ScopeKindName())
    31  	if err != nil {
    32  		return coupon.Discount{}, coupon.Scope{}, err
    33  	}
    34  	scope, err := coupon.NewScope(scopeKind, t.ScopeTargetID())
    35  	if err != nil {
    36  		return coupon.Discount{}, coupon.Scope{}, err
    37  	}
    38  
    39  	return discount, scope, nil
    40  }
