    45  // DefineCampaign は、admin がキャンペーンを 1 件定義します。

    49  func (u *usecase) DefineCampaign(

    62  
    63  	code, err := campaign.NewCode(params.Code)

    67

    69  		DiscountKindName:  params.DiscountKind,

    71  		DiscountMaxAmount: params.MaxAmount,
    72  		ScopeKindName:     params.ScopeKind,
    73  		ScopeTargetID:     params.ScopeTargetID,

    75  		UsableFrom:        params.UsableFrom,
    76  		ExpiresAt:         params.CouponExpiresAt,
    77  	})
    78  	if err != nil {
    79  		return CampaignView{}, err
    80  	}
    81  	if _, _, err = resolveTemplate(template); err != nil {

    84

    89  
    90  	c, err := campaign.New(id, campaign.Attributes{
    91  		Code:         code,
    92  		Template:     template,
    93  		StartsAt:     params.StartsAt,
    94  		EndsAt:       params.EndsAt,
    95  		TotalLimit:   params.TotalLimit,
    96  		PerUserLimit: params.PerUserLimit,
    97  	})
    98  	if err != nil {
    99  		return CampaignView{}, err
   100  	}
   101  
   102  	if err = u.campaignRepo.Create(ctx, c); err != nil {
   103  		return CampaignView{}, err
   104  	}
   105  
   106  	return toCampaignView(c), nil
   107  }
