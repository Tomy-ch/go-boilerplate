    39  	code string,
    40  ) (ClaimedCouponView, error) {
    41  	ctx, endSpan := u.tracer.Start(ctx)
    42  	defer endSpan()
    43  
    44  	userID, err := requireUserID(authn)
    45  	if err != nil {
    46  		return ClaimedCouponView{}, err
    47  	}
    48  
    49  	// 形式が不正なコードも、存在しないコードと同じ扱いにします。区別できると、
    50  	// 形式の当たりだけを先に絞り込めてしまいます。
    51  	claimCode, err := campaign.NewCode(code)
    52  	if err != nil {
    53  		return ClaimedCouponView{}, unclaimable(campaign.ErrCodeUnknown)
    54  	}
    55  
    56  	couponID, err := uuid.New()
    57  	if err != nil {
    58  		return ClaimedCouponView{}, xerrors.Wrap(err, "failed to generate coupon id")
    59  	}
    60  	claimID, err := uuid.New()
    61  	if err != nil {
    62  		return ClaimedCouponView{}, xerrors.Wrap(err, "failed to generate claim id")
    63  	}
    64  
    65  	now := u.clock.Now()
    66  
    67  	var issued *coupon.Coupon
    68  	err = u.txm.Do(ctx, func(ctx context.Context) error {
    69  		c, cerr := u.claimWithin(ctx, claimInput{
    70  			Code:      claimCode,
    71  			UserID:    userID,
    72  			CouponID:  couponID,
    73  			ClaimID:   claimID,
    74  			ClaimedAt: now,
    75  		})
    76  		if cerr != nil {
    77  			return cerr
    78  		}
    79  		issued = c
    80  
    81  		return nil
    82  	})
    83  	if err != nil {
    84  		return ClaimedCouponView{}, err
    85  	}
    86  
    87  	return toClaimedCouponView(issued), nil
    88  }

    98  // claimWithin は、開かれたトランザクションの中で受け取りを成立させ、発行したクーポンを返します。
    99  //
   100  // 呼び出し順がそのまま直列化の前提になります。ロック（LockByCode）を最初に取り、
   101  // 枚数を数えるのはその後です（ADR-0036 (ordered-pessimistic-row-locks) の決定 2）。
   102  func (u *usecase) claimWithin(ctx context.Context, in claimInput) (*coupon.Coupon, error) {
   103  	c, err := u.campaignRepo.LockByCode(ctx, in.Code)
   104  	if err != nil {
   105  		if xerrors.Is(err, apperror.ErrNotFound) {
   106  			return nil, unclaimable(campaign.ErrCodeUnknown)
   107  		}
   108  
   109  		return nil, err
   110  	}
   111  
   112  	claimed, err := u.campaignRepo.CountClaims(ctx, campaign.ClaimCountParams{
   113  		CampaignID: c.ID(),
   114  		UserID:     in.UserID,
   115  	})
   116  	if err != nil {
   117  		return nil, err
   118  	}
   119  
   120  	// 受け取りの事実は集約ルートだけが生む。記録を得られたことが、上限と期間の判定を
   121  	// 通ったことと同じ意味になる。
   122  	record, err := c.Claim(campaign.ClaimParams{
   123  		ClaimedAt:     in.ClaimedAt,
   124  		ClaimedByUser: claimed,
   125  		ClaimID:       in.ClaimID,
   126  		UserID:        in.UserID,
   127  		CouponID:      in.CouponID,
   128  	})
   129  	if err != nil {
   130  		return nil, unclaimable(err)
   131  	}
   132  
   133  	attrs, err := couponAttributesFor(c, in.UserID, in.ClaimedAt)
   134  	if err != nil {
   135  		return nil, err
   136  	}
   137  
   138  	issued, err := coupon.New(in.CouponID, attrs)
   139  	if err != nil {
   140  		return nil, err
   141  	}
   142  	if err = u.couponRepo.Create(ctx, issued); err != nil {
   143  		return nil, err
   144  	}
   145  	if err = u.campaignRepo.RecordClaim(ctx, record); err != nil {
   146  		return nil, err
   147  	}
   148  
   149  	return issued, nil
   150  }
