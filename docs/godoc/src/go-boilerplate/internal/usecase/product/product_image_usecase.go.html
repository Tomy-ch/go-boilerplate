    57  	ext, ok := allowedImageContentTypes[params.ContentType]

    61  	if int64(len(params.Data)) > u.maxUploadBytes {
    62  		return ProductImageView{}, xerrors.Wrap(apperror.ErrPayloadTooLarge, "image exceeds the maximum upload size")
    63  	}
    64  
    65  	id, err := uuid.New()
    66  	if err != nil {
    67  		return ProductImageView{}, xerrors.Wrap(err, "failed to generate object key")
    68  	}
    69  	key := imageKeyPrefix + id.String() + "." + ext
    70  
    71  	path, err := u.storage.Put(ctx, objectstorage.PutObject{
    72  		Key:          key,
    73  		Body:         params.Data,
    74  		ContentType:  params.ContentType,
    75  		CacheControl: imageCacheControl,
    76  	})
    77  	if err != nil {
    78  		return ProductImageView{}, err
    79  	}
    80  
    81  	return ProductImageView{Path: string(path)}, nil
