   180  func parseProductPriceFilter(name, value string) (*money.Price, error) {
   181  	if len(value) > maxProductPriceFilterLength {
   182  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, name+" is too long")
   183  	}
   184  	parsed, err := decimal.Parse(value)
   185  	if err != nil {
   186  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, name+" must be a non-negative decimal")
   187  	}
   188  	price, err := money.NewPrice(parsed)
   189  	if err != nil {
   190  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, name+" must be a non-negative decimal")
   191  	}
   192  	return &price, nil
   193  }

   195  // validateMasterFilter は、マスタ参照の絞り込みが非推奨の ID と後継のコードで二重に指定されていないことを
   196  // 検証します。両方が来た場合にどちらかを優先すると、送ったのに無視された側が静かに効かなくなるため拒否します。
   197  func validateMasterFilter(filter SearchFilter) error {
   198  	if filter.CategoryID != nil && len(filter.CategoryCodes) > 0 {
   199  		return xerrors.Wrap(apperror.ErrInvalidArgument, "categoryId and categoryCodes must not be specified together")
   200  	}
   201  	if filter.StatusID != nil && len(filter.StatusCodes) > 0 {
   202  		return xerrors.Wrap(apperror.ErrInvalidArgument, "statusId and statusCodes must not be specified together")
   203  	}
   204  	return nil
   205  }

   207  func parseProductListRange(filter SearchFilter) (productListRange, error) {
   208  	if err := validateMasterFilter(filter); err != nil {
   209  		return productListRange{}, err
   210  	}
   211  
   212  	result := productListRange{minQuantity: filter.MinQuantity, maxQuantity: filter.MaxQuantity}
   213  	if filter.MinPrice != nil {
   214  		minPrice, err := parseProductPriceFilter("minPrice", *filter.MinPrice)
   215  		if err != nil {
   216  			return productListRange{}, err
   217  		}
   218  		result.minPrice = minPrice
   219  	}
   220  	if filter.MaxPrice != nil {
   221  		maxPrice, err := parseProductPriceFilter("maxPrice", *filter.MaxPrice)
   222  		if err != nil {
   223  			return productListRange{}, err
   224  		}
   225  		result.maxPrice = maxPrice
   226  	}
   227  	if result.minPrice != nil && result.maxPrice != nil &&
   228  		result.minPrice.Decimal().Cmp(result.maxPrice.Decimal()) > 0 {
   229  		return productListRange{}, xerrors.Wrap(apperror.ErrInvalidArgument, "minPrice must not exceed maxPrice")
   230  	}
   231  	if result.minQuantity != nil && *result.minQuantity < 0 {
   232  		return productListRange{}, xerrors.Wrap(apperror.ErrInvalidArgument, "minQuantity must be non-negative")
   233  	}
   234  	if result.maxQuantity != nil && *result.maxQuantity < 0 {
   235  		return productListRange{}, xerrors.Wrap(apperror.ErrInvalidArgument, "maxQuantity must be non-negative")
   236  	}
   237  	if result.minQuantity != nil && result.maxQuantity != nil && *result.minQuantity > *result.maxQuantity {
   238  		return productListRange{}, xerrors.Wrap(apperror.ErrInvalidArgument, "minQuantity must not exceed maxQuantity")
   239  	}
   240  	return result, nil
   241  }

   243  func (u *usecase) ListProducts(ctx context.Context, params ListProductsParams) (ProductListView, error) {
   244  	if params.Cursor == nil {
   245  		return ProductListView{}, xerrors.Wrap(apperror.ErrInvalidArgument, "cursor must not be nil")
   246  	}
   247  
   248  	rangeFilter, err := parseProductListRange(params.SearchFilter)
   249  	if err != nil {
   250  		return ProductListView{}, err
   251  	}
   252  
   253  	ctx, endSpan := u.tracer.Start(ctx)
   254  	defer endSpan()
   255  
   256  	after, err := decodeProductCursor(params.Cursor)
   257  	if err != nil {
   258  		return ProductListView{}, err
   259  	}
   260  
   261  	domainParams := product.ListParams{
   262  		SearchFilter: toDomainSearchFilter(params.SearchFilter, rangeFilter),
   263  		Limit:        params.Cursor.Limit32() + 1,
   264  		Ascending:    params.Ascending,
   265  	}
   266  	if after != nil {
   267  		publishedAt := after.publishedAt
   268  		id := after.id
   269  		domainParams.AfterPublishedAt = &publishedAt
   270  		domainParams.AfterID = &id
   271  	}
   272  
   273  	products, err := u.repo.FindPublishedList(ctx, domainParams)
   274  	if err != nil {
   275  		return ProductListView{}, err
   276  	}
   277  	if err := ensurePublished(products); err != nil {
   278  		return ProductListView{}, err
   279  	}
   280  
   281  	limit := params.Cursor.Limit()
   282  	hasNext := len(products) > limit
   283  	if hasNext {
   284  		products = products[:limit]
   285  	}
   286  
   287  	items := make([]ProductView, len(products))
   288  	for i, p := range products {
   289  		items[i] = toProductView(p)
   290  	}
   291  
   292  	var nextCursor *string
   293  	if hasNext && len(products) > 0 {
   294  		// len 判定は防御的な安全弁。hasNext は len > limit なので limit >= 1 の下では冗長だが、
   295  		// limit の下限保証は paging.NewCursor 依存であり、ゼロ値 Cursor 混入時の products[-1] panic を防ぐ。
   296  		encoded := encodeProductCursor(products[len(products)-1])
   297  		nextCursor = &encoded
   298  	}
   299  
   300  	return ProductListView{Items: items, NextCursor: nextCursor}, nil
   301  }
