   210  func parseProductPriceFilter(name, value string) (*money.Price, error) {
   211  	if len(value) > maxProductPriceFilterLength {
   212  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, name+" is too long")
   213  	}
   214  	parsed, err := decimal.Parse(value)
   215  	if err != nil {
   216  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, name+" must be a non-negative decimal")
   217  	}
   218  	price, err := money.NewPrice(parsed)
   219  	if err != nil {
   220  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, name+" must be a non-negative decimal")
   221  	}
   222  	return &price, nil
   223  }

   225  // validateMasterFilter は、マスタ参照の絞り込みが非推奨の ID と後継のコードで二重に指定されていないことを
   226  // 検証します。両方が来た場合にどちらかを優先すると、送ったのに無視された側が静かに効かなくなるため拒否します。
   227  func validateMasterFilter(filter SearchFilter) error {
   228  	if filter.CategoryID != nil && len(filter.CategoryCodes) > 0 {
   229  		return xerrors.Wrap(apperror.ErrInvalidArgument, "categoryId and categoryCodes must not be specified together")
   230  	}
   231  	if filter.StatusID != nil && len(filter.StatusCodes) > 0 {
   232  		return xerrors.Wrap(apperror.ErrInvalidArgument, "statusId and statusCodes must not be specified together")
   233  	}
   234  	return nil
   235  }

   237  func parseProductListRange(filter SearchFilter) (productListRange, error) {
   238  	if err := validateMasterFilter(filter); err != nil {
   239  		return productListRange{}, err
   240  	}
   241  
   242  	result := productListRange{minQuantity: filter.MinQuantity, maxQuantity: filter.MaxQuantity}
   243  	if filter.MinPrice != nil {
   244  		minPrice, err := parseProductPriceFilter("minPrice", *filter.MinPrice)
   245  		if err != nil {
   246  			return productListRange{}, err
   247  		}
   248  		result.minPrice = minPrice
   249  	}
   250  	if filter.MaxPrice != nil {
   251  		maxPrice, err := parseProductPriceFilter("maxPrice", *filter.MaxPrice)
   252  		if err != nil {
   253  			return productListRange{}, err
   254  		}
   255  		result.maxPrice = maxPrice
   256  	}
   257  	if result.minPrice != nil && result.maxPrice != nil &&
   258  		result.minPrice.Decimal().Cmp(result.maxPrice.Decimal()) > 0 {
   259  		return productListRange{}, xerrors.Wrap(apperror.ErrInvalidArgument, "minPrice must not exceed maxPrice")
   260  	}
   261  	if result.minQuantity != nil && *result.minQuantity < 0 {
   262  		return productListRange{}, xerrors.Wrap(apperror.ErrInvalidArgument, "minQuantity must be non-negative")
   263  	}
   264  	if result.maxQuantity != nil && *result.maxQuantity < 0 {
   265  		return productListRange{}, xerrors.Wrap(apperror.ErrInvalidArgument, "maxQuantity must be non-negative")
   266  	}
   267  	if result.minQuantity != nil && result.maxQuantity != nil && *result.minQuantity > *result.maxQuantity {
   268  		return productListRange{}, xerrors.Wrap(apperror.ErrInvalidArgument, "minQuantity must not exceed maxQuantity")
   269  	}
   270  	return result, nil
   271  }

   273  func (u *usecase) ListProducts(
   274  	ctx context.Context, authn *auth.Authn, params ListProductsParams,
   275  ) (ProductListView, error) {
   276  	if params.Cursor == nil {
   277  		return ProductListView{}, xerrors.Wrap(apperror.ErrInvalidArgument, "cursor must not be nil")
   278  	}
   279  
   280  	rangeFilter, err := parseProductListRange(params.SearchFilter)
   281  	if err != nil {
   282  		return ProductListView{}, err
   283  	}
   284  
   285  	ctx, endSpan := u.tracer.Start(ctx)
   286  	defer endSpan()
   287  
   288  	if err := u.authorizeUnpublishedRead(ctx, authn, params.IncludeUnpublished); err != nil {
   289  		return ProductListView{}, err
   290  	}
   291  
   292  	// 母集団が変わると並び順の軸も変わるため、取得とカーソル生成は同じ枝で揃えます。
   293  	var (
   294  		products product.Products
   295  		encode   func(last *product.Product) string
   296  	)
   297  	if params.IncludeUnpublished {
   298  		products, err = u.findAllPage(ctx, params, rangeFilter)
   299  		encode = encodeAllProductCursor
   300  	} else {
   301  		products, err = u.findPublishedPage(ctx, params, rangeFilter)
   302  		encode = encodeProductCursor
   303  	}
   304  	if err != nil {
   305  		return ProductListView{}, err
   306  	}
   307  
   308  	limit := params.Cursor.Limit()
   309  	hasNext := len(products) > limit
   310  	if hasNext {
   311  		products = products[:limit]
   312  	}
   313  
   314  	items := make([]ProductView, len(products))
   315  	for i, p := range products {
   316  		items[i] = toProductView(p)
   317  	}
   318  
   319  	var nextCursor *string
   320  	if hasNext && len(products) > 0 {
   321  		// len 判定は防御的な安全弁。hasNext は len > limit なので limit >= 1 の下では冗長だが、
   322  		// limit の下限保証は paging.NewCursor 依存であり、ゼロ値 Cursor 混入時の products[-1] panic を防ぐ。
   323  		encoded := encode(products[len(products)-1])
   324  		nextCursor = &encoded
   325  	}
   326  
   327  	return ProductListView{Items: items, NextCursor: nextCursor}, nil
   328  }

   331  	ctx context.Context, authn *auth.Authn, params CountProductsParams,
   332  ) (ProductCountView, error) {
   333  	rangeFilter, err := parseProductListRange(params.SearchFilter)
   334  	if err != nil {

   338  	ctx, endSpan := u.tracer.Start(ctx)

   352  	}
   353  	if err != nil {
   354  		return ProductCountView{}, err
   355  	}
   356

   377  // GetProduct は、存在秘匿を Repository が返す NotFound に委ね、usecase 側では公開判定を再実装しません。
   378  // Repository のエラーをそのまま伝播させることで、未存在と非公開が 404 として区別不能に保たれます。
   379  // 未公開を含める指定の可否は商品を引く前に判定するため、拒否から商品の存在有無は分かりません。
   380  func (u *usecase) GetProduct(
   381  	ctx context.Context, authn *auth.Authn, params GetProductParams,
   382  ) (ProductView, error) {
   383  	ctx, endSpan := u.tracer.Start(ctx)
   384  	defer endSpan()
   385  
   386  	if err := u.authorizeUnpublishedRead(ctx, authn, params.IncludeUnpublished); err != nil {
   387  		return ProductView{}, err
   388  	}
   389  
   390  	if params.IncludeUnpublished {
   391  		p, err := u.repo.FindByID(ctx, params.ID)
   392  		if err != nil {
   393  			return ProductView{}, err
   394  		}
   395  
   396  		return toProductView(p), nil
   397  	}
   398  
   399  	p, err := u.repo.FindPublishedByID(ctx, params.ID)
   400  	if err != nil {
   401  		return ProductView{}, err
   402  	}
   403  	if err := ensurePublished(product.Products{p}); err != nil {
   404  		return ProductView{}, err
   405  	}
   406  
   407  	return toProductView(p), nil
   408  }

   410  // authorizeUnpublishedRead は、未公開商品を含む参照の可否を判定します。
   411  // 既定の可視範囲（公開済みのみ）は認証を要さないため、includeUnpublished が false のときは何も課しません。
   412  // 一覧・一致件数・詳細はいずれも同じ能力を要求するため、判定はこの 1 箇所に集約します。
   413  func (u *usecase) authorizeUnpublishedRead(ctx context.Context, authn *auth.Authn, includeUnpublished bool) error {
   414  	if !includeUnpublished {
   415  		return nil
   416  	}
   417  	if authn == nil {
   418  		return apperror.ErrUnauthenticated
   419  	}
