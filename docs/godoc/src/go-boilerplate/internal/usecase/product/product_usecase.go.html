   132  func (u *usecase) ListProducts(ctx context.Context, params ListProductsParams) (ProductListView, error) {
   133  	if params.Cursor == nil {
   134  		return ProductListView{}, xerrors.Wrap(apperror.ErrInvalidArgument, "cursor must not be nil")
   135  	}
   136  
   137  	ctx, endSpan := u.tracer.Start(ctx)
   138  	defer endSpan()
   139  
   140  	after, err := decodeProductCursor(params.Cursor)
   141  	if err != nil {
   142  		return ProductListView{}, err
   143  	}
   144  
   145  	domainParams := product.ListParams{
   146  		Limit:      params.Cursor.Limit32() + 1,
   147  		Ascending:  params.Ascending,
   148  		CategoryID: params.CategoryID,
   149  		StatusID:   params.StatusID,
   150  		Keyword:    params.Keyword,
   151  	}
   152  	if after != nil {
   153  		publishedAt := after.publishedAt
   154  		id := after.id
   155  		domainParams.AfterPublishedAt = &publishedAt
   156  		domainParams.AfterID = &id
   157  	}
   158  
   159  	products, err := u.repo.FindPublishedList(ctx, domainParams)
   160  	if err != nil {
   161  		return ProductListView{}, err
   162  	}
   163  	if err := ensurePublished(products); err != nil {
   164  		return ProductListView{}, err
   165  	}
   166  
   167  	limit := params.Cursor.Limit()
   168  	hasNext := len(products) > limit
   169  	if hasNext {
   170  		products = products[:limit]
   171  	}
   172  
   173  	items := make([]ProductView, len(products))
   174  	for i, p := range products {
   175  		items[i] = toProductView(p)
   176  	}
   177  
   178  	var nextCursor *string
   179  	if hasNext && len(products) > 0 {
   180  		// len 判定は防御的な安全弁。hasNext は len > limit なので limit >= 1 の下では冗長だが、
   181  		// limit の下限保証は paging.NewCursor 依存であり、ゼロ値 Cursor 混入時の products[-1] panic を防ぐ。
   182  		encoded := encodeProductCursor(products[len(products)-1])
   183  		nextCursor = &encoded
   184  	}
   185  
   186  	return ProductListView{Items: items, NextCursor: nextCursor}, nil
   187  }

   189  // GetProduct は、存在秘匿を Repository が返す NotFound に委ね、usecase 側では公開判定を再実装しません。
   190  // Repository のエラーをそのまま伝播させることで、未存在と非公開が 404 として区別不能に保たれます。
   191  func (u *usecase) GetProduct(ctx context.Context, id uuid.UUID) (ProductView, error) {
   192  	ctx, endSpan := u.tracer.Start(ctx)
   193  	defer endSpan()
   194  
   195  	p, err := u.repo.FindPublishedByID(ctx, id)
   196  	if err != nil {
   197  		return ProductView{}, err
   198  	}
   199  	if err := ensurePublished(product.Products{p}); err != nil {
   200  		return ProductView{}, err
   201  	}
   202  
   203  	return toProductView(p), nil
   204  }

   206  // ensurePublished は、Repository が公開中として返した商品が、ドメインの定義でも公開中であることを確かめます。
   207  //
   208  // 絞り込みを実行するのは SQL ですが、「公開中」を定義するのは Product.IsPublished です。両者が食い違えば、
   209  // 一方だけが変更されたということであり、それは黙って落としてよい行ではなく表に出すべき欠陥です。
   210  func ensurePublished(products product.Products) error {
   211  	for _, p := range products {
   212  		if !p.IsPublished() {
   213  			return xerrors.Wrap(errUnpublishedInPublishedRead, p.ID().String())
   214  		}
   215  	}
   216  	return nil
   217  }
