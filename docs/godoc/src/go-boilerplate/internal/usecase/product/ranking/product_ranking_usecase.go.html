    95  // GetQuantityRanking は、件数の正規化を usecase 側の入力方針として引き受けます。QueryService へは
    96  // 正規化済みの値だけが渡るため、範囲外の件数が集計側へ到達する経路はありません。
    97  func (u *usecase) GetQuantityRanking(ctx context.Context, params GetRankingParams) (QuantityRankingView, error) {
    98  	ctx, endSpan := u.tracer.Start(ctx)
    99  	defer endSpan()
   100  
   101  	results, err := u.qs.ListQuantityRanking(ctx, toQueryParams(params))
   102  	if err != nil {
   103  		return QuantityRankingView{}, err
   104  	}
   105  
   106  	for _, r := range results {
   107  		if verr := ensurePublished(r.ProductID, r.PublishedAt); verr != nil {
   108  			return QuantityRankingView{}, verr
   109  		}
   110  	}
   111  
   112  	items := make([]QuantityRankingItemView, len(results))
   113  	for i, r := range results {
   114  		items[i] = QuantityRankingItemView{
   115  			ProductID:    r.ProductID,
   116  			Name:         r.Name,
   117  			Price:        r.Price,
   118  			SoldQuantity: r.SoldQuantity,
   119  		}
   120  	}
   121  
   122  	return QuantityRankingView{Rankings: items}, nil
   123  }
