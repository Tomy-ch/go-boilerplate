    76  // 正規化済みの値だけが渡るため、未知の期間や範囲外の件数が集計側へ到達する経路はありません。
    77  func (u *usecase) GetProductsRanking(ctx context.Context, params GetRankingParams) (RankingView, error) {
    78  	ctx, endSpan := u.tracer.Start(ctx)
    79  	defer endSpan()
    80  
    81  	results, err := u.qs.ListRanking(ctx, query.RankingQueryParams{
    82  		Period: normalizePeriod(params.Period),
    83  		Limit:  paging.NewLimit(ptr.To(params.Limit), rankingLimitPolicy).Value(),
    84  	})
    85  	if err != nil {
    86  		return RankingView{}, err
    87  	}
    88  
    89  	if verr := ensurePublished(results); verr != nil {
    90  		return RankingView{}, verr
    91  	}
    92  
    93  	items := make([]RankingItemView, len(results))
    94  	for i, r := range results {
    95  		items[i] = RankingItemView{
    96  			ProductID:    r.ProductID,
    97  			Name:         r.Name,
    98  			Price:        r.Price,
    99  			SoldQuantity: r.SoldQuantity,
   100  		}
   101  	}
   102  
   103  	return RankingView{Rankings: items}, nil
   104  }

   106  // ensurePublished は、集計結果の各行が「公開中」の定義を満たすことをドメインの述語で確かめます。
   107  // 集約を再構築しない経路のため突き合わせるのは公開日時だけですが、定義の所在はドメインに保たれます。
   108  // 外れた行はドリフトなので、黙って落とさずエラーにします。
   109  func ensurePublished(results []query.RankingResult) error {
   110  	for _, r := range results {
   111  		if !product.IsPublished(r.PublishedAt) {
   112  			return xerrors.Wrap(errUnpublishedInRanking, r.ProductID.String())
   113  		}
   114  	}
   115  
   116  	return nil
   117  }
