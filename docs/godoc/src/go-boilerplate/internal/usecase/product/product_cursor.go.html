    25  // decodeProductCursor は、cursor の不透明キー列を keyset 境界（productCursor）へ解釈します。
    26  // 先頭ページ（カーソル無し）の場合は nil を返します。キーの個数・型が不正な場合は ErrInvalidArgument を返します。
    27  func decodeProductCursor(cursor *paging.Cursor) (*productCursor, error) {
    28  	if !cursor.HasCursor() {
    29  		return nil, nil //nolint:nilnil // 先頭ページは境界なし（nil）を正常値として返す
    30  	}
    31  
    32  	keys := cursor.Keys()
    33  	if len(keys) != productCursorKeyCount {
    34  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "invalid cursor: expected 2 keys")
    35  	}
    36  
    37  	publishedAt, err := time.Parse(time.RFC3339Nano, keys[0])
    38  	if err != nil {
    39  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "invalid cursor: published_at is not RFC3339Nano")
    40  	}
    41  
    42  	id, err := uuid.Parse(keys[1])
    43  	if err != nil {
    44  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "invalid cursor: id is not a valid UUID")
    45  	}
    46  
    47  	return &productCursor{publishedAt: publishedAt, id: id}, nil
    48  }
