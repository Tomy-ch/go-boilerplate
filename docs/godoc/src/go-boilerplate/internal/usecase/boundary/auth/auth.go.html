    26  func New(
    27  	subject string,
    28  	provider string,
    29  	scopes []string,
    30  	claims map[string]any,
    31  ) (*Authn, error) {
    32  	trimmedSubject := strings.TrimSpace(subject)
    33  	if trimmedSubject == "" {
    34  		return nil, ErrUnauthenticatedSubjectMissing
    35  	}
    36  
    37  	a := &Authn{
    38  		subject:  trimmedSubject,
    39  		provider: provider,
    40  		scopes:   slices.Clone(scopes),
    41  		claims:   maps.Clone(claims),
    42  	}
    43  
    44  	// subject が UUID なら id を埋める（変換できない場合は nil のまま）
    45  	if id, err := uuid.Parse(trimmedSubject); err == nil {
    46  		a.id = &id
    47  	}
    48  
    49  	return a, nil
    50  }
