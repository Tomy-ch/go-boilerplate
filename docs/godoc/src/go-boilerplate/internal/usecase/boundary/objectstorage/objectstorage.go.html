    14  // Storage は、オブジェクトを保存・列挙・削除する中立な境界です。
    15  type Storage interface {
    16  	// Put は、obj をそのキー配下へ保存し、保存されたパス（オブジェクトキー）を返します。
    17  	// 返された Path は上位が永続化する対象であり、表示 URL は上位（フロント）が
    18  	// 配信ベース URL と組み合わせて別途組み立てます。
    19  	// 失敗時は apperror sentinel（ErrUnavailable 等）を返します。
    20  	Put(ctx context.Context, obj PutObject) (Path, error)
    21  	// List は、query に一致するオブジェクトを 1 ページ分列挙します。
    22  	// 続きがある場合は ListResult.NextCursor が非空になり、それを次の ListQuery.Cursor に渡すと続きを取得できます。
    23  	// 失敗時は apperror sentinel（ErrUnavailable 等）を返します。
    24  	List(ctx context.Context, query ListQuery) (ListResult, error)
    25  	// Delete は、keys のオブジェクトをまとめて削除します。keys が空の場合は何もしません。
    26  	// 存在しないキーはエラーにならず、同じ keys で再実行しても結果は変わりません。
    27  	// 失敗時は apperror sentinel（ErrUnavailable 等）を返します。
    28  	Delete(ctx context.Context, keys []string) error
    29  }

    32  type PutObject struct {

    34  	Key string

    36  	Body []byte

    38  	ContentType string

    40  	// 空文字なら設定しません。キャッシュ可否はキーの不変性に依存するため、呼び出し側が決めます。
    41  	CacheControl string
    42  }

    50  	Cursor string
