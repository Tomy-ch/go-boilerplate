    10  // Consumer は、pull-ack クラスのキューに対する seam です。
    11  //
    12  // per-message の再配送遅延は first-class な port capability です。engine が
    13  // 再配送 backoff の policy（ReceiveCount からの指数 + jitter）を決め、adapter は native 機構
    14  // （SQS ChangeMessageVisibility 等）でそれを honor します。これは broker 非依存の
    15  // circuit（poll ループ全体の intake throttle）とは別レイヤで、両者は併存します。
    16  type Consumer interface {
    17  	// Receive は long-poll で最大 limit 件を取得します。ctx 完了 or メッセージ到着までブロックします。
    18  	Receive(ctx context.Context, limit int) ([]Message, error)
    19  	// Ack は処理成功後にのみ呼びます（process → ack を厳守）。
    20  	Ack(ctx context.Context, m Message) error
    21  	// Nack はメッセージを即時に再配送へ戻します（遅延なし）。
    22  	Nack(ctx context.Context, m Message) error
    23  	// NackWithBackoff は、最低 d だけ遅延させてからメッセージを再配送へ戻します。
    24  	// adapter は native 機構（SQS ChangeMessageVisibility 等）で遅延を honor します。
    25  	// d<=0 は Nack と等価です。delay は best-effort ではなく port が要求する capability です。
    26  	NackWithBackoff(ctx context.Context, m Message, d time.Duration) error
    27  	// Extend は処理 lease/visibility を延長します（長時間 handler のハートビート）。
    28  	Extend(ctx context.Context, m Message, d time.Duration) error
    29  }
