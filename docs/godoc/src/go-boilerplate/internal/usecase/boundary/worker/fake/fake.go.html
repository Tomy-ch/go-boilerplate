    25  // Fake は、worker.Consumer と worker.FailureHandler を実装する in-memory テストダブルです。
    26  // メッセージの投入・再配送・失敗注入と、Ack/Nack/Extend/Fail の呼び出し記録を提供します。
    27  type Fake struct {
    28  	mu sync.Mutex
    29  
    30  	queue    []worker.Message          // 受信待ちのメッセージ
    31  	inflight map[string]worker.Message // 受信済み・Ack/Nack 待ち（ID キー）
    32  	delivery map[string]int            // ID ごとの配送回数（再配送で増加）
    33  
    34  	acked     []string
    35  	nacked    []string
    36  	extends   map[string]int // ID ごとの Extend 呼び出し回数
    37  	extendErr error          // 設定時、Extend が常にこのエラーを返す（H2）
    38  	failed    []FailedRecord
    39  
    40  	receiveErrs []error       // 注入された Receive エラー（先頭から消費）
    41  	notify      chan struct{} // long-poll 起床用のブロードキャスト
    42  }

    54  // Receive は、最大 max 件を取得します。キューが空の場合は投入 or ctx 完了までブロックします。
    55  func (f *Fake) Receive(ctx context.Context, limit int) ([]worker.Message, error) {
    56  	for {
    57  		f.mu.Lock()
    58  		if len(f.receiveErrs) > 0 {
    59  			err := f.receiveErrs[0]
    60  			f.receiveErrs = f.receiveErrs[1:]
    61  			f.mu.Unlock()
    62  			return nil, err
    63  		}
    64  		if len(f.queue) > 0 {
    65  			n := min(limit, len(f.queue))
    66  			out := make([]worker.Message, 0, n)
    67  			for range n {
    68  				m := f.queue[0]
    69  				f.queue = f.queue[1:]
    70  				f.delivery[m.ID]++
    71  				m.ReceiveCount = f.delivery[m.ID]
    72  				f.inflight[m.ID] = m
    73  				out = append(out, m)
    74  			}
    75  			f.mu.Unlock()
    76  			return out, nil
    77  		}
    78  		notify := f.notify
    79  		f.mu.Unlock()
    80  
    81  		select {
    82  		case <-ctx.Done():
    83  			return nil, ctx.Err()
    84  		case <-notify:
    85  			// メッセージ投入 or 再配送で起床。ループ先頭で再評価する。
    86  		}
    87  	}
    88  }
