    25  // Fake は、worker.Consumer と worker.FailureHandler を実装する in-memory テストダブルです。
    26  // メッセージの投入・再配送・失敗注入と、Ack/Nack/Extend/Fail の呼び出し記録を提供します。

    56  // Receive は、最大 limit 件を取得します。キューが空の場合は投入 or ctx 完了までブロックします。
    57  func (f *Fake) Receive(ctx context.Context, limit int) ([]worker.Message, error) {
    58  	for {
    59  		f.mu.Lock()
    60  		if len(f.receiveErrs) > 0 {
    61  			err := f.receiveErrs[0]
    62  			f.receiveErrs = f.receiveErrs[1:]
    63  			f.mu.Unlock()
    64  			return nil, err
    65  		}
    66  		if len(f.queue) > 0 {
    67  			n := min(limit, len(f.queue))
    68  			out := make([]worker.Message, 0, n)
    69  			for range n {
    70  				m := f.queue[0]
    71  				f.queue = f.queue[1:]
    72  				f.delivery[m.ID]++
    73  				m.ReceiveCount = f.delivery[m.ID]
    74  				f.inflight[m.ID] = m
    75  				out = append(out, m)
    76  			}
    77  			f.mu.Unlock()
    78  			return out, nil
    79  		}
    80  		notify := f.notify
    81  		f.mu.Unlock()
    82  
    83  		select {
    84  		case <-ctx.Done():
    85  			return nil, ctx.Err()
    86  		case <-notify:
    87  			// メッセージ投入 or 再配送で起床。ループ先頭で再評価する。
    88  		}
    89  	}
    90  }
