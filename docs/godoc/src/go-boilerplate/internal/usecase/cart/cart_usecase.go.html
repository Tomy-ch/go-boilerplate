   187  // resolveCart は、主体から表示すべきカートを解決します。
   188  // 主体を持たない・カートが無い・有効期限を過ぎているのいずれも「表示するカートが無い」であり、
   189  // found=false を返します。いずれも呼び出し側の失敗ではありません。
   190  func (u *usecase) resolveCart(ctx context.Context, subject Subject, now time.Time) (*cart.Cart, bool, error) {
   191  	c, found, err := u.findCart(ctx, subject)
   192  	if err != nil {
   193  		if xerrors.Is(err, apperror.ErrNotFound) {
   194  			return nil, false, nil
   195  		}
   196  		return nil, false, err
   197  	}
   198  	if !found || c.IsExpired(now) {
   199  		return nil, false, nil
   200  	}
   201  	return c, true, nil
   202  }

   204  // findCart は、主体の種類に応じてカートを引きます。主体を持たない場合は found=false を返します。
   205  // 認証済みユーザーはゲストセッションより優先されます。
   206  func (u *usecase) findCart(ctx context.Context, subject Subject) (*cart.Cart, bool, error) {
   207  	switch {
   208  	case subject.UserID != nil:
   209  		c, err := u.cartRepo.FindByOwnerID(ctx, *subject.UserID)
   210  		return c, err == nil, err
   211  	case subject.SessionToken != nil:
   212  		token, err := cart.NewSessionToken(*subject.SessionToken)
   213  		if err != nil {
   214  			return nil, false, err
   215  		}
   216  		c, err := u.cartRepo.FindBySessionToken(ctx, token)
   217  		return c, err == nil, err
   218  	default:
   219  		return nil, false, nil
   220  	}
   221  }

   223  // findProducts は、明細が指す商品を ID で引き当てた表を返します。
   224  // 引けなかった商品は表に現れません。明細が無い場合は問い合わせません。
   225  func (u *usecase) findProducts(
   226  	ctx context.Context, items []cart.CartItem,
   227  ) (map[uuid.UUID]*product.Product, error) {
   228  	if len(items) == 0 {
   229  		return map[uuid.UUID]*product.Product{}, nil
   230  	}
   231  
   232  	ids := make([]uuid.UUID, 0, len(items))
   233  	for _, item := range items {
   234  		ids = append(ids, item.ProductID())
   235  	}
   236  
   237  	products, err := u.productRepo.FindByIDs(ctx, ids)
   238  	if err != nil {
   239  		return nil, err
   240  	}
   241  
   242  	found := make(map[uuid.UUID]*product.Product, len(products))
   243  	for _, p := range products {
   244  		found[p.ID()] = p
   245  	}
   246  	return found, nil
   247  }

   249  // buildView は、明細を再評価して出力 DTO を組み立て、あわせてカートの状態を進めます。
   250  // 提示した価格の記録と有効期限の延長までを含むため、呼び出し側は戻り値を返す前にカートを保存します。
   251  //
   252  // 判定も合算も提示価格を書き換える前に済ませます（docs/spec/cart/domain.md の Cart.Subtotal）。
   253  //
   254  // SessionToken は埋めません。トークンを新しく発行したかどうかを知るのはカートを解決した側だけです。
   255  func (u *usecase) buildView(ctx context.Context, c *cart.Cart, now time.Time) (CartView, error) {
   256  	items := c.Items()
   257  	products, err := u.findProducts(ctx, items)
   258  	if err != nil {
   259  		return CartView{}, err
   260  	}
   261  
   262  	views, seen := evaluateItems(items, products)
   263  	subtotal, serr := c.Subtotal(toSnapshots(products))
   264  	if serr != nil {
   265  		return CartView{}, serr
   266  	}
   267  
   268  	c.MarkSeen(seen)
   269  	c.Touch(now, cartTTL)
   270  
   271  	expiresAt := c.ExpiresAt()
   272  	return CartView{Items: views, SubtotalAmount: subtotal, ExpiresAt: &expiresAt}, nil
   273  }

   275  // evaluateItems は、明細を商品の現在値と突き合わせ、明細ごとの View と、提示した価格の表を返します。
   276  // 価格の表には引けた商品のぶんだけが入ります（引けなかった明細の提示価格は据え置かれます）。
   277  func evaluateItems(
   278  	items []cart.CartItem, products map[uuid.UUID]*product.Product,
   279  ) ([]CartItemView, map[uuid.UUID]money.Price) {
   280  	views := make([]CartItemView, 0, len(items))
   281  	seen := make(map[uuid.UUID]money.Price, len(items))
   282  
   283  	for _, item := range items {
   284  		p := products[item.ProductID()]
   285  		views = append(views, evaluateItem(item, p))
   286  		if p != nil {
   287  			seen[p.ID()] = p.Price()
   288  		}
   289  	}
   290  
   291  	return views, seen
   292  }

   294  // toSnapshots は、引けた商品を再評価用の観測値へ切り出します。
   295  // 引けなかった商品は含めません（この表に無い明細の扱いは docs/spec/cart/domain.md の Cart.Subtotal）。
   296  func toSnapshots(products map[uuid.UUID]*product.Product) map[uuid.UUID]cart.ProductSnapshot {
   297  	snapshots := make(map[uuid.UUID]cart.ProductSnapshot, len(products))
   298  	for id, p := range products {
   299  		snapshots[id] = cart.NewProductSnapshot(p.Quantity(), p.Price(), p.IsPublished())
   300  	}
   301  
   302  	return snapshots
   303  }

   305  // evaluateItem は、明細 1 件の再評価を cart.CartItem.Evaluate へ委ね、結果を出力 DTO の語彙へ写します。
   306  func evaluateItem(item cart.CartItem, p *product.Product) CartItemView {
   307  	view := CartItemView{ProductID: item.ProductID(), Quantity: item.Quantity()}
   308  
   309  	var snapshot *cart.ProductSnapshot
   310  	if p != nil {
   311  		name, price := p.Name(), p.Price().Decimal()
   312  		view.ProductName, view.UnitPrice = &name, &price
   313  		if image, ok := p.PrimaryImage(); ok {
   314  			imagePath := image.ImagePath()
   315  			view.ImagePath = &imagePath
   316  		}
   317  		s := cart.NewProductSnapshot(p.Quantity(), p.Price(), p.IsPublished())
   318  		snapshot = &s
   319  	}
   320  
   321  	evaluation := item.Evaluate(snapshot)
   322  	view.Issues = toItemIssues(evaluation.Issues())
   323  	view.AvailableQuantity = evaluation.AvailableQuantity()
   324  
   325  	return view
   326  }

   329  func toItemIssues(issues []cart.Issue) []ItemIssue {
   330  	converted := make([]ItemIssue, len(issues))
   331  	for i, issue := range issues {
   332  		converted[i] = toItemIssue(issue)
   333  	}

   337  // toItemIssue は、再評価結果 1 件を出力 DTO の語彙へ写します。
   338  // 対応を持たない値は写像できないため、黙って既定値へ倒さず panic で異常を知らせます。
   339  func toItemIssue(issue cart.Issue) ItemIssue {
   340  	switch issue {
   341  	case cart.IssueNotFound:
   342  		return ItemIssueNotFound
   343  	case cart.IssueUnpublished:
   344  		return ItemIssueUnpublished
   345  	case cart.IssueOutOfStock:
   346  		return ItemIssueOutOfStock
   347  	case cart.IssueInsufficientStock:
   348  		return ItemIssueInsufficientStock
   349  	case cart.IssuePriceIncreased:
   350  		return ItemIssuePriceIncreased
   351  	case cart.IssuePriceDecreased:
   352  		return ItemIssuePriceDecreased
   353  	default:
   354  		panic(fmt.Sprintf("cart: unknown item issue: %q", issue))
   355  	}
   356  }
