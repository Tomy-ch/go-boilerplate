    14  // maxSetItemAttempts は、カートの作成が競合したときに解決からやり直す上限です
    15  // （docs/spec/cart/usecase.md の SetItem）。
    16  const maxSetItemAttempts = 2

    18  // errCartCreationRace は、カートを作ろうとして他の要求に先を越されたことを表します。
    19  // この層の中だけで使い、外へは apperror.ErrConflict として出ます。
    20  var errCartCreationRace = xerrors.New("cart creation lost a race")
    21  
    22  // SetItem は、作成の衝突をトランザクションごとやり直します（docs/spec/cart/usecase.md の SetItem）。
    23  //
    24  // この本体に外部副作用を足してはなりません。やり直しで二重に実行されます
    25  // （ADR-0035 (transaction-retry-idempotent-callers)）。
    26  func (u *usecase) SetItem(ctx context.Context, params SetItemParams) (CartView, error) {
    27  	ctx, endSpan := u.tracer.Start(ctx)
    28  	defer endSpan()
    29  
    30  	var raced error
    31  	for range maxSetItemAttempts {
    32  		view, err := tx.DoWithResult(ctx, u.txm, func(ctx context.Context) (CartView, error) {
    33  			return u.setItem(ctx, params)
    34  		})
    35  		if !xerrors.Is(err, errCartCreationRace) {
    36  			return view, err
    37  		}
    38  		raced = err
    39  	}
    40  
    41  	return CartView{}, xerrors.Wrap(raced, "cart creation kept losing the race")
    42  }

    44  // setItem は、1 トランザクション分の処理です。
    45  func (u *usecase) setItem(ctx context.Context, params SetItemParams) (CartView, error) {
    46  	now := u.clock.Now()
    47  
    48  	c, issuedToken, err := u.resolveOrCreateCart(ctx, params.Subject, now)
    49  	if err != nil {
    50  		return CartView{}, err
    51  	}
    52  
    53  	if aerr := u.ensureProductAvailable(ctx, params.ProductID); aerr != nil {
    54  		return CartView{}, aerr
    55  	}
    56  
    57  	itemID, err := uuid.New()
    58  	if err != nil {
    59  		return CartView{}, xerrors.Wrap(err, "failed to generate cart item id")
    60  	}
    61  	if serr := c.SetItem(cart.SetItemAttributes{
    62  		ItemID:    itemID,
    63  		ProductID: params.ProductID,
    64  		Quantity:  params.Quantity,
    65  	}, now); serr != nil {
    66  		return CartView{}, serr
    67  	}
    68  
    69  	view, verr := u.buildView(ctx, c, now)
    70  	if verr != nil {
    71  		return CartView{}, verr
    72  	}
    73  	if uerr := u.cartRepo.Update(ctx, c); uerr != nil {
    74  		return CartView{}, uerr
    75  	}
    76  
    77  	view.SessionToken = issuedToken
    78  	return view, nil
    79  }

    92  // resolveOrCreateCart は、主体のカートを永続化済みの状態で返します。カートを持たない主体には作ります。
    93  // 有効期限を過ぎたカートは明細を捨てて空から始めます（docs/spec/cart/usecase.md の SetItem）。
    94  //
    95  // 2 つ目の戻り値は、このときゲストトークンを新しく発行した場合だけ非 nil です。
    96  func (u *usecase) resolveOrCreateCart(
    97  	ctx context.Context, subject Subject, now time.Time,
    98  ) (*cart.Cart, *string, error) {
    99  	if subject.UserID != nil {
   100  		return u.resolveOwnerCart(ctx, *subject.UserID, now)
   101  	}
   102  	return u.resolveGuestCart(ctx, subject.SessionToken, now)

   105  // resolveOwnerCart は、所有者が確定したカートを解決します。
   106  // ユーザー 1 人につきカートは高々 1 件のため、期限切れでも作り直せません
   107  // （docs/spec/cart/usecase.md の SetItem）。
   108  //
   109  // 解決とロックの間にカートが消えていた場合は、引けなかった場合と同じく作り直します。
   110  // この op は 404 を宣言していません。
   111  func (u *usecase) resolveOwnerCart(
   112  	ctx context.Context, userID uuid.UUID, now time.Time,
   113  ) (*cart.Cart, *string, error) {
   114  	found, err := u.cartRepo.FindByOwnerID(ctx, userID)
   115  	if err != nil {
   116  		if !xerrors.Is(err, apperror.ErrNotFound) {
   117  			return nil, nil, err
   118  		}
   119  		return u.createOwnerCart(ctx, userID, now)
   120  	}
   121  
   122  	c, lerr := u.cartRepo.LockByID(ctx, found.ID())
   123  	if lerr != nil {
   124  		if !xerrors.Is(lerr, apperror.ErrNotFound) {
   125  			return nil, nil, lerr
   126  		}
   127  		return u.createOwnerCart(ctx, userID, now)
   128  	}
   129  	if c.IsExpired(now) {
   130  		c.Clear()
   131  	}
   132  	return c, nil, nil
   133  }

   135  // resolveGuestCart は、ゲストのカートを解決します。
   136  // 提示されたトークンで引けなかった場合と、引けたが期限切れだった場合は、どちらも採番し直します
   137  // （docs/spec/cart/usecase.md の SetItem）。
   138  //
   139  // 解決とロックの間にカートが消えていた場合も同じく採番し直します。引き継ぎ（MergeOnLogin）が
   140  // ゲストカートを行ごと消すため、この窓は実際に開きます。この op は 404 を宣言していません。
   141  func (u *usecase) resolveGuestCart(
   142  	ctx context.Context, presented *string, now time.Time,
   143  ) (*cart.Cart, *string, error) {
   144  	if presented == nil {
   145  		return u.createGuestCart(ctx, now)
   146  	}
   147  
   148  	token, err := cart.NewSessionToken(*presented)
   149  	if err != nil {
   150  		return nil, nil, err
   151  	}
   152  
   153  	found, ferr := u.cartRepo.FindBySessionToken(ctx, token)
   154  	if ferr != nil {
   155  		if !xerrors.Is(ferr, apperror.ErrNotFound) {
   156  			return nil, nil, ferr
   157  		}
   158  		return u.createGuestCart(ctx, now)
   159  	}
   160  
   161  	c, lerr := u.cartRepo.LockByID(ctx, found.ID())
   162  	if lerr != nil {
   163  		if !xerrors.Is(lerr, apperror.ErrNotFound) {
   164  			return nil, nil, lerr
   165  		}
   166  		return u.createGuestCart(ctx, now)
   167  	}
   168  	if c.IsExpired(now) {
   169  		return u.createGuestCart(ctx, now)
   170  	}
   171  	return c, nil, nil
   172  }

   174  // createOwnerCart は、所有者が確定した空のカートを作ります。
   175  func (u *usecase) createOwnerCart(
   176  	ctx context.Context, userID uuid.UUID, now time.Time,
   177  ) (*cart.Cart, *string, error) {
   178  	id, err := uuid.New()
   179  	if err != nil {
   180  		return nil, nil, xerrors.Wrap(err, "failed to generate cart id")
   181  	}
   182  
   183  	c, nerr := cart.NewForOwner(id, cart.Attributes{OwnerID: &userID, ExpiresAt: now.Add(cartTTL)})
   184  	if nerr != nil {
   185  		return nil, nil, nerr
   186  	}
   187  	if cerr := u.cartRepo.Create(ctx, c); cerr != nil {
   188  		return nil, nil, markCreationRace(cerr)
   189  	}
   190  	return c, nil, nil
   191  }

   193  // createGuestCart は、新しいトークンを発行して空のゲストカートを作り、そのトークンを返します。
   194  func (u *usecase) createGuestCart(ctx context.Context, now time.Time) (*cart.Cart, *string, error) {
   195  	raw, err := u.tokenGen.Generate()
   196  	if err != nil {
   197  		return nil, nil, xerrors.Wrap(err, "failed to generate session token")
   198  	}
   199  
   200  	token, terr := cart.NewSessionToken(raw)
   201  	if terr != nil {
   202  		return nil, nil, terr
   203  	}
   204  
   205  	id, ierr := uuid.New()
   206  	if ierr != nil {
   207  		return nil, nil, xerrors.Wrap(ierr, "failed to generate cart id")
   208  	}
   209  
   210  	c, nerr := cart.NewForGuest(id, token, now.Add(cartTTL))
   211  	if nerr != nil {
   212  		return nil, nil, nerr
   213  	}
   214  	if cerr := u.cartRepo.Create(ctx, c); cerr != nil {
   215  		return nil, nil, markCreationRace(cerr)
   216  	}
   217  	return c, &raw, nil
   218  }

   220  // markCreationRace は、カート作成の衝突だけをやり直しの対象として印を付けます。
   221  // 衝突を一律にやり直すと、やり直しても解消しない要求までもう一度走らせることになります。
   222  func markCreationRace(err error) error {
   223  	if xerrors.Is(err, apperror.ErrConflict) {
   224  		return xerrors.Join(errCartCreationRace, err)
   225  	}
   226  	return err
   227  }
