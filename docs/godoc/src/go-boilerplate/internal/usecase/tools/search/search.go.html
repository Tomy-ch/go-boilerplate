    20  func ParseSearchTokens(keyword *string, maxTokens int) []string {
    21  	if keyword == nil || *keyword == "" {
    22  		return []string{}
    23  	}

    25  		maxTokens = DefaultMaxTokens
    26  	}

    37  	return limit(unique, maxTokens)

    48  func dedupePreserveOrder(ss []string) []string {
    49  	seen := make(map[string]struct{}, len(ss))
    50  	out := make([]string, 0, len(ss))
    51  
    52  	for _, t := range ss {
    53  		if _, ok := seen[t]; ok {
    54  			continue
    55  		}

    57  		out = append(out, t)
    58  	}
    59  	return out
