    25  // NewCursor は、不透明カーソル文字列（after）と取得件数（first）から Cursor を生成します。
    26  //
    27  //	after が nil または空文字の場合は先頭ページとして扱い、keys は空になります。
    28  //	after の形式が不正な場合は apperror.ErrInvalidArgument を返します。
    29  //	first[limit] の補完・クランプ規約は offset 版（NewPageFrom1Based）と共通で、
    30  //	0以下または nil の場合は defaultPerPage、maxPerPage を超える場合は maxPerPage を使用します。
    31  //	keyset はページ番号を持たないため、offset 版のような最大ページ数エラーは発生しません。
    32  func NewCursor(after *string, first *int) (*Cursor, error) {
    33  	limit := defaultPerPage
    34  	if first != nil && *first > 0 {
    35  		limit = *first
    36  	}
    37  	if limit > maxPerPage {
    38  		limit = maxPerPage
    39  	}
    40  
    41  	keys, err := decodeCursor(after)
    42  	if err != nil {
    43  		return nil, err
    44  	}
    45  
    46  	return &Cursor{
    47  		limit: limit,
    48  		keys:  keys,
    49  	}, nil
    50  }

    93  //
    94  //	nil または空文字は先頭ページとみなし、nil を返します。
    95  //	base64/JSON の形式不正、または空タプルの場合は apperror.ErrInvalidArgument を返します。
    96  func decodeCursor(after *string) ([]string, error) {
    97  	if after == nil || *after == "" {
    98  		return nil, nil
    99  	}
   100  
   101  	raw, err := base64.RawURLEncoding.DecodeString(*after)
   102  	if err != nil {
   103  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, fmt.Sprintf("invalid cursor encoding: %v", err))
   104  	}
   105  
   106  	var keys []string
   107  	if err := json.Unmarshal(raw, &keys); err != nil {
   108  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, fmt.Sprintf("invalid cursor payload: %v", err))
   109  	}
   110  	if len(keys) == 0 {
   111  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "invalid cursor: empty keyset")
   112  	}
   113  
   114  	return keys, nil
   115  }
