    30  // NewPagingFrom1Based は、ページ番号と1ページあたりの件数から取得上限とオフセットを計算して Paging を返します。
    31  //
    32  //	ページ番号が1未満の場合は1に、1ページあたりの件数が0以下の場合はデフォルト値を使用します。
    33  //	page[offsetの係数]は最小値1で、maxPage を超える場合はエラーを返します。
    34  //	perPage[limit]は、0以下の場合はdefaultPerPage値：最大値をmaxPerPageまで許容します。
    35  func NewPagingFrom1Based(page, perPage *int) (*Paging, error) {
    36  	limit := defaultPerPage
    37  	if perPage != nil && *perPage > 0 {
    38  		limit = *perPage
    39  	}
    40  	if limit > maxPerPage {
    41  		limit = maxPerPage
    42  	}
    43  
    44  	pg := minPage
    45  	if page != nil && *page > 0 {
    46  		pg = *page
    47  	}
    48  	if pg > maxPage {
    49  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, fmt.Sprintf("invalid page number: %d, max page is %d", pg, maxPage))
    50  	}
    51  
    52  	return &Paging{
    53  		limit:  limit,
    54  		offset: (pg - 1) * limit,
    55  	}, nil
    56  }
