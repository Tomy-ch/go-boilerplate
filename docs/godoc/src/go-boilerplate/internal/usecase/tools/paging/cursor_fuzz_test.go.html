     9  // FuzzNewCursor は、NewCursor が任意の不透明カーソル文字列に対して panic せず、
    10  // 受理した場合はソートキーのタプルを取り出せることを検証します。
    11  //
    12  //	カーソルは URL に載る外部入力であり、base64 と JSON の二段復号を通るため、
    13  //	どちらの層の異常も NewCursor がエラーとして閉じ込める必要があります。
    14  func FuzzNewCursor(f *testing.F) {
    15  	for _, seed := range []string{
    16  		"",
    17  		EncodeCursor("2026-01-01T00:00:00Z"),
    18  		EncodeCursor("2026-01-01T00:00:00Z", "b1d4e0f2-3c5a-4b6d-8e7f-1a2b3c4d5e6f"),
    19  		"not-base64!!",
    20  		"e30",
    21  		"W10",
    22  		"////",
    23  	} {
    24  		f.Add(seed)
    25  	}
    26  
    27  	f.Fuzz(func(t *testing.T, after string) {
    28  		c, err := NewCursor(&after, nil)
    29  		if err != nil {
    30  			return
    31  		}
    32  
    33  		require.NotNil(t, c)
    34  		require.LessOrEqual(t, c.Limit(), maxPerPage)
    35  		require.Positive(t, c.Limit())
    36  
    37  		// 受理されたカーソルのキーは再エンコードして再解釈しても同じタプルへ戻る。
    38  		if c.HasCursor() {
    39  			keys := c.Keys()
    40  			round := EncodeCursor(keys...)
    41  			back, err := NewCursor(&round, nil)
    42  			require.NoError(t, err)
    43  			require.Equal(t, keys, back.Keys())
    44  		}
    45  	})
    46  }
