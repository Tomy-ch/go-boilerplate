    51  func (u *emitUsecase) Emit(ctx context.Context, in EmitInput) (uuid.UUID, error) {
    52  	ctx, endSpan := u.tracer.Start(ctx)
    53  	defer endSpan()
    54  
    55  	headers := make(map[string]string, len(in.Headers)+1)
    56  	for k, v := range in.Headers {
    57  		// 呼び出し側契約（EmitInput.Headers の doc）に加えた defense-in-depth として、
    58  		// egress の起点であるここで、誤って混入した既知の機微ヘッダを保守的に落とします。
    59  		if httpheader.IsSensitive(k) {
    60  			continue
    61  		}
    62  		headers[k] = v
    63  	}
    64  	// emit span の trace context を traceparent として載せる（消費側が同一 trace に繋がる）。
    65  	observability.InjectTraceContextToCarrier(ctx, headers)
    66  
    67  	var headerBytes []byte
    68  	if len(headers) > 0 {
    69  		b, err := json.Marshal(headers)
    70  		if err != nil {
    71  			return uuid.UUID{}, xerrors.Join(apperror.ErrInternal, xerrors.Wrap(err, "failed to encode outbox headers"))
    72  		}
    73  		headerBytes = b
    74  	}
    75  
    76  	return u.store.Insert(ctx, outboxbndry.EmitParams{
    77  		AggregateType: in.AggregateType,
    78  		AggregateID:   in.AggregateID,
    79  		EventType:     in.EventType,
    80  		Payload:       in.Payload,
    81  		Headers:       headerBytes,
    82  	})
    83  }
