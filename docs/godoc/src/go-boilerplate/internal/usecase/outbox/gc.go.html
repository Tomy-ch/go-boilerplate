    37  // SweepPublished は、retention より古い published 行を batchSize 件ずつ削除し、合計削除件数を返します。
    38  func (g *gcUsecase) SweepPublished(ctx context.Context, batchSize int32) (int64, error) {
    39  	if batchSize <= 0 {
    40  		batchSize = DefaultGCBatchSize
    41  	}
    42  	cutoff := g.clock.Now().Add(-g.retention)
    43  
    44  	var total int64
    45  	for {
    46  		deleted, err := g.store.DeletePublished(ctx, cutoff, batchSize)
    47  		if err != nil {
    48  			return total, err
    49  		}
    50  		total += deleted
    51  		// バッチが満たなかった = もう対象行は無い。
    52  		if deleted < int64(batchSize) {
    53  			return total, nil
    54  		}
    55  	}
