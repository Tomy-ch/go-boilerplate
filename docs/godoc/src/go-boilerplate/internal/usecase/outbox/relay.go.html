    87  // pending 行が無ければ 0 を記録します。

    89  	ctx, endSpan := u.tracer.Start(ctx)
    90  	defer endSpan()
    91  
    92  	createdAt, ok, err := u.store.OldestPendingCreatedAt(ctx)
    93  	if err != nil {
    94  		return err
    95  	}
    96  
    97  	var lag time.Duration
    98  	if ok {
    99  		if lag = u.clock.Now().Sub(createdAt); lag < 0 {
   100  			lag = 0

   102  	}

   107  // RelayBatch は、最大 batchSize 件の pending 行を 1 tx で claim → publish → mark し、結果を返します。
   108  func (u *relayUsecase) RelayBatch(ctx context.Context, batchSize int32) (RelayResult, error) {
   109  	ctx, endSpan := u.tracer.Start(ctx)
   110  	defer endSpan()
   111  
   112  	if batchSize <= 0 {
   113  		batchSize = DefaultBatchSize
   114  	}
   115  
   116  	// claim〜mark を 1 tx に収めることで FOR UPDATE SKIP LOCKED の行ロックを
   117  	// publish 完了まで保持し、多インスタンスでの二重 publish を防ぐ。
   118  	return tx.DoWithResult(ctx, u.txm, func(ctx context.Context) (RelayResult, error) {
   119  		msgs, err := u.store.ClaimPending(ctx, batchSize)
   120  		if err != nil {
   121  			return RelayResult{}, err
   122  		}
   123  		res := RelayResult{Claimed: len(msgs)}
   124  		for i := range msgs {
   125  			published, derr := u.deliver(ctx, msgs[i])
   126  			if derr != nil {
   127  				return RelayResult{}, derr
   128  			}
   129  			if published {
   130  				res.Published++
   131  			}
   132  		}
   133  		return res, nil
   134  	})
   135  }

   137  // deliver は、1 件を publish し、結果に応じて published / failed / dead をマークします。
   138  // publish 成功なら published=true を返します。publish 失敗は tx を巻き戻さず（次 poll の再送に委ねる）
   139  // published=false・error=nil を返し、DB マークの失敗のみエラーを返します。
   140  func (u *relayUsecase) deliver(ctx context.Context, m outboxbndry.PendingMessage) (bool, error) {
   141  	perr := u.publisher.Publish(ctx, publisher.Message{
   142  		MessageID: m.MessageID,
   143  		EventType: m.EventType,
   144  		Payload:   m.Payload,
   145  		Headers:   decodeHeaders(m.Headers),
   146  	})
   147  	if perr == nil {
   148  		return true, u.store.MarkPublished(ctx, m.ID)
   149  	}
   150  
   151  	attempts, ferr := u.store.MarkFailed(ctx, m.ID, perr.Error())
   152  	if ferr != nil {
   153  		return false, ferr
   154  	}
   155  	if attempts >= u.maxAttempts {
   156  		if derr := u.store.MarkDead(ctx, m.ID); derr != nil {
   157  			return false, derr
   158  		}
   159  		u.metrics.IncDead(ctx)
   160  		u.logging.Named(relayLoggerName).Warn(
   161  			"outbox message marked dead after reaching max attempts",
   162  			logging.String(logging.MessageIDKey, m.MessageID.String()),
   163  			logging.String(logging.EventTypeKey, m.EventType),
   164  			logging.Error(logging.JobErrorKey, perr),
   165  		)
   166  	}
   167  	return false, nil
   168  }
