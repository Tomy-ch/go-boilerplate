    88  	ctx, endSpan := u.tracer.Start(ctx)
    89  	defer endSpan()
    90  
    91  	createdAt, ok, err := u.store.OldestPendingCreatedAt(ctx)

   106  func (u *relayUsecase) RelayBatch(ctx context.Context, batchSize int32) (RelayResult, error) {
   107  	ctx, endSpan := u.tracer.Start(ctx)
   108  	defer endSpan()
   109  
   110  	if batchSize <= 0 {
   111  		batchSize = DefaultBatchSize
   112  	}
   113  
   114  	// claim〜mark を同一 tx 内で完結させることで、多インスタンス間での二重 publish を防ぐ。
   115  	return tx.DoWithResult(ctx, u.txm, func(ctx context.Context) (RelayResult, error) {
   116  		msgs, err := u.store.ClaimPending(ctx, batchSize)
   117  		if err != nil {
   118  			return RelayResult{}, err
   119  		}
   120  		res := RelayResult{Claimed: len(msgs)}
   121  		for i := range msgs {
   122  			published, derr := u.deliver(ctx, msgs[i])
   123  			if derr != nil {
   124  				return RelayResult{}, derr
   125  			}
   126  			if published {
   127  				res.Published++
   128  			}
   129  		}
   130  		return res, nil
   131  	})
   132  }

   134  // deliver は、1 件を publish し、結果に応じて published / failed / dead をマークします。
   135  // publish 成功なら published=true を返します。publish 失敗は tx を巻き戻さず（次 poll の再送に委ねる）
   136  // published=false・error=nil を返し、DB マークの失敗のみエラーを返します。
   137  func (u *relayUsecase) deliver(ctx context.Context, m outboxbndry.PendingMessage) (bool, error) {
   138  	perr := u.publisher.Publish(ctx, publisher.Message{
   139  		MessageID: m.MessageID,
   140  		EventType: m.EventType,
   141  		Payload:   m.Payload,
   142  		Headers:   decodeHeaders(m.Headers),
   143  	})
   144  	if perr == nil {
   145  		return true, u.store.MarkPublished(ctx, m.ID)
   146  	}
   147  
   148  	attempts, ferr := u.store.MarkFailed(ctx, m.ID, perr.Error())
   149  	if ferr != nil {
   150  		return false, ferr
   151  	}
   152  	if attempts >= u.maxAttempts {
   153  		if derr := u.store.MarkDead(ctx, m.ID); derr != nil {
   154  			return false, derr
   155  		}
   156  		u.metrics.IncDead(ctx)
   157  		u.logging.Named(relayLoggerName).Warn(
   158  			ctx,
   159  			"outbox message marked dead after reaching max attempts",
   160  			logging.String(logging.MessageIDKey, m.MessageID.String()),
   161  			logging.String(logging.EventTypeKey, m.EventType),
   162  			logging.Error(logging.JobErrorKey, perr),
   163  		)
   164  	}
   165  	return false, nil
   166  }
