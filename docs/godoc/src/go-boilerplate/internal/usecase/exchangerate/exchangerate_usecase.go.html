    72  // Convert は、base 通貨建ての金額を quote 通貨へ換算し、任意で表示通貨の参考換算額を返します。
    73  func (u *usecase) Convert(ctx context.Context, in ConvertInput) (*ConvertResult, error) {
    74  	ctx, endSpan := u.tracer.Start(ctx)
    75  	defer endSpan()
    76  
    77  	rate, err := u.gateway.GetRate(ctx, in.Base, in.Quote)
    78  	if err != nil {
    79  		return nil, err
    80  	}
    81  
    82  	result := &ConvertResult{Converted: in.Amount * rate.Value}
    83  
    84  	// 参考レートの取得に失敗しても degrade（Reference = nil）とし、本体換算は継続させる。
    85  	if in.DisplayCurrency != nil {
    86  		if refRate, refErr := u.gateway.GetRate(ctx, in.Base, *in.DisplayCurrency); refErr == nil {
    87  			amountMinor := int64(math.Round(in.Amount * baseMinorUnitScale))
    88  			result.Reference = BuildReferenceAmount(refRate, amountMinor)
    89  		}
    90  	}
    91  	return result, nil
    92  }

    94  // BuildReferenceAmount は、レートと base 最小単位額から参考換算額を組み立てます。
    95  // referenceAmount の組み立てを 1 箇所へ集約し、後続 API（purchases 等）から再利用します。
    96  func BuildReferenceAmount(rate *boundary.Rate, amountMinor int64) *ReferenceAmount {
    97  	return &ReferenceAmount{
    98  		Currency: rate.Quote,
    99  		Amount:   money.ApplyRateHalfUp(amountMinor, rate.Value, baseMinorUnitScale),
   100  		Rate:     rate.Value,
