    40  func TestBuildReferenceAmount(t *testing.T) {
    41  	t.Parallel()
    42  
    43  	t.Run("正常系", func(t *testing.T) {
    44  		t.Parallel()
    45  
    46  		t.Run("レートと金額から参考換算額を最小単位整数で組み立てる", func(t *testing.T) {
    47  			t.Parallel()
    48  			rate := &boundary.Rate{Base: "USD", Quote: "JPY", Value: decimaltestkit.MustParse(t, "150.5"), Date: "2026-07-21"}
    49  			ref, err := exchangerate.BuildReferenceAmount(rate, decimal.FromInt(100))
    50  			require.NoError(t, err)
    51  			require.NotNil(t, ref)
    52  			assert.Equal(t, int64(15050), ref.Amount)
    53  			assert.Equal(t, "JPY", ref.Currency)
    54  			assert.Equal(t, "150.5", ref.Rate.String())
    55  			assert.Equal(t, "2026-07-21", ref.RateDate)
    56  		})
    57  	})
    58  
    59  	t.Run("異常系", func(t *testing.T) {
    60  		t.Parallel()
    61  
    62  		t.Run("最小単位整数が int64 の範囲を超える場合はエラーを返す", func(t *testing.T) {
    63  			t.Parallel()
    64  			rate := &boundary.Rate{Base: "USD", Quote: "JPY", Value: decimal.FromInt(1)}
    65  			_, err := exchangerate.BuildReferenceAmount(rate, decimaltestkit.MustParse(t, "1e19"))
    66  			require.ErrorIs(t, err, decimal.ErrOverflow)
    67  		})
    68  	})

    94  		t.Run("displayCurrency指定時は参考換算額を整数で返す", func(t *testing.T) {
    95  			t.Parallel()
    96  
    97  			ctrl := gomock.NewController(t)
    98  			gw := mock_exchangerate.NewMockGateway(ctrl)
    99  			// 本体換算 base→quote（USD→EUR）
   100  			gw.EXPECT().
   101  				GetRate(gomock.Any(), "USD", "EUR").
   102  				Return(&boundary.Rate{Base: "USD", Quote: "EUR", Value: decimaltestkit.MustParse(t, "0.92")}, nil)
   103  			// 参考換算 base→display（USD→JPY）
   104  			gw.EXPECT().
   105  				GetRate(gomock.Any(), "USD", "JPY").
   106  				Return(&boundary.Rate{Base: "USD", Quote: "JPY", Value: decimaltestkit.MustParse(t, "150.5"), Date: "2026-07-21"}, nil)
   107  
   108  			uc := exchangerate.New(gw, logging.NewTestLogger(t), observability.NewNoopTracerFactory(t))
   109  			got, err := uc.Convert(context.Background(), exchangerate.ConvertInput{
   110  				Base: "USD", Quote: "EUR", Amount: decimal.FromInt(100), DisplayCurrency: new("JPY"),
   111  			})
   112  
   113  			require.NoError(t, err)
   114  			require.NotNil(t, got.Reference)
   115  			// 100 USD × 150.5 = 15050 JPY（JPY は最小単位 0 桁）
   116  			assert.Equal(t, int64(15050), got.Reference.Amount)
   117  			assert.Equal(t, "JPY", got.Reference.Currency)
   118  			assert.Equal(t, "2026-07-21", got.Reference.RateDate)
   119  			assert.Equal(t, "150.5", got.Reference.Rate.String())
   120  		})
