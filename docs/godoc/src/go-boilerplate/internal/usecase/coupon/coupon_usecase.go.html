   148  // ListApplicableToMyCart は、カートの明細を対象にクーポンごとの値引き額を求め、0 になるものを落とします。
   149  // 対象明細の絞り込みは docs/spec/usecase/coupon.md の Workflow を参照。
   150  //
   151  // 最低購入金額に満たないクーポンは [coupon.Coupon.DiscountFor] が 0 を返すため、
   152  // 対象明細が 1 件も無い場合と同じ経路で落ちます。
   153  func (u *usecase) ListApplicableToMyCart(ctx context.Context, authn *auth.Authn) ([]CartCouponView, error) {
   154  	ctx, endSpan := u.tracer.Start(ctx)
   155  	defer endSpan()
   156  
   157  	userID, err := requireUserID(authn)
   158  	if err != nil {
   159  		return nil, err
   160  	}
   161  
   162  	coupons, err := u.couponRepo.FindByUserID(ctx, userID)
   163  	if err != nil {
   164  		return nil, err
   165  	}
   166  	if len(coupons) == 0 {
   167  		return []CartCouponView{}, nil
   168  	}
   169  
   170  	now := u.clock.Now()
   171  
   172  	lines, err := u.buildLines(ctx, userID, now)
   173  	if err != nil {
   174  		return nil, err
   175  	}
   176  	if len(lines) == 0 {
   177  		return []CartCouponView{}, nil
   178  	}
   179  
   180  	views := make([]CartCouponView, 0, len(coupons))
   181  	for _, c := range coupons {
   182  		if c.IsUsed() || c.IsExpired(now) || c.IsNotYetUsable(now) {
   183  			continue
   184  		}
   185  
   186  		amount, derr := c.DiscountFor(lines)
   187  		if derr != nil {
   188  			return nil, derr
   189  		}
   190  		if amount <= 0 {
   191  			continue
   192  		}
   193  
   194  		views = append(views, CartCouponView{Coupon: toCouponView(c), DiscountAmount: amount})
   195  	}
   196  
   197  	return views, nil
   198  }

   200  // buildLines は、カートの明細のうち購入できるものだけをクーポンの対象明細へ写します。
   201  // カートを持たない場合と期限切れの場合は、いずれも空を返します。
   202  //
   203  // Repository は期限切れのカートも返し、期限の判定は Cart.IsExpired が持ちます
   204  // （internal/domain/cart/cart_repository.go）。期限切れのカートは購入へ進めないため、
   205  // カート表示（internal/usecase/cart）と同じく「表示するカートが無い」に畳みます。
   206  func (u *usecase) buildLines(
   207  	ctx context.Context, userID uuid.UUID, now time.Time,
   208  ) ([]coupon.Line, error) {
   209  	c, err := u.cartRepo.FindByOwnerID(ctx, userID)
   210  	if err != nil {
   211  		if xerrors.Is(err, apperror.ErrNotFound) {
   212  			return nil, nil
   213  		}
   214  		return nil, err
   215  	}
   216  	if c.IsExpired(now) {
   217  		return nil, nil
   218  	}
   219  
   220  	items := c.Items()
   221  	if len(items) == 0 {
   222  		return nil, nil
   223  	}
   224  
   225  	ids := make([]uuid.UUID, len(items))
   226  	for i, item := range items {
   227  		ids[i] = item.ProductID()
   228  	}
   229  	products, err := u.productRepo.FindByIDs(ctx, ids)
   230  	if err != nil {
   231  		return nil, err
   232  	}
   233  
   234  	byID := make(map[uuid.UUID]*product.Product, len(products))
   235  	for _, p := range products {
   236  		byID[p.ID()] = p
   237  	}
   238  
   239  	lines := make([]coupon.Line, 0, len(items))
   240  	for _, item := range items {
   241  		p, ok := byID[item.ProductID()]
   242  		if !ok {
   243  			continue
   244  		}
   245  
   246  		snapshot := cart.NewProductSnapshot(cart.ProductSnapshotAttributes{
   247  			Quantity:     p.Quantity(),
   248  			Price:        p.Price(),
   249  			Published:    p.IsPublished(),
   250  			Discontinued: p.IsDiscontinued(),
   251  		})
   252  		if len(item.Evaluate(&snapshot).Issues()) > 0 {
   253  			continue
   254  		}
   255  
   256  		lines = append(lines, coupon.NewLine(coupon.LineAttributes{
   257  			ProductID:  p.ID(),
   258  			CategoryID: p.Category().ID(),
   259  			Subtotal:   p.Price().Decimal().Mul(decimal.FromInt(int64(item.Quantity()))),
   260  		}))
   261  	}
   262  
   263  	return lines, nil
   264  }

   267  func toCouponView(c *coupon.Coupon) CouponView {
   268  	return CouponView{

   273  		ScopeTargetID:     c.Scope().TargetID(),
   274  		MaxAmount:         c.Discount().MaxAmount(),
   275  		MinPurchaseAmount: c.MinPurchaseAmount(),
   276  		UsableFrom:        c.UsableFrom(),
   277  		ExpiresAt:         c.ExpiresAt(),
   278  		UsedAt:            c.UsedAt(),
   279  		IssuedAt:          c.IssuedAt(),
