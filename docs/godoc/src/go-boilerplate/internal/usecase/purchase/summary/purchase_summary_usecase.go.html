   106  func (u *usecase) GetPurchaseSummary(
   107  	ctx context.Context, authn *auth.Authn, params GetSummaryParams,
   108  ) (SummaryView, error) {
   109  	ctx, endSpan := u.tracer.Start(ctx)
   110  	defer endSpan()
   111  
   112  	if authn == nil {
   113  		return SummaryView{}, xerrors.Wrap(apperror.ErrUnauthenticated, "requires authenticated user")
   114  	}
   115  	userID, err := authn.UserID()
   116  	if err != nil {
   117  		return SummaryView{}, xerrors.Wrap(err, "failed to get user ID from authenticator")
   118  	}
   119  
   120  	if err = validateGroupBy(params.GroupBy); err != nil {
   121  		return SummaryView{}, err
   122  	}
   123  
   124  	statuses, err := u.qs.SummarizeByUserID(ctx, userID, params.Window)
   125  	if err != nil {
   126  		return SummaryView{}, err
   127  	}
   128  
   129  	itemsTotal, groups, err := u.summarizeItems(ctx, userID, params.Window, params.GroupBy)
   130  	if err != nil {
   131  		return SummaryView{}, err
   132  	}
   133  
   134  	view := toSummaryView(statuses)
   135  	view.ItemsTotal = itemsTotal
   136  	view.Groups = groups
   137  	return view, nil
   138  }

   143  	ctx context.Context, userID uuid.UUID, window timewindow.Window, groupBy []GroupKind,

   161  // validateGroupBy は、グループ化単位の指定を検証します。未知の単位と同一単位の重複を拒否します。
   162  // 重複を通すと同じ階層が入れ子になり、内訳の総和が親と一致しなくなります。
   163  func validateGroupBy(groupBy []GroupKind) error {
   164  	seen := make(map[GroupKind]struct{}, len(groupBy))
   165  	for _, g := range groupBy {
   166  		if g != GroupByCategory && g != GroupByProduct {
   167  			return xerrors.Wrap(apperror.ErrInvalidArgument, fmt.Sprintf("unknown groupBy: %s", g))
   168  		}
   169  		if _, dup := seen[g]; dup {
   170  			return xerrors.Wrap(apperror.ErrInvalidArgument, fmt.Sprintf("groupBy %s appears more than once", g))
   171  		}
   172  		seen[g] = struct{}{}
   173  	}
   174  	return nil
   175  }

   177  // toSummaryView は、ステータス別の集計結果を総計へ畳み込みつつ出力 DTO へ写像します。
   178  // 総計をステータス別集計から導くことで、総件数・合計金額と内訳が同一スナップショットで整合します。
   179  func toSummaryView(results []query.PurchaseStatusSummaryReadModel) SummaryView {
   180  	view := SummaryView{StatusBreakdown: make([]StatusCountView, len(results))}
   181  	for i, r := range results {
   182  		view.TotalCount += r.Count
   183  		view.TotalAmount += r.TotalAmount
   184  		view.StatusBreakdown[i] = StatusCountView{
   185  			StatusID:    r.StatusID,
   186  			StatusCode:  r.StatusCode,
   187  			StatusName:  r.StatusName,
   188  			Count:       r.Count,
   189  			TotalAmount: r.TotalAmount,
   190  		}
   191  	}
   192  	return view
   193  }

   195  // toGroups は、商品単位の集計行を groupBy の指定順に入れ子のマップへ畳み込み、その総和も返します。
   196  // 総和を同じ行から導くことで、明細金額の合計と内訳が必ず一致します。
   197  func toGroups(
   198  	rows []query.PurchaseItemSummaryReadModel, groupBy []GroupKind,
   199  ) (decimal.Decimal, map[string]GroupNodeView) {
   200  	groups := make(map[string]GroupNodeView, len(rows))
   201  	total := decimal.Decimal{}
   202  	for _, row := range rows {
   203  		groups = accumulate(groups, levelsOf(row, groupBy), row.ItemsTotal)
   204  		total = total.Add(row.ItemsTotal)
   205  	}
   206  	return total, groups
   207  }

   209  // levelsOf は、集計行を groupBy の指定順にたどるためのキーと表示名の並びへ写像します。
   210  // カテゴリ名は商品カテゴリマスタで一意なためそのままキーにでき、商品名は一意でないため ID をキーにします
   211  // （同名の別商品を 1 つのグループへ畳み込まないため）。
   212  func levelsOf(row query.PurchaseItemSummaryReadModel, groupBy []GroupKind) []groupLevel {
   213  	levels := make([]groupLevel, 0, len(groupBy))
   214  	for _, g := range groupBy {
   215  		switch g {
   216  		case GroupByCategory:
   217  			levels = append(levels, groupLevel{key: row.CategoryName, name: row.CategoryName})
   218  		case GroupByProduct:
   219  			levels = append(levels, groupLevel{key: row.ProductID.String(), name: row.ProductName})
   220  		}
   221  	}
   222  	return levels
   223  }
   224  
   225  // accumulate は、1 行の金額を levels がたどる各階層のノードへ加算し、更新後のマップを返します。
   226  // どの階層のノードもその配下の総和を持つため、内訳の合計は常に親と一致します。
   227  func accumulate(groups map[string]GroupNodeView, levels []groupLevel, amount decimal.Decimal) map[string]GroupNodeView {
   228  	if len(levels) == 0 {
   229  		return groups
   230  	}
   231  	if groups == nil {
   232  		groups = make(map[string]GroupNodeView, 1)
   233  	}
   234  	level := levels[0]
   235  	node := groups[level.key]
   236  	node.Name = level.name
   237  	node.ItemsTotal = node.ItemsTotal.Add(amount)
   238  	node.Groups = accumulate(node.Groups, levels[1:], amount)
   239  	groups[level.key] = node
   240  	return groups
   241  }
