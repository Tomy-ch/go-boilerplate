    57  func (u *usecase) GetPurchaseSummary(ctx context.Context, authn *auth.Authn) (SummaryView, error) {
    58  	ctx, endSpan := u.tracer.Start(ctx)
    59  	defer endSpan()
    60  
    61  	if authn == nil {
    62  		return SummaryView{}, xerrors.Wrap(apperror.ErrUnauthenticated, "requires authenticated user")
    63  	}
    64  	userID, err := authn.UserID()
    65  	if err != nil {
    66  		return SummaryView{}, xerrors.Wrap(err, "failed to get user ID from authenticator")
    67  	}
    68  
    69  	results, err := u.qs.SummarizeByUserID(ctx, userID)
    70  	if err != nil {
    71  		return SummaryView{}, err
    72  	}
    73  
    74  	return toSummaryView(results), nil
    75  }

    77  // toSummaryView は、ステータス別の集計結果を総計へ畳み込みつつ出力 DTO へ写像します。
    78  // 総計をステータス別集計から導くことで、総件数・合計金額と内訳が同一スナップショットで整合します。
    79  func toSummaryView(results []query.PurchaseStatusSummaryReadModel) SummaryView {
    80  	view := SummaryView{StatusBreakdown: make([]StatusCountView, len(results))}
    81  	for i, r := range results {
    82  		view.TotalCount += r.Count
    83  		view.TotalAmount += r.TotalAmount
    84  		view.StatusBreakdown[i] = StatusCountView{
    85  			StatusID:    r.StatusID,
    86  			StatusName:  r.StatusName,
    87  			Count:       r.Count,
    88  			TotalAmount: r.TotalAmount,
    89  		}
    90  	}
    91  	return view
