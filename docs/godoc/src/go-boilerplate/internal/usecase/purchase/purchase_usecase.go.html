   242  // newPurchaseDraft は、購入・購入コード・各明細の ID を採番し、ドメイン入力と商品 ID 列を組み立てます。
   243  // 採番はトランザクションの外で行うため、リトライされても同じ ID が再利用されることはありません。
   244  func newPurchaseDraft(details []DetailParam) (*purchaseDraft, error) {
   245  	purchaseID, err := uuid.New()
   246  	if err != nil {
   247  		return nil, xerrors.Wrap(err, "failed to generate purchase id")
   248  	}
   249  	codeUUID, err := uuid.New()
   250  	if err != nil {
   251  		return nil, xerrors.Wrap(err, "failed to generate purchase code")
   252  	}
   253  
   254  	draft := &purchaseDraft{
   255  		purchaseID: purchaseID,
   256  		code:       codeUUID.String(),
   257  		inputs:     make([]purchase.DetailInput, len(details)),
   258  		productIDs: make([]uuid.UUID, len(details)),
   259  	}
   260  	for i, d := range details {
   261  		detailID, derr := uuid.New()
   262  		if derr != nil {
   263  			return nil, xerrors.Wrap(derr, "failed to generate purchase detail id")
   264  		}
   265  		draft.inputs[i] = purchase.DetailInput{ID: detailID, ProductID: d.ProductID, Quantity: d.Quantity}
   266  		draft.productIDs[i] = d.ProductID
   267  	}
   268  	return draft, nil
   269  }

   271  func (u *usecase) CreatePurchase(ctx context.Context, params CreatePurchaseParams) (PurchaseView, error) {
   272  	ctx, endSpan := u.tracer.Start(ctx)
   273  	defer endSpan()
   274  
   275  	draft, err := newPurchaseDraft(params.Details)
   276  	if err != nil {
   277  		return PurchaseView{}, err
   278  	}
   279  
   280  	var created *purchase.Purchase
   281  	// nested で最外 tx に乗る（tx 所有については docs/spec/purchase/usecase.md 冒頭を参照）。
   282  	if txErr := u.txm.Do(ctx, func(ctx context.Context) error {
   283  		// ロック順序（ユーザー行 → 商品行、id 昇順）は docs/spec/purchase/usecase.md の Workflow を参照。
   284  		if uerr := u.ensurePurchaserActive(ctx, params.UserID); uerr != nil {
   285  			return uerr
   286  		}
   287  
   288  		// 在庫ロックは商品集約の読み取りなので、購入の書き込みポートではなく商品 Repository を通す。
   289  		products, lerr := u.productRepo.LockByIDs(ctx, draft.productIDs)
   290  		if lerr != nil {
   291  			return lerr
   292  		}
   293  		locked := make([]purchase.LockedProduct, len(products))
   294  		for i, p := range products {
   295  			locked[i] = purchase.NewLockedProduct(p.ID(), p.Price(), p.Quantity())
   296  		}
   297  
   298  		entity, nerr := purchase.New(draft.purchaseID, draft.code, params.UserID, draft.inputs, locked)
   299  		if nerr != nil {
   300  			return nerr
   301  		}
   302  
   303  		if cerr := u.cmd.CreatePurchase(ctx, entity); cerr != nil {
   304  			return cerr
   305  		}
   306  
   307  		payload, perr := event.BuildCreated(entity)
   308  		if perr != nil {
   309  			return perr
   310  		}
   311  		if _, eerr := u.emit.Emit(ctx, outbox.EmitInput{
   312  			AggregateType: aggregateType,
   313  			AggregateID:   draft.purchaseID.String(),
   314  			EventType:     event.TypeCreated,
   315  			Payload:       payload,
   316  		}); eerr != nil {
   317  			return eerr
   318  		}
   319  
   320  		// 書き込み後、Repository 経由で再検証する（README の Verifying infrastructure against the domain）。
   321  		reread, rerr := u.repo.FindByID(ctx, draft.purchaseID)
   322  		if rerr != nil {
   323  			return rerr
   324  		}
   325  		created = reread
   326  		return nil
   327  	}); txErr != nil {
   328  		return PurchaseView{}, txErr
   329  	}
   330  
   331  	return toPurchaseView(created), nil
   332  }

   334  func (u *usecase) CancelPurchase(ctx context.Context, params CancelPurchaseParams) (CancelPurchaseView, error) {
   335  	ctx, endSpan := u.tracer.Start(ctx)
   336  	defer endSpan()
   337  
   338  	now := u.clock.Now()
   339  
   340  	var detail *purchase.Detail
   341  	// tx 境界は PayPurchase のコメントを参照。CommandService（cmd）を使う理由・二重キャンセル対策は
   342  	// docs/spec/purchase/usecase.md § PATCH キャンセル を参照。
   343  	if txErr := u.txm.Do(ctx, func(ctx context.Context) error {
   344  		locked, lerr := u.cmd.LockPurchase(ctx, params.PurchaseCode)
   345  		if lerr != nil {
   346  			return lerr
   347  		}
   348  
   349  		// 存在秘匿のため NotFound へ畳む理由は CancelPurchase の doc コメントを参照。
   350  		if locked.UserID() != params.UserID {
   351  			return xerrors.Wrap(apperror.ErrNotFound, "purchase not found")
   352  		}
   353  
   354  		domainEvent, cerr := locked.Cancel(now)
   355  		if cerr != nil {
   356  			return cerr
   357  		}
   358  
   359  		if perr := u.cmd.CancelPurchase(ctx, locked); perr != nil {
   360  			return perr
   361  		}
   362  
   363  		payload, berr := event.BuildCanceled(locked)
   364  		if berr != nil {
   365  			return berr
   366  		}
   367  		eventType, terr := event.WireType(domainEvent.Type())
   368  		if terr != nil {
   369  			return terr
   370  		}
   371  		if _, eerr := u.emit.Emit(ctx, outbox.EmitInput{
   372  			AggregateType: aggregateType,
   373  			AggregateID:   locked.ID().String(),
   374  			EventType:     eventType,
   375  			Payload:       payload,
   376  		}); eerr != nil {
   377  			return eerr
   378  		}
   379  
   380  		reread, rerr := u.repo.FindDetailByID(ctx, locked.ID())
   381  		if rerr != nil {
   382  			return rerr
   383  		}
   384  		detail = reread
   385  		return nil
   386  	}); txErr != nil {
   387  		return CancelPurchaseView{}, txErr
   388  	}
   389  
   390  	return toCancelPurchaseView(detail), nil
   391  }

   393  func (u *usecase) PayPurchase(ctx context.Context, params PayPurchaseParams) (PayPurchaseView, error) {
   394  	ctx, endSpan := u.tracer.Start(ctx)
   395  	defer endSpan()
   396  
   397  	now := u.clock.Now()
   398  
   399  	var detail *purchase.Detail
   400  	// この Do が最外 tx（本経路は Idempotency-Key を配線しない）。CommandService を使わない理由・
   401  	// 二重支払い対策は docs/spec/purchase/usecase.md § PATCH 支払い を参照。
   402  	if txErr := u.txm.Do(ctx, func(ctx context.Context) error {
   403  		locked, lerr := u.repo.LockByCode(ctx, params.PurchaseCode)
   404  		if lerr != nil {
   405  			return lerr
   406  		}
   407  
   408  		// 存在秘匿のため NotFound へ畳む理由は CancelPurchase の doc コメントを参照。
   409  		if locked.UserID() != params.UserID {
   410  			return xerrors.Wrap(apperror.ErrNotFound, "purchase not found")
   411  		}
   412  
   413  		domainEvent, perr := locked.Pay(now)
   414  		if perr != nil {
   415  			return perr
   416  		}
   417  
   418  		if perr := u.repo.UpdatePaid(ctx, locked); perr != nil {
   419  			return perr
   420  		}
   421  
   422  		payload, berr := event.BuildPaid(locked)
   423  		if berr != nil {
   424  			return berr
   425  		}
   426  		eventType, terr := event.WireType(domainEvent.Type())
   427  		if terr != nil {
   428  			return terr
   429  		}
   430  		if _, eerr := u.emit.Emit(ctx, outbox.EmitInput{
   431  			AggregateType: aggregateType,
   432  			AggregateID:   locked.ID().String(),
   433  			EventType:     eventType,
   434  			Payload:       payload,
   435  		}); eerr != nil {
   436  			return eerr
   437  		}
   438  
   439  		reread, rerr := u.repo.FindDetailByID(ctx, locked.ID())
   440  		if rerr != nil {
   441  			return rerr
   442  		}
   443  		detail = reread
   444  		return nil
   445  	}); txErr != nil {
   446  		return PayPurchaseView{}, txErr
   447  	}
   448  
   449  	return toPayPurchaseView(detail), nil
   450  }

   458  	if authn == nil {
   459  		return ShipPurchaseView{}, apperror.ErrUnauthenticated
