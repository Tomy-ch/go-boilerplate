    73  // 区分ごとの必須指定の欠落・書式不正、および終了日が開始日より前の場合は ErrInvalidArgument を返します。
    74  func Resolve(spec Spec, now time.Time, loc *time.Location) (Window, error) {

    80  	case KindRecent:

    84  	default:

    89  // resolveMonth は、暦月指定を月初日から月末日までの対象期間へ解決します。
    90  // 月末日は翌月の月初日から 1 日戻して求めるため、月の日数と閏年を場合分けせずに済みます。
    91  func resolveMonth(spec Spec, loc *time.Location) (Window, error) {
    92  	if spec.Month == nil {
    93  		return Window{}, xerrors.Wrap(apperror.ErrInvalidArgument, "period=month requires month")
    94  	}
    95  	first, err := time.ParseInLocation(monthLayout, *spec.Month, loc)
    96  	if err != nil {
    97  		return Window{}, xerrors.Wrap(apperror.ErrInvalidArgument, "month must be in YYYY-MM format: "+err.Error())
    98  	}
    99  	return newWindow(first, first.AddDate(0, 1, -1)), nil
   100  }

   102  // resolveRange は、開始日・終了日の指定を対象期間へ解決します。
   103  func resolveRange(spec Spec, loc *time.Location) (Window, error) {
   104  	if spec.From == nil || spec.To == nil {
   105  		return Window{}, xerrors.Wrap(apperror.ErrInvalidArgument, "period=range requires both from and to")
   106  	}
   107  	from, to := dateOnly(*spec.From, loc), dateOnly(*spec.To, loc)
   108  	if to.Before(from) {
   109  		return Window{}, xerrors.Wrap(apperror.ErrInvalidArgument, "to must not be before from")
   110  	}
   111  	return newWindow(from, to), nil
   112  }

   114  // resolveRecent は、相対指定を今日を終了日とする対象期間へ解決します。
   115  // 開始日は今日から Days 日前で、両端を含むため対象は Days + 1 日分になります。
   116  func resolveRecent(spec Spec, now time.Time, loc *time.Location) (Window, error) {
   117  	if spec.Days == nil {
   118  		return Window{}, xerrors.Wrap(apperror.ErrInvalidArgument, "period=recent requires days")
   119  	}
   120  	if *spec.Days < minRecentDays {
   121  		return Window{}, xerrors.Wrap(
   122  			apperror.ErrInvalidArgument,
   123  			fmt.Sprintf("days must be %d or greater, got %d", minRecentDays, *spec.Days),
   124  		)
   125  	}
   126  	// 今日は loc の暦日で判定するため、現在時刻を loc へ移してから年月日を取る。
   127  	today := dateOnly(now.In(loc), loc)
   128  	return newWindow(today.AddDate(0, 0, -*spec.Days), today), nil
   129  }

   132  func newWindow(from, to time.Time) Window {
   133  	return Window{from: from, to: to, filtered: true}

   136  // dateOnly は、t が表す年月日の開始時刻を loc のゾーンで返します。年月日は t 自身のロケーションで解釈し、
   137  // loc は返り値のゾーンとしてのみ用います。利用者が指定した暦日をそのまま保つためで、t を loc へ変換すると
   138  // UTC より西のロケーションでは前日へずれます。
   139  func dateOnly(t time.Time, loc *time.Location) time.Time {
   140  	return time.Date(t.Year(), t.Month(), t.Day(), 0, 0, 0, 0, loc)
