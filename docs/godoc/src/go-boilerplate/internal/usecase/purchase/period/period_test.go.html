    13  // testLoc は、注入されたロケーションが使われることを設定値と独立に固定するための、UTC から離れた固定ゾーンです。

    22  }
    23  
    24  // utcDate は、OpenAPI の date が解析される形（UTC 深夜）で testNow と同じ年月の暦日を返します。

    26  	return time.Date(testNow.Year(), testNow.Month(), day, 0, 0, 0, 0, time.UTC)
