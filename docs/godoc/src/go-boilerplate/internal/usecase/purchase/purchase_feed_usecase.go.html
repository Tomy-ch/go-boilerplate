    45  // 対象が無い場合は Repository が空一覧を返すため、他ユーザーの購入が混ざる経路は存在しません。

    47  	if cursor == nil {
    48  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "cursor must not be nil")
    49  	}
    50  
    51  	ctx, endSpan := u.tracer.Start(ctx)
    52  	defer endSpan()

    55  	if err != nil {
    56  		return nil, err
    57  	}
    58  
    59  	params := purchase.ListFeedParams{Limit: cursor.Limit32() + 1}
    60  	if after != nil {
    61  		params.AfterOrderedAt = &after.orderedAt
    62  		params.AfterID = &after.id
    63  	}
    64

    66  	if err != nil {
    67  		return nil, err
    68  	}
    69  
    70  	limit := cursor.Limit()
    71  	hasNext := len(feed) > limit
    72  	if hasNext {
    73  		feed = feed[:limit]
    74  	}
    75  
    76  	items := make([]PurchaseSummaryView, len(feed))
    77  	for i, f := range feed {
    78  		items[i] = PurchaseSummaryView{
    79  			Code:        f.Code,
    80  			TotalAmount: f.TotalAmount,
    81  			StatusID:    f.StatusID,
    82  			StatusName:  f.StatusName,
    83  			OrderedAt:   f.OrderedAt,
    84  		}
    85  	}
    86  
    87  	var nextCursor *string
    88  	if hasNext && len(feed) > 0 {
    89  		// len 判定は防御的な安全弁。hasNext は len > limit なので limit >= 1 の下では冗長だが、
    90  		// limit の下限保証は paging.NewCursor 依存であり、ゼロ値 Cursor 混入時の feed[-1] panic を防ぐ。

    92  		nextCursor = &encoded
    93  	}
    94

    98  // decodePurchaseCursor は、cursor の不透明キー列を keyset 境界（purchaseCursor）へ解釈します。
    99  // 先頭ページ（カーソル無し）の場合は nil を返します。キーの個数・型が不正な場合は ErrInvalidArgument を返します。
   100  func decodePurchaseCursor(cursor *paging.Cursor) (*purchaseCursor, error) {
   101  	if !cursor.HasCursor() {
   102  		return nil, nil //nolint:nilnil // 先頭ページは境界なし（nil）を正常値として返す
   103  	}
   104  
   105  	keys := cursor.Keys()
   106  	if len(keys) != purchaseCursorKeyCount {
   107  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, fmt.Sprintf("invalid cursor: expected %d keys", purchaseCursorKeyCount))
   108  	}
   109  
   110  	orderedAt, err := time.Parse(time.RFC3339Nano, keys[0])
   111  	if err != nil {
   112  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "invalid cursor: ordered_at is not RFC3339Nano")
   113  	}
   114  
   115  	id, err := uuid.Parse(keys[1])
   116  	if err != nil {
   117  		return nil, xerrors.Wrap(apperror.ErrInvalidArgument, "invalid cursor: id is not a valid UUID")
   118  	}
   119  
   120  	return &purchaseCursor{orderedAt: orderedAt, id: id}, nil
   121  }
