    18  // CommandService は、購入に伴う複数集約（商品在庫 / 購入 / 購入明細）への
    19  // 原子的な書き込みを定義します。金額計算・売り越し判定・スナップショットはドメインが持ち、本サービスは
    20  // 「決定済みの書き込みの実行」のみを担います（QueryService の write 側対称物）。
    21  //
    22  // ここに載せてよいのは、集約を読み込んで保存する形では表現できない書き込みだけです。すなわち相対更新・
    23  // 集合演算・ロックを取らずに原子性を得る操作。集約を読んで変更して保存できるものは Repository に載ります。
    24  // この線引きが無いと、CommandService は SQL を直接書きたいときの抜け道に劣化します。
    25  //
    26  // 本サービスが強制する条件は、ドメインの不変条件から導出されたものでなければならず、独立に書き起こしては
    27  // いけません。同じ規則が 2 箇所に別々に書かれると、片方だけ変わっても誰も気づけなくなります。
    28  type CommandService interface {
    29  	// CreatePurchase は、在庫の減算・購入の作成・明細の作成を、渡された ctx のトランザクション内で
    30  	// 原子的に実行します。在庫減算は防御的に売り越しを弾きます。
    31  	CreatePurchase(ctx context.Context, p *purchase.Purchase) error
    32  	// LockPurchase は、対象の購入を悲観ロックして明細込みで再構築し返します。
    33  	// キャンセルの状態遷移の競合（同一購入への並行キャンセル）をこのロックで直列化します。
    34  	// 存在しない場合は NotFound を返します。
    35  	LockPurchase(ctx context.Context, id uuid.UUID) (*purchase.Purchase, error)
    36  	// CancelPurchase は、キャンセルに伴う在庫復元（明細分の加算）と購入の状態遷移（→ キャンセル）を、
    37  	// 渡された ctx のトランザクション内で原子的に実行します。在庫加算は相対更新で売り越しを生みません。
    38  	CancelPurchase(ctx context.Context, p *purchase.Purchase) error
