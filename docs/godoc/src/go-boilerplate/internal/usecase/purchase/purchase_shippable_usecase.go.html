    64  // ListShippablePurchases は、admin 認可のうえ、発送可能な購入をまとめて発送してよい組に分けて返します。
    65  func (u *usecase) ListShippablePurchases(
    66  	ctx context.Context, authn *auth.Authn, params ListShippablePurchasesParams,
    67  ) (PurchaseShippableListView, error) {
    68  	ctx, endSpan := u.tracer.Start(ctx)
    69  	defer endSpan()
    70  
    71  	if authn == nil {
    72  		return PurchaseShippableListView{}, apperror.ErrUnauthenticated
    73  	}
    74  	if err := u.authorizer.Authorize(
    75  		ctx, authn, authz.ActionPurchaseListShippable, authz.NewResource("purchase", nil),
    76  	); err != nil {
    77  		return PurchaseShippableListView{}, err
    78  	}
    79  
    80  	limit := paging.NewLimit(ptr.To(params.Limit), shippableLimitPolicy)
    81  	purchases, err := u.repo.FindShippable(ctx, limit.Value32())
    82  	if err != nil {
    83  		return PurchaseShippableListView{}, err
    84  	}
    85  	// 絞り込みを実行するのは SQL だが、発送可能を定義するのは Purchase.IsShippable。両者の乖離を表に出す。
    86  	for _, p := range purchases {
    87  		if !p.IsShippable() {
    88  			return PurchaseShippableListView{}, xerrors.Wrap(errNotShippableInShippableRead, p.ID().String())
    89  		}
    90  	}
    91  
    92  	groups := dispatch.GroupForDispatch(purchases)
    93  
    94  	views := make([]DispatchGroupView, len(groups))
    95  	for i, group := range groups {
    96  		views[i] = toDispatchGroupView(group)
    97  	}
    98  
    99  	return PurchaseShippableListView{Groups: views}, nil
   100  }

   102  // toDispatchGroupView は、まとめ発送の組を出力 DTO へ写像します。
   103  // 組は同一購入者で分けられているため、購入者 ID は先頭の購入から取ります。
   104  func toDispatchGroupView(group purchase.Purchases) DispatchGroupView {
   105  	items := make([]ShippablePurchaseView, len(group))
   106  	for i, p := range group {
   107  		items[i] = ShippablePurchaseView{
   108  			ID:          p.ID(),
   109  			Code:        p.Code(),
   110  			TotalAmount: p.TotalAmount(),
   111  			OrderedAt:   p.OrderedAt(),
   112  		}
   113  	}
   114  	return DispatchGroupView{UserID: group[0].UserID(), Purchases: items}
   115  }
