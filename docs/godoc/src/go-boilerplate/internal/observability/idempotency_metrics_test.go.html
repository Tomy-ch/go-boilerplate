   264  // attributeOf は、指定 counter の最初のデータ点から指定キーの属性値を取り出します。
   265  func attributeOf(t *testing.T, rm metricdata.ResourceMetrics, name, key string) string {
   266  	t.Helper()
   267  	for _, sm := range rm.ScopeMetrics {
   268  		for _, m := range sm.Metrics {
   269  			if m.Name != name {
   270  				continue
   271  			}
   272  			sum, ok := m.Data.(metricdata.Sum[int64])
   273  			require.True(t, ok)
   274  			require.NotEmpty(t, sum.DataPoints)
   275  			v, ok := sum.DataPoints[0].Attributes.Value(attribute.Key(key))
   276  			require.True(t, ok)
   277  			return v.AsString()
   278  		}
   279  	}
   280  	t.Fatalf("metric %s not found", name)
   281  	return ""
   282  }

   304  // counterValueOf は、指定 counter の全データ点の合計値を返します。
   305  func counterValueOf(t *testing.T, rm metricdata.ResourceMetrics, name string) int64 {
   306  	t.Helper()
   307  	for _, sm := range rm.ScopeMetrics {
   308  		for _, m := range sm.Metrics {
   309  			if m.Name != name {
   310  				continue
   311  			}
   312  			sum, ok := m.Data.(metricdata.Sum[int64])
   313  			require.True(t, ok)
   314  			var total int64
   315  			for _, dp := range sum.DataPoints {
   316  				total += dp.Value
   317  			}
   318  			return total
   319  		}
   320  	}
   321  	t.Fatalf("metric %s not found", name)
   322  	return 0
