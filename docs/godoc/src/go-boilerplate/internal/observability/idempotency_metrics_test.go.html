    76  // attributeOf は、指定 counter の最初のデータ点から指定キーの属性値を取り出します。
    77  func attributeOf(t *testing.T, rm metricdata.ResourceMetrics, name, key string) string {
    78  	t.Helper()
    79  	for _, sm := range rm.ScopeMetrics {
    80  		for _, m := range sm.Metrics {
    81  			if m.Name != name {
    82  				continue
    83  			}
    84  			sum, ok := m.Data.(metricdata.Sum[int64])
    85  			require.True(t, ok)
    86  			require.NotEmpty(t, sum.DataPoints)
    87  			v, ok := sum.DataPoints[0].Attributes.Value(attribute.Key(key))
    88  			require.True(t, ok)
    89  			return v.AsString()
    90  		}
    91  	}
    92  	t.Fatalf("metric %s not found", name)
    93  	return ""
    94  }

   128  // counterValueOf は、指定 counter の全データ点の合計値を返します。
   129  func counterValueOf(t *testing.T, rm metricdata.ResourceMetrics, name string) int64 {
   130  	t.Helper()
   131  	for _, sm := range rm.ScopeMetrics {
   132  		for _, m := range sm.Metrics {
   133  			if m.Name != name {
   134  				continue
   135  			}
   136  			sum, ok := m.Data.(metricdata.Sum[int64])
   137  			require.True(t, ok)
   138  			var total int64
   139  			for _, dp := range sum.DataPoints {
   140  				total += dp.Value
   141  			}
   142  			return total
   143  		}
   144  	}
   145  	t.Fatalf("metric %s not found", name)
   146  	return 0
   147  }

   316  func TestIdempotencyMetrics_IncExpiredCleanupFailure(t *testing.T) {
   317  	t.Parallel()
   318  
   319  	t.Run("正常系", func(t *testing.T) {
   320  		t.Parallel()
   321  
   322  		t.Run("専用カウンタへ job=idempotency_gc で計上しper-requestのfailuresへは計上しない", func(t *testing.T) {
   323  			t.Parallel()
   324  
   325  			im, reader := newIdempotencyMetricsForTest(t)
   326  
   327  			im.IncExpiredCleanupFailure(context.Background())
   328  
   329  			rm := collectIdempotencyMetrics(t, reader)
   330  			// GC 失敗は per-request の failures ではなく専用カウンタへ計上され、
   331  			// operation_id="unknown" を帯びず job ラベルで成功と対称になる。
   332  			assert.Equal(t, "idempotency_gc", attributeOf(t, rm, "idempotency.expired_cleanup_failure", "job"))
   333  			assert.Equal(t, int64(1), counterValueOf(t, rm, "idempotency.expired_cleanup_failure"))
   334  			// 併せて per-request の idempotency.failures には一切 emit されないことを固定する
   335  			// （GC 失敗を専用カウンタと failures の両方へ計上する二重計上の回帰を検出する）。
   336  			assert.False(t, metricPresent(rm, "idempotency.failures"),
   337  				"GC 失敗が per-request failures へ二重計上されないこと")
   338  		})
   339  	})
