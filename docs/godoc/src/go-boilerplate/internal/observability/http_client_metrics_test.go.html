   200  func gaugeValueOf(t *testing.T, rm metricdata.ResourceMetrics, name string) int64 {
   201  	t.Helper()

   208  // attrOfAny は、Sum / Histogram / Gauge いずれの metric でも最初のデータ点から指定キーの属性値を取り出します。
   209  func attrOfAny(t *testing.T, rm metricdata.ResourceMetrics, name, key string) string {
   210  	t.Helper()
   211  	var attrs attribute.Set
   212  	switch d := metricByName(t, rm, name).Data.(type) {
   213  	case metricdata.Sum[int64]:
   214  		require.NotEmpty(t, d.DataPoints)
   215  		attrs = d.DataPoints[0].Attributes
   216  	case metricdata.Histogram[float64]:
   217  		require.NotEmpty(t, d.DataPoints)
   218  		attrs = d.DataPoints[0].Attributes
   219  	case metricdata.Gauge[int64]:
   220  		require.NotEmpty(t, d.DataPoints)
   221  		attrs = d.DataPoints[0].Attributes
   222  	default:
   223  		t.Fatalf("unsupported metric data type for %s", name)
   224  	}
   225  	v, ok := attrs.Value(attribute.Key(key))
   226  	require.True(t, ok)
   227  	return v.AsString()
   228  }

   230  func TestNewHTTPClientTransport(t *testing.T) {
   231  	t.Parallel()
   232  
   233  	t.Run("正常系", func(t *testing.T) {
   234  		t.Parallel()
   235  
   236  		t.Run("生成したRoundTripperでリクエストを送信できる", func(t *testing.T) {
   237  			t.Parallel()
   238  
   239  			srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
   240  				w.WriteHeader(http.StatusOK)
   241  			}))
   242  			t.Cleanup(srv.Close)
   243  
   244  			transport := observability.NewHTTPClientTransport(
   245  				tracenoop.NewTracerProvider(), observability.NewTextMapPropagator(),
   246  			)
   247  			require.NotNil(t, transport)
   248  
   249  			// httptest は loopback のため、SSRF ガードを通すには private 許可フラグが要る。
   250  			ctx := observability.ContextWithAllowPrivateNetwork(context.Background(), true)
   251  			client := &http.Client{Transport: transport.RoundTripper()}
   252  			req, err := http.NewRequestWithContext(ctx, http.MethodGet, srv.URL, nil)
   253  			require.NoError(t, err)
   254  
   255  			resp, err := client.Do(req)
   256  			require.NoError(t, err)
   257  			t.Cleanup(func() { _ = resp.Body.Close() })
   258  			assert.Equal(t, http.StatusOK, resp.StatusCode)
   259  		})
   260  	})
