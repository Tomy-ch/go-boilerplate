    75  // attrOfAny は、Sum / Histogram / Gauge いずれの metric でも最初のデータ点から指定キーの属性値を取り出します。
    76  func attrOfAny(t *testing.T, rm metricdata.ResourceMetrics, name, key string) string {
    77  	t.Helper()
    78  	var attrs attribute.Set
    79  	switch d := metricByName(t, rm, name).Data.(type) {
    80  	case metricdata.Sum[int64]:
    81  		require.NotEmpty(t, d.DataPoints)
    82  		attrs = d.DataPoints[0].Attributes
    83  	case metricdata.Histogram[float64]:
    84  		require.NotEmpty(t, d.DataPoints)
    85  		attrs = d.DataPoints[0].Attributes
    86  	case metricdata.Gauge[int64]:
    87  		require.NotEmpty(t, d.DataPoints)
    88  		attrs = d.DataPoints[0].Attributes
    89  	default:
    90  		t.Fatalf("unsupported metric data type for %s", name)
    91  	}
    92  	v, ok := attrs.Value(attribute.Key(key))
    93  	require.True(t, ok)
    94  	return v.AsString()
    95  }

    98  	t.Parallel()
    99  
   100  	t.Run("正常系", func(t *testing.T) {
   101  		t.Parallel()
   102  
   103  		t.Run("生成したRoundTripperでリクエストを送信できる", func(t *testing.T) {
   104  			t.Parallel()
   105  
   106  			srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
   107  				w.WriteHeader(http.StatusOK)
   108  			}))
   109  			t.Cleanup(srv.Close)
   110  
   111  			transport := observability.NewHTTPClientTransport(
   112  				tracenoop.NewTracerProvider(), observability.NewTextMapPropagator(),
   113  			)
   114  			require.NotNil(t, transport)
   115  
   116  			// httptest は loopback のため、SSRF ガードを通すには private 許可フラグが要る。
   117  			ctx := observability.ContextWithAllowPrivateNetwork(context.Background(), true)
   118  			client := &http.Client{Transport: transport.RoundTripper()}
   119  			req, err := http.NewRequestWithContext(ctx, http.MethodGet, srv.URL, nil)
   120  			require.NoError(t, err)
   121  
   122  			resp, err := client.Do(req)
   123  			require.NoError(t, err)
   124  			t.Cleanup(func() { _ = resp.Body.Close() })
   125  			assert.Equal(t, http.StatusOK, resp.StatusCode)
   126  		})
   127  	})

   151  			t.Parallel()

   154  			ctx := context.Background()
   155  
   156  			hm.RecordRequest(ctx, "outbox", "2xx")
   157  			hm.RecordRequest(ctx, "outbox", "5xx")
   158  
   159  			rm := collectHTTPClientMetrics(t, reader)
   160  			// status_class ごとに系列が分かれる（合計 2、データ点も 2）。
   161  			assert.Equal(t, int64(2), counterValueOf(t, rm, "httpclient.requests"))
   162  			sum, ok := metricByName(t, rm, "httpclient.requests").Data.(metricdata.Sum[int64])
