     9  // maxOutboundRedirects は、追従を許す最大リダイレクト回数です。
    10  const maxOutboundRedirects = 10

    12  // OutboundHTTPClient は、SSRF ガード付き transport を持つ外部 SDK 向けの HTTP クライアントです。
    13  // 自前の substrate（infrastructure/httpclient）を通せない SDK へ渡すための型で、素の *http.Client と
    14  // 取り違えないよう名前を持たせています。
    15  type OutboundHTTPClient struct {
    16  	*http.Client
    17  }
    18  
    19  // policyStampingRoundTripper は、通過する全リクエストへ固定の SSRF ガード方針を適用する RoundTripper です。
    20  type policyStampingRoundTripper struct {
    21  	inner               http.RoundTripper
    22  	allowPrivateNetwork bool
    23  }
    24  
    25  // NewOutboundHTTPClient は、SSRF ガード付き transport を持つ OutboundHTTPClient を返します。
    26  //
    27  // SDK は独自のリトライとタイムアウトを持つため、ここで与えるのは transport だけで、breaker や
    28  // budget は載せません。
    29  //
    30  // allowPrivateNetwork はこの client を通る全リクエストへ一律に適用します。SDK は呼び出し側の ctx を
    31  // そのままリクエストへ渡すため、呼び出しごとに ctx へ積む形にすると、積み忘れた経路が黙って既定値で
    32  // 通ってしまいます。link-local（クラウドメタデータ等）の拒否はフラグに依らず常に効きます。
    33  //
    34  // 同じ t から方針の異なる client を作らないでください。判定は dial の直前に行う一方、transport は
    35  // コネクションプールを持つため、許可側が張った接続を拒否側が再利用すると判定を経ずに通ります。
    36  func NewOutboundHTTPClient(t *HTTPClientTransport, allowPrivateNetwork bool) *OutboundHTTPClient {
    37  	return &OutboundHTTPClient{
    38  		Client: &http.Client{
    39  			Transport: policyStampingRoundTripper{
    40  				inner:               t.RoundTripper(),
    41  				allowPrivateNetwork: allowPrivateNetwork,
    42  			},
    43  			CheckRedirect: limitedRedirectForOutbound,
    44  		},
    45  	}
    46  }
    47  
    48  // RoundTrip は、方針を ctx へ積んだ clone を内側へ委譲します。
    49  func (rt policyStampingRoundTripper) RoundTrip(req *http.Request) (*http.Response, error) {
    50  	// http.RoundTripper は呼び出し元の Request を変更してはならないため clone する。
    51  	return rt.inner.RoundTrip(
    52  		req.Clone(ContextWithAllowPrivateNetwork(req.Context(), rt.allowPrivateNetwork)),
    53  	)
    54  }
    55  
    56  // limitedRedirectForOutbound は、メソッドと本文を保つ 307 / 308 だけ追従し、他は最終レスポンス
    57  // （3xx）をそのまま返します。
    58  //
    59  // 全面禁止にしないのは、AWS SDK 既定のクライアントが同じ範囲で追従するためです。HTTPClient を
    60  // 明示した時点で SDK 側の追従は効かなくなるので、ここで同じ範囲を持たないと、S3 が 307 を返す
    61  // 場面（作成直後のバケット等）で操作が失敗するようになります。
    62  // 追従先も同じ transport を通るため、link-local / private 宛ての判定は追従後の接続にも効きます。
    63  func limitedRedirectForOutbound(req *http.Request, via []*http.Request) error {
    64  	if req.Response == nil || len(via) >= maxOutboundRedirects {
    65  		return http.ErrUseLastResponse
    66  	}
    67  	switch req.Response.StatusCode {
    68  	case http.StatusTemporaryRedirect, http.StatusPermanentRedirect:
    69  		return nil
    70  	default:
    71  		return http.ErrUseLastResponse
    72  	}
    73  }

    75  // NewDisabledOutboundHTTPClient は、スパンを一切送出しない OutboundHTTPClient を返します。
