    16  // IdempotencyMetrics は、冪等性サブシステムの判定結果・内部失敗・GC 削除を計上する計装一式です。
    17  // usecase 層の idempotency.Metrics および idempotency.GCMetrics を実装します。
    18  //
    19  // 高カーディナリティ・秘匿値（Idempotency-Key / scope / fingerprint / PII / raw error）は
    20  // ラベルに載せません。許可ラベルは operation_id / result / phase / job のみです。
    21  type IdempotencyMetrics struct {
    22  	requests              metric.Int64Counter
    23  	failures              metric.Int64Counter
    24  	expiredCleanup        metric.Int64Counter
    25  	expiredCleanupFailure metric.Int64Counter
    26  }

    28  // NewIdempotencyMetrics は、注入された MeterProvider から冪等性計装一式を生成します。
    29  // いずれかの計装生成失敗で error を返します。
    30  func NewIdempotencyMetrics(mp metric.MeterProvider) (*IdempotencyMetrics, error) {
    31  	b := &meterBuilder{m: mp.Meter(idempotencyMeterName)}
    32  	im := &IdempotencyMetrics{
    33  		requests:              b.counter("idempotency.requests", "冪等性判定結果数(result=hit/miss/conflict/fingerprint_mismatch 別)"),
    34  		failures:              b.counter("idempotency.failures", "冪等性のリクエスト内部失敗数(phase=claim/complete 別)"),
    35  		expiredCleanup:        b.counter("idempotency.expired_cleanup", "GC が削除した失効キー件数(job=idempotency_gc)"),
    36  		expiredCleanupFailure: b.counter("idempotency.expired_cleanup_failure", "GC 削除バッチの失敗回数(job=idempotency_gc)"),
    37  	}
    38  	if b.err != nil {
    39  		return nil, b.err
    40  	}
    41  	return im, nil
    42  }
