    83  			t.Parallel()
    84  
    85  			ctx := context.Background()
    86  			reader := sdkmetric.NewManualReader()

    88  
    89  			wm, err := observability.NewWorkerMetrics(provider)
    90  			require.NoError(t, err)

    92  			// メソッドごとに区別できる測定値を与える（manual reader は測定のある計装のみ出力）。
    93  			wm.Received(ctx, 5)
    94  			wm.Processed(ctx)
    95  			wm.Failed(ctx)
    96  			wm.Retried(ctx)
    97  			wm.DLQ(ctx)
    98  			wm.PollError(ctx)
    99  			wm.ExtendError(ctx)
   100  			wm.RecordLatencyMs(ctx, 12.5)
   101  			wm.InFlightAdd(ctx, 3)
   102  
   103  			var rm metricdata.ResourceMetrics
   104  			require.NoError(t, reader.Collect(ctx, &rm))
   105  
   106  			names := map[string]bool{}
   107  			for _, sm := range rm.ScopeMetrics {
   108  				for _, m := range sm.Metrics {
   109  					names[m.Name] = true
   110  				}
   111  			}
   112  
   113  			for _, want := range []string{
   114  				"worker.received", "worker.processed", "worker.failed", "worker.retried",
   115  				"worker.dlq", "worker.poll_errors", "worker.extend_errors",
   116  				"worker.processing_latency_ms", "worker.in_flight",
   117  			} {
   118  				assert.Contains(t, names, want)
   119  			}
   120  
   121  			// 記録値そのものを検証する（名前存在のみの疑似陽性を防ぐ）。
   122  			assert.Equal(t, int64(5), counterValueOf(t, rm, "worker.received"))
   123  			assert.Equal(t, int64(1), counterValueOf(t, rm, "worker.processed"))
   124  			assert.Equal(t, int64(3), counterValueOf(t, rm, "worker.in_flight"))
   125  			assert.InDelta(t, 12.5, histogramSumOf(t, rm, "worker.processing_latency_ms"), 1e-9)
   126  		})
   127  	})

   161  // histogramSumOf は、指定 histogram の最初のデータ点の Sum を返します。
   162  func histogramSumOf(t *testing.T, rm metricdata.ResourceMetrics, name string) float64 {
   163  	t.Helper()
   164  	for _, sm := range rm.ScopeMetrics {
   165  		for _, m := range sm.Metrics {
   166  			if m.Name != name {
   167  				continue
   168  			}
   169  			h, ok := m.Data.(metricdata.Histogram[float64])
   170  			require.True(t, ok)
   171  			require.NotEmpty(t, h.DataPoints)
   172  			return h.DataPoints[0].Sum
   173  		}
   174  	}
   175  	t.Fatalf("metric %s not found", name)
   176  	return 0
   177  }

   179  // metricNamesOf は、収集結果に現れた metric 名を昇順で返します。
   180  // manual reader は測定のあった計装のみ出力するため、「どの計装へ計上されたか」を排他的に固定できます。
   181  func metricNamesOf(t *testing.T, rm metricdata.ResourceMetrics) []string {
   182  	t.Helper()
   183  	var names []string
   184  	for _, sm := range rm.ScopeMetrics {
   185  		for _, m := range sm.Metrics {
   186  			names = append(names, m.Name)
   187  		}
   188  	}
   189  	sort.Strings(names)
   190  	return names
   191  }
