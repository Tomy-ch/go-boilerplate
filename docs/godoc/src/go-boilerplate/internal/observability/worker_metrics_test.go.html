    76  	t.Parallel()

    78  	t.Run("正常系", func(t *testing.T) {
    79  		t.Parallel()

    81  		t.Run("engine 所有の全 metric が登録され記録できる", func(t *testing.T) {
    82  			t.Parallel()
    83  
    84  			ctx := context.Background()
    85  			reader := sdkmetric.NewManualReader()
    86  			provider := sdkmetric.NewMeterProvider(sdkmetric.WithReader(reader))
    87  
    88  			wm, err := observability.NewWorkerMetrics(provider)
    89  			require.NoError(t, err)
    90  
    91  			// メソッドごとに区別できる測定値を与える（manual reader は測定のある計装のみ出力）。
    92  			wm.Received(ctx, 5)
    93  			wm.Processed(ctx)
    94  			wm.Failed(ctx)
    95  			wm.Retried(ctx)
    96  			wm.DLQ(ctx)
    97  			wm.PollError(ctx)
    98  			wm.ExtendError(ctx)
    99  			wm.RecordLatencyMs(ctx, 12.5)
   100  			wm.InFlightAdd(ctx, 3)
   101  
   102  			var rm metricdata.ResourceMetrics
   103  			require.NoError(t, reader.Collect(ctx, &rm))
   104  
   105  			names := map[string]bool{}
   106  			for _, sm := range rm.ScopeMetrics {
   107  				for _, m := range sm.Metrics {
   108  					names[m.Name] = true
   109  				}
   110  			}
   111  
   112  			for _, want := range []string{
   113  				"worker.received", "worker.processed", "worker.failed", "worker.retried",
   114  				"worker.dlq", "worker.poll_errors", "worker.extend_errors",
   115  				"worker.processing_latency_ms", "worker.in_flight",
   116  			} {
   117  				assert.Contains(t, names, want)
   118  			}
   119  
   120  			// 記録値そのものを検証する（名前存在のみの疑似陽性を防ぐ）。
   121  			assert.Equal(t, int64(5), counterValueOf(t, rm, "worker.received"))
   122  			assert.Equal(t, int64(1), counterValueOf(t, rm, "worker.processed"))
   123  			assert.Equal(t, int64(3), counterValueOf(t, rm, "worker.in_flight"))
   124  			assert.InDelta(t, 12.5, histogramSumOf(t, rm, "worker.processing_latency_ms"), 1e-9)
   125  		})
   126  	})

   160  // histogramSumOf は、指定 histogram の最初のデータ点の Sum を返します。
   161  func histogramSumOf(t *testing.T, rm metricdata.ResourceMetrics, name string) float64 {
   162  	t.Helper()
   163  	for _, sm := range rm.ScopeMetrics {
   164  		for _, m := range sm.Metrics {
   165  			if m.Name != name {
   166  				continue
   167  			}
   168  			h, ok := m.Data.(metricdata.Histogram[float64])
   169  			require.True(t, ok)
   170  			require.NotEmpty(t, h.DataPoints)
   171  			return h.DataPoints[0].Sum
   172  		}
   173  	}
   174  	t.Fatalf("metric %s not found", name)
   175  	return 0
   176  }
