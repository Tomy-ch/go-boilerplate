   151  func Test_newHTTPClientTransport(t *testing.T) {
   152  	t.Parallel()
   153  
   154  	t.Run("正常系", func(t *testing.T) {
   155  		t.Parallel()
   156  
   157  		t.Run("注入したdial controlを経由して宛先へ接続する", func(t *testing.T) {
   158  			t.Parallel()
   159  
   160  			srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
   161  				w.WriteHeader(http.StatusNoContent)
   162  			}))
   163  			defer srv.Close()
   164  
   165  			var gotAddr string
   166  			control := func(_ context.Context, _, address string, _ syscall.RawConn) error {
   167  				gotAddr = address
   168  				return nil
   169  			}
   170  
   171  			transport := newHTTPClientTransport(noop.NewTracerProvider(), NewTextMapPropagator(), control)
   172  			req, err := http.NewRequestWithContext(context.Background(), http.MethodGet, srv.URL, nil)
   173  			require.NoError(t, err)
   174  
   175  			resp, err := (&http.Client{Transport: transport.RoundTripper()}).Do(req)
   176  			require.NoError(t, err)
   177  			defer func() { _ = resp.Body.Close() }()
   178  
   179  			assert.Equal(t, http.StatusNoContent, resp.StatusCode)
   180  			// 既定の guardedDialControl ではなく注入した control が dial 判定に使われている。
   181  			assert.Equal(t, srv.Listener.Addr().String(), gotAddr)
   182  		})
   183  	})
   184  
   185  	t.Run("異常系", func(t *testing.T) {
   186  		t.Parallel()
   187  
   188  		t.Run("dial controlが拒否した宛先へは接続せずエラーを伝播する", func(t *testing.T) {
   189  			t.Parallel()
   190  
   191  			srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
   192  				w.WriteHeader(http.StatusNoContent)
   193  			}))
   194  			defer srv.Close()
   195  
   196  			control := func(context.Context, string, string, syscall.RawConn) error { return errTestDialBlocked }
   197  
   198  			transport := newHTTPClientTransport(noop.NewTracerProvider(), NewTextMapPropagator(), control)
   199  			req, err := http.NewRequestWithContext(context.Background(), http.MethodGet, srv.URL, nil)
   200  			require.NoError(t, err)
   201  
   202  			resp, err := (&http.Client{Transport: transport.RoundTripper()}).Do(req)
   203  			require.ErrorIs(t, err, errTestDialBlocked)
   204  			assert.Nil(t, resp)
   205  		})
   206  	})
