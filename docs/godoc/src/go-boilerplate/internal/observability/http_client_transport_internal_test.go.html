    13  func TestGuardedDialControl(t *testing.T) {
    14  	t.Parallel()
    15  
    16  	allow := ContextWithAllowPrivateNetwork(context.Background(), true)
    17  	deny := ContextWithAllowPrivateNetwork(context.Background(), false)
    18  
    19  	t.Run("正常系", func(t *testing.T) {
    20  		t.Parallel()
    21  
    22  		t.Run("グローバルアドレスは許可する", func(t *testing.T) {
    23  			t.Parallel()
    24  			require.NoError(t, guardedDialControl(deny, "tcp", "93.184.216.34:443", nil))
    25  		})
    26  
    27  		t.Run("private許可フラグありならloopback/privateを許可する", func(t *testing.T) {
    28  			t.Parallel()
    29  			require.NoError(t, guardedDialControl(allow, "tcp", "127.0.0.1:8080", nil))
    30  			require.NoError(t, guardedDialControl(allow, "tcp", "10.0.0.5:80", nil))
    31  			require.NoError(t, guardedDialControl(allow, "tcp", "192.168.1.10:80", nil))
    32  			// CGNAT(RFC 6598) も private 扱いで、フラグありなら許可する。
    33  			require.NoError(t, guardedDialControl(allow, "tcp", "100.64.0.1:80", nil))
    34  		})
    35  
    36  		t.Run("CGNAT帯の外(100.128.x)は許可する", func(t *testing.T) {
    37  			t.Parallel()
    38  			// 100.128.0.0 は /10 の外＝グローバル扱いで通ること（過剰ブロック防止）。
    39  			require.NoError(t, guardedDialControl(deny, "tcp", "100.128.0.1:80", nil))
    40  		})
    41  	})
    42  
    43  	t.Run("異常系", func(t *testing.T) {
    44  		t.Parallel()
    45  
    46  		t.Run("リンクローカル(メタデータ)はフラグに関わらず拒否する", func(t *testing.T) {
    47  			t.Parallel()
    48  			require.Error(t, guardedDialControl(allow, "tcp", "169.254.169.254:80", nil))
    49  			require.Error(t, guardedDialControl(deny, "tcp", "169.254.169.254:80", nil))
    50  		})
    51  
    52  		t.Run("private許可フラグなしならloopback/privateを拒否する", func(t *testing.T) {
    53  			t.Parallel()
    54  			require.Error(t, guardedDialControl(deny, "tcp", "127.0.0.1:8080", nil))
    55  			require.Error(t, guardedDialControl(deny, "tcp", "10.0.0.5:80", nil))
    56  			require.Error(t, guardedDialControl(deny, "tcp", "192.168.1.10:80", nil))
    57  		})
    58  
    59  		t.Run("private許可フラグなしならCGNAT(100.64.0.0/10)も拒否する", func(t *testing.T) {
    60  			t.Parallel()
    61  			// Go の IsPrivate は CGNAT を含まないため、明示判定で塞ぐ。境界も検証。
    62  			require.Error(t, guardedDialControl(deny, "tcp", "100.64.0.1:80", nil))
    63  			require.Error(t, guardedDialControl(deny, "tcp", "100.127.255.254:80", nil))
    64  		})
    65  
    66  		t.Run("予約/将来利用帯はprivate許可フラグありでも拒否する", func(t *testing.T) {
    67  			t.Parallel()
    68  			// RFC 5737 TEST-NET-1（192.0.2.0/24）— ドキュメント/テスト用、正当な宛先にならない。
    69  			require.Error(t, guardedDialControl(allow, "tcp", "192.0.2.1:80", nil))
    70  			// RFC 5737 TEST-NET-2（198.51.100.0/24）— ドキュメント/テスト用、正当な宛先にならない。
    71  			require.Error(t, guardedDialControl(allow, "tcp", "198.51.100.5:80", nil))
    72  			// RFC 5737 TEST-NET-3（203.0.113.0/24）— ドキュメント/テスト用、正当な宛先にならない。
    73  			require.Error(t, guardedDialControl(allow, "tcp", "203.0.113.5:80", nil))
    74  			// RFC 1112/6890 将来予約（240.0.0.0/4）— 現実の宛先として到達不能。
    75  			require.Error(t, guardedDialControl(allow, "tcp", "240.0.0.1:80", nil))
    76  			// RFC 2544 ベンチマーク測定用（198.18.0.0/15）— 本番トラフィックの宛先にならない。
    77  			require.Error(t, guardedDialControl(allow, "tcp", "198.18.0.1:80", nil))
    78  			// RFC 6890 IETF プロトコル割当（192.0.0.0/24）— 一般宛て通信には使われない。
    79  			require.Error(t, guardedDialControl(allow, "tcp", "192.0.0.1:80", nil))
    80  			// RFC 3849 IPv6 ドキュメント用（2001:db8::/32）— テスト/文書専用で実到達不能。
    81  			require.Error(t, guardedDialControl(allow, "tcp", "[2001:db8::1]:443", nil))
    82  		})
    83  	})

    95  func TestConditionalPropagator(t *testing.T) {
    96  	t.Parallel()
    97  
    98  	prop := conditionalPropagator{inner: propagation.TraceContext{}}
    99  
   100  	t.Run("正常系", func(t *testing.T) {
   101  		t.Parallel()
   102  
   103  		t.Run("フラグ未設定なら通常どおり伝搬する", func(t *testing.T) {
   104  			t.Parallel()
   105  
   106  			carrier := propagation.MapCarrier{}
   107  			prop.Inject(newSampledContext(), carrier)
   108  			assert.NotEmpty(t, carrier.Get("traceparent"))
   109  		})
   110  
   111  		t.Run("伝搬有効フラグでは伝搬する", func(t *testing.T) {
   112  			t.Parallel()
   113  
   114  			ctx := ContextWithTracePropagation(newSampledContext(), true)
   115  			carrier := propagation.MapCarrier{}
   116  			prop.Inject(ctx, carrier)
   117  			assert.NotEmpty(t, carrier.Get("traceparent"))
   118  		})
   119  
   120  		t.Run("Extract は内側 propagator へ委譲し trace context を復元する", func(t *testing.T) {
   121  			t.Parallel()
   122  
   123  			// 内側と同じ propagator で carrier を作り、Extract が委譲されていることを確認する。
   124  			carrier := propagation.MapCarrier{}
   125  			propagation.TraceContext{}.Inject(newSampledContext(), carrier)
   126  
   127  			got := prop.Extract(context.Background(), carrier)
   128  
   129  			sc := trace.SpanContextFromContext(got)
   130  			assert.True(t, sc.IsValid())
   131  			want := trace.SpanContextFromContext(newSampledContext())
   132  			assert.Equal(t, want.TraceID().String(), sc.TraceID().String())
   133  			assert.Equal(t, want.SpanID().String(), sc.SpanID().String())
   134  		})
   135  
   136  		t.Run("Fields は内側 propagator のキー集合を委譲する", func(t *testing.T) {
   137  			t.Parallel()
   138  
   139  			assert.Equal(t, propagation.TraceContext{}.Fields(), prop.Fields())
   140  		})
   141  	})
   142  
   143  	t.Run("異常系", func(t *testing.T) {
   144  		t.Parallel()
   145  
   146  		t.Run("伝搬無効フラグでは外部へ注入しない", func(t *testing.T) {
   147  			t.Parallel()
   148  
   149  			ctx := ContextWithTracePropagation(newSampledContext(), false)
   150  			carrier := propagation.MapCarrier{}
   151  			prop.Inject(ctx, carrier)
   152  			assert.Empty(t, carrier.Get("traceparent"))
   153  		})
