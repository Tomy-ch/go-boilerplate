    13  func Test_guardedDialControl(t *testing.T) {
    14  	t.Parallel()
    15  
    16  	allow := ContextWithAllowPrivateNetwork(context.Background(), true)
    17  	deny := ContextWithAllowPrivateNetwork(context.Background(), false)
    18  
    19  	t.Run("正常系", func(t *testing.T) {
    20  		t.Parallel()
    21  
    22  		t.Run("グローバルアドレスは許可する", func(t *testing.T) {
    23  			t.Parallel()
    24  			require.NoError(t, guardedDialControl(deny, "tcp", "93.184.216.34:443", nil))
    25  		})
    26  
    27  		t.Run("private許可フラグありならloopback/privateを許可する", func(t *testing.T) {
    28  			t.Parallel()
    29  			require.NoError(t, guardedDialControl(allow, "tcp", "127.0.0.1:8080", nil))
    30  			require.NoError(t, guardedDialControl(allow, "tcp", "10.0.0.5:80", nil))
    31  			require.NoError(t, guardedDialControl(allow, "tcp", "192.168.1.10:80", nil))
    32  			// CGNAT(RFC 6598) も private 扱いで、フラグありなら許可する。
    33  			require.NoError(t, guardedDialControl(allow, "tcp", "100.64.0.1:80", nil))
    34  		})
    35  
    36  		t.Run("CGNAT帯の外(100.128.x)は許可する", func(t *testing.T) {
    37  			t.Parallel()
    38  			// 100.128.0.0 は /10 の外＝グローバル扱いで通ること（過剰ブロック防止）。
    39  			require.NoError(t, guardedDialControl(deny, "tcp", "100.128.0.1:80", nil))
    40  		})
    41  	})
    42  
    43  	t.Run("異常系", func(t *testing.T) {
    44  		t.Parallel()
    45  
    46  		t.Run("ポート無しアドレスはSplitHostPortエラーを返す", func(t *testing.T) {
    47  			t.Parallel()
    48  			// net.SplitHostPort が失敗するアドレス（ポート区切り無し）はそのままエラーを返す。
    49  			require.Error(t, guardedDialControl(deny, "tcp", "noport", nil))
    50  		})
    51  
    52  		t.Run("IPリテラルでないホスト名はfail-closeで拒否する", func(t *testing.T) {
    53  			t.Parallel()
    54  			// パース不能なアドレスは許可せず拒否する（fail-close）。
    55  			require.Error(t, guardedDialControl(allow, "tcp", "example.com:80", nil))
    56  			require.Error(t, guardedDialControl(deny, "tcp", "example.com:80", nil))
    57  		})
    58  
    59  		t.Run("zone付きIPv6リンクローカルはフラグに関わらず拒否する", func(t *testing.T) {
    60  			t.Parallel()
    61  			// fe80::1%eth0 は zone 付きでも link-local と判定してブロックする（判定回避の穴を塞ぐ）。
    62  			require.Error(t, guardedDialControl(allow, "tcp", "[fe80::1%eth0]:80", nil))
    63  			require.Error(t, guardedDialControl(deny, "tcp", "[fe80::1%eth0]:80", nil))
    64  		})
    65  
    66  		t.Run("IPv4-mappedのIPv6リテラルでも予約帯/CGNATを拒否する", func(t *testing.T) {
    67  			t.Parallel()
    68  			// netip.Prefix.Contains は family 不一致（IPv4 prefix vs 4-in-6）で false を返すため、
    69  			// Unmap による正規化が退行すると ::ffff: 形式で判定を迂回できてしまう。
    70  			require.Error(t, guardedDialControl(allow, "tcp", "[::ffff:240.0.0.1]:80", nil))
    71  			require.Error(t, guardedDialControl(deny, "tcp", "[::ffff:100.64.0.1]:80", nil))
    72  		})
    73  
    74  		t.Run("リンクローカル(メタデータ)はフラグに関わらず拒否する", func(t *testing.T) {
    75  			t.Parallel()
    76  			require.Error(t, guardedDialControl(allow, "tcp", "169.254.169.254:80", nil))
    77  			require.Error(t, guardedDialControl(deny, "tcp", "169.254.169.254:80", nil))
    78  		})
    79  
    80  		t.Run("private許可フラグなしならloopback/privateを拒否する", func(t *testing.T) {
    81  			t.Parallel()
    82  			require.Error(t, guardedDialControl(deny, "tcp", "127.0.0.1:8080", nil))
    83  			require.Error(t, guardedDialControl(deny, "tcp", "10.0.0.5:80", nil))
    84  			require.Error(t, guardedDialControl(deny, "tcp", "192.168.1.10:80", nil))
    85  		})
    86  
    87  		t.Run("private許可フラグなしならCGNAT(100.64.0.0/10)も拒否する", func(t *testing.T) {
    88  			t.Parallel()
    89  			// Go の IsPrivate は CGNAT を含まないため、明示判定で塞ぐ。境界も検証。
    90  			require.Error(t, guardedDialControl(deny, "tcp", "100.64.0.1:80", nil))
    91  			require.Error(t, guardedDialControl(deny, "tcp", "100.127.255.254:80", nil))
    92  		})
    93  
    94  		t.Run("予約/将来利用帯はprivate許可フラグありでも拒否する", func(t *testing.T) {
    95  			t.Parallel()
    96  			// RFC 5737 TEST-NET-1（192.0.2.0/24）— ドキュメント/テスト用、正当な宛先にならない。
    97  			require.Error(t, guardedDialControl(allow, "tcp", "192.0.2.1:80", nil))
    98  			// RFC 5737 TEST-NET-2（198.51.100.0/24）— ドキュメント/テスト用、正当な宛先にならない。
    99  			require.Error(t, guardedDialControl(allow, "tcp", "198.51.100.5:80", nil))
   100  			// RFC 5737 TEST-NET-3（203.0.113.0/24）— ドキュメント/テスト用、正当な宛先にならない。
   101  			require.Error(t, guardedDialControl(allow, "tcp", "203.0.113.5:80", nil))
   102  			// RFC 1112/6890 将来予約（240.0.0.0/4）— 現実の宛先として到達不能。
   103  			require.Error(t, guardedDialControl(allow, "tcp", "240.0.0.1:80", nil))
   104  			// RFC 2544 ベンチマーク測定用（198.18.0.0/15）— 本番トラフィックの宛先にならない。
   105  			require.Error(t, guardedDialControl(allow, "tcp", "198.18.0.1:80", nil))
   106  			// RFC 1122/6890「このネットワーク」（0.0.0.0/8）— 0.0.0.0 ちょうど以外も拒否する。
   107  			require.Error(t, guardedDialControl(allow, "tcp", "0.0.0.1:80", nil))
   108  			// 0.0.0.0 ちょうどは IsUnspecified 経路で拒否される（/8 追加前からの既存挙動）。
   109  			require.Error(t, guardedDialControl(allow, "tcp", "0.0.0.0:80", nil))
   110  			// RFC 6890 IETF プロトコル割当（192.0.0.0/24）— 一般宛て通信には使われない。
   111  			require.Error(t, guardedDialControl(allow, "tcp", "192.0.0.1:80", nil))
   112  			// RFC 3849 IPv6 ドキュメント用（2001:db8::/32）— テスト/文書専用で実到達不能。
   113  			require.Error(t, guardedDialControl(allow, "tcp", "[2001:db8::1]:443", nil))
   114  		})
   115  	})

   127  func Test_conditionalPropagator(t *testing.T) {
   128  	t.Parallel()
   129  
   130  	prop := conditionalPropagator{inner: propagation.TraceContext{}}
   131  
   132  	t.Run("正常系", func(t *testing.T) {
   133  		t.Parallel()
   134  
   135  		t.Run("フラグ未設定なら通常どおり伝搬する", func(t *testing.T) {
   136  			t.Parallel()
   137  
   138  			carrier := propagation.MapCarrier{}
   139  			prop.Inject(newSampledContext(), carrier)
   140  			assert.NotEmpty(t, carrier.Get("traceparent"))
   141  		})
   142  
   143  		t.Run("伝搬有効フラグでは伝搬する", func(t *testing.T) {
   144  			t.Parallel()
   145  
   146  			ctx := ContextWithTracePropagation(newSampledContext(), true)
   147  			carrier := propagation.MapCarrier{}
   148  			prop.Inject(ctx, carrier)
   149  			assert.NotEmpty(t, carrier.Get("traceparent"))
   150  		})
   151  
   152  		t.Run("Extract は内側 propagator へ委譲し trace context を復元する", func(t *testing.T) {
   153  			t.Parallel()
   154  
   155  			// 内側と同じ propagator で carrier を作り、Extract が委譲されていることを確認する。
   156  			carrier := propagation.MapCarrier{}
   157  			propagation.TraceContext{}.Inject(newSampledContext(), carrier)
   158  
   159  			got := prop.Extract(context.Background(), carrier)
   160  
   161  			sc := trace.SpanContextFromContext(got)
   162  			assert.True(t, sc.IsValid())
   163  			want := trace.SpanContextFromContext(newSampledContext())
   164  			assert.Equal(t, want.TraceID().String(), sc.TraceID().String())
   165  			assert.Equal(t, want.SpanID().String(), sc.SpanID().String())
   166  		})
   167  
   168  		t.Run("Fields は内側 propagator のキー集合を委譲する", func(t *testing.T) {
   169  			t.Parallel()
   170  
   171  			assert.Equal(t, propagation.TraceContext{}.Fields(), prop.Fields())
   172  		})
   173  	})
   174  
   175  	t.Run("異常系", func(t *testing.T) {
   176  		t.Parallel()
   177  
   178  		t.Run("伝搬無効フラグでは外部へ注入しない", func(t *testing.T) {
   179  			t.Parallel()
   180  
   181  			ctx := ContextWithTracePropagation(newSampledContext(), false)
   182  			carrier := propagation.MapCarrier{}
   183  			prop.Inject(ctx, carrier)
   184  			assert.Empty(t, carrier.Get("traceparent"))
   185  		})
   186  	})
