    56  
    57  	t.Run("異常系", func(t *testing.T) {
    58  		t.Parallel()
    59  
    60  		t.Run("TracerProviderのShutdownが失敗した場合は結合エラーとして伝播する", func(t *testing.T) {
    61  			t.Parallel()
    62  
    63  			wantErr := errors.New("span exporter shutdown failed")
    64  			// バッチャ経由で失敗する exporter を仕込み、tp.Shutdown がエラーを返すようにする。
    65  			tp := sdktrace.NewTracerProvider(sdktrace.WithBatcher(errSpanExporter{err: wantErr}))
    66  			mp := sdkmetric.NewMeterProvider()
    67  			lp := sdklog.NewLoggerProvider()
    68  
    69  			// errors.Join は全引数（tp/mp/lp の Shutdown）を評価してから結合するため、
    70  			// tp が失敗しても mp / lp の Shutdown は必ず呼ばれる（Go の引数評価保証）。
    71  			err := NewProviderShutdowner(tp, mp, lp).Shutdown(context.Background())
    72  
    73  			require.ErrorContains(t, err, wantErr.Error())
    74  		})
    75  	})
