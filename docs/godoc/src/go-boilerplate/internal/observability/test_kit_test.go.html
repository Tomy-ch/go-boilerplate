   143  	t.Parallel()
   144  
   145  	t.Run("正常系", func(t *testing.T) {
   146  		t.Parallel()
   147  
   148  		t.Run("SSRFガードを無効化し loopback 宛ての実接続を許可する", func(t *testing.T) {
   149  			t.Parallel()
   150  
   151  			srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
   152  				w.WriteHeader(http.StatusNoContent)
   153  			}))
   154  			defer srv.Close()
   155  
   156  			transport := NewNoopHTTPClientTransport(t)
   157  			require.NotNil(t, transport)
   158  
   159  			// permissive な dial control により loopback(httptest) 宛ての実 dial が拒否されず接続できる。
   160  			req, err := http.NewRequestWithContext(context.Background(), http.MethodGet, srv.URL, nil)
   161  			require.NoError(t, err)
   162  			resp, err := (&http.Client{Transport: transport.RoundTripper()}).Do(req)
   163  			require.NoError(t, err)
   164  			defer func() { _ = resp.Body.Close() }()
   165  
   166  			assert.Equal(t, http.StatusNoContent, resp.StatusCode)
   167  		})
   168  	})
