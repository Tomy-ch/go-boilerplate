   126  	t.Parallel()
   127  
   128  	t.Run("正常系", func(t *testing.T) {
   129  		t.Parallel()
   130  
   131  		t.Run("SSRFガードを無効化し loopback 宛ての実接続を許可する", func(t *testing.T) {
   132  			t.Parallel()
   133  
   134  			srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
   135  				w.WriteHeader(http.StatusNoContent)
   136  			}))
   137  			defer srv.Close()
   138  
   139  			transport := NewNoopHTTPClientTransport(t)
   140  			require.NotNil(t, transport)
   141  
   142  			// permissive な dial control により loopback(httptest) 宛ての実 dial が拒否されず接続できる。
   143  			req, err := http.NewRequestWithContext(context.Background(), http.MethodGet, srv.URL, nil)
   144  			require.NoError(t, err)
   145  			resp, err := (&http.Client{Transport: transport.RoundTripper()}).Do(req)
   146  			require.NoError(t, err)
   147  			defer func() { _ = resp.Body.Close() }()
   148  
   149  			assert.Equal(t, http.StatusNoContent, resp.StatusCode)
   150  		})
   151  	})
