    65  //
    66  // span の開始・終了時に Infoレベルでログ出力を行う。
    67  //
    68  // 呼び出し元は、関数 fn の実行結果とエラーを受け取ることができる。
    69  //
    70  // 典型的な呼び出し方:
    71  //
    72  //	ctx, result, err := layerTracer.RunWithSpan(ctx, "usecase", "user", "FullName", func(ctx context.Context) (T, error) {
    73  //	    // 処理内容

    75  func RunWithSpan[T any](
    76  	parentCtx context.Context,
    77  	lt LayerTracer,
    78  	layer layerName,
    79  	pkg, funcName string,
    80  	fn func(ctx context.Context) (T, error),
    81  ) (context.Context, T, error) {
    82  	spanName := BuildSpanName(string(layer), pkg, funcName)
    83  
    84  	tc, childCtx, end := StartSpanWithParent(parentCtx, lt, spanName)
    85  	start := time.Now()
    86  
    87  	obsIn := logging.ObservabilityFieldsInput{
    88  		SpanName:     spanName,
    89  		Layer:        string(layer),
    90  		PkgName:      pkg,
    91  		FuncName:     funcName,
    92  		TraceID:      tc.TraceID(),
    93  		ParentSpanID: tc.ParentSpanID(),
    94  		SpanID:       tc.SpanID(),
    95  	}
    96  	obsIn.EventType = SpanEventStart
    97  
    98  	lt.logSpanEvent(callSkip, obsIn)
    99  
   100  	defer func() {
   101  		event := SpanEventEnd
   102  		obsIn.EventType = event
   103  		obsIn.Latency = time.Since(start)
   104  		lt.logSpanEvent(callSkip+1, obsIn)
   105  		end()
   106  	}()
   107  
   108  	v, err := fn(childCtx)
   109  	if err != nil {
   110  		var zero T
   111  		return childCtx, zero, err
   112  	}
   113  
   114  	return childCtx, v, nil
   115  }

   133  func (lt LayerTracer) startSpan(

   136  	opts ...trace.SpanStartOption,
   137  ) (context.Context, func()) {

   142  
   143  	obsIn := logging.ObservabilityFieldsInput{
   144  		SpanName:     spanName,

   146  		PkgName:      lt.pkgName,
   147  		FuncName:     lt.funcName,
   148  		EventAt:      time.Now(),
