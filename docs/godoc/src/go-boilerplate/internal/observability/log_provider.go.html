    19  func NewLoggerProvider(
    20  	obsCfg *config.ObservabilityConfig, epCfg *config.EndpointConfig, res *resource.Resource,
    21  ) (*sdklog.LoggerProvider, error) {
    22  	opts := []sdklog.LoggerProviderOption{sdklog.WithResource(res)}
    23  
    24  	if obsCfg.LogsEnabled() {
    25  		exporter, err := newLogExporter(context.Background(), obsCfg, epCfg)

    31
