    43  // ensureOTLPPath は OTLP HTTP のエンドポイント URL に signal パスが無ければ defaultPath を補う。
    44  // otlp*http の WithEndpointURL は path 無し（"" / "/"）の URL に既定 path を補わずルートへ送って 404 に
    45  // なる版があるため、空・ルートのときは defaultPath を明示する。
    46  func ensureOTLPPath(rawURL, defaultPath string) string {
    47  	u, err := url.Parse(rawURL)
    48  	if err != nil || (u.Path != "" && u.Path != "/") {
    49  		return rawURL
    50  	}
    51  	u.Path = defaultPath
    52  	return u.String()
    53  }

   101  // NewMeterProvider は MeterProvider をグローバル登録し、構築した MeterProvider を返す。
   102  // MetricsEnabled が真の場合は Go ランタイムメトリクスの収集 goroutine も開始する。
   103  func NewMeterProvider(obsCfg *config.ObservabilityConfig, res *resource.Resource) (*sdkmetric.MeterProvider, error) {
   104  	opts := []sdkmetric.Option{sdkmetric.WithResource(res)}
   105  
   106  	if obsCfg.MetricsEnabled() {
   107  		reader, err := newMetricReader(context.Background(), obsCfg)
   108  		if err != nil {
   109  			return nil, xerrors.Wrap(err, "failed to build metric reader")
   110  		}
   111  		opts = append(opts, sdkmetric.WithReader(reader))
   112  	}
   113  
   114  	mp := sdkmetric.NewMeterProvider(opts...)
   115  
   116  	if obsCfg.MetricsEnabled() {
   117  		if err := runtime.Start(runtime.WithMeterProvider(mp)); err != nil {
   118  			// mp は既に PeriodicReader（送出 goroutine）を抱えているため、失敗時は Shutdown して回収する。
   119  			_ = mp.Shutdown(context.Background())
   120  			return nil, xerrors.Wrap(err, "failed to start runtime metrics")
   121  		}
   122  	}
   123  
   124  	otel.SetMeterProvider(mp)
   125  
   126  	return mp, nil
   127  }

   129  // newSpanExporter は OBS_OTLP_PROTOCOL / OBS_OTLP_ENDPOINT から OTLP SpanExporter を構築する。
   130  // endpoint 未指定時は OTLP のデフォルト（localhost:4318 / :4317）に従う。
   131  func newSpanExporter(ctx context.Context, obsCfg *config.ObservabilityConfig) (sdktrace.SpanExporter, error) {
   132  	switch obsCfg.OTLPProtocol() {
   133  	case protocolGRPC:
   134  		var opts []otlptracegrpc.Option
   135  		if ep := obsCfg.OTLPEndpoint(); ep != "" {
   136  			opts = append(opts, otlptracegrpc.WithEndpointURL(ep))
   137  		}
   138  		return otlptracegrpc.New(ctx, opts...)
   139  	case protocolHTTP, "":
   140  		var opts []otlptracehttp.Option
   141  		if ep := obsCfg.OTLPEndpoint(); ep != "" {
   142  			opts = append(opts, otlptracehttp.WithEndpointURL(ensureOTLPPath(ep, otlpTracesPath)))
   143  		}
   144  		return otlptracehttp.New(ctx, opts...)
   145  	default:
   146  		return nil, errInvalidOTLPProtocol
   147  	}
   148  }
