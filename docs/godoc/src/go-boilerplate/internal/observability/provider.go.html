    46  // SpanExporter は標準 OTEL_* env から構築し、送出先未指定時は no-op となる。
    47  func TracerProvider(reg lifecycle.Registrar, res *resource.Resource) (trace.TracerProvider, error) {
    48  	exporter, err := autoexport.NewSpanExporter(context.Background(), autoexport.WithFallbackSpanExporter(newNoopSpanExporter))
    49  	if err != nil {
    50  		return nil, xerrors.Wrap(err, "failed to build span exporter")
    51  	}
    52  
    53  	opts := []sdktrace.TracerProviderOption{sdktrace.WithResource(res)}
    54  	if _, isNoop := exporter.(noopSpanExporter); !isNoop {
    55  		opts = append(opts, sdktrace.WithBatcher(exporter))
    56  	}
    57  
    58  	tp := sdktrace.NewTracerProvider(opts...)
    59  
    60  	otel.SetTracerProvider(tp)
    61  	otel.SetTextMapPropagator(propagation.NewCompositeTextMapPropagator(
    62  		propagation.TraceContext{},
    63  		propagation.Baggage{},
    64  	))
    65  
    66  	reg.RegisterStop(tp.Shutdown)
    67  
    68  	return tp, nil
    69  }

    83  
    84  	if _, isNoop := reader.(*sdkmetric.ManualReader); !isNoop {
    85  		if err := runtime.Start(runtime.WithMeterProvider(mp)); err != nil {
    86  			return nil, xerrors.Wrap(err, "failed to start runtime metrics")
    87  		}
    88  	}

    92  
    93  	return mp, nil
