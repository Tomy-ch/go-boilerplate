    11  const httpClientMeterName = "go-boilerplate/httpclient"

    13  // HTTPClientMetrics は、外部 HTTP client の計装一式です
    14  // （RED＝リクエスト数・エラー数・レイテンシに、retry 回数・処理中リクエスト数・breaker 状態 gauge を加えたもの）。
    15  //
    16  // downstream には registry で解決される固定の Downstream 名など低カーディナリティな識別子のみを
    17  // 渡します。生の URL やリクエスト固有の値を渡すとメトリクスのカーディナリティが際限なく増えます。
    18  type HTTPClientMetrics struct {
    19  	requests     metric.Int64Counter
    20  	errors       metric.Int64Counter
    21  	retries      metric.Int64Counter
    22  	latencyMs    metric.Float64Histogram
    23  	inFlight     metric.Int64UpDownCounter
    24  	breakerState metric.Int64Gauge
    25  }
    26  
    27  // NewHTTPClientMetrics は、注入された MeterProvider から substrate の計装一式を生成します。
    28  // いずれかの計装生成失敗で error を返します。
    29  func NewHTTPClientMetrics(mp metric.MeterProvider) (*HTTPClientMetrics, error) {
    30  	b := &meterBuilder{m: mp.Meter(httpClientMeterName)}
    31  	hm := &HTTPClientMetrics{
    32  		requests:     b.counter("httpclient.requests", "完了した外部 HTTP リクエスト数(ステータスクラス別)"),
    33  		errors:       b.counter("httpclient.errors", "失敗した外部 HTTP リクエスト数(理由別)"),
    34  		retries:      b.counter("httpclient.retries", "リトライした回数"),
    35  		latencyMs:    b.histogram("httpclient.request_latency_ms", "リクエスト全体の所要時間(ミリ秒)"),
    36  		inFlight:     b.upDownCounter("httpclient.in_flight", "処理中(送信済み・未完了)のリクエスト数"),
    37  		breakerState: b.gauge("httpclient.breaker_state", "circuit breaker の状態(0:closed 1:half-open 2:open)"),
    38  	}
    39  	if b.err != nil {
    40  		return nil, b.err
    41  	}
    42  	return hm, nil
    43  }
