   103  // LabelValues は、metricName の counter に付いた labelKey の値を返します（未計上なら空）。
   104  // 返るのは counter のデータポイント単位、すなわち計上された label 組であり、記録回数ではありません
   105  // （同一 label 組へ何回記録しても cumulative counter の 1 データポイントに集約されます）。
   106  // metricName が int64 counter でない場合は、テストキットの誤用としてテストを失敗させます。
   107  func (o *ObservedHTTPClientMetrics) LabelValues(t *testing.T, metricName, labelKey string) []string {
   108  	t.Helper()
   109  
   110  	var rm metricdata.ResourceMetrics
   111  	if err := o.reader.Collect(context.Background(), &rm); err != nil {
   112  		t.Fatalf("failed to collect http client metrics: %v", err)
   113  	}
   114  
   115  	values, ok := counterLabelValues(rm, metricName, labelKey)
   116  	if !ok {
   117  		t.Fatalf("metric %s is not an int64 counter", metricName)
   118  	}
   119  	return values
   120  }
   121  
   122  // counterLabelValues は、rm から metricName の int64 counter を探し、各データポイントの labelKey の値を返します。
   123  // metricName の指標が int64 counter でない場合は ok=false を返します（未計上の場合は空・ok=true）。
   124  func counterLabelValues(rm metricdata.ResourceMetrics, metricName, labelKey string) ([]string, bool) {
   125  	var values []string
   126  	for _, sm := range rm.ScopeMetrics {
   127  		for _, m := range sm.Metrics {
   128  			if m.Name != metricName {
   129  				continue
   130  			}
   131  			sum, ok := m.Data.(metricdata.Sum[int64])
   132  			if !ok {
   133  				return nil, false
   134  			}
   135  			for _, dp := range sum.DataPoints {
   136  				if v, found := dp.Attributes.Value(attribute.Key(labelKey)); found {
   137  					values = append(values, v.String())
   138  				}
   139  			}
   140  		}
   141  	}
   142  	return values, true
   143  }
