    91  	// otel.SetTracerProvider / SetTextMapPropagator をグローバルに触るため Parallel 不可。
    92  
    93  	t.Run("正常系", func(t *testing.T) {
    94  		t.Run("Registrarにシャットダウンを登録し、伝播器を設定してグローバルなTracerProviderを返す", func(t *testing.T) {
    95  			ctrl := gomock.NewController(t)
    96  
    97  			mockReg := mock_lifecycle.NewMockRegistrar(ctrl)
    98  			var shutdownFunc func(context.Context) error
    99  			dummy := func(context.Context) error { return nil }
   100  			mockReg.EXPECT().RegisterStop(gomock.AssignableToTypeOf(dummy)).Do(func(args ...any) {
   101  				shutdownFunc = args[0].(func(context.Context) error)
   102  			}).Times(1)
   103  
   104  			tp, err := TracerProvider(mockReg, newTestResource(t))
   105  
   106  			require.NoError(t, err)
   107  			require.NotNil(t, tp)
   108  			_, ok := tp.(*sdktrace.TracerProvider)
   109  			assert.True(t, ok)
   110  			assert.Equal(t, tp, otel.GetTracerProvider())
   111  
   112  			// W3C TraceContext + Baggage の伝播器が登録されていること。
   113  			fields := otel.GetTextMapPropagator().Fields()
   114  			assert.Contains(t, fields, "traceparent")
   115  			assert.Contains(t, fields, "baggage")
   116  
   117  			require.NotNil(t, shutdownFunc)
   118  			require.NoError(t, shutdownFunc(context.Background()))
   119  		})
   120  	})
   121  
   122  	t.Run("異常系", func(t *testing.T) {
   123  		t.Run("不正なOTEL_TRACES_EXPORTERが指定された場合はエラーを返す", func(t *testing.T) {
