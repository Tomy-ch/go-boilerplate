    40  	t.Helper()
    41  	ctx, cancel := context.WithTimeout(context.Background(), 200*time.Millisecond)
    42  	defer cancel()
    43  	_ = mp.Shutdown(ctx)

   108  	t.Run("正常系", func(t *testing.T) {
   109  		t.Run("trace有効(http)で伝播器を設定しグローバルなTracerProviderを構築する", func(t *testing.T) {
   110  			prevTP, prevProp := otel.GetTracerProvider(), otel.GetTextMapPropagator()
   111  			t.Cleanup(func() {
   112  				otel.SetTracerProvider(prevTP)
   113  				otel.SetTextMapPropagator(prevProp)
   114  			})
   115  
   116  			tp, err := NewTracerProvider(newTestObsCfg(t), newTestResource(t))
   117  
   118  			require.NoError(t, err)
   119  			require.NotNil(t, tp)
   120  			assert.Same(t, tp, otel.GetTracerProvider())
   121  
   122  			// W3C TraceContext + Baggage の伝播器が登録されていること。
   123  			fields := otel.GetTextMapPropagator().Fields()
   124  			assert.Contains(t, fields, "traceparent")
   125  			assert.Contains(t, fields, "baggage")
   126  
   127  			require.NoError(t, tp.Shutdown(context.Background()))
   128  		})
   129

   139  			require.NoError(t, err)
