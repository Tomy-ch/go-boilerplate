    87  func Test_NewTracerProvider(t *testing.T) {
    88  	// otel.SetTracerProvider / SetTextMapPropagator をグローバルに触るため Parallel 不可。
    89  
    90  	t.Run("正常系", func(t *testing.T) {
    91  		t.Run("伝播器を設定してグローバルなTracerProviderを構築し、Shutdown可能な具象を返す", func(t *testing.T) {
    92  			prevTP, prevProp := otel.GetTracerProvider(), otel.GetTextMapPropagator()
    93  			t.Cleanup(func() {
    94  				otel.SetTracerProvider(prevTP)
    95  				otel.SetTextMapPropagator(prevProp)
    96  			})
    97  
    98  			tp, err := NewTracerProvider(newTestResource(t))
    99  
   100  			require.NoError(t, err)
   101  			require.NotNil(t, tp)
   102  			assert.Same(t, tp, otel.GetTracerProvider())
   103  
   104  			// W3C TraceContext + Baggage の伝播器が登録されていること。
   105  			fields := otel.GetTextMapPropagator().Fields()
   106  			assert.Contains(t, fields, "traceparent")
   107  			assert.Contains(t, fields, "baggage")
   108  
   109  			// ライフサイクル登録は di 層に移譲したため、ここでは Shutdown が呼べることのみ確認する。
   110  			require.NoError(t, tp.Shutdown(context.Background()))
   111  		})
   112  
   113  		t.Run("OTEL_TRACES_EXPORTERがnoneの場合もno-opとして構築しエラーを返さない", func(t *testing.T) {
   114  			t.Setenv("OTEL_TRACES_EXPORTER", "none")
   115  			prevTP := otel.GetTracerProvider()
   116  			t.Cleanup(func() { otel.SetTracerProvider(prevTP) })
   117  
   118  			tp, err := NewTracerProvider(newTestResource(t))
   119  
   120  			require.NoError(t, err)
   121  			require.NotNil(t, tp)
   122  			require.NoError(t, tp.Shutdown(context.Background()))
   123  		})
   124  	})
   125  
   126  	t.Run("異常系", func(t *testing.T) {
   127  		t.Run("不正なOTEL_TRACES_EXPORTERが指定された場合はエラーを返す", func(t *testing.T) {
   128  			t.Setenv("OTEL_TRACES_EXPORTER", "invalid-exporter")
