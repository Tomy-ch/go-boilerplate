    40  	t.Helper()

    42  	defer cancel()
    43  	_ = mp.Shutdown(ctx)

    89  	t.Run("正常系", func(t *testing.T) {
    90  		t.Run("trace有効(http)で伝播器を設定しグローバルなTracerProviderを構築する", func(t *testing.T) {
    91  			prevTP, prevProp := otel.GetTracerProvider(), otel.GetTextMapPropagator()
    92  			t.Cleanup(func() {
    93  				otel.SetTracerProvider(prevTP)
    94  				otel.SetTextMapPropagator(prevProp)
    95  			})
    96  
    97  			tp, err := NewTracerProvider(newTestObsCfg(t), newTestResource(t))
    98  
    99  			require.NoError(t, err)
   100  			require.NotNil(t, tp)
   101  			assert.Same(t, tp, otel.GetTracerProvider())
   102  
   103  			// W3C TraceContext + Baggage の伝播器が登録されていること。
   104  			fields := otel.GetTextMapPropagator().Fields()
   105  			assert.Contains(t, fields, "traceparent")
   106  			assert.Contains(t, fields, "baggage")
   107  
   108  			require.NoError(t, tp.Shutdown(context.Background()))
   109  		})
   110  
   111  		t.Run("trace有効(grpc)でも構築できる", func(t *testing.T) {
   112  			prevTP := otel.GetTracerProvider()
   113  			t.Cleanup(func() { otel.SetTracerProvider(prevTP) })
   114  
   115  			obsCfg := newTestObsCfg(t)
   116  			obsCfg.SetObservabilityOTLPProtocol(t, protocolGRPC)
   117  
   118  			tp, err := NewTracerProvider(obsCfg, newTestResource(t))
   119  
   120  			require.NoError(t, err)
   121  			require.NotNil(t, tp)
   122  			require.NoError(t, tp.Shutdown(context.Background()))
   123  		})

   125  		t.Run("trace無効ならBatcherを付けずに構築する", func(t *testing.T) {

   128  
   129  			obsCfg := newTestObsCfg(t)
   130  			obsCfg.SetObservabilityTracesExporter(t, "")
   131  
   132  			tp, err := NewTracerProvider(obsCfg, newTestResource(t))
