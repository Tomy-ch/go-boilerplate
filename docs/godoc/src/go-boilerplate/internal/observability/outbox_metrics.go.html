    12  // OutboxMetrics は、outbox relay 所有の outbox 固有計装です。
    13  // publish 自体の count/latency/error は httpclient(downstream=outbox) の計装で賄うため、
    14  // ここでは outbox 固有の SLI（lag）と dead 数のみを持ちます。
    15  type OutboxMetrics struct {
    16  	lagSeconds metric.Int64Gauge
    17  	dead       metric.Int64Counter
    18  }
    19  
    20  // NewOutboxMetrics は、注入された MeterProvider から outbox relay の計装一式を生成します。
    21  func NewOutboxMetrics(mp metric.MeterProvider) (*OutboxMetrics, error) {
    22  	b := &meterBuilder{m: mp.Meter(outboxMeterName)}
    23  	om := &OutboxMetrics{
    24  		lagSeconds: b.gauge("outbox.lag_seconds", "最古 pending 行の経過秒数（SLI=outbox lag）"),
    25  		dead:       b.counter("outbox.dead", "max attempts 到達で dead 化したメッセージ数"),
    26  	}
    27  	if b.err != nil {
    28  		return nil, b.err
    29  	}
    30  	return om, nil
    31  }
