    33  // NewWorkerMetrics は、注入された MeterProvider から worker engine の計装一式を生成します。
    34  // いずれかの計装生成失敗で error を返します。
    35  func NewWorkerMetrics(mp metric.MeterProvider) (*WorkerMetrics, error) {
    36  	b := &meterBuilder{m: mp.Meter(workerMeterName)}
    37  	wm := &WorkerMetrics{
    38  		received:     b.counter("worker.received", "受信したメッセージ数"),
    39  		processed:    b.counter("worker.processed", "正常処理したメッセージ数"),
    40  		failed:       b.counter("worker.failed", "処理に失敗したメッセージ数"),
    41  		retried:      b.counter("worker.retried", "Nack して再配送へ戻したメッセージ数"),
    42  		dlq:          b.counter("worker.dlq", "FailureHandler へ退避したメッセージ数"),
    43  		pollErrors:   b.counter("worker.poll_errors", "Receive のエラー回数"),
    44  		extendErrors: b.counter("worker.extend_errors", "Extend(ハートビート)のエラー回数"),
    45  		latencyMs:    b.histogram("worker.processing_latency_ms", "Handle の処理時間(ミリ秒)"),
    46  		inFlight:     b.upDownCounter("worker.in_flight", "処理中(受信済み・未確定)のメッセージ数"),
    47  	}
    48  	if b.err != nil {
    49  		return nil, b.err
    50  	}
    51  	return wm, nil
    52  }
