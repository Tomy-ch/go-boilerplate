    15  // drainMetrics は、emit が channel へ送出したメトリクスを全件読み出します。
    16  // emit は consumer が読むまでブロックし得るため、送出側を別 goroutine に置きます。
    17  func drainMetrics(t *testing.T, emit func(chan<- prometheus.Metric)) []prometheus.Metric {
    18  	t.Helper()
    19  	ch := make(chan prometheus.Metric)
    20  	go func() {
    21  		emit(ch)
    22  		close(ch)
    23  	}()
    24  	var got []prometheus.Metric
    25  	for m := range ch {
    26  		got = append(got, m)
    27  	}
    28  	return got
    29  }

    31  // labeledValues は、メトリクスを label 組（値の連結）→ 値の map に畳み込みます。
    32  func labeledValues(t *testing.T, metrics []prometheus.Metric) map[string]float64 {
    33  	t.Helper()
    34  	got := map[string]float64{}
    35  	for _, m := range metrics {
    36  		var pb dto.Metric
    37  		require.NoError(t, m.Write(&pb))
    38  		var key strings.Builder
    39  		for _, l := range pb.GetLabel() {
    40  			key.WriteString(l.GetName() + "=" + l.GetValue() + ";")
    41  		}
    42  		got[key.String()] = pb.GetGauge().GetValue()
    43  	}
    44  	return got
    45  }

   158  				for _, l := range pb.GetLabel() {
   159  					if l.GetName() == "queue" {
   160  						byQueue[l.GetValue()] = pb.GetCounter().GetValue()
   161  					}
   162  				}
   163  			}
