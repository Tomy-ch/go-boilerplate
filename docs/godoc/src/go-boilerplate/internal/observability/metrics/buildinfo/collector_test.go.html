    27  func gatherLabels(t *testing.T, c *Collector) (map[string]string, float64) {
    28  	t.Helper()
    29  
    30  	reg := prometheus.NewRegistry()
    31  	require.NoError(t, reg.Register(c))
    32  
    33  	families, err := reg.Gather()
    34  	require.NoError(t, err)
    35  
    36  	var target *dto.MetricFamily
    37  	for _, mf := range families {
    38  		if mf.GetName() == metricName {
    39  			target = mf
    40  			break
    41  		}
    42  	}
    43  	require.NotNil(t, target)
    44  	assert.Equal(t, dto.MetricType_GAUGE, target.GetType())
    45  	require.Len(t, target.GetMetric(), 1)
    46  
    47  	metric := target.GetMetric()[0]
    48  	labels := make(map[string]string, len(metric.GetLabel()))
    49  	for _, lp := range metric.GetLabel() {
    50  		labels[lp.GetName()] = lp.GetValue()
    51  	}
    52  	return labels, metric.GetGauge().GetValue()
    53  }
