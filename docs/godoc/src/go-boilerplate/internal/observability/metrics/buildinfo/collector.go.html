     1  // Package buildinfo は、アプリケーションのビルド・バージョン・ランタイム情報を
     2  // Prometheus の info gauge (app_build_info) として公開する機能を提供します。
     3  //
     4  // 値は常に 1 であり、意味はラベル (service / environment / version / revision /
     5  // build_date / go_version) に持たせます。これらの値はプロセス起動後に変化しない
     6  // ため、ラベル値は DI 結線時に一度だけ解決・正規化して保持し、スクレイプごとの

    20  // Collector は、ビルド情報を info gauge として公開する prometheus.Collector です。
    21  type Collector struct {
    22  	desc        *prometheus.Desc
    23  	labelValues []string
    24  }

    26  // NewCollector は、ラベル値を結線時に一度だけ解決・正規化して Collector を生成します。
    27  func NewCollector(appCfg *config.ApplicationConfig, bi system.BuildInfo) *Collector {
    28  	// ラベル名と値を 1 箇所で対にし、位置依存の取り違え（version に revision 値が
    29  	// 入る等のサイレントな順序ずれ）を構造的に排除します。
    30  	pairs := []struct{ name, value string }{
    31  		{labelService, normalize(appCfg.Name())},
    32  		{labelEnvironment, normalize(appCfg.Env())},
    33  		{labelVersion, normalize(bi.Version())},
    34  		{labelRevision, normalize(bi.Revision())},
    35  		{labelBuildDate, normalize(bi.BuildDate())},
    36  		{labelGoVersion, normalize(runtime.Version())},
    37  	}
    38  	names := make([]string, len(pairs))
    39  	values := make([]string, len(pairs))
    40  	for i, p := range pairs {
    41  		names[i] = p.name
    42  		values[i] = p.value
    43  	}
    44  	return &Collector{
    45  		desc:        prometheus.NewDesc(metricName, metricHelp, names, nil),
    46  		labelValues: values,
    47  	}
    48  }

    67  // register は、指定された Registerer に Collector を登録します。
    68  //
    69  // 重複登録は AlreadyRegisteredError を無視して安全にスキップします。
    70  func register(reg prometheus.Registerer, c *Collector) error {
    71  	err := reg.Register(c)
    72  	if err != nil {
    73  		var alreadyRegisteredErr prometheus.AlreadyRegisteredError
    74  		if xerrors.As(err, &alreadyRegisteredErr) {
    75  			return nil
    76  		}
    77  		return err
    78  	}
    79  	return nil
    80  }
