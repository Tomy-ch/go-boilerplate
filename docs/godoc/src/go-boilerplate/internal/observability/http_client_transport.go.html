   108  // newGuardedBaseTransport は、指定の dial control を備えた base transport を返します。
   109  func newGuardedBaseTransport(control dialControl) *http.Transport {
   110  	base := &http.Transport{}
   111  	if t, ok := http.DefaultTransport.(*http.Transport); ok {
   112  		base = t.Clone()
   113  	}
   114  	dialer := &net.Dialer{ControlContext: control}

   119  // guardedDialControl は、名前解決後の実接続先 IP を検査する SSRF ガードです（DNS rebinding も防ぎます）。
   120  // link-local / unspecified / bogon予約帯は常に拒否し、loopback / private / CGNAT は ctx フラグで許可されない限り拒否します。
   121  func guardedDialControl(ctx context.Context, _, address string, _ syscall.RawConn) error {
   122  	host, _, err := net.SplitHostPort(address)
   123  	if err != nil {
   124  		return err
   125  	}
   126  	ip := net.ParseIP(host)
   127  	if ip == nil {
   128  		return nil
   129  	}
   130  	if ip.IsLinkLocalUnicast() || ip.IsLinkLocalMulticast() || ip.IsUnspecified() {
   131  		return xerrors.New("ssrf guard: blocked address " + host)
   132  	}
   133  	// bogon/予約帯は正当な宛先にならないため allowPrivateNetwork フラグに関わらず常時拒否する。
   134  	for _, n := range reservedNets {
   135  		if n.Contains(ip) {
   136  			return xerrors.New("ssrf guard: blocked reserved address " + host)
   137  		}
   138  	}
   139  	if !allowPrivateNetworkFromContext(ctx) && (ip.IsLoopback() || ip.IsPrivate() || cgnatNet.Contains(ip)) {
   140  		return xerrors.New("ssrf guard: blocked private/loopback address " + host)
   141  	}
   142  	return nil
   143  }
