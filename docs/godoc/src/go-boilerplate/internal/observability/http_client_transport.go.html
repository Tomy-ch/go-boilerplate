    91  // newGuardedBaseTransport は、指定の dial control を備えた base transport を返します。
    92  func newGuardedBaseTransport(control dialControl) *http.Transport {
    93  	base := &http.Transport{}
    94  	if t, ok := http.DefaultTransport.(*http.Transport); ok {
    95  		base = t.Clone()

    97  	dialer := &net.Dialer{ControlContext: control}

   102  // guardedDialControl は、名前解決後の実接続先 IP を検査する SSRF ガードです（DNS rebinding も防ぎます）。
   103  // link-local / unspecified は常に拒否し、loopback / private は ctx フラグで許可されない限り拒否します。
   104  func guardedDialControl(ctx context.Context, _, address string, _ syscall.RawConn) error {
   105  	host, _, err := net.SplitHostPort(address)
   106  	if err != nil {
   107  		return err
   108  	}
   109  	ip := net.ParseIP(host)
   110  	if ip == nil {
   111  		return nil
   112  	}
   113  	if ip.IsLinkLocalUnicast() || ip.IsLinkLocalMulticast() || ip.IsUnspecified() {
   114  		return xerrors.New("ssrf guard: blocked address " + host)
   115  	}
   116  	if !allowPrivateNetworkFromContext(ctx) && (ip.IsLoopback() || ip.IsPrivate()) {
   117  		return xerrors.New("ssrf guard: blocked private/loopback address " + host)
   118  	}
   119  	return nil
   120  }
