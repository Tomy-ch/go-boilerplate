   105  // newHTTPClientTransport は、dial control を差し替え可能にした内部コンストラクタです。
   106  //
   107  // transport チェーンは外側から redact(退避) → otelhttp(span生成) → restore(復元) → guardedBase の順です。
   108  // otelhttp は span の url.full を req.URL.String() から算出しクエリ・フラグメント込みで記録するため、otelhttp へ
   109  // 渡す前にこれらを退避・除去し、実送信直前に復元することで span からのみ落とします（実リクエストは無改変）。
   110  //
   111  // 不変条件: otelhttp の base には URL をエラーメッセージへ整形しない素の RoundTripper（*http.Transport）を渡します。
   112  // エラー時 otelhttp は span へ err.Error() を記録するため、base が *url.Error（URL 込み）を返す層（例: http.Client
   113  // でのラップ）だと復元後のクエリがエラー span へ漏れます。この境界を跨ぐ RoundTripper を挟む改修は避けてください。
   114  func newHTTPClientTransport(
   115  	tp trace.TracerProvider, propagator propagation.TextMapPropagator, control dialControl,
   116  ) *HTTPClientTransport {
   117  	rt := otelhttp.NewTransport(
   118  		urlSecretRestoringRoundTripper{base: newGuardedBaseTransport(control)},
   119  		otelhttp.WithTracerProvider(tp),
   120  		otelhttp.WithMeterProvider(metricnoop.NewMeterProvider()),
   121  		otelhttp.WithPropagators(conditionalPropagator{inner: propagator}),
   122  	)
   123  	return &HTTPClientTransport{rt: spanURLRedactingRoundTripper{inner: rt}}
   124  }

   126  // RoundTrip は、クエリ・フラグメントを ctx へ退避し URL から除去した clone を otelhttp へ渡します。
   127  func (rt spanURLRedactingRoundTripper) RoundTrip(req *http.Request) (*http.Response, error) {
   128  	if req.URL == nil || (req.URL.RawQuery == "" && req.URL.Fragment == "" && req.URL.RawFragment == "") {
   129  		return rt.inner.RoundTrip(req)
   130  	}
   131  	// http.RoundTripper は呼び出し元の Request を変更してはならないため clone する。
   132  	parts := redactedURLParts{
   133  		rawQuery:    req.URL.RawQuery,
   134  		fragment:    req.URL.Fragment,
   135  		rawFragment: req.URL.RawFragment,
   136  	}
   137  	cloned := req.Clone(context.WithValue(req.Context(), spanQueryRedactionKey{}, parts))
   138  	cloned.URL.RawQuery = ""
   139  	cloned.URL.Fragment = ""
   140  	cloned.URL.RawFragment = ""
   141  	return rt.inner.RoundTrip(cloned)
   142  }

   187  // newGuardedBaseTransport は、指定の dial control を備えた base transport を返します。
   188  func newGuardedBaseTransport(control dialControl) *http.Transport {
   189  	base := &http.Transport{}
   190  	if t, ok := http.DefaultTransport.(*http.Transport); ok {
   191  		base = t.Clone()
   192  	}
   193  	dialer := &net.Dialer{ControlContext: control}
   194  	base.DialContext = dialer.DialContext
   195  	// 不変条件: proxy 経由では dial 先が宛先ではなく proxy になり、guardedDialControl の宛先 IP 検査が
   196  	// 素通りする（SSRF ガードの無効化）。DefaultTransport から継承した環境変数由来の Proxy を無効化し、
   197  	// 宛先へ直結してガードを常に宛先 IP に効かせる（ADR-0020 の最終宛先 IP 検査と整合）。
   198  	// 運用注意: 直接 egress を遮断し forward proxy 必須にした環境では outbound HTTP が全断する
   199  	// （HTTP_PROXY 注入では復活せず、ネットワーク層での吸収が必要）。
   200  	base.Proxy = nil
   201  	return base
   202  }

   204  // guardedDialControl は、名前解決後の実接続先 IP を検査する SSRF ガードです（DNS rebinding も防ぎます）。
   205  // パース不能なアドレスは fail-close で拒否し（zone 付き IPv6 リテラルによる link-local 判定回避を塞ぐ）、
   206  // link-local / unspecified / bogon予約帯は常に拒否、loopback / private / CGNAT は ctx フラグで許可されない限り拒否します。
   207  func guardedDialControl(ctx context.Context, _, address string, _ syscall.RawConn) error {
   208  	host, _, err := net.SplitHostPort(address)
   209  	if err != nil {
   210  		return xerrors.Wrap(err, "ssrf guard: blocked malformed address "+address)
   211  	}
   212  	addr, err := netip.ParseAddr(host)
   213  	if err != nil {
   214  		return xerrors.Wrap(errSSRFUnparsableAddress, host)
   215  	}
   216  	// zone を落として family を正規化し、Prefix 判定を IPv4-mapped/zone 付きでも取りこぼさないようにする。
   217  	addr = addr.Unmap().WithZone("")
   218  	// 埋め込み IPv4 を取り出し、以降の loopback/private/予約帯判定を実宛先に効かせる。
   219  	if nat64WellKnownPrefix.Contains(addr) {
   220  		b := addr.As16()
   221  		addr = netip.AddrFrom4([4]byte{b[12], b[13], b[14], b[15]})
   222  	}
   223  	if addr.IsLinkLocalUnicast() || addr.IsLinkLocalMulticast() || addr.IsUnspecified() {
   224  		return xerrors.Wrap(errSSRFBlockedAddress, host)
   225  	}
   226  	// bogon/予約帯は正当な宛先にならないため allowPrivateNetwork フラグに関わらず常時拒否する。
   227  	for _, p := range reservedPrefixes {
   228  		if p.Contains(addr) {
   229  			return xerrors.Wrap(errSSRFReservedAddress, host)
   230  		}
   231  	}
   232  	if !allowPrivateNetworkFromContext(ctx) && (addr.IsLoopback() || addr.IsPrivate() || cgnatPrefix.Contains(addr)) {
   233  		return xerrors.Wrap(errSSRFPrivateAddress, host)
   234  	}
   235  	return nil
   236  }
