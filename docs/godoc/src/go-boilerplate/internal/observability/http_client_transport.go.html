   110  // newGuardedBaseTransport は、指定の dial control を備えた base transport を返します。
   111  func newGuardedBaseTransport(control dialControl) *http.Transport {
   112  	base := &http.Transport{}
   113  	if t, ok := http.DefaultTransport.(*http.Transport); ok {
   114  		base = t.Clone()
   115  	}
   116  	dialer := &net.Dialer{ControlContext: control}

   119  }

   121  // guardedDialControl は、名前解決後の実接続先 IP を検査する SSRF ガードです（DNS rebinding も防ぎます）。
   122  // パース不能なアドレスは fail-close で拒否し（zone 付き IPv6 リテラルによる link-local 判定回避を塞ぐ）、
   123  // link-local / unspecified / bogon予約帯は常に拒否、loopback / private / CGNAT は ctx フラグで許可されない限り拒否します。
   124  func guardedDialControl(ctx context.Context, _, address string, _ syscall.RawConn) error {
   125  	host, _, err := net.SplitHostPort(address)
   126  	if err != nil {
   127  		return err
   128  	}
   129  	addr, err := netip.ParseAddr(host)
   130  	if err != nil {
   131  		return xerrors.New("ssrf guard: blocked unparsable address " + host)
   132  	}
   133  	// zone を落として family を正規化し、Prefix 判定を IPv4-mapped/zone 付きでも取りこぼさないようにする。
   134  	addr = addr.Unmap().WithZone("")
   135  	if addr.IsLinkLocalUnicast() || addr.IsLinkLocalMulticast() || addr.IsUnspecified() {
   136  		return xerrors.New("ssrf guard: blocked address " + host)
   137  	}
   138  	// bogon/予約帯は正当な宛先にならないため allowPrivateNetwork フラグに関わらず常時拒否する。
   139  	for _, p := range reservedPrefixes {
   140  		if p.Contains(addr) {
   141  			return xerrors.New("ssrf guard: blocked reserved address " + host)
   142  		}
   143  	}
   144  	if !allowPrivateNetworkFromContext(ctx) && (addr.IsLoopback() || addr.IsPrivate() || cgnatPrefix.Contains(addr)) {
   145  		return xerrors.New("ssrf guard: blocked private/loopback address " + host)
   146  	}
   147  	return nil
   148  }
