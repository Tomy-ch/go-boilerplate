    15  	t.Parallel()

    24

    34  			})

    38  
    39  			got := extractFromCarrier(context.Background(), map[string]string(carrier), prop)
    40  
    41  			gsc := trace.SpanContextFromContext(got)

    86  				SpanID:     spanID,
    87  				TraceFlags: trace.FlagsSampled,
    88  			})
    89  			ctx := trace.ContextWithSpanContext(context.Background(), sc)
    90  
    91  			attrs := map[string]string{}

    98  		})
    99  	})
