    15  func Test_extractFromCarrier(t *testing.T) {
    16  	t.Parallel()
    17  
    18  	prop := propagation.TraceContext{}
    19  
    20  	t.Run("正常系", func(t *testing.T) {
    21  		t.Parallel()
    22  
    23  		t.Run("attrs の traceparent から trace context を継続する", func(t *testing.T) {
    24  			t.Parallel()
    25  
    26  			traceID, err := trace.TraceIDFromHex("0123456789abcdef0123456789abcdef")
    27  			require.NoError(t, err)
    28  			spanID, err := trace.SpanIDFromHex("0123456789abcdef")
    29  			require.NoError(t, err)
    30  			sc := trace.NewSpanContext(trace.SpanContextConfig{
    31  				TraceID:    traceID,
    32  				SpanID:     spanID,
    33  				TraceFlags: trace.FlagsSampled,
    34  				Remote:     true,
    35  			})
    36  			// producer 側が attrs に traceparent を載せた状態を作る
    37  			carrier := propagation.MapCarrier{}
    38  			prop.Inject(trace.ContextWithSpanContext(context.Background(), sc), carrier)
    39  
    40  			got := extractFromCarrier(context.Background(), map[string]string(carrier), prop)
    41  
    42  			gsc := trace.SpanContextFromContext(got)
    43  			require.True(t, gsc.HasTraceID())
    44  			assert.Equal(t, traceID, gsc.TraceID())
    45  		})
    46  
    47  		t.Run("公開関数はグローバル伝播器で attrs の traceparent から trace を復元する", func(t *testing.T) {
    48  			t.Parallel()
    49  
    50  			// 公開関数は otel.GetTextMapPropagator() を参照するため、TraceContext を一時登録する。
    51  			prev := otel.GetTextMapPropagator()
    52  			otel.SetTextMapPropagator(propagation.TraceContext{})
    53  			t.Cleanup(func() { otel.SetTextMapPropagator(prev) })
    54  
    55  			traceID, err := trace.TraceIDFromHex("0123456789abcdef0123456789abcdef")
    56  			require.NoError(t, err)
    57  			spanID, err := trace.SpanIDFromHex("0123456789abcdef")
    58  			require.NoError(t, err)
    59  			sc := trace.NewSpanContext(trace.SpanContextConfig{
    60  				TraceID:    traceID,
    61  				SpanID:     spanID,
    62  				TraceFlags: trace.FlagsSampled,
    63  				Remote:     true,
    64  			})
    65  			carrier := propagation.MapCarrier{}
    66  			propagation.TraceContext{}.Inject(trace.ContextWithSpanContext(context.Background(), sc), carrier)
    67  
    68  			got := ExtractFromCarrier(context.Background(), map[string]string(carrier))
    69  
    70  			gsc := trace.SpanContextFromContext(got)
    71  			require.True(t, gsc.HasTraceID())
    72  			assert.Equal(t, traceID, gsc.TraceID())
    73  		})
    74  	})
    75  
    76  	t.Run("異常系", func(t *testing.T) {
    77  		t.Parallel()
    78  
    79  		t.Run("attrs が空なら trace context は付与されない", func(t *testing.T) {
    80  			t.Parallel()
    81  
    82  			got := extractFromCarrier(context.Background(), nil, prop)
    83  
    84  			assert.False(t, trace.SpanContextFromContext(got).HasTraceID())
    85  		})
    86  
    87  		t.Run("公開関数はグローバル伝播器を用い、attrs が空なら ctx をそのまま返す", func(t *testing.T) {
    88  			t.Parallel()
    89  
    90  			ctx := context.Background()
    91  
    92  			// グローバル伝播器を変更せず公開経路（otel.GetTextMapPropagator 利用）を通す。
    93  			assert.Equal(t, ctx, ExtractFromCarrier(ctx, nil))
    94  		})
    95  	})

   121  	t.Parallel()
   122  
   123  	prop := propagation.TraceContext{}
   124  
   125  	t.Run("正常系", func(t *testing.T) {
   126  		t.Parallel()
   127  
   128  		t.Run("ctx の trace context を attrs へ traceparent として書き込む", func(t *testing.T) {
   129  			t.Parallel()
   130  
   131  			traceID, err := trace.TraceIDFromHex("0123456789abcdef0123456789abcdef")
   132  			require.NoError(t, err)
   133  			spanID, err := trace.SpanIDFromHex("0123456789abcdef")
   134  			require.NoError(t, err)
   135  			sc := trace.NewSpanContext(trace.SpanContextConfig{
   136  				TraceID:    traceID,
   137  				SpanID:     spanID,
   138  				TraceFlags: trace.FlagsSampled,
   139  			})
   140  			ctx := trace.ContextWithSpanContext(context.Background(), sc)
   141  
   142  			attrs := map[string]string{}
   143  			injectToCarrier(ctx, attrs, prop)
   144  
   145  			require.Contains(t, attrs, "traceparent")
   146  			// inject した attrs から復元すると同一 traceID に繋がる（往復確認）。
   147  			got := extractFromCarrier(context.Background(), attrs, prop)
   148  			assert.Equal(t, traceID, trace.SpanContextFromContext(got).TraceID())
   149  		})
   150  	})
   151  
   152  	t.Run("異常系", func(t *testing.T) {
   153  		t.Parallel()
   154  
   155  		t.Run("attrs が nil なら何もしない", func(t *testing.T) {
   156  			t.Parallel()
   157  
   158  			assert.NotPanics(t, func() { injectToCarrier(context.Background(), nil, prop) })
   159  		})
   160  
   161  		t.Run("公開関数は TraceContext 限定の伝播器を用いる", func(t *testing.T) {
   162  			t.Parallel()
   163  
   164  			// グローバル伝播器に依らず TraceContext 限定で inject する公開経路を通す。
   165  			assert.NotPanics(t, func() { InjectTraceContextToCarrier(context.Background(), map[string]string{}) })
   166  		})
   167  
   168  		t.Run("公開関数は trace context を inject しつつ baggage は転送しない", func(t *testing.T) {
   169  			t.Parallel()
   170  
   171  			// span が無いと traceparent も書かれず NotContains が疑似陽性になるため、
   172  			// span 有りで traceparent が書かれた上で baggage だけが落ちることを検証する。
   173  			traceID, err := trace.TraceIDFromHex("0123456789abcdef0123456789abcdef")
   174  			require.NoError(t, err)
   175  			spanID, err := trace.SpanIDFromHex("0123456789abcdef")
   176  			require.NoError(t, err)
   177  			sc := trace.NewSpanContext(trace.SpanContextConfig{
   178  				TraceID:    traceID,
   179  				SpanID:     spanID,
   180  				TraceFlags: trace.FlagsSampled,
   181  			})
   182  			ctx := trace.ContextWithSpanContext(context.Background(), sc)
   183  
   184  			member, err := baggage.NewMember("tenant", "acme")
   185  			require.NoError(t, err)
   186  			bag, err := baggage.New(member)
   187  			require.NoError(t, err)
   188  			ctx = baggage.ContextWithBaggage(ctx, bag)
   189  
   190  			attrs := map[string]string{}
   191  			InjectTraceContextToCarrier(ctx, attrs)
   192  
   193  			assert.Contains(t, attrs, "traceparent")
   194  			assert.NotContains(t, attrs, "baggage")
   195  		})
   196  	})
