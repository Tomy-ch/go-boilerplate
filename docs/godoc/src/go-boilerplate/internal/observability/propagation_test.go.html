   112  func TestExtractFromCarrier(t *testing.T) {
   113  	t.Run("正常系", func(t *testing.T) {
   114  		t.Run("グローバル伝播器で attrs の traceparent から trace を復元する", func(t *testing.T) {
   115  			// 公開関数は otel.GetTextMapPropagator() を参照するため、TraceContext を一時登録する。
   116  			prev := otel.GetTextMapPropagator()
   117  			otel.SetTextMapPropagator(propagation.TraceContext{})
   118  			t.Cleanup(func() { otel.SetTextMapPropagator(prev) })
   119  
   120  			sc := testSampledSpanContext(true)
   121  			traceID := sc.TraceID()
   122  			spanID := sc.SpanID()
   123  			carrier := propagation.MapCarrier{}
   124  			propagation.TraceContext{}.Inject(trace.ContextWithSpanContext(context.Background(), sc), carrier)
   125  
   126  			got := ExtractFromCarrier(context.Background(), map[string]string(carrier))
   127  
   128  			gsc := trace.SpanContextFromContext(got)
   129  			assert.True(t, gsc.HasTraceID())
   130  			assert.Equal(t, traceID, gsc.TraceID())
   131  			assert.Equal(t, spanID, gsc.SpanID())
   132  		})
   133  	})
   134  
   135  	t.Run("異常系", func(t *testing.T) {
   136  		t.Run("attrs が空なら ctx をそのまま返す", func(t *testing.T) {
   137  			ctx := context.Background()
   138  
   139  			// グローバル伝播器を変更せず公開経路（otel.GetTextMapPropagator 利用）を通す。
   140  			assert.Equal(t, ctx, ExtractFromCarrier(ctx, nil))
   141  		})
   142  	})

   146  func TestInjectTraceContextToCarrier(t *testing.T) {
   147  	t.Run("正常系", func(t *testing.T) {
   148  		t.Run("trace context を inject しつつ baggage は転送しない", func(t *testing.T) {
   149  			// span が無いと traceparent も書かれず NotContains が疑似陽性になるため、
   150  			// span 有りで traceparent が書かれた上で baggage だけが落ちることを検証する。
   151  			sc := testSampledSpanContext(false)
   152  			ctx := trace.ContextWithSpanContext(context.Background(), sc)
   153  
   154  			member, err := baggage.NewMember("tenant", "acme")
   155  			require.NoError(t, err)
   156  			bag, err := baggage.New(member)
   157  			require.NoError(t, err)
   158  			ctx = baggage.ContextWithBaggage(ctx, bag)
   159  
   160  			attrs := map[string]string{}
   161  			InjectTraceContextToCarrier(ctx, attrs)
   162  
   163  			assert.Contains(t, attrs, "traceparent")
   164  			assert.NotContains(t, attrs, "baggage")
   165  		})
   166  
   167  		t.Run("グローバル伝播器に依らず TraceContext 限定で inject する", func(t *testing.T) {
   168  			// グローバル伝播器を Baggage 単独に差し替えても traceparent が書かれる。
   169  			// すなわち traceContextPropagator 固定で otel.GetTextMapPropagator を見ない。
   170  			prev := otel.GetTextMapPropagator()
   171  			otel.SetTextMapPropagator(propagation.Baggage{})
   172  			t.Cleanup(func() { otel.SetTextMapPropagator(prev) })
   173  
   174  			sc := testSampledSpanContext(false)
   175  
   176  			attrs := map[string]string{}
   177  			InjectTraceContextToCarrier(trace.ContextWithSpanContext(context.Background(), sc), attrs)
   178  
   179  			assert.Contains(t, attrs, "traceparent")
   180  		})
   181  	})
   182  
   183  	t.Run("異常系", func(t *testing.T) {
   184  		t.Run("attrs が nil なら何もせずパニックしない", func(t *testing.T) {
   185  			assert.NotPanics(t, func() { InjectTraceContextToCarrier(newSampledContext(), nil) })
   186  		})
   187  	})
