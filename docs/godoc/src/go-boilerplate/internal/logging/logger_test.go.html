   105  // newBufLogger は、書き込み内容を検証可能なバッファ付き logger を生成する。
   106  func newBufLogger() (*logger, *bytes.Buffer) {
   107  	var buf bytes.Buffer
   108  	enc := zapcore.NewConsoleEncoder(zap.NewDevelopmentEncoderConfig())
   109  	core := zapcore.NewCore(enc, zapcore.AddSync(&buf), zapcore.DebugLevel)
   110  	return &logger{log: zap.New(core)}, &buf
   111  }
