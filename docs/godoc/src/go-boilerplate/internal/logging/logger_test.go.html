   106  			t.Parallel()
   107  
   108  			var buf bytes.Buffer
   109  			enc := zapcore.NewJSONEncoder(zapcore.EncoderConfig{MessageKey: "msg"})
   110  			inner := zapcore.NewCore(enc, zapcore.AddSync(&buf), zapcore.DebugLevel)
   111  			gated := levelGatedCore{Core: inner, min: zapcore.WarnLevel}
   112  
   113  			got := gated.With([]zapcore.Field{zap.String("svc", "demo")})
   114  
   115  			gc, ok := got.(levelGatedCore)
   116  			require.True(t, ok)
   117  			assert.Equal(t, zapcore.WarnLevel, gc.min)
   118  			assert.False(t, gc.Enabled(zapcore.InfoLevel))
   119  			assert.True(t, gc.Enabled(zapcore.WarnLevel))
   120  
   121  			require.NoError(t, gc.Write(zapcore.Entry{Level: zapcore.WarnLevel, Message: "hi"}, nil))
   122  			var m map[string]any
   123  			require.NoError(t, json.Unmarshal(bytes.TrimRight(buf.Bytes(), "\n"), &m))
   124  			assert.Equal(t, "demo", m["svc"])
   125  		})
   126  	})

   237  func newBufLogger() (*logger, *bytes.Buffer) {
   238  	var buf bytes.Buffer
   239  	enc := zapcore.NewConsoleEncoder(zap.NewDevelopmentEncoderConfig())
   240  	core := zapcore.NewCore(enc, zapcore.AddSync(&buf), zapcore.DebugLevel)
   241  	return &logger{log: zap.New(core)}, &buf
