    39  		})

    42  			t.Parallel()

    46  			// 追加 core 自体は Debug まで通すが、元 Logger は Info のため Debug はゲートされる。
    47  			extra := zapcore.NewCore(enc, zapcore.AddSync(&buf), zapcore.DebugLevel)
    48

    51  
    52  			got.Debug("gated-out")

   153  // newBufLogger は、書き込み内容を検証可能なバッファ付き logger を生成する。
   154  func newBufLogger() (*logger, *bytes.Buffer) {
   155  	var buf bytes.Buffer
   156  	enc := zapcore.NewConsoleEncoder(zap.NewDevelopmentEncoderConfig())
   157  	core := zapcore.NewCore(enc, zapcore.AddSync(&buf), zapcore.DebugLevel)
   158  	return &logger{log: zap.New(core)}, &buf
   159  }
