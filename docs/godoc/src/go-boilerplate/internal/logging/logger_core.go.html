    19  // mustOpenSink は path の WriteSyncer を返します。開けない場合は panic します。
    20  func mustOpenSink(path string) zapcore.WriteSyncer {
    21  	ws, _, err := zap.Open(path)
    22  	if err != nil {
    23  		panic("logging: failed to open sink " + path + ": " + err.Error())
    24  	}
    25  	return ws
    26  }

    60  // buildLogger は encoder と出力先から Logger を構築します。
    61  // jsonArrayStacktrace=true のとき、stacktrace を行配列として出力します。
    62  func buildLogger(
    63  	enc zapcore.Encoder, ws zapcore.WriteSyncer, level, stacktraceLevel zapcore.Level, jsonArrayStacktrace bool,
    64  ) Logger {
    65  	core := zapcore.NewCore(enc, ws, zap.NewAtomicLevelAt(level))
    66  	if jsonArrayStacktrace {
    67  		core = wrapStacktraceCore(core, stacktraceKey)
    68  	}
    69  	zl := zap.New(core,
    70  		zap.AddCaller(),
    71  		zap.AddStacktrace(stacktraceLevel),
    72  		zap.ErrorOutput(stderrSyncer),
    73  	)
    74  	return &logger{log: zl}
    75  }
