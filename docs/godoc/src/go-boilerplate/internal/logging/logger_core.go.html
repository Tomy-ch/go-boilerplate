    61  // buildLogger は、zap.Config から Logger を構築する共通処理です。
    62  // Build 失敗時は zap が nil の *zap.Logger を返すため、それを包んだ
    63  // Logger（中身 nil）を返すと初回ログ出力で nil 参照 panic になる。
    64  // よってエラー時は Logger を返さず、nil とエラーのみを返す。
    65  //
    66  // stacktrace は JSON 出力時のみ行配列に変換する（wrapStacktraceCore）。
    67  // console エンコーダは Entry.Stack を独自に整形（改行+インデント）するため、
    68  // wrap すると console 出力が一行 JSON ダンプ化して可読性が落ちるので適用しない。
    69  // StacktraceKey 未設定の Config（テスト用最小構成など）でも wrap をスキップする。
    70  func buildLogger(cfg zap.Config, stacktraceLevel zapcore.Level) (Logger, error) {
    71  	opts := []zap.Option{
    72  		zap.AddCaller(),
    73  		zap.AddStacktrace(stacktraceLevel),
    74  	}
    75  	if stacktraceKey := cfg.EncoderConfig.StacktraceKey; cfg.Encoding == "json" && stacktraceKey != "" {
    76  		opts = append(opts, zap.WrapCore(func(core zapcore.Core) zapcore.Core {
    77  			return wrapStacktraceCore(core, stacktraceKey)
    78  		}))
    79  	}
    80  	zl, err := cfg.Build(opts...)
    81  	if err != nil {
    82  		return nil, err
    83  	}
    84  
    85  	return &logger{log: zl}, nil
    86  }
