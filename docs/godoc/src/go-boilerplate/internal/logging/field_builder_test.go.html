   319  // fieldKeys は、Field スライスからキー文字列の一覧を抽出するテストヘルパーです。
   320  func fieldKeys(fs []*Field) []string {
   321  	keys := make([]string, 0, len(fs))
   322  	for _, f := range fs {
   323  		keys = append(keys, f.key)
   324  	}
   325  	return keys
