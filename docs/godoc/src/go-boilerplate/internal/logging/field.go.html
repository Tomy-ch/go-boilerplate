   122  // Stacktrace は、エラースタックトレースのログフィールドを作成します。
   123  // JSON ビューア（Grafana / Loki 等）で改行が可読に表示されるよう、
   124  // 単一文字列ではなく行ごとに分割した []string を保持します。
   125  func Stacktrace(key fieldKey, err error) *Field {
   126  	return &Field{
   127  		key:          string(key),
   128  		kind:         fieldStrings,
   129  		stringsValue: SplitStackLines(xerrors.StackTrace(err)),
   130  	}
   131  }
   132  
   133  // SplitStackLines は、スタックトレース文字列を改行で分割し行配列に変換します。
   134  // 末尾の空行は除去し、空文字列または改行のみの入力では nil を返します。
   135  // 配列化により行境界とネスト深さが構造として表現されるため、cockroachdb/errors の
   136  // `%+v` が各行頭に付与するガター（`  | ` の空白・パイプや `\t<file>:<line>` の先頭タブ）は
   137  // 冗長になる。各行頭の空白・`|`・タブをまとめて除去し、内容だけを残す。
   138  func SplitStackLines(s string) []string {
   139  	trimmed := strings.TrimRight(s, "\n")
   140  	if trimmed == "" {
   141  		return nil
   142  	}
   143  	lines := strings.Split(trimmed, "\n")
   144  	for i, line := range lines {
   145  		lines[i] = strings.TrimLeft(line, " |\t")
   146  	}
   147  	return lines
   148  }
