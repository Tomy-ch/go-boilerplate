   177  // BuildSQLEndFields は、SQLの終了時点のログ出力用の Field スライスを構築します。
   178  func (l *logFieldBuilder) BuildSQLEndFields(sql SQLFieldsEndInput) []*Field {
   179  	compact := l.buildCompactQuery(sql.Query)
   180  
   181  	fields := append(l.buildEventHeader(EventTypeEnd, sql.EventAt),
   182  		String(LayerKey, sql.Layer),
   183  		String(PackageKey, sql.PkgName),
   184  		String(FunctionKey, sql.FuncName),
   185  		String(SpanNameKey, sql.SpanName),
   186  		String(RawQueryKey, sql.Query),
   187  		String(QueryCompactKey, compact),
   188  		DurationMs(LatencyKey, sql.Latency),
   189  	)
   190  
   191  	if len(sql.Args) > 0 {
   192  		fields = append(fields,
   193  			Int(QueryArgsCountKey, len(sql.Args)),
   194  		)
   195  	}
   196  
   197  	if sql.Err != nil {
   198  		fields = append(fields, Error(InternalErrorKey, sql.Err))
   199  	}
   200  
   201  	return l.appendTraceSpanFields(fields, sql.TraceID, sql.SpanID, sql.ParentSpanID)
   202  }

   232  // appendTraceSpanFields は、obs が有効なとき trace/span を fields に追加する。
   233  func (l *logFieldBuilder) appendTraceSpanFields(
   234  	fields []*Field,
   235  	traceID string,
   236  	spanID string,
   237  	parentSpanID string,
   238  ) []*Field {
   239  	if !l.obsCfg.Enabled() || traceID == "" || spanID == "" {
   240  		return fields
   241  	}
   242  
   243  	fields = append(fields,
   244  		String(TraceIDKey, traceID),
   245  		String(SpanIDKey, spanID),
   246  	)
   247  
   248  	if parentSpanID == "" {
   249  		return fields
   250  	}
   251  
   252  	return append(fields,
   253  		String(ParentSpanIDKey, parentSpanID),
   254  	)
   255  }

   257  // buildCompactQuery は、SQL クエリを1行の短縮表現に変換します。
   258  func (*logFieldBuilder) buildCompactQuery(q string) string {
   259  	if q == "" {
   260  		return ""
   261  	}
   262  
   263  	s := strings.ReplaceAll(q, "\n", " ")
   264  	s = strings.ReplaceAll(s, "\t", " ")
   265  
   266  	s = strings.Join(strings.Fields(s), " ")
   267  
   268  	return s
   269  }
