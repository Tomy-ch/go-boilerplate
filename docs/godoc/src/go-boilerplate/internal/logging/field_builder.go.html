   133  // BuildSQLEndFields は、SQLの終了時点のログ出力用の Field スライスを構築します。
   134  func (l *logFieldBuilder) BuildSQLEndFields(sql SQLFieldsEndInput) []*Field {
   135  	compact := l.buildCompactQuery(sql.Query)
   136  
   137  	fields := append(l.buildEventHeader(EventTypeEnd, sql.EventAt),
   138  		String(LayerKey, sql.Layer),
   139  		String(PackageKey, sql.PkgName),
   140  		String(FunctionKey, sql.FuncName),
   141  		String(SpanNameKey, sql.SpanName),
   142  		String(RawQueryKey, sql.Query),
   143  		String(QueryCompactKey, compact),
   144  		DurationMs(LatencyKey, sql.Latency),
   145  	)
   146  
   147  	if len(sql.Args) > 0 {
   148  		fields = append(fields,
   149  			Int(QueryArgsCountKey, len(sql.Args)),
   150  		)
   151  	}
   152  
   153  	if sql.Err != nil {
   154  		fields = append(fields, Error(InternalErrorKey, sql.Err))
   155  	}
   156  
   157  	return l.appendTraceSpanFields(fields, sql.TraceID, sql.SpanID, sql.ParentSpanID)
   158  }

   170  // appendTraceSpanFields は、obs が有効なとき trace/span を fields に追加する。
   171  func (l *logFieldBuilder) appendTraceSpanFields(
   172  	fields []*Field,
   173  	traceID string,
   174  	spanID string,
   175  	parentSpanID string,
   176  ) []*Field {
   177  	if !l.obsCfg.Enabled() || traceID == "" || spanID == "" {
   178  		return fields
   179  	}
   180  
   181  	fields = append(fields,
   182  		String(TraceIDKey, traceID),
   183  		String(SpanIDKey, spanID),
   184  	)
   185  
   186  	if parentSpanID == "" {
   187  		return fields
   188  	}
   189  
   190  	return append(fields,
   191  		String(ParentSpanIDKey, parentSpanID),
   192  	)
   193  }

   195  // buildCompactQuery は、SQL クエリを1行の短縮表現に変換します。
   196  func (*logFieldBuilder) buildCompactQuery(q string) string {
   197  	if q == "" {
   198  		return ""
   199  	}
   200  
   201  	s := strings.ReplaceAll(q, "\n", " ")
   202  	s = strings.ReplaceAll(s, "\t", " ")
   203  
   204  	s = strings.Join(strings.Fields(s), " ")
   205  
   206  	return s
   207  }
