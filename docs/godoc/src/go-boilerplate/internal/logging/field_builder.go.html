   178  // BuildSQLEndFields は、SQLの終了時点のログ出力用の Field スライスを構築します。
   179  func (l *logFieldBuilder) BuildSQLEndFields(sql SQLFieldsEndInput) []*Field {
   180  	compact := l.buildCompactQuery(sql.Query)
   181  
   182  	fields := append(l.buildEventHeader(EventTypeEnd, sql.EventAt),
   183  		String(LayerKey, sql.Layer),
   184  		String(PackageKey, sql.PkgName),
   185  		String(FunctionKey, sql.FuncName),
   186  		String(SpanNameKey, sql.SpanName),
   187  		String(RawQueryKey, sql.Query),
   188  		String(QueryCompactKey, compact),
   189  		DurationMs(LatencyKey, sql.Latency),
   190  	)
   191  
   192  	if len(sql.Args) > 0 {
   193  		fields = append(fields,
   194  			Int(QueryArgsCountKey, len(sql.Args)),
   195  		)
   196  	}
   197  
   198  	if sql.Err != nil {
   199  		fields = append(fields, Error(InternalErrorKey, sql.Err))
   200  	}
   201  
   202  	return l.appendTraceSpanFields(fields, sql.TraceID, sql.SpanID, sql.ParentSpanID)
   203  }

   233  // appendTraceSpanFields は、obs が有効なとき trace/span を fields に追加する。
   234  func (l *logFieldBuilder) appendTraceSpanFields(
   235  	fields []*Field,
   236  	traceID string,
   237  	spanID string,
   238  	parentSpanID string,
   239  ) []*Field {
   240  	if !l.obsCfg.Enabled() || traceID == "" || spanID == "" {
   241  		return fields
   242  	}
   243  
   244  	fields = append(fields,
   245  		String(TraceIDKey, traceID),
   246  		String(SpanIDKey, spanID),
   247  	)
   248  
   249  	if parentSpanID == "" {
   250  		return fields
   251  	}
   252  
   253  	return append(fields,
   254  		String(ParentSpanIDKey, parentSpanID),
   255  	)
   256  }

   258  // buildCompactQuery は、SQL クエリを1行の短縮表現に変換します。
   259  func (*logFieldBuilder) buildCompactQuery(q string) string {
   260  	if q == "" {
   261  		return ""
   262  	}
   263  
   264  	s := strings.ReplaceAll(q, "\n", " ")
   265  	s = strings.ReplaceAll(s, "\t", " ")
   266  
   267  	s = strings.Join(strings.Fields(s), " ")
   268  
   269  	return s
   270  }
