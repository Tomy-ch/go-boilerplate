     8  // stacktraceArrayCore は zap が自動付与する stacktrace（埋め込み \n\t の単一文字列）を、
     9  // 行ごとの []string に変換して emit する zapcore.Core ラッパです。
    10  // Grafana / Loki などの JSON ビューアでは配列要素ごとに改行表示されるため、
    11  // 単一文字列のときに発生する \n のリテラル表示問題を避けられます。
    12  //
    13  // 前提: 本実装は単一の leaf Core をラップする用途のみを想定します。
    14  // Check は内側 Core の Check に委譲せず自身のみ登録するため、内側が Sampler や
    15  // 複数子の Tee の場合に Check フェーズの副作用（Sampler のカウンタ更新など）が
    16  // 失われます。現状の NewJSONLogger / NewConsoleLogger は Sampling を
    17  // 使わず Tee も組まないため到達しません。将来サンプリング等を導入する場合は
    18  // 本ラッパを Check/Write 二段階契約に沿って再設計してください。
    19  type stacktraceArrayCore struct {
    20  	zapcore.Core
    21  
    22  	key string
    23  }
    24  
    25  // wrapStacktraceCore は、与えられた Core を stacktraceArrayCore でラップします。
    26  // key は出力先キー名（EncoderConfig.StacktraceKey と一致させる）。
    27  func wrapStacktraceCore(core zapcore.Core, key string) zapcore.Core {
    28  	return &stacktraceArrayCore{Core: core, key: key}
    29  }
    30  
    31  // With は内側 Core の With を伝播しつつ、自身のラップ属性を維持します。
    32  func (c *stacktraceArrayCore) With(fs []zapcore.Field) zapcore.Core {
    33  	return &stacktraceArrayCore{Core: c.Core.With(fs), key: c.key}
    34  }
    35  
    36  // Check は、レベルが有効な場合に自身を CheckedEntry に登録します。
    37  func (c *stacktraceArrayCore) Check(ent zapcore.Entry, ce *zapcore.CheckedEntry) *zapcore.CheckedEntry {
    38  	if c.Enabled(ent.Level) {
    39  		return ce.AddCore(ent, c)
    40  	}
    41  	return ce
    42  }
    43  
    44  // Write は、Entry.Stack を行配列に変換した zap.Strings フィールドへ差し替えます。
    45  // 元の Entry.Stack はクリアして、エンコーダが二重に文字列出力するのを防ぎます。
    46  func (c *stacktraceArrayCore) Write(ent zapcore.Entry, fs []zapcore.Field) error {
    47  	if ent.Stack != "" {
    48  		fs = append(fs, zap.Strings(c.key, SplitStackLines(ent.Stack)))
    49  		ent.Stack = ""
    50  	}
    51  	return c.Core.Write(ent, fs)
    52  }
