    10  // Logger は、アプリ全体が使うロガーのインターフェースです。
    11  type Logger interface {
    12  	// Debug はデバッグレベルのログを出力する。
    13  	Debug(msg string, fields ...*Field)
    14  	// Info は情報レベルのログを出力する。
    15  	Info(msg string, fields ...*Field)
    16  	// Warn は警告レベルのログを出力する。

    18  	// Error はエラーレベルのログを出力する。
    19  	Error(msg string, fields ...*Field)

    21  	Named(name string) Logger
    22  	// CallerSkip は、コールスタックのスキップ数を設定した新しいLoggerを返す。
    23  	CallerSkip(skip int) Logger
    24  }

    64  // convertFields は Field のスライスを zap.Field のスライスに変換する。
    65  func (l *logger) convertFields(fs []*Field) []zap.Field {
    66  	zfs := make([]zap.Field, 0, len(fs))
    67  
    68  	for _, f := range fs {
    69  		switch f.kind {
    70  		case fieldString:
    71  			zfs = append(zfs, zap.String(f.key, f.stringValue))
    72  		case fieldStrings:
    73  			zfs = append(zfs, zap.Strings(f.key, f.stringsValue))
    74  		case fieldInt:
    75  			zfs = append(zfs, zap.Int(f.key, f.intValue))
    76  		case fieldInt64:
    77  			zfs = append(zfs, zap.Int64(f.key, f.int64Value))
    78  		case fieldFloat64:
    79  			zfs = append(zfs, zap.Float64(f.key, f.float64Value))
    80  		case fieldBool:
    81  			zfs = append(zfs, zap.Bool(f.key, f.boolValue))
    82  		case fieldError:
    83  			zfs = append(zfs, zap.NamedError(f.key, f.errorValue))
    84  		case fieldAny:
    85  			zfs = append(zfs, zap.Any(f.key, f.anyValue))
    86  		default:
    87  			// fieldUnknown（ゼロ値）や将来追加され case 漏れした kind が
    88  			// 到達する位置。バグ検知の余地を残しつつ zap.Any で安全に出力する。
    89  			zfs = append(zfs, zap.Any(f.key, f.anyValue))
    90  		}
    91  	}
    92  
    93  	return zfs
    94  }
