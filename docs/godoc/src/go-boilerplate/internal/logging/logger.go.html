    54  // WithCore は、既存 Logger に追加のログ core を、元 Logger と同じ最小レベルでゲートして
    55  // Tee した新しい Logger を返します。core が nil の場合は、元の Logger をそのまま返します。
    56  func WithCore(l Logger, core LogCore) Logger {
    57  	if core == nil {
    58  		return l
    59  	}
    60  	// 本パッケージの具象 *logger だけが zap core を内包するため、Tee 対象も *logger に限定する。
    61  	// それ以外の Logger 実装（テスト用 fake 等）は core を持たないので、ゲートできず素通しする。
    62  	base, ok := l.(*logger)
    63  	if !ok {
    64  		return l
    65  	}
    66  	gated := levelGatedCore{Core: core, min: base.log.Level()}
    67  
    68  	return &logger{
    69  		log: base.log.WithOptions(zap.WrapCore(func(c zapcore.Core) zapcore.Core {
    70  			return zapcore.NewTee(c, gated)
    71  		})),
    72  		extract: base.extract,
    73  	}
    74  }

   138  // injectTrace は、TraceExtractor が trace を返す場合に trace_id / span_id を
   139  // フィールド先頭へ注入する。extractor 未注入（nil）や span 無効時は fields をそのまま返す。
   140  func (l *logger) injectTrace(ctx context.Context, fields []*Field) []*Field {
   141  	if l.extract == nil {
   142  		return fields
   143  	}
   144  	traceID, spanID, ok := l.extract(ctx)
   145  	if !ok {
   146  		return fields
   147  	}
   148  
   149  	out := make([]*Field, 0, len(fields)+traceFieldCount)
   150  	out = append(out, String(TraceIDKey, traceID), String(SpanIDKey, spanID))
   151  	return append(out, fields...)
   152  }

   154  // convertFields は Field のスライスを zap.Field のスライスに変換する。
   155  func (l *logger) convertFields(fs []*Field) []zap.Field {
   156  	zfs := make([]zap.Field, 0, len(fs))
   157  
   158  	for _, f := range fs {
   159  		switch f.kind {
   160  		case fieldString:
   161  			zfs = append(zfs, zap.String(f.key, f.stringValue))
   162  		case fieldStrings:
   163  			zfs = append(zfs, zap.Strings(f.key, f.stringsValue))
   164  		case fieldInt:
   165  			zfs = append(zfs, zap.Int(f.key, f.intValue))
   166  		case fieldInt64:
   167  			zfs = append(zfs, zap.Int64(f.key, f.int64Value))
   168  		case fieldFloat64:
   169  			zfs = append(zfs, zap.Float64(f.key, f.float64Value))
   170  		case fieldBool:
   171  			zfs = append(zfs, zap.Bool(f.key, f.boolValue))
   172  		case fieldError:
   173  			zfs = append(zfs, zap.NamedError(f.key, f.errorValue))
   174  		case fieldAny:
   175  			zfs = append(zfs, zap.Any(f.key, f.anyValue))
   176  		default:
   177  			// fieldUnknown（ゼロ値）や将来追加され case 漏れした kind が
   178  			// 到達する位置。バグ検知の余地を残しつつ zap.Any で安全に出力する。
   179  			zfs = append(zfs, zap.Any(f.key, f.anyValue))
   180  		}
   181  	}
   182  
   183  	return zfs
   184  }
