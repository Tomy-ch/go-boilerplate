   174  // driveInFlight は、url へ接続してハンドラを進行中（非アイドル接続）にし、進入を待ちます。
   175  func driveInFlight(t *testing.T, url string, entered <-chan struct{}, release chan struct{}) {
   176  	t.Helper()
   177  	go sendUntilConnected(url)
   178  	select {
   179  	case <-entered:
   180  	case <-time.After(3 * time.Second):
   181  		close(release)
   182  		t.Fatal("進行中ハンドラへ到達しなかった")
   183  	}
   184  }

   186  // sendUntilConnected は、bind 完了まで接続を試み、接続できたら応答を読み切ります。
   187  func sendUntilConnected(url string) {
   188  	deadline := time.Now().Add(3 * time.Second)
   189  	for time.Now().Before(deadline) {
   190  		req, err := http.NewRequestWithContext(context.Background(), http.MethodGet, url, nil)
   191  		if err != nil {
   192  			return
   193  		}
   194  		res, err := http.DefaultClient.Do(req)
   195  		if err == nil {
   196  			_, _ = io.Copy(io.Discard, res.Body)
   197  			_ = res.Body.Close()
   198  			return
   199  		}
   200  		time.Sleep(5 * time.Millisecond)
   201  	}
   202  }
