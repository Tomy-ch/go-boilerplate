    15  // RunWorkerWith は、provide が返す開始・停止関数で worker を実行します。
    16  // grace（APP_SHUTDOWN_TIMEOUT）は停止猶予の単一軸で、停止 context の deadline に用います。
    17  func RunWorkerWith(
    18  	ctx context.Context,
    19  	name string,
    20  	args []string,
    21  	grace time.Duration,

    28  // runWorker は、worker を起動して常駐させ、SIGTERM(ctx 完了) もしくは engine 自走停止で
    29  // グレースフルに停止します。job と異なり完了 channel を待ち続ける常駐型です。
    30  func runWorker(ctx context.Context, name string, args []string, grace time.Duration, start StartFunc, stop StopFunc) error {
    31  	done := start(ctx, name, args)
    32  
    33  	var (
    34  		runErr      error
    35  		selfStopped bool
    36  	)
    37  	select {
    38  	case runErr = <-done:
    39  		// engine が自走停止（Fatal / unknown worker）。done は消費済み。
    40  		selfStopped = true
    41  	case <-ctx.Done():
    42  		// SIGTERM。engine を drain して停止し、停止後に結果を確認する。
    43  	}
    44  
    45  	gracefulStop(ctx, grace, stop)
    46  
    47  	// SIGTERM 経路では engine の実際の終了結果を必ず待ち切る。
    48  	// grace が engine の drain 完了より先に満了して OnStop が早期 return しても、
    49  	// engine goroutine は drain 後に必ず done へ結果を書く（DrainTimeout で有界。
    50  	// 起動時 validation で DrainTimeout < grace を保証）。
    51  	// 非ブロッキングで default を取ると、その間に発生した Fatal を取りこぼし exit 0 になってしまう。
    52  	if !selfStopped {
    53  		runErr = <-done
    54  	}
    55  	return runErr
    56  }
    57  
    58  // gracefulStop は、停止開始時点から grace の猶予を与えて後始末を実行します。
    59  // ctx は SIGTERM で既にキャンセル済みのため、停止用 context はキャンセルだけ切り離して
    60  // （trace/baggage は引き継ぎつつ）作り直します。
    61  func gracefulStop(ctx context.Context, grace time.Duration, stop StopFunc) {
    62  	stopCtx, cancel := context.WithTimeout(context.WithoutCancel(ctx), grace)
    63  	defer cancel()
    64  	_ = stop(stopCtx)
    65  }
