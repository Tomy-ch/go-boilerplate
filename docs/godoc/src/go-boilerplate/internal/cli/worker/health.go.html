    19  // NewHealthServer は、liveness(/healthz) と readiness(/readyz) を公開する health listener の
    20  // 開始・終了関数を返します。ready は readiness 判定（例: engine.Healthy）を渡します。
    21  // metrics サーバーと異なり専用の ServeMux を使い、pprof/metrics と衝突しません。
    22  func NewHealthServer(addr string, ready func() bool, logger logging.Logger) (func(), func(context.Context)) {
    23  	srv := &http.Server{
    24  		Addr:              addr,
    25  		Handler:           healthMux(ready),
    26  		ReadHeaderTimeout: healthReadHeaderTimeout,
    27  		ReadTimeout:       healthReadTimeout,
    28  		WriteTimeout:      healthWriteTimeout,
    29  		IdleTimeout:       healthIdleTimeout,
    30  	}
    31  
    32  	start := func() {
    33  		go func() {
    34  			if err := srv.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
    35  				logger.Named("worker.health").Error("health server error", logging.Error(logging.ErrorKey, err))
    36  			}
    37  		}()
    38  	}
    39  	stop := func(ctx context.Context) {
    40  		if err := srv.Shutdown(ctx); err != nil {
    41  			logger.Named("worker.health").Error("health server shutdown error", logging.Error(logging.ErrorKey, err))
    42  		}
    43  	}
    44  	return start, stop
    45  }

    47  // healthMux は、liveness(/healthz) と readiness(/readyz) を提供する ServeMux を構築します。
    48  func healthMux(ready func() bool) *http.ServeMux {
    49  	mux := http.NewServeMux()
    50  	mux.HandleFunc("/healthz", func(w http.ResponseWriter, _ *http.Request) {
    51  		w.WriteHeader(http.StatusOK)
    52  		_, _ = w.Write([]byte("ok"))
    53  	})
    54  	mux.HandleFunc("/readyz", func(w http.ResponseWriter, _ *http.Request) {
    55  		if ready() {
    56  			w.WriteHeader(http.StatusOK)
    57  			_, _ = w.Write([]byte("ready"))
    58  			return
    59  		}
    60  		w.WriteHeader(http.StatusServiceUnavailable)
    61  		_, _ = w.Write([]byte("not ready"))
    62  	})
    63  	return mux
    64  }
