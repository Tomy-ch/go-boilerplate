   220  	t.Parallel()
   221  
   222  	t.Run("正常系", func(t *testing.T) {
   223  		t.Parallel()
   224  
   225  		t.Run("NilVersionは未適用として扱いそのままDownする", func(t *testing.T) {
   226  			t.Parallel()
   227  
   228  			ctrl := gomock.NewController(t)
   229  			m := mock_migrate.NewMockMigrator(ctrl)
   230  			m.EXPECT().Version().Return(uint(0), false, migrate.ErrNilVersion)
   231  			m.EXPECT().Down().Return(nil)
   232  
   233  			require.NoError(t, executeMigrateFullDown(m))
   234  		})
   235  
   236  		t.Run("dirty時はForceで整合を取り直してからDownする", func(t *testing.T) {
   237  			t.Parallel()
   238  
   239  			ctrl := gomock.NewController(t)
   240  			m := mock_migrate.NewMockMigrator(ctrl)
   241  			m.EXPECT().Version().Return(uint(5), true, nil)
   242  			m.EXPECT().Force(5).Return(nil)
   243  			m.EXPECT().Down().Return(nil)
   244  
   245  			require.NoError(t, executeMigrateFullDown(m))
   246  		})
   247  
   248  		t.Run("Downが無変更ならErrNoChangeを握りつぶし成功扱いとする", func(t *testing.T) {
   249  			t.Parallel()
   250  
   251  			ctrl := gomock.NewController(t)
   252  			m := mock_migrate.NewMockMigrator(ctrl)
   253  			m.EXPECT().Version().Return(uint(2), false, nil)
   254  			m.EXPECT().Down().Return(migrate.ErrNoChange)
   255  
   256  			require.NoError(t, executeMigrateFullDown(m))
   257  		})
   258  
   259  		t.Run("dirty時にForce後のDownがErrNoChangeでも成功扱いとする", func(t *testing.T) {
   260  			t.Parallel()
   261  
   262  			// dirty=true → Force 成功 → Down(ErrNoChange) の経路で
   263  			// ErrNoChange 握りつぶしが Force 後にも効くことを検証する。
   264  			ctrl := gomock.NewController(t)
   265  			m := mock_migrate.NewMockMigrator(ctrl)
   266  			m.EXPECT().Version().Return(uint(3), true, nil)
   267  			m.EXPECT().Force(3).Return(nil)
   268  			m.EXPECT().Down().Return(migrate.ErrNoChange)
   269  
   270  			require.NoError(t, executeMigrateFullDown(m))
   271  		})
   272  	})
   273  
   274  	t.Run("異常系", func(t *testing.T) {
   275  		t.Parallel()
   276  
   277  		t.Run("Version取得がErrNilVersion以外で失敗した場合はエラーを返す", func(t *testing.T) {
   278  			t.Parallel()
   279  
   280  			ctrl := gomock.NewController(t)

   282  			m.EXPECT().Version().Return(uint(0), false, errBoom)

   285  		})
   286  
   287  		t.Run("dirty時のForceが失敗した場合はエラーを返す", func(t *testing.T) {
   288  			t.Parallel()
   289  
   290  			ctrl := gomock.NewController(t)
   291  			m := mock_migrate.NewMockMigrator(ctrl)
   292  			m.EXPECT().Version().Return(uint(5), true, nil)
   293  			m.EXPECT().Force(5).Return(errBoom)
   294  
   295  			require.ErrorIs(t, executeMigrateFullDown(m), errBoom)
   296  		})
   297  
   298  		t.Run("dirty時にバージョンがintへ変換できない場合はエラーを返す", func(t *testing.T) {
   299  			t.Parallel()
   300  
   301  			ctrl := gomock.NewController(t)
   302  			m := mock_migrate.NewMockMigrator(ctrl)
   303  			// MaxInt を超える uint は safecast でオーバーフロー検出され、Force/Down へ進まない。
   304  			m.EXPECT().Version().Return(uint(math.MaxUint64), true, nil)
   305  
   306  			require.Error(t, executeMigrateFullDown(m))
   307  		})
   308  	})
