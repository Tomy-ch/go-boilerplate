    14  // MigrateDownRun は、データベースマイグレーションを Down 方向に適用します。
    15  // steps=0 なら全マイグレーションをロールバックし、正数なら段数分だけ戻します。steps が負の場合はエラーを返します。dirty 状態の DB は Force で整合を取ってから Down します。
    16  func MigrateDownRun(steps int, database string, logger logging.Logger, newMigrator MigratorFactory) error {
    17  	// CLI 起動系のため trace span は無い。context.Background() を用いる。
    18  	ctx := context.Background()
    19  
    20  	if steps < 0 {
    21  		// 負値を許すと符号反転で Up 方向へ進んでしまうため、Down コマンドでは弾きます。
    22  		err := xerrors.New(fmt.Sprintf("steps must be zero or positive, got %d", steps))
    23  		logger.Named("migrateDownRun").Error(ctx, "invalid steps", logging.Error(logging.ErrorKey, err))
    24  		return err
    25  	}
    26  
    27  	m, err := newMigrator(database)
    28  	if err != nil {
    29  		logger.Named("migrateDownRun.buildMigrateInstance").Error(ctx, "failed to create migrate instance",
    30  			logging.Error(logging.ErrorKey, err),
    31  		)
    32  		return err
    33  	}
    34  
    35  	if steps == 0 {
    36  		logger.Named("migrateDownRun").Info(ctx, "running full migration down")
    37  		if err := executeMigrateFullDown(m); err != nil {
    38  			logger.Named("migrateDownRun.executeMigrateFullDown").Error(ctx, "down migration failed",
    39  				logging.Error(logging.ErrorKey, err),
    40  			)
    41  			return err
    42  		}
    43  	} else {
    44  		logger.Named("migrateDownRun").Info(ctx, "running migration down steps", logging.Int("steps", steps))
    45  		if err := executeMigrateDownSteps(m, steps); err != nil {
    46  			logger.Named("migrateDownRun.executeMigrateDownSteps").Error(ctx, "down migration steps failed",
    47  				logging.Error(logging.ErrorKey, err),
    48  			)
    49  			return err
    50  		}
    51  	}
    52  
    53  	logger.Named("migrateDownRun").Info(ctx, "✅ migration down completed")
    54  	return nil
    55  }

    57  // executeMigrateDownSteps は、現在位置から steps 段だけ Down します。無変更（ErrNoChange）は成功扱いです。
    58  func executeMigrateDownSteps(m Migrator, steps int) error {
    59  	// golang-migrate の Steps は負数を渡すとその段数だけ Down するため、検証済みの正値を反転します。
    60  	if err := m.Steps(-steps); err != nil && !xerrors.Is(err, migrate.ErrNoChange) {
    61  		return err
    62  	}
    63  	return nil
    64  }

    66  // executeMigrateFullDown は、マイグレーションを全てダウングレードして、DBを初期状態に戻します。
    67  func executeMigrateFullDown(m Migrator) error {
    68  	// dirty 状態のままでは Down できないため、現在バージョンで整合を取り直してから巻き戻します。
    69  	v, dirty, err := m.Version()
    70  	if err != nil && !xerrors.Is(err, migrate.ErrNilVersion) {
    71  		return err
    72  	}
    73  	if dirty {
    74  		safeValue, err := safecast.UintToInt(v)
    75  		if err != nil {
    76  			return err
    77  		}
    78  		if err := m.Force(safeValue); err != nil {
    79  			return err
    80  		}
    81  	}
    82  	if err := m.Down(); err != nil && !xerrors.Is(err, migrate.ErrNoChange) {
    83  		return err
    84  	}
    85  	return nil
    86  }
