    14  // MigrateDownRun は、マイグレーションをダウングレードするための実行関数です。
    15  func MigrateDownRun(steps int, database string, logger logging.Logger, newMigrator MigratorFactory) error {
    16  	if steps < 0 {
    17  		// 負値を許すと符号反転で Up 方向へ進んでしまうため、Down コマンドでは弾きます。
    18  		err := xerrors.New(fmt.Sprintf("steps must be zero or positive, got %d", steps))
    19  		logger.Named("migrateDownRun").Error("invalid steps", logging.Error(logging.ErrorKey, err))
    20  		return err
    21  	}
    22  
    23  	m, err := newMigrator(database)
    24  	if err != nil {
    25  		logger.Named("migrateDownRun.buildMigrateInstance").Error("failed to create migrate instance",
    26  			logging.Error(logging.ErrorKey, err),
    27  		)
    28  		return err
    29  	}
    30  
    31  	if steps == 0 {
    32  		logger.Named("migrateDownRun").Info("running full migration down")
    33  		if err := executeMigrateFullDown(m); err != nil {
    34  			logger.Named("migrateDownRun.executeMigrateFullDown").Error("down migration failed",
    35  				logging.Error(logging.ErrorKey, err),
    36  			)
    37  			return err
    38  		}
    39  	} else {
    40  		logger.Named("migrateDownRun").Info("running migration down steps", logging.Int("steps", steps))
    41  		if err := executeMigrateDownSteps(m, steps); err != nil {
    42  			logger.Named("migrateDownRun.executeMigrateDownSteps").Error("down migration steps failed",
    43  				logging.Error(logging.ErrorKey, err),
    44  			)
    45  			return err
    46  		}
    47  	}
    48  
    49  	logger.Named("migrateDownRun").Info("✅ migration down completed")
    50  	return nil
    51  }

    53  // executeMigrateDownSteps は、現在位置から steps 段だけ Down します。無変更（ErrNoChange）は成功扱いです。
    54  func executeMigrateDownSteps(m Migrator, steps int) error {
    55  	// golang-migrate の Steps は負数を渡すとその段数だけ Down するため、検証済みの正値を反転します。
    56  	if err := m.Steps(-steps); err != nil && !errors.Is(err, migrate.ErrNoChange) {
    57  		return err
    58  	}
    59  	return nil
    60  }

    62  // executeMigrateFullDown は、マイグレーションを全てダウングレードして、DBを初期状態に戻します。
    63  func executeMigrateFullDown(m Migrator) error {
    64  	// dirty 状態のままでは Down できないため、現在バージョンで整合を取り直してから巻き戻します。
    65  	v, dirty, err := m.Version()
    66  	if err != nil && !errors.Is(err, migrate.ErrNilVersion) {
    67  		return err
    68  	}
    69  	if dirty {
    70  		safeValue, err := safecast.UintToInt(v)
    71  		if err != nil {
    72  			return err
    73  		}
    74  		if err := m.Force(safeValue); err != nil {
    75  			return err
    76  		}
    77  	}
    78  	if err := m.Down(); err != nil && !errors.Is(err, migrate.ErrNoChange) {
    79  		return err
    80  	}
    81  	return nil
    82  }
