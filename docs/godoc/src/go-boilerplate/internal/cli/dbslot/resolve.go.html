   110  // Resolve は、git 文脈と .gobp-db-slot から解決済みの値を組み立てます。
   111  // git リポジトリなのに構成を読めなかった場合だけエラーにします（inspectGit を参照）。
   112  func (r *Resolver) Resolve(ctx context.Context) (Values, error) {
   113  	gitCtx, err := r.inspectGit(ctx)
   114  	if err != nil {
   115  		return Values{}, err
   116  	}
   117  
   118  	slot := readSlotFile(filepath.Join(r.cfg.Root, ".gobp-db-slot"))
   119  
   120  	values := Values{
   121  		Git:        gitCtx,
   122  		SlotHeld:   slot["DB_NAME_LOCAL"] != "",
   123  		DBLocal:    orDefault(slot["DB_NAME_LOCAL"], defaultDBLocal),
   124  		DBTest:     orDefault(slot["DB_NAME_TEST"], defaultDBTest),
   125  		AppProject: orDefault(slot["SERVE_PROJECT"], "gobp-app-"+filepath.Base(r.cfg.Root)),
   126  		AuthIssuer: "http://localhost:" + orDefault(slot["MOCK_AUTH_HOST_PORT"], strconv.Itoa(r.cfg.MockAuthBase)),
   127  	}
   128  
   129  	// 共有インフラを奪い合う相手が居るのはリンク worktree のときだけ。単一 checkout では
   130  	// compose 本来の「up は定義変更へ再収束する」契約を捨てる理由がないため空にします。
   131  	if gitCtx == GitLinkedWorktree {
   132  		values.InfraNoRecreate = noRecreateFlag
   133  	}
   134  
   135  	return values, nil
   136  }

   138  // RequireOwner は、所有データベースを持たない状態（リンク worktree かつスロット未取得）を
   139  // 検出してエラーにします。主 checkout・ツールランナー内・CI は素通りします。
   140  //
   141  // 素通りさせる 3 つのうち、ツールランナー内と CI は「git 文脈を確かめられない」側に属します。
   142  // git を引けないことを一律に素通りの理由にすると、リンク worktree で git が失敗したときまで
   143  // 素通りし、1 つのデータベースを 2 箇所から触る状態へ黙って落ちます（pool.mk の不変条件）。
   144  // そのため「git 実行ファイルが無い」「リポジトリでない」は素通り、「リポジトリなのに読めない」は
   145  // 失敗として扱います。
   146  func (r *Resolver) RequireOwner(ctx context.Context) error {
   147  	values, err := r.Resolve(ctx)
   148  	if err != nil {
   149  		return err
   150  	}
   151  
   152  	if values.Git != GitLinkedWorktree || values.SlotHeld {
   153  		return nil
   154  	}
   155  
   156  	_, _ = fmt.Fprint(r.out, "❌ この worktree は DB スロットを取得していないため、所有するデータベースがありません。\n"+
   157  		"   1 つのデータベースを複数の checkout から触らないよう、既定の local / test へは\n"+
   158  		"   フォールバックしません。\n"+
   159  		"   → make slot-acquire でスロットを取得してください（make slot-status で空きを確認）。\n")
   160  
   161  	return errNoDatabaseOwner
   162  }

   189  // inspectGit は、git 文脈を判定します。
   190  // git を引けなかったときは、.git の有無で「リポジトリでない」と「リポジトリなのに読めない」を
   191  // 分けます。git のエラーメッセージはロケールで変わるため、メッセージは判定材料にしません。
   192  func (r *Resolver) inspectGit(ctx context.Context) (GitContext, error) {
   193  	if err := r.probe.LookGit(); err != nil {
   194  		return GitUnavailable, nil //nolint:nilerr // git が無いのは異常ではなく「判定できない文脈」であり、素通りが正しい
   195  	}
   196  
   197  	out, err := r.probe.GitDirs(ctx, r.cfg.Root)
   198  	if err != nil {
   199  		if r.probe.HasGitEntry(r.cfg.Root) {
   200  			return 0, xerrors.Wrap(errGitLayoutUnreadable, err.Error())
   201  		}
   202  
   203  		return GitNoRepository, nil
   204  	}
   205  
   206  	fields := strings.Fields(out)
   207  	if len(fields) != gitDirsFields {
   208  		return 0, xerrors.Wrap(errGitLayoutUnreadable, "unexpected `git rev-parse` output: "+strconv.Quote(out))
   209  	}
   210  
   211  	if r.absDir(fields[0]) == r.absDir(fields[1]) {
   212  		return GitMainCheckout, nil
   213  	}
   214  
   215  	return GitLinkedWorktree, nil
   216  }

   227  // RenderEnv は、解決済みの値を make のレシピが eval できる KEY=VALUE へ整形します。
   228  func RenderEnv(v Values) string {
   229  	pairs := [][2]string{
   230  		{"GOBP_GIT_CONTEXT", v.Git.String()},
   231  		{"DB_LOCAL", v.DBLocal},
   232  		{"DB_TEST", v.DBTest},
   233  		{"APP_PROJECT", v.AppProject},
   234  		{"AUTH_ISSUER", v.AuthIssuer},
   235  		{"INFRA_NO_RECREATE", v.InfraNoRecreate},
   236  	}
   237  
   238  	var sb strings.Builder
   239  	for _, kv := range pairs {
   240  		fmt.Fprintf(&sb, "%s='%s'\n", kv[0], strings.ReplaceAll(kv[1], "'", `'\''`))
   241  	}
   242  
   243  	return sb.String()
   244  }

   246  // RenderValues は、解決済みの値を人間向けの一覧へ整形します。
   247  func RenderValues(v Values) string {
   248  	var sb strings.Builder
   249  
   250  	sb.WriteString("\n--- この checkout の解決済みの値 ---\n")
   251  	fmt.Fprintf(&sb, "git context       : %s\n", v.Git)
   252  	fmt.Fprintf(&sb, "slot held         : %s\n", yesNo(v.SlotHeld))
   253  	fmt.Fprintf(&sb, "DB_LOCAL          : %s\n", v.DBLocal)
   254  	fmt.Fprintf(&sb, "DB_TEST           : %s\n", v.DBTest)
   255  	fmt.Fprintf(&sb, "APP_PROJECT       : %s\n", v.AppProject)
   256  	fmt.Fprintf(&sb, "AUTH_ISSUER       : %s\n", v.AuthIssuer)
   257  	fmt.Fprintf(&sb, "INFRA_NO_RECREATE : %s\n", orDefault(v.InfraNoRecreate, "（渡さない）"))
   258  
   259  	return sb.String()
   260  }

   262  // realGitProbe は、ホストの git を使う GitProbe を返します。
   263  func realGitProbe() GitProbe {
   264  	return GitProbe{
   265  		LookGit: func() error {
   266  			_, err := exec.LookPath("git")
   267  
   268  			return err
   269  		},
   270  		GitDirs: func(ctx context.Context, root string) (string, error) {
   271  			cmd := exec.CommandContext(ctx, "git", "rev-parse", "--git-dir", "--git-common-dir")
   272  			cmd.Dir = root
   273  			out, err := cmd.Output()
   274  
   275  			return string(out), err
   276  		},
   277  		HasGitEntry: hasGitEntry,
   278  	}
   279  }

   281  // hasGitEntry は、dir か その祖先に .git（ディレクトリ、またはリンク worktree の .git ファイル）が
   282  // あるかを返します。git の失敗が「リポジトリでない」ためか「読み取れなかった」ためかを、
   283  // git のエラーメッセージに頼らず区別するために使います（メッセージはロケールで変わります）。
   284  func hasGitEntry(dir string) bool {
   285  	for {
   286  		if _, err := os.Lstat(filepath.Join(dir, ".git")); err == nil {
   287  			return true
   288  		}
   289  
   290  		parent := filepath.Dir(dir)
   291  		if parent == dir {
   292  			return false
   293  		}
   294  
   295  		dir = parent
   296  	}
   297  }

   299  // readSlotFile は、.gobp-db-slot の KEY=VALUE を読み取ります。
   300  // ファイルが無い状態は「スロット未取得」という正常な状態なので、空の map を返します。
   301  func readSlotFile(path string) map[string]string {
   302  	values := make(map[string]string)
   303  
   304  	b, err := os.ReadFile(path) //nolint:gosec // 読むのは自 worktree の .gobp-db-slot のみ
   305  	if err != nil {
   306  		return values
   307  	}
   308  
   309  	for line := range strings.SplitSeq(string(b), "\n") {
   310  		key, value, found := strings.Cut(strings.TrimSpace(line), "=")
   311  		if !found {
   312  			continue
   313  		}
   314  
   315  		values[key] = value
   316  	}
   317  
   318  	return values
   319  }
