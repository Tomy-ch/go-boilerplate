    59  func TestRunServer(t *testing.T) {
    60  	t.Parallel()
    61  
    62  	cfg := config.MockConfigForTest(t)
    63  	appCfg := config.NewApplicationConfig(cfg)
    64  	shutdownTimeout := appCfg.ShutdownTimeout()
    65  
    66  	t.Run("正常系", func(t *testing.T) {
    67  		t.Parallel()
    68  
    69  		t.Run("シグナル受信後に起動しグレースフルシャットダウンが実行される", func(t *testing.T) {
    70  			t.Parallel()
    71  
    72  			ctx, cancel := context.WithCancel(context.Background())
    73  
    74  			var startCalled bool
    75  			//nolint:unparam // RunServer のシグネチャ func(context.Context) error に合わせる必要がある
    76  			start := func(context.Context) error {
    77  				startCalled = true
    78  				return nil
    79  			}
    80  
    81  			var (
    82  				stopMetricsCalled bool
    83  				stopMetricsCtx    context.Context
    84  			)
    85  			stopMetrics := func(c context.Context) {
    86  				stopMetricsCalled = true
    87  				stopMetricsCtx = c
    88  			}
    89  
    90  			var (
    91  				stopDeadline    time.Time
    92  				stopHasDeadline bool
    93  				stopCtx         context.Context
    94  			)
    95  			//nolint:unparam // RunServer のシグネチャ func(context.Context) error に合わせる必要がある
    96  			stop := func(c context.Context) error {
    97  				stopCtx = c
    98  				stopDeadline, stopHasDeadline = c.Deadline()
    99  				return nil
   100  			}
   101  
   102  			errCh := make(chan error, 1)
   103  			go func() {
   104  				errCh <- RunServer(ctx, shutdownTimeout, start, stop, stopMetrics)
   105  			}()
   106  
   107  			// 起動から一定時間「稼働」させてからシグナル受信相当のキャンセルを行う。
   108  			// 停止タイムアウトが「停止開始時点」から計測されること（起動時点ではない）を検証するため、
   109  			// 稼働時間（120ms）を許容誤差（60ms）より十分大きく取る。
   110  			time.Sleep(120 * time.Millisecond)
   111  			stopStart := time.Now()
   112  			cancel()
   113  
   114  			require.NoError(t, <-errCh)
   115  
   116  			assert.True(t, startCalled, "startApp が呼ばれること")
   117  			assert.True(t, stopMetricsCalled, "stopMetrics が呼ばれること")
   118  			require.True(t, stopHasDeadline, "stopApp は期限付き context を受け取ること")
   119  			// 停止用 context の deadline が「停止開始時点 + ShutdownTimeout」付近であること
   120  			// （= 稼働時間に消費されていないこと。b0a8e21 の回帰防止）。
   121  			assert.WithinDuration(t, stopStart.Add(shutdownTimeout), stopDeadline, 60*time.Millisecond)
   122  			// メトリクス停止とアプリ停止は同一の停止用 context を共有すること。
   123  			assert.Equal(t, stopCtx, stopMetricsCtx)
   124  		})
