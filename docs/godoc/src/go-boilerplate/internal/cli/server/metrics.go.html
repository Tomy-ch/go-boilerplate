    22  // metricsServer は、メトリクス/pprof（DefaultServeMux）を公開する補助 HTTP サーバーを生成します。
    23  func metricsServer(mtcCfg *config.MetricsConfig) *http.Server {
    24  	return &http.Server{
    25  		// pprof / metrics は DefaultServeMux に登録される前提で補助サーバーとして公開します。
    26  		Addr:              mtcCfg.Host() + ":" + strconv.Itoa(mtcCfg.Port()),
    27  		Handler:           http.DefaultServeMux,
    28  		ReadHeaderTimeout: readHeaderTimeout,
    29  		ReadTimeout:       readTimeout,
    30  		WriteTimeout:      writeTimeout,
    31  		IdleTimeout:       idleTimeout,
    32  	}
    33  }

    35  // NewMetricsServer は、メトリクスサーバーの開始および終了関数を生成します。
    36  func NewMetricsServer(mtcCfg *config.MetricsConfig, logger logging.Logger) (func(), func(ctx context.Context)) {
    37  	metricsSrv := metricsServer(mtcCfg)
    38  
    39  	start := func() {
    40  		go func() {
    41  			logListenError(logger, metricsSrv.ListenAndServe())
    42  		}()
    43  	}
    44  	end := func(ctx context.Context) {
    45  		// 停止失敗でアプリ本体を巻き込まないよう、panic せずログに留める（start 側の goroutine と同方針）。
    46  		if err := metricsSrv.Shutdown(ctx); err != nil {
    47  			logger.Named("metrics.Shutdown").Error(ctx, "metrics server shutdown error", logging.Error(logging.ErrorKey, err))
    48  		}
    49  	}
    50  	return start, end
    51  }

    53  // logListenError は、補助サーバーの ListenAndServe 戻り値を判定し、正常停止(ErrServerClosed)以外の
    54  // エラーのみをログ記録します。bind 失敗等でアプリ本体を巻き込まないよう、panic せずログに留めます。
    55  func logListenError(logger logging.Logger, err error) {
    56  	if err != nil && !xerrors.Is(err, http.ErrServerClosed) {
    57  		// start goroutine 内で request ctx を持たないため context.Background() を用いる。
    58  		logger.Named("metrics.ListenAndServe").Error(context.Background(), "metrics server error", logging.Error(logging.ErrorKey, err))
    59  	}
    60  }
