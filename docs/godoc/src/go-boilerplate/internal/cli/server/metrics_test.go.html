    97  			mtcCfg.SetMetricsPort(t, 0)
    98  
    99  			start, end := NewMetricsServer(mtcCfg, logging.NewTestLogger(t))
   100  			require.NotNil(t, start)
   101  			require.NotNil(t, end)
   102  
   103  			assert.NotPanics(t, func() {
   104  				start()
   105  				end(context.Background())
   106  			})
   107  		})
   108  
   109  		t.Run("起動していないサーバーへの停止でもpanicしない", func(t *testing.T) {
   110  			t.Parallel()
   111  
   112  			cfg := config.MockConfigForTest(t)
   113  			mtcCfg := config.NewMetricsConfig(cfg)
   114  
   115  			_, end := NewMetricsServer(mtcCfg, logging.NewTestLogger(t))
   116  
   117  			// 起動していないサーバーへの Shutdown はエラーにならず、panic しないこと
   118  			// （goroutine 内 panic 廃止の方針に沿う）。
