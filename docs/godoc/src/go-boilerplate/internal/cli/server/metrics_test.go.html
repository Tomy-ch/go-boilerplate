   163  // blockingHandler は、最初の呼び出しで entered を通知し release まで待機するハンドラを返します。
   164  func blockingHandler(entered chan<- struct{}, release <-chan struct{}) http.HandlerFunc {
   165  	return func(w http.ResponseWriter, _ *http.Request) {
   166  		select {
   167  		case entered <- struct{}{}:
   168  		default:
   169  		}
   170  		<-release
   171  		w.WriteHeader(http.StatusOK)
   172  	}
   173  }

   175  // reservePort は、host と同じ family で bind 可能なポートを予約し確定ポートを返します。
   176  func reservePort(t *testing.T, host string) int {
   177  	t.Helper()
   178  	ln, err := (&net.ListenConfig{}).Listen(t.Context(), "tcp", host+":0")
   179  	require.NoError(t, err)
   180  	_, portStr, err := net.SplitHostPort(ln.Addr().String())
   181  	require.NoError(t, err)
   182  	port, err := strconv.Atoi(portStr)
   183  	require.NoError(t, err)
   184  	require.NoError(t, ln.Close())
   185  	return port
   186  }

   188  // driveInFlight は、url へ接続してハンドラを進行中（非アイドル接続）にし、進入を待ちます。
   189  func driveInFlight(t *testing.T, url string, entered <-chan struct{}, release chan struct{}) {
   190  	t.Helper()
   191  	go sendUntilConnected(url)
   192  	select {
   193  	case <-entered:
   194  	case <-time.After(3 * time.Second):
   195  		close(release)
   196  		t.Fatal("進行中ハンドラへ到達しなかった")
   197  	}
   198  }

   200  // sendUntilConnected は、bind 完了まで接続を試み、接続できたら応答を読み切ります。
   201  func sendUntilConnected(url string) {
   202  	deadline := time.Now().Add(3 * time.Second)
   203  	for time.Now().Before(deadline) {
   204  		req, err := http.NewRequestWithContext(context.Background(), http.MethodGet, url, nil)
   205  		if err != nil {
   206  			return
   207  		}
   208  		res, err := http.DefaultClient.Do(req)
   209  		if err == nil {
   210  			_, _ = io.Copy(io.Discard, res.Body)
   211  			_ = res.Body.Close()
   212  			return
   213  		}
   214  		time.Sleep(5 * time.Millisecond)
   215  	}
   216  }
