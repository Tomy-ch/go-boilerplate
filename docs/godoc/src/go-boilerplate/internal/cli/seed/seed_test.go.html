   132  				db.EXPECT().Exec(gomock.Any(), "SELECT b;").Return(pgconn.CommandTag{}, nil),
   133  			)
   134  
   135  			err := runSeeds(context.Background(), fsys, db, logging.NewTestLogger(t), []string{"b.sql", "a.sql"})
   136  			require.NoError(t, err)
   137  		})
   138  	})
