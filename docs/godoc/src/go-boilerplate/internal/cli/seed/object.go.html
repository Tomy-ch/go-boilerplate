    42  // RunObjectSeed は、objectSeedPlace 配下のファイルをオブジェクトストレージへ保存します。
    43  // objectSeedPlace からの相対パスがそのままオブジェクトキーになるため、置いたディレクトリ構造が
    44  // キーの構造になります（例 storage/seed/products/a.webp → products/a.webp）。
    45  //
    46  // endpoint が空の環境では何もしません。対象が 1 件も無い場合も同様で、いずれも成功として扱います。
    47  func RunObjectSeed(logger logging.Logger, fsys fs.FS, endpoint string, put PutObjectFunc) error {
    48  	ctx := context.Background()
    49  	log := logger.Named("objectSeedRun")
    50  
    51  	// endpoint が空は SDK 既定のエンドポイント解決、すなわち実在の AWS S3 を指す。シード用の
    52  	// ファイルを実環境のバケットへ流し込まないよう、互換エンドポイントを明示している場合だけ投入する。
    53  	if endpoint == "" {
    54  		log.Info(ctx, "object storage endpoint is empty, skipping object seed")
    55  		return nil
    56  	}
    57  
    58  	files, err := collectSeedObjects(ctx, log, fsys)
    59  	if err != nil {
    60  		return err
    61  	}
    62  	if len(files) == 0 {
    63  		log.Info(ctx, "no seed object found, skipping object seed")
    64  		return nil
    65  	}
    66  
    67  	return putSeedObjects(ctx, log, fsys, put, files)
    68  }

    70  // collectSeedObjects は、投入対象のファイルを昇順で列挙します。
    71  // ドットファイルは対象外とし、対応していない拡張子は警告して除外します。
    72  func collectSeedObjects(ctx context.Context, log logging.Logger, fsys fs.FS) ([]string, error) {
    73  	entries, err := fsys.Glob(objectSeedPlace + "/*/*")
    74  	if err != nil {
    75  		log.Error(ctx, "failed to glob seed objects", logging.Error(logging.ErrorKey, err))
    76  		return nil, err
    77  	}
    78  
    79  	files := make([]string, 0, len(entries))
    80  	for _, e := range entries {
    81  		name := path.Base(e)
    82  		if strings.HasPrefix(name, ".") {
    83  			continue
    84  		}
    85  		if _, ok := objectSeedContentTypes[strings.ToLower(path.Ext(name))]; !ok {
    86  			log.Warn(ctx, "unsupported seed object extension, skipping", logging.String("file", e))
    87  			continue
    88  		}
    89  		files = append(files, e)
    90  	}
    91  	sort.Strings(files)
    92  
    93  	return files, nil
    94  }

    96  // putSeedObjects は、対象を順に投入します。1 件の失敗で打ち切らず、全件を試みたうえで
    97  // エラーをまとめて返します。
    98  func putSeedObjects(ctx context.Context, log logging.Logger, fsys fs.FS, put PutObjectFunc, files []string) error {
    99  	var seedErr error
   100  	for _, f := range files {
   101  		if err := putSeedObject(ctx, log, fsys, put, f); err != nil {
   102  			seedErr = xerrors.Join(seedErr, err)
   103  		}
   104  	}
   105  	if seedErr != nil {
   106  		return seedErr
   107  	}
   108  	log.Info(ctx, "✅ object seeding completed", logging.Int("count", len(files)))
   109  
   110  	return nil
   111  }

   113  // putSeedObject は、1 つのファイルを相対パスと同じキーで保存します。
   114  func putSeedObject(ctx context.Context, log logging.Logger, fsys fs.FS, put PutObjectFunc, filePath string) error {
   115  	data, err := fsys.ReadFile(filePath)
   116  	if err != nil {
   117  		log.Error(ctx, "failed to read seed object", logging.String("file", filePath), logging.Error(logging.ErrorKey, err))
   118  		return err
   119  	}
   120  
   121  	key := strings.TrimPrefix(filePath, objectSeedPlace+"/")
   122  	if perr := put(ctx, ObjectToPut{
   123  		Key:         key,
   124  		Body:        data,
   125  		ContentType: objectSeedContentTypes[strings.ToLower(path.Ext(filePath))],
   126  	}); perr != nil {
   127  		log.Error(ctx, "failed to put seed object", logging.String("key", key), logging.Error(logging.ErrorKey, perr))
   128  		return perr
   129  	}
   130  
   131  	log.Info(ctx, "seed object put", logging.String("key", key))
   132  
   133  	return nil
   134  }
