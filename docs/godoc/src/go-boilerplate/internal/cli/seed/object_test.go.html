    22  // recordingPut は、保存を依頼されたオブジェクトを recorded へ記録する PutObjectFunc を返します。
    23  func recordingPut(recorded *[]ObjectToPut) PutObjectFunc {
    24  	return func(_ context.Context, obj ObjectToPut) error {
    25  		*recorded = append(*recorded, obj)
    26  		return nil
    27  	}
    28  }

    31  func failingPut(err error) PutObjectFunc {
    32  	return func(context.Context, ObjectToPut) error { return err }
