    62  func (osFileSystem) ListSubDirNames(base string) ([]string, error) {
    63  	ents, err := os.ReadDir(base)
    64  	if err != nil {
    65  		return nil, err
    66  	}
    67  	dirs := make([]string, 0, len(ents))
    68  	for _, e := range ents {
    69  		if e.IsDir() {
    70  			dirs = append(dirs, e.Name())
    71  		}
    72  	}
    73  	sort.Strings(dirs)
    74  	return dirs, nil

    91  func (osFileSystem) FindSQLFiles(dir string) ([]string, error) {
    92  	var files []string
    93  	err := filepath.WalkDir(dir, func(path string, d os.DirEntry, err error) error {
    94  		if err != nil {
    95  			return err
    96  		}
    97  		if d.IsDir() {
    98  			return nil
    99  		}
   100  		if filepath.Ext(path) == ".sql" {
   101  			files = append(files, path)
   102  		}
   103  		return nil
   104  	})
   105  	if err != nil {
   106  		return nil, err
   107  	}
   108  	sort.Strings(files)
   109  	return files, nil
   110  }

   137  // RunMerge は、DMLファイルをマージして、カテゴリごとに単一ファイルにまとめます。
   138  func RunMerge(ctx context.Context, gen *Generator, targetType string) error {
   139  	logger := gen.logger
   140  
   141  	categories, err := gen.fs.ListSubDirNames(gen.dmlTypeRootAbs(targetType))
   142  	if err != nil {
   143  		logger.CallerSkip(gen.callerSkipCount).Named("mergedml.listDirs").Error("failed to list directories",
   144  			logging.Error(logging.ErrorKey, err),
   145  		)
   146  		return err
   147  	}
   148  	if len(categories) == 0 {
   149  		logger.CallerSkip(gen.callerSkipCount).Named("mergedml.NoCategories").Info(
   150  			"no categories found under dml directory, cleanup only",
   151  			logging.String("dml_path", gen.dmlRootDir+targetType),
   152  		)
   153  
   154  		// ★ 0件でも stale を消す（=このtypeの生成物を全消しに近い挙動）
   155  		if err := gen.cleanupStaleGeneratedFiles(nil, targetType); err != nil {
   156  			logger.CallerSkip(gen.callerSkipCount).Named("mergedml.cleanupStaleGeneratedFiles").Error(
   157  				"failed to cleanup stale generated sql files",
   158  				logging.Error(logging.ErrorKey, err),
   159  			)
   160  			return err
   161  		}
   162  		return nil
   163  	}
   164  
   165  	eg, egCtx := errgroup.WithContext(ctx)
   166  	sem := semaphore.NewWeighted(int64(resolveConcurrencyConst()))
   167  
   168  	for _, category := range categories {
   169  		cat := category
   170  		eg.Go(func() error {
   171  			if err := sem.Acquire(egCtx, 1); err != nil {
   172  				return err
   173  			}
   174  			defer sem.Release(1)
   175  
   176  			return gen.buildCategorySQLFile(cat, targetType)
   177  		})
   178  	}
   179  
   180  	if err := eg.Wait(); err != nil {
   181  		gen.logger.CallerSkip(gen.callerSkipCount).Named("mergedml.buildMergedQueries").Error("failed to build merged sql files",
   182  			logging.Error(logging.ErrorKey, err),
   183  		)
   184  		return err
   185  	}
   186  
   187  	if err := gen.cleanupStaleGeneratedFiles(categories, targetType); err != nil {
   188  		gen.logger.CallerSkip(gen.callerSkipCount).Named("mergedml.cleanupStaleGeneratedFiles").Error(
   189  			"failed to cleanup stale generated sql files",
   190  			logging.Error(logging.ErrorKey, err),
   191  		)
   192  		return err
   193  	}
   194  
   195  	return nil
   196  }

   203  // buildCategorySQLFile は、指定されたカテゴリのSQLファイルを連結して1つのSQLファイルにまとめます。
   204  func (g *Generator) buildCategorySQLFile(category, targetType string) error {
   205  	// 入力走査も workDir 起点で統一する。CWD 起点だと CWD != workDir のとき走査結果が 0 件になり、
   206  	// 「入力なし」分岐に入って workDir 配下の生成物を誤って削除してしまうため。
   207  	dmlDir := filepath.Join(g.workDir, g.dmlRootDir, targetType, category)
   208  
   209  	files, err := g.fs.FindSQLFiles(dmlDir)
   210  	if err != nil {
   211  		return err
   212  	}
   213  
   214  	outName := fmt.Sprintf("%s_%s.gen.sql", category, targetType) // 例: prefecture_repository.gen.sql
   215  	// 出力先も workDir 起点で統一し、相対/絶対の二系統を排除する。
   216  	dstPath := filepath.Join(g.workDir, g.genRootDir, outName)
   217  
   218  	if len(files) == 0 {
   219  		// 「カテゴリは存在するが SQL が空」の場合は、生成物を消すのが最新状態とみなします。
   220  		if err := g.ensureUnderDir(dstPath); err != nil {
   221  			return err
   222  		}
   223  
   224  		if err := g.fs.Remove(dstPath); err != nil && !os.IsNotExist(err) {
   225  			return err
   226  		}
   227  
   228  		g.logger.CallerSkip(g.callerSkipCount).Named("mergedml.buildCategorySQLFile").Info(
   229  			"no sql files found, remove generated file if exists",
   230  			logging.String("category", category),
   231  			logging.String("type", targetType),
   232  			logging.String("dst", dstPath),
   233  		)
   234  		return nil
   235  	}
   236  
   237  	g.logger.CallerSkip(g.callerSkipCount).Named("mergedml.buildCategorySQLFile").Info("build merged sql for sqlc",
   238  		logging.String("category", category),
   239  		logging.String("type", targetType),
   240  		logging.String("dst", dstPath),
   241  		logging.Int("files", len(files)),
   242  	)
   243  
   244  	if err := g.ensureUnderDir(dstPath); err != nil {
   245  		return err
   246  	}
   247  
   248  	// 連結結果はメモリ上に構築し、成功時のみ一括書き込みします（部分生成物を残さない）。
   249  	var buf bytes.Buffer
   250  	for _, fpath := range files {
   251  		// 由来が分かるように見出しを入れる（sqlcはSQLコメントなら無害）
   252  		_, _ = fmt.Fprintf(&buf, "\n-- === source: %s ===\n", filepath.ToSlash(fpath))
   253  
   254  		b, err := g.fs.ReadFile(fpath)
   255  		if err != nil {
   256  			return err
   257  		}
   258  		buf.Write(b)
   259  
   260  		// ファイル末尾に改行が無いケースでも連結が壊れないように
   261  		if len(b) > 0 && b[len(b)-1] != '\n' {
   262  			buf.WriteByte('\n')
   263  		}
   264  	}
   265  
   266  	return g.fs.WriteFile(dstPath, buf.Bytes(), genFilePerm)
   267  }

   269  // ensureUnderDir は path が baseDir 配下かを検証します。
   270  func (g *Generator) ensureUnderDir(path string) error {
   271  	// 誤ったパス解決や path traversal により、想定外の場所を操作しないようにします。
   272  	absPath, err := filepath.Abs(path)
   273  	if err != nil {
   274  		return err
   275  	}
   276  
   277  	absBase, err := filepath.Abs(filepath.Join(g.workDir, g.genRootDir))
   278  	if err != nil {
   279  		return err
   280  	}
   281  
   282  	rel, err := filepath.Rel(absBase, absPath)
   283  	if err != nil {
   284  		return err
   285  	}
   286  	rel = filepath.Clean(rel)
   287  	if rel == ".." || strings.HasPrefix(rel, ".."+string(os.PathSeparator)) {
   288  		return fmt.Errorf("path is outside of baseDir: path=%s base=%s", absPath, absBase)
   289  	}
   290  	return nil
   291  }

   293  // resolveConcurrencyConst は、SQLCの同時実行数を解決します。
   294  //
   295  //   - 上限: min(runtime.NumCPU(), maxSQLCConcurrency)
   296  //   - 下限: minSQLCConcurrency
   297  //
   298  // Docker/CIで全コア占有を避けるため、固定上限を設けています。
   299  func resolveConcurrencyConst() int {
   300  	return resolveConcurrency(runtime.NumCPU())
   301  }

   303  // resolveConcurrency は、CPU 数を [minSQLCConcurrency, maxSQLCConcurrency] にクランプして同時実行数を返します。
   304  func resolveConcurrency(numCPU int) int {
   305  	return max(min(numCPU, maxSQLCConcurrency), minSQLCConcurrency)

   308  func (g *Generator) cleanupStaleGeneratedFiles(categories []string, targetType string) error {
   309  	keep := make(map[string]struct{}, len(categories))
   310  	for _, cat := range categories {
   311  		keep[fmt.Sprintf("%s_%s.gen.sql", cat, targetType)] = struct{}{}
   312  	}
   313  
   314  	genAbs := filepath.Join(g.workDir, g.genRootDir) // /app/database/gen
   315  	names, err := g.fs.ListGenFileNames(genAbs)
   316  	if err != nil {
   317  		return err
   318  	}
   319  
   320  	suffix := fmt.Sprintf("_%s.gen.sql", targetType)
   321  
   322  	for _, name := range names {
   323  		if !strings.HasSuffix(name, suffix) {
   324  			continue
   325  		}
   326  
   327  		if _, ok := keep[name]; ok {
   328  			continue
   329  		}
   330  
   331  		full := filepath.Join(genAbs, name)
   332  
   333  		if err := g.ensureUnderDir(full); err != nil {
   334  			return err
   335  		}
   336  
   337  		if err := g.fs.Remove(full); err != nil {
   338  			return err
   339  		}
   340  
   341  		g.logger.CallerSkip(g.callerSkipCount).Named("mergedml.cleanupStaleGeneratedFiles").Info(
   342  			"remove stale generated sql",
   343  			logging.String("file", filepath.ToSlash(full)),
   344  			logging.String("type", targetType),
   345  		)
   346  	}
   347  
   348  	return nil
   349  }
