    78  // dumpSchema は、ダンプコマンドを実行してスキーマのDDLを取得し、sqlc 向けに整形して schema.gen.sql として保存します。
    79  func (g *Generator) dumpSchema(ctx context.Context, dbURL, password string) error {
    80  	args := append([]string{dbURL}, g.dumpArgs...)
    81  	env := []string{"PGPASSWORD=" + password}
    82  
    83  	g.logger.CallerSkip(g.callerSkipCount).Named(dumpSchemaLoggerName).Info(ctx, "start pg_dump schema",
    84  		logging.String("out", g.schemaRelPath),
    85  	)
    86  
    87  	out, err := g.runner.Output(ctx, g.workDir, env, g.dumpCommand, args)
    88  	if err != nil {
    89  		g.logger.CallerSkip(g.callerSkipCount).Named(dumpSchemaLoggerName).Warn(ctx, "pg_dump failed",
    90  			logging.String("out", g.schemaRelPath),
    91  			logging.Error(logging.ErrorKey, err),
    92  		)
    93  		return xerrors.Wrap(err, "pg_dump failed")
    94  	}
    95  
    96  	schemaAbs := filepath.Join(g.workDir, g.schemaRelPath)
    97  	if err := g.fs.WriteFile(schemaAbs, sanitizeSchema(out), g.permission); err != nil {
    98  		return xerrors.Wrap(err, "failed to write schema file")
    99  	}
   100  
   101  	g.logger.CallerSkip(g.callerSkipCount).Named(dumpSchemaLoggerName).Info(ctx, "pg_dump schema completed",
   102  		logging.String("out", g.schemaRelPath),
   103  	)
   104  
   105  	return nil
   106  }

   108  // sanitizeSchema は、pg_dump 出力を sqlc 向けに整形して返します。
   109  // trimPrefixes 一致のメタ行に加え、空行（空白のみ・元から空）も除去します（空行除去まで含むのは意図的）。
   110  func sanitizeSchema(raw []byte) []byte {
   111  	lines := strings.Split(string(raw), "\n")
   112  	out := make([]string, 0, len(lines))
   113  	for _, ln := range lines {
   114  		trim := strings.TrimSpace(ln)
   115  		for _, prefix := range trimPrefixes {
   116  			if strings.HasPrefix(trim, prefix) {
   117  				trim = ""
   118  				break
   119  			}
   120  		}
   121  		if trim == "" {
   122  			continue
   123  		}
   124  		out = append(out, ln)
   125  	}
   126  
   127  	return []byte(strings.Join(out, "\n"))
   128  }
