    36  func TestGenerator_dumpSchema(t *testing.T) {
    37  	t.Parallel()
    38  
    39  	schemaAbs := filepath.Join(testWorkDir, "database/gen/schema.gen.sql")
    40  
    41  	t.Run("正常系", func(t *testing.T) {
    42  		t.Parallel()
    43  
    44  		t.Run("pg_dumpの出力を整形してschemaファイルへ書き込む", func(t *testing.T) {
    45  			t.Parallel()
    46  			ctrl := gomock.NewController(t)
    47  			fs := mock_fs.NewMockFS(ctrl)
    48  			runner := mock_exec.NewMockRunner(ctrl)
    49  
    50  			// メタコマンド行と末尾空行を含む生ダンプが、整形後のバイト列で 1 度だけ書き込まれることを検証する。
    51  			out := []byte("\\connect mydb\nCREATE TABLE users (id int);\n")
    52  			wantWritten := []byte("CREATE TABLE users (id int);")
    53  			// DSN を先頭に dumpSubArgs を連結した引数と、PGPASSWORD を含む env が
    54  			// そのまま runner へ渡ることを具体値で検証する。
    55  			wantEnv := []string{"PGPASSWORD=pw"}
    56  			wantArgs := []string{"postgres://dsn", "--schema-only", "--no-owner", "--no-privileges", "--format=plain"}
    57  			runner.EXPECT().Output(gomock.Any(), testWorkDir, wantEnv, "pg_dump", wantArgs).Return(out, nil)
    58  			fs.EXPECT().WriteFile(schemaAbs, wantWritten, os.FileMode(schemaFilePerm)).Return(nil)
    59  
    60  			g := newTestGenerator(t, fs, runner)
    61  			require.NoError(t, g.dumpSchema(context.Background(), "postgres://dsn", "pw"))
    62  		})
    63  	})
    64  
    65  	t.Run("異常系", func(t *testing.T) {
    66  		t.Parallel()
    67  
    68  		t.Run("pg_dump失敗時はWriteFileを呼ばずエラー", func(t *testing.T) {
    69  			t.Parallel()
    70  			ctrl := gomock.NewController(t)
    71  			fs := mock_fs.NewMockFS(ctrl)
    72  			runner := mock_exec.NewMockRunner(ctrl)
    73  
    74  			runner.EXPECT().Output(gomock.Any(), gomock.Any(), gomock.Any(), gomock.Any(), gomock.Any()).Return(nil, xerrors.New("pg_dump failed"))
    75  
    76  			g := newTestGenerator(t, fs, runner)
    77  			require.Error(t, g.dumpSchema(context.Background(), "postgres://dsn", "pw"))
    78  		})

    96  	t.Parallel()
    97  
    98  	t.Run("正常系", func(t *testing.T) {
    99  		t.Parallel()
   100  
   101  		t.Run("メタコマンドとバージョンコメントと空行を除去しDDLは残す", func(t *testing.T) {
   102  			t.Parallel()
   103  
   104  			input := []byte("" +
   105  				"\\connect mydb\n" +
   106  				"-- Dumped from database version 14.1\n" +
   107  				"-- Dumped by pg_dump version 14.1\n" +
   108  				"\n" +
   109  				"CREATE TABLE users (id int);\n" +
   110  				"\n" +
   111  				"CREATE TABLE items (id int);\n")
   112  
   113  			got := string(sanitizeSchema(input))
   114  			assert.Contains(t, got, "CREATE TABLE users (id int);")
   115  			assert.Contains(t, got, "CREATE TABLE items (id int);")
   116  			assert.NotContains(t, got, "\\connect")
   117  			assert.NotContains(t, got, "Dumped from database version")
   118  			assert.NotContains(t, got, "Dumped by pg_dump version")
   119  			assert.NotContains(t, got, "\n\n")
   120  		})
   121  
   122  		t.Run("整形対象が無い入力はDDL行がそのまま残る", func(t *testing.T) {
   123  			t.Parallel()
   124  
   125  			got := sanitizeSchema([]byte("CREATE TABLE x (id int);\nCREATE TABLE y (id int);\n"))
   126  			assert.Equal(t, "CREATE TABLE x (id int);\nCREATE TABLE y (id int);", string(got))
   127  		})
   128  	})
