    29  // runJob は、ジョブ実行のオーケストレーション（タイムアウト分岐と停止処理）を行います。
    30  func runJob(
    31  	ctx context.Context,
    32  	name string,
    33  	args []string,
    34  	timeout time.Duration,
    35  	start StartFunc,
    36  	stop StopFunc,
    37  ) error {
    38  	done := start(ctx, name, args)
    39  
    40  	if timeout <= 0 {
    41  		err := <-done
    42  		gracefulStop(ctx, stop)
    43  		return err
    44  	}
    45  
    46  	waitCtx, cancel := context.WithTimeout(ctx, timeout)
    47  	defer cancel()
    48  
    49  	select {
    50  	case err := <-done:
    51  		gracefulStop(ctx, stop)
    52  		return err
    53  	case <-waitCtx.Done():
    54  		// waitCtx は ctx の子。タイムアウト(DeadlineExceeded)・親キャンセル(Canceled)の両方で発火する。
    55  		// 停止は期限切れの waitCtx ではなく専用 context を作り直して猶予を与える。
    56  		gracefulStop(ctx, stop)
    57  		return waitCtx.Err()
    58  	}
    59  }

    61  // gracefulStop は、停止開始時点から stopTimeout の猶予を与えて後始末（app.Stop）を実行します。
    62  func gracefulStop(ctx context.Context, stop StopFunc) {

    64  	defer cancel()
