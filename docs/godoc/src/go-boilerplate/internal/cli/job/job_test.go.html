    20  // makeStart は、固定の done チャネルを返すフェイクの開始関数を生成します。
    21  func makeStart(done <-chan error) StartFunc {
    22  	return func(_ context.Context, _ string, _ []string) <-chan error {

    27  func (r *recordingStop) fn() StopFunc {

    31  		return nil
    32  	}
