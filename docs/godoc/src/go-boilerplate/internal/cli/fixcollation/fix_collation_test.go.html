    62  					Return(nil, nil),
    63  				runner.EXPECT().Output(gomock.Any(), workDir, gomock.Any(), psqlCommand,
    64  					[]string{"postgres://dsn", "-v", "ON_ERROR_STOP=1", "-c", "ALTER DATABASE local REFRESH COLLATION VERSION;"}).
    65  					Return(nil, nil),
    66  			)
    67  
    68  			err := fixCollation(context.Background(), runner, logging.NewTestLogger(t), "postgres://dsn", "pw", "local")
