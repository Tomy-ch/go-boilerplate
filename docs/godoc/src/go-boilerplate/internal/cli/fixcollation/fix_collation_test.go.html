   120  			t.Parallel()
   121  			ctrl := gomock.NewController(t)
   122  			runner := mock_exec.NewMockRunner(ctrl)
   123  
   124  			// 2 文 (REINDEX → ALTER) が psql 経由で順序通り実行され、 ON_ERROR_STOP=1 が常に
   125  			// 付与され、 SQL 内容自体がデータベース名込みで一致することを検証する。
   126  			gomock.InOrder(
   127  				runner.EXPECT().Output(gomock.Any(), workDir, gomock.Any(), psqlCommand,
   128  					[]string{"postgres://dsn", "-v", "ON_ERROR_STOP=1", "-c", "REINDEX DATABASE local;"}).
   129  					Return(nil, nil),
   130  				runner.EXPECT().Output(gomock.Any(), workDir, gomock.Any(), psqlCommand,
   131  					[]string{"postgres://dsn", "-v", "ON_ERROR_STOP=1", "-c", "ALTER DATABASE local REFRESH COLLATION VERSION;"}).
   132  					Return(nil, nil),
   133  			)
   134  
   135  			err := fixCollation(context.Background(), runner, logging.NewTestLogger(t), "postgres://dsn", "pw", "local")

   197  			runner := mock_exec.NewMockRunner(ctrl)

   199  			// REINDEX DATABASE は接続中のデータベースしか対象にできないため、設定が local を指していても
   200  			// 指定した template1 へ接続し直すことを、psql へ渡す接続 URL で固定する。

   212  				return "postgres://user@localhost:5432/local?sslmode=disable", "pw", nil
   213  			}
