    34  	if err := validateDatabaseName(database); err != nil {

    51  // validateDatabaseName は、許可済みの開発・テスト用 DB 名のみを受け付けます。

    53  func validateDatabaseName(name string) error {
    54  	switch name {
    55  	case "local", "test", "template1":

    57  	}

    60  	}

    67  func withDatabase(dsn, name string) (string, error) {

    70  		return "", xerrors.Wrap(errInvalidDSN, dsn)

    78  func fixCollation(ctx context.Context, runner exec.Runner, logger logging.Logger, dbURL, password, database string) error {
    79  	log := logger.CallerSkip(callerSkipCount).Named("fixcollation")
    80  	log.Info(ctx, "start collation fix", logging.String("database", database))
    81  
    82  	sqlStatements := []string{
    83  		fmt.Sprintf("REINDEX DATABASE %s;", database),
    84  		fmt.Sprintf("ALTER DATABASE %s REFRESH COLLATION VERSION;", database),
    85  	}
    86  
    87  	env := []string{"PGPASSWORD=" + password}
    88  	// 依存順序があるため、照合順序修正 SQL は並列ではなく順番に流します。
    89  	for _, sql := range sqlStatements {
    90  		args := []string{dbURL, "-v", "ON_ERROR_STOP=1", "-c", sql}
    91  		if _, err := runner.Output(ctx, workDir, env, psqlCommand, args); err != nil {
    92  			log.Error(ctx, "psql command failed",
    93  				logging.String("database", database),
    94  				logging.String("sql", sql),
    95  				logging.Error(logging.ErrorKey, err),
    96  			)
    97  			return xerrors.Wrap(err, "psql command failed")
    98  		}
    99  	}
   100  
   101  	log.Info(ctx, "collation fix completed successfully", logging.String("database", database))
   102  	return nil
   103  }
