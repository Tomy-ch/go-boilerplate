    54  type Outbox struct {
    55  	// Publisher は publish 先の種別（"http" / "sqs"）です。publish 先は環境ティアではなく
    56  	// デプロイ先の判断で決まるため、ENV 分岐ではなく明示の判別子で切り替える。
    57  	// 未知の値は DI で起動エラーにする（fail-closed）。

    62  	BatchSize    int           `env:"BATCH_SIZE"    envDefault:"100"`
    63  	// Queue* は PUBLISHER=sqs のときだけ使う。未設定のまま sqs を選ぶと adapter 構築時に落とす。
    64  	QueueEndpoint        string `env:"QUEUE_ENDPOINT"          envDefault:""`
    65  	QueueRegion          string `env:"QUEUE_REGION"            envDefault:""`
    66  	QueueURL             string `env:"QUEUE_URL"               envDefault:""`

    71  // ConsumerQueue は worker が consume する broker（SQS 互換）の adapter 設定を保持する。
    72  // engine-core の Worker は broker 非依存と定めているため、broker 語彙を持つ設定は別軸に置く。

    82  	// DLQURL は Permanent メッセージの退避先であり、滞留量の収集対象でもある。

    87  	// WaitTimeSeconds は long-poll の待機秒数（0〜20）。上限にすると空ポーリングの往復が最も減る。
    88  	WaitTimeSeconds int32 `env:"WAIT_TIME_SECONDS" envDefault:"20"`

    90  	VisibilityTimeout int32 `env:"VISIBILITY_TIMEOUT" envDefault:"30"`
