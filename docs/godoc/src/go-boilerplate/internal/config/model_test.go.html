   475  func TestObservabilityConfig_Enabled(t *testing.T) {
   476  	t.Parallel()
   477  
   478  	cases := []struct {
   479  		name        string
   480  		traces      string
   481  		metrics     string
   482  		logs        string
   483  		wantTraces  bool
   484  		wantMetrics bool
   485  		wantLogs    bool
   486  		wantEnabled bool
   487  	}{
   488  		{"正常系_全部otlpなら全て有効", "otlp", "otlp", "otlp", true, true, true, true},
   489  		{"正常系_traceのみ有効", "otlp", "", "", true, false, false, true},
   490  		{"正常系_metricのみ有効", "", "otlp", "", false, true, false, true},
   491  		{"正常系_logのみ有効", "", "", "otlp", false, false, true, true},
   492  		{"正常系_全部空なら無効", "", "", "", false, false, false, false},
   493  		{"正常系_noneは無効として扱う", "none", "none", "none", false, false, false, false},
   494  		{"正常系_大文字NONEも無効として扱う", "NONE", "None", "nOnE", false, false, false, false},
   495  	}
   496
