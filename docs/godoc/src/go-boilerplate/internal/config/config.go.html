   147  // validateServerConfig は、サーバー設定を検証します。
   148  func validateServerConfig(srvCfg Server) error {
   149  	if err := validatePortRange(srvCfg.Port, ErrInvalidPortRange); err != nil {
   150  		return err
   151  	}
   152  
   153  	if srvCfg.ReadHeaderTimeout <= 0 {
   154  		return ErrInvalidReadHeaderTimeout
   155  	}
   156  
   157  	if srvCfg.ReadTimeout <= 0 {
   158  		return ErrInvalidReadTimeout
   159  	}
   160  
   161  	if srvCfg.WriteTimeout <= 0 {
   162  		return ErrInvalidWriteTimeout
   163  	}
   164  
   165  	if srvCfg.IdleTimeout <= 0 {
   166  		return ErrInvalidIdleTimeout
   167  	}
   168  
   169  	if srvCfg.ReadHeaderTimeout > srvCfg.ReadTimeout {
   170  		return ErrReadHeaderTimeoutExceedsReadTimeout
   171  	}
   172  	return nil
   173  }

   197  // validateSecurityConfig は、セキュリティ設定を検証します。
   198  // CIDR は解析が検証を兼ねるため、ここでは扱わず New の parseCIDR に委ねる。
   199  func validateSecurityConfig(secCfg Security) error {
   200  	if len(secCfg.AllowedOrigins) == 0 {
   201  		return ErrEmptyAllowedOrigins
   202  	}
   203  
   204  	if secCfg.BcryptCost < bcrypt.MinCost || bcrypt.MaxCost < secCfg.BcryptCost {
   205  		return ErrInvalidBcryptCost
   206  	}
   207  
   208  	for _, origin := range secCfg.AllowedOrigins {
   209  		parsedURL, err := url.Parse(origin)
   210  		if err != nil {
   211  			return ErrHTTPOnlyAllowedForLocalhost
   212  		}
   213  		// スキームは url.Parse で小文字正規化されるが、念のため EqualFold で大小無視判定する。
   214  		if strings.EqualFold(parsedURL.Scheme, "http") &&
   215  			parsedURL.Hostname() != "localhost" && parsedURL.Hostname() != "127.0.0.1" {
   216  			return ErrHTTPOnlyAllowedForLocalhost
   217  		}
   218  	}
   219  
   220  	return nil
   221  }
