   180  // 実効モードが production かつ埋め込み env の APP_ENV が非本番（deny-list）の場合、

   218  func validateServerConfig(srvCfg Server) error {
   219  	if err := validatePortRange(srvCfg.Port, ErrInvalidPortRange); err != nil {
   220  		return err
   221  	}
   222  
   223  	if srvCfg.ReadHeaderTimeout <= 0 {
   224  		return ErrInvalidReadHeaderTimeout
   225  	}
   226  
   227  	if srvCfg.ReadTimeout <= 0 {
   228  		return ErrInvalidReadTimeout
   229  	}
   230  
   231  	if srvCfg.WriteTimeout <= 0 {
   232  		return ErrInvalidWriteTimeout
   233  	}
   234  
   235  	if srvCfg.IdleTimeout <= 0 {
   236  		return ErrInvalidIdleTimeout
   237  	}
   238  
   239  	if srvCfg.ReadHeaderTimeout > srvCfg.ReadTimeout {
   240  		return ErrReadHeaderTimeoutExceedsReadTimeout
   241  	}
   242  
   243  	if srvCfg.WriteTimeout < srvCfg.RequestTimeout {
   244  		return ErrWriteTimeoutBelowRequestTimeout
   245  	}
   246  	return nil
   247  }

   291  // validateSecurityConfig は、セキュリティ設定を検証します。
   292  // CIDR は解析が検証を兼ねるため、ここでは扱わず New の parseCIDR に委ねる。
   293  func validateSecurityConfig(secCfg Security) error {
   294  	if len(secCfg.AllowedOrigins) == 0 {
   295  		return ErrEmptyAllowedOrigins
   296  	}
   297  
   298  	if secCfg.BcryptCost < bcrypt.MinCost || bcrypt.MaxCost < secCfg.BcryptCost {
   299  		return ErrInvalidBcryptCost
   300  	}
   301  
   302  	for _, origin := range secCfg.AllowedOrigins {
   303  		parsedURL, err := url.Parse(origin)
   304  		if err != nil {
   305  			return ErrHTTPOnlyAllowedForLocalhost
   306  		}
   307  		// スキームは url.Parse で小文字正規化されるが、念のため EqualFold で大小無視判定する。
   308  		if strings.EqualFold(parsedURL.Scheme, "http") &&
   309  			parsedURL.Hostname() != "localhost" && parsedURL.Hostname() != "127.0.0.1" {
   310  			return ErrHTTPOnlyAllowedForLocalhost
   311  		}
   312  	}
   313  
   314  	return nil
   315  }
