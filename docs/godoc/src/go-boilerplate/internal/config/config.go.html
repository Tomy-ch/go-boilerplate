   217  // validateSecurityConfig は、セキュリティ設定を検証します。
   218  // CIDR は解析が検証を兼ねるため、ここでは扱わず New の parseCIDR に委ねる。
   219  func validateSecurityConfig(secCfg Security) error {
   220  	if len(secCfg.AllowedOrigins) == 0 {
   221  		return ErrEmptyAllowedOrigins
   222  	}
   223  
   224  	if secCfg.BcryptCost < bcrypt.MinCost || bcrypt.MaxCost < secCfg.BcryptCost {
   225  		return ErrInvalidBcryptCost
   226  	}
   227  
   228  	for _, origin := range secCfg.AllowedOrigins {
   229  		parsedURL, err := url.Parse(origin)
   230  		if err != nil {
   231  			return ErrHTTPOnlyAllowedForLocalhost
   232  		}
   233  		// スキームは url.Parse で小文字正規化されるが、念のため EqualFold で大小無視判定する。
   234  		if strings.EqualFold(parsedURL.Scheme, "http") &&
   235  			parsedURL.Hostname() != "localhost" && parsedURL.Hostname() != "127.0.0.1" {
   236  			return ErrHTTPOnlyAllowedForLocalhost
   237  		}
   238  	}
   239  
   240  	return nil
   241  }
