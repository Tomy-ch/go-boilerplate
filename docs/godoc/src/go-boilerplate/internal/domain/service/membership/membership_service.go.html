     1  // Package membership は、ユーザーの在籍と購入の進行状態にまたがる規則を表すドメインサービスです。
     2  //
     3  // 「在籍していないユーザーは購入できない」と「進行中の購入が残っているユーザーは退会できない」は、
     4  // 一つの不変条件（ユーザーと進行中の購入を切り離してはならない）を購入側と退会側から見た姿です。
     5  // ユーザー集約は購入の進行状態を知らず、購入集約は購入者の在籍を知らないため、この規則はどちらの
     6  // 集約の自然な責務にもなりません。状態を持たず、読み込み済みの状態だけを受け取って可否を返すため、
     7  // 集約の外に置くドメインサービスとして表現します。
     8  //
     9  // 判定に必要な状態の取得（ロック・問い合わせ）と、その順序・トランザクション境界は Usecase の
    10  // 責務です。本パッケージは I/O を持たず、context.Context も受け取りません。
    11  package membership

    14  	"go-boilerplate/internal/apperror"
    15  	"go-boilerplate/internal/domain/purchase"
    16  	"go-boilerplate/internal/domain/user"
    17  	"go-boilerplate/pkg/xerrors"
    18  )

    20  var (
    21  	// ErrPurchaserWithdrawn は、退会済みの購入者による購入を拒否したことを表します。
    22  	ErrPurchaserWithdrawn = xerrors.Wrap(apperror.ErrConflict, "purchaser is withdrawn")
    23  	// ErrInProgressPurchaseExists は、進行中の購入が残っているユーザーの退会を拒否したことを表します。
    24  	ErrInProgressPurchaseExists = xerrors.Wrap(apperror.ErrConflict, "user has in-progress purchases")

    27  // EnsurePurchasable は、購入者が購入してよい状態かを判定します。
    28  // 在籍していない（退会済みの）購入者には ErrPurchaserWithdrawn を返します。
    29  func EnsurePurchasable(purchaser *user.User) error {
    30  	if !purchaser.IsActive() {
    31  		return ErrPurchaserWithdrawn
    32  	}
    33  	return nil
    34  }

    36  // EnsureWithdrawable は、ユーザーが退会してよい状態かを判定します。statuses は、そのユーザーの
    37  // 購入が取っているステータスです（重複の有無は問いません）。進行中の購入が 1 件でも残っている
    38  // 場合は ErrInProgressPurchaseExists を返します。
    39  //
    40  // 進行中かどうかの定義は購入集約が持つため（Status.IsTerminal の否定）ここでは再掲しません。
    41  // 終端でない新しいステータスが増えた場合も、既定で進行中として扱われます。
    42  // 既に退会しているユーザーは退会の対象になり得ないため、user.ErrAlreadyDeleted を返します。
    43  func EnsureWithdrawable(withdrawing *user.User, statuses []purchase.Status) error {
    44  	if !withdrawing.IsActive() {
    45  		return user.ErrAlreadyDeleted
    46  	}
    47  	for _, status := range statuses {
    48  		if !status.IsTerminal() {
    49  			return ErrInProgressPurchaseExists
    50  		}
    51  	}
    52  	return nil
    53  }
