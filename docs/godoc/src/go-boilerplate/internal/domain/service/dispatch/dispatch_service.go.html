     1  // Package dispatch は、発送待ちの購入をまとめて発送してよい組へ分ける規則を表すドメインサービスです。
     2  //
     3  // 「この購入は発送可能か」は 1 件の購入自身の状態だけで決まるため Purchase のメソッド（IsShippable）
     4  // ですが、「これらのうちどれとどれを 1 便にまとめてよいか」は集合についての問いで、ある 1 件についての
     5  // 答えが集合に他のどれが含まれるかに依存します。どの Purchase 1 件のメソッドにもなり得ないため、
     6  // 購入集約の自然な責務にはなりません。状態を持たず、読み込み済みの状態だけを受け取って組を返すため、
     7  // 集約の外に置くドメインサービスとして表現します。
     8  //
     9  // 語るのは購入集約だけで、import するのも internal/domain/purchase だけです。この置き場が集約の
    10  // import を許すのは許可であって要求ではありません（internal/domain/README.md）。
    11  //
    12  // 対象の取得（絞り込み・件数の上限）と、取得した行が実際に発送可能かの検証、および
    13  // トランザクション境界は Usecase の責務です。本パッケージは I/O を持たず、context.Context も
    14  // 受け取りません。

    25  // GroupForDispatch は、発送待ちの購入を、まとめて発送してよい組に分けます。
    26  // 同一の購入者宛てであることを軸とし、同じ購入者の購入は 1 つの組にまとまります。
    27  //
    28  // 受け取るのは発送可能な購入の集合であることを前提とし、本関数は発送可能性を検証しません
    29  // （その定義は Purchase.IsShippable が持ち、取得した行との突き合わせは Usecase が行います）。
    30  // 同梱可能期間などの追加条件は現時点では持ちません。
    31  //
    32  // 結果は入力の並び順に依存せず一意に定まります。組の中は注文日時の古い順（同時刻は購入 ID の昇順）、
    33  // 組同士はその組の最も古い購入の同じ順序で並びます。購入が 1 件だけの購入者もその 1 件からなる組に
    34  // なります。purchases が空の場合は空を返します。
    35  func GroupForDispatch(purchases purchase.Purchases) []purchase.Purchases {
    36  	byPurchaser := make(map[uuid.UUID]purchase.Purchases, len(purchases))
    37  	for _, p := range purchases {
    38  		byPurchaser[p.UserID()] = append(byPurchaser[p.UserID()], p)
    39  	}
    40  
    41  	groups := make([]purchase.Purchases, 0, len(byPurchaser))
    42  	for _, group := range byPurchaser {
    43  		slices.SortFunc(group, compareDispatchOrder)
    44  		groups = append(groups, group)
    45  	}
    46  	slices.SortFunc(groups, func(a, b purchase.Purchases) int {
    47  		return compareDispatchOrder(a[0], b[0])
    48  	})
    49  
    50  	return groups
    51  }

    53  // compareDispatchOrder は、購入 a と b の発送順を比較します。注文日時の古い順、同時刻は購入 ID の
    54  // 昇順です。同時刻の並びを ID で決めるのは、順序を一意にするためです。
    55  func compareDispatchOrder(a, b *purchase.Purchase) int {
    56  	if !a.OrderedAt().Equal(b.OrderedAt()) {
    57  		return a.OrderedAt().Compare(b.OrderedAt())
    58  	}
    59  	return cmp.Compare(a.ID().String(), b.ID().String())
    60  }
