    55  // codesOf は、組の並びを購入コードの二重スライスへ落とします。順序の検証を読みやすくするためです。
    56  func codesOf(groups []purchase.Purchases) [][]string {
    57  	codes := make([][]string, 0, len(groups))
    58  	for _, group := range groups {
    59  		groupCodes := make([]string, 0, len(group))
    60  		for _, p := range group {
    61  			groupCodes = append(groupCodes, p.Code())
    62  		}
    63  		codes = append(codes, groupCodes)
    64  	}
    65  	return codes
    66  }
