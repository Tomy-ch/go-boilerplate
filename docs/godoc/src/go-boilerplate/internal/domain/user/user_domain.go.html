    60  func New(id uuid.UUID, attrs Attributes) (*User, error) {

    63  	}
    64  
    65  	emailVO, postalCodeVO, err := validateProfileFields(attrs.Profile)

    70  	if attrs.UpdatedAt.Before(attrs.CreatedAt) {

   169  func (u *User) MarkAsDeleted(deletedAt time.Time) error {
   170  	if err := u.ensureNotDeleted(); err != nil {

   191  // ensureUpdatedAt は、更新日時が createdAt 以降かつ現在の updatedAt 以降（単調非減少）であることを検証します。
   192  func (u *User) ensureUpdatedAt(updatedAt time.Time) error {
   193  	if updatedAt.Before(u.createdAt) {
   194  		return xerrors.Wrap(ErrInvalidUpdatedAt, "updatedAt must be after or equal to createdAt")
   195  	}
   196  	if updatedAt.Before(u.updatedAt) {

   203  func validateDeletedAt(deletedAt, createdAt, updatedAt time.Time) error {
   204  	if deletedAt.Before(createdAt) {
   205  		return xerrors.Wrap(ErrInvalidDeletedAt, "deletedAt must be after or equal to createdAt")
   206  	}

   217  func validateProfileFields(profile Profile) (Email, PostalCode, error) {

   240  		fields = append(fields, FieldPrefectureID)
   241  	}
   242  	if ok, msg := stringkit.ValidateInRange(profile.City, minLength, maxCityLength); !ok {
   243  		errs = append(errs, xerrors.Wrap(ErrInvalidCity, msg))
   244  		fields = append(fields, FieldCity)
   245  	}
   246  	if ok, msg := stringkit.ValidateInRange(profile.Street, minLength, maxStreetLength); !ok {
   247  		errs = append(errs, xerrors.Wrap(ErrInvalidStreet, msg))

   249  	}
   250  	if profile.Building != nil {
