   140  // 増減後の在庫が負になる場合は、在庫を変更せずに検証エラーを返します。

   142  func (p *Product) AdjustStock(delta int) error {
   143  	// 増減の途中結果は在庫の表現範囲を超えうるため、検証を通すまでは広い幅で保持します。
   144  	adjusted := int64(p.quantity) + int64(delta)
   145  	if err := validateQuantity(adjusted); err != nil {
   146  		return err
   147  	}
   148  
   149  	p.quantity = int(adjusted)
   150  
   151  	return nil
