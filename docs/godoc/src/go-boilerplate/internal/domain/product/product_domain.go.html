    69  func newProduct(id uuid.UUID, attrs Attributes, version int, createdAt time.Time) (*Product, error) {
    70  	if id.IsNil() {
    71  		return nil, xerrors.Wrap(ErrInvalidID, "id is required")
    72  	}
    73  	if err := validateAttributes(attrs); err != nil {
    74  		return nil, err
    75  	}
    76  	if version < initialVersion {
    77  		return nil, xerrors.Wrap(
    78  			ErrInvalidVersion,
    79  			fmt.Sprintf("version must be %d or greater, got %d", initialVersion, version),
    80  		)
    81  	}
    82  	if createdAt.IsZero() {
    83  		return nil, xerrors.Wrap(ErrInvalidCreatedAt, "createdAt is required")
    84  	}
    85  
    86  	return &Product{
    87  		id:                    id,
    88  		name:                  attrs.Name,
    89  		description:           ptr.Copy(attrs.Description),
    90  		price:                 attrs.Price,
    91  		quantity:              attrs.Quantity,
    92  		stockWarningThreshold: ptr.Copy(attrs.StockWarningThreshold),
    93  		status:                attrs.Status,
    94  		category:              attrs.Category,
    95  		publishedAt:           ptr.Copy(attrs.PublishedAt),
    96  		images:                sortImagesByDisplaySort(attrs.Images),
    97  		createdAt:             createdAt,
    98  		version:               version,
    99  	}, nil
   100  }

   102  // validateAttributes は、商品属性の不変条件を検証します。生成時と更新時で同一の条件を課します。
   103  func validateAttributes(attrs Attributes) error {
   104  	if ok, msg := stringkit.ValidateInRange(attrs.Name, minNameLength, maxNameLength); !ok {
   105  		return xerrors.Wrap(ErrInvalidName, msg)
   106  	}
   107  	if err := validateQuantity(int64(attrs.Quantity)); err != nil {
   108  		return err
   109  	}
   110  	if err := validateStockWarningThreshold(attrs.StockWarningThreshold); err != nil {
   111  		return err
   112  	}
   113  	if attrs.Status.id.IsNil() {
   114  		return xerrors.Wrap(ErrInvalidStatusID, "status is required")
   115  	}
   116  	if attrs.Category.id.IsNil() {
   117  		return xerrors.Wrap(ErrInvalidCategoryID, "category is required")
   118  	}
   119  	return validateImages(attrs.Images)
   120  }

   143  // AdjustStock は、在庫数を delta の分だけ増減します。delta は正で補充、負で差し引きを表します。
   144  // 増減後の在庫が負になる場合は、在庫を変更せずに検証エラーを返します。
   145  // バージョンの採番タイミングは Update と同じです。
   146  func (p *Product) AdjustStock(delta int) error {
   147  	// 増減の途中結果は在庫の表現範囲を超えうるため、検証を通すまでは広い幅で保持します。
   148  	adjusted := int64(p.quantity) + int64(delta)
   149  	if err := validateQuantity(adjusted); err != nil {
   150  		return err
   151  	}
   152  
   153  	p.quantity = int(adjusted)
   154  
   155  	return nil
   156  }
