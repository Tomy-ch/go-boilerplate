    71  // Evaluate は、明細を商品の観測値と突き合わせて再評価結果を返します。
    72  // snapshot が nil の場合は商品を引けなかったことを表し、IssueNotFound だけが立ちます。
    73  func (i CartItem) Evaluate(snapshot *ProductSnapshot) Evaluation {
    74  	if snapshot == nil {
    75  		return Evaluation{issues: []Issue{IssueNotFound}}
    76  	}
    77  
    78  	issues := make([]Issue, 0, maxIssuesPerItem)
    79  	if !snapshot.published {
    80  		issues = append(issues, IssueUnpublished)
    81  	}
    82  
    83  	var availableQuantity *int
    84  	switch stock := snapshot.quantity; {
    85  	case stock <= 0:
    86  		issues = append(issues, IssueOutOfStock)
    87  	case stock < i.quantity:
    88  		issues = append(issues, IssueInsufficientStock)
    89  		availableQuantity = &stock
    90  	}
    91  
    92  	if i.lastSeenPrice != nil {
    93  		switch snapshot.price.Decimal().Cmp(i.lastSeenPrice.Decimal()) {
    94  		case 1:
    95  			issues = append(issues, IssuePriceIncreased)
    96  		case -1:
    97  			issues = append(issues, IssuePriceDecreased)
    98  		}
    99  	}
   100  
   101  	return Evaluation{issues: issues, availableQuantity: availableQuantity}
   102  }
