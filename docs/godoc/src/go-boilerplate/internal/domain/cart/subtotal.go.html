     9  // Subtotal は、購入可能な明細だけを合算して決済スケールの整数へ落とします。
    10  // 丸めは合算の後に一度だけ行い、明細ごとの丸め誤差が積み上がらないようにします。
    11  //
    12  // snapshots は商品 ID ごとの観測値で、引けなかった商品は含みません。含まれない明細と、
    13  // 突き合わせで issue が立った明細は合算に入りません。
    14  //
    15  // 幅を超えるのは明細が積み上がった結果に限られます（単価 1 件が決済スケールへ落とせることは
    16  // money.Price の構築時に保証されるため）。合計を偽らずに返す術が無いため、エラーを返します。
    17  func (c *Cart) Subtotal(snapshots map[uuid.UUID]ProductSnapshot) (int64, error) {
    18  	sum := decimal.FromInt(0)
    19  
    20  	for _, item := range c.items {
    21  		snapshot, ok := snapshots[item.ProductID()]
    22  		if !ok || !item.Evaluate(&snapshot).hasNoIssue() {
    23  			continue
    24  		}
    25  		sum = sum.Add(snapshot.Price().Decimal().Mul(decimal.FromInt(int64(item.Quantity()))))
    26  	}
    27  
    28  	subtotal, err := sum.ToScaledInt64(subtotalMinorUnitDigits)
    29  	if err != nil {
    30  		return 0, xerrors.Join(ErrSubtotalOutOfRange, err)
    31  	}
    32  
    33  	return subtotal, nil
    34  }
