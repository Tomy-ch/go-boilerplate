   123  func (d Discount) WithMaxAmount(maxAmount int64) (Discount, error) {
   124  	if d.kind != DiscountKindRate {
   125  		return Discount{}, xerrors.Wrap(ErrInvalidMaxAmount, "max amount is only meaningful for a rate discount")

   128  		return Discount{}, xerrors.Wrap(ErrInvalidMaxAmount, "max amount must be positive")
   129  	}

   161  // Apply は、対象額に対して差し引く額を価格スケールの十進量で返します。
   162  //
   163  // 「いくら引くか」の定義はこのメソッドが持ちます。定額は対象額を上限に切り詰め、定率は対象額に
   164  // 率を掛けます。どちらも対象額を超えないため、請求額が負になることはありません。
   165  // 対象額が 0 以下なら差し引く額も 0 です。
   166  //
   167  // 丸めません。決済スケールへの丸めは [Coupon.DiscountFor] が 1 箇所で行います
   168  // （ADR-0038 (two-scale-quantity-model)）。
   169  func (d Discount) Apply(eligible decimal.Decimal) decimal.Decimal {
   170  	if eligible.Sign() <= 0 {
   171  		return decimal.FromInt(0)
   172  	}
   173  
   174  	switch d.kind {
   175  	case DiscountKindFlat:
   176  		if d.value.Cmp(eligible) > 0 {
   177  			return eligible
   178  		}
   179  		return d.value
   180  	case DiscountKindRate:
   181  		return eligible.Mul(d.value)
   182  	default:
   183  		return decimal.FromInt(0)
   184  	}
   185  }
