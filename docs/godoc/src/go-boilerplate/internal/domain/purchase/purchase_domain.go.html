   116  // validateDetails は、購入集約が保持する明細の集合が満たすべき不変条件を検証します。
   117  // 明細 ID / 商品 ID が未設定、数量が最小値未満の場合はそれぞれ ErrInvalidID / ErrInvalidQuantity を、
   118  // 同一商品 ID が重複する場合は ErrDuplicateProductID を返します。
   119  // 空の集合は呼び出し元が ErrEmptyDetails で扱うため、ここでは許容します。
   120  func validateDetails(details []PurchaseDetail) error {
   121  	seen := make(map[uuid.UUID]struct{}, len(details))
   122  	for _, d := range details {
   123  		if d.id.IsNil() {
   124  			return xerrors.Wrap(ErrInvalidID, "detail id is required")
   125  		}
   126  		if d.productID.IsNil() {
   127  			return xerrors.Wrap(ErrInvalidID, "detail product id is required")
   128  		}
   129  		if d.quantity < minQuantity {
   130  			return xerrors.Wrap(
   131  				ErrInvalidQuantity, fmt.Sprintf("quantity must be %d or greater, got %d", minQuantity, d.quantity),
   132  			)
   133  		}
   134  		if _, dup := seen[d.productID]; dup {
   135  			return xerrors.Wrap(ErrDuplicateProductID, fmt.Sprintf("product %s appears more than once", d.productID))
   136  		}
   137  		seen[d.productID] = struct{}{}
   138  	}
   139  
   140  	return nil
   141  }

   143  // New は、購入明細の入力とロック済み在庫から購入集約を生成します。
   144  // 明細が空、同一 productID が重複、数量が最小値未満、明細に対応するロック済み商品が無い場合はそれぞれ
   145  // 検証エラー（422）を返し、要求数量がロック済み在庫を超える場合は ErrInsufficientStock（409）を返します。
   146  // 単価は対応するロック済み商品の価格スナップショットとし、金額（小計・税・送料・合計）を整数で計算します
   147  // （税・送料の丸めは切り捨てで本メソッド内 1 箇所に集約）。ステータスは「未処理」で生成します。
   148  func New(
   149  	id uuid.UUID,
   150  	code string,
   151  	userID uuid.UUID,
   152  	inputs []DetailInput,
   153  	locked []LockedProduct,
   154  ) (*Purchase, error) {
   155  	if id.IsNil() {
   156  		return nil, xerrors.Wrap(ErrInvalidID, "id is required")
   157  	}
   158  	if code == "" {
   159  		return nil, xerrors.Wrap(ErrInvalidCode, "code is required")
   160  	}
   161  	if userID.IsNil() {
   162  		return nil, xerrors.Wrap(ErrInvalidUserID, "userID is required")
   163  	}
   164  	if len(inputs) == 0 {
   165  		return nil, ErrEmptyDetails
   166  	}
   167  
   168  	lockedByID := make(map[uuid.UUID]LockedProduct, len(locked))
   169  	for _, l := range locked {
   170  		lockedByID[l.id] = l
   171  	}
   172  
   173  	seen := make(map[uuid.UUID]struct{}, len(inputs))
   174  	details := make([]PurchaseDetail, 0, len(inputs))
   175  	// 小計は価格スケール（ドル decimal）で正確に積算し、最後に決済スケール（整数セント）へ切り捨てる。
   176  	subtotalDollars := decimal.FromInt(0)
   177  	for _, in := range inputs {
   178  		if in.ID.IsNil() {
   179  			return nil, xerrors.Wrap(ErrInvalidID, "detail id is required")
   180  		}
   181  		if in.Quantity < minQuantity {
   182  			return nil, xerrors.Wrap(ErrInvalidQuantity, fmt.Sprintf("quantity must be %d or greater, got %d", minQuantity, in.Quantity))
   183  		}
   184  		if _, dup := seen[in.ProductID]; dup {
   185  			return nil, xerrors.Wrap(ErrDuplicateProductID, fmt.Sprintf("product %s appears more than once", in.ProductID))
   186  		}
   187  		seen[in.ProductID] = struct{}{}
   188  
   189  		l, ok := lockedByID[in.ProductID]
   190  		if !ok {
   191  			return nil, xerrors.Wrap(ErrProductNotFound, fmt.Sprintf("product %s is not available", in.ProductID))
   192  		}
   193  		if in.Quantity > l.quantity {
   194  			return nil, xerrors.Wrap(
   195  				ErrInsufficientStock,
   196  				fmt.Sprintf("product %s: requested %d, in stock %d", in.ProductID, in.Quantity, l.quantity),
   197  			)
   198  		}
   199  
   200  		details = append(details, PurchaseDetail{
   201  			id:        in.ID,
   202  			productID: in.ProductID,
   203  			quantity:  in.Quantity,
   204  			unitPrice: l.price,
   205  		})
   206  		subtotalDollars = subtotalDollars.Add(l.price.Decimal().Mul(decimal.FromInt(int64(in.Quantity))))
   207  	}
   208  
   209  	// 決済スケールへの丸め（切り捨て）はこの 1 箇所のみ。以降の税・合計は整数セントで計算する。
   210  	subtotalCents, err := subtotalDollars.Truncate(minorUnitDigits).ToScaledInt64(minorUnitDigits)
   211  	if err != nil {
   212  		return nil, xerrors.Wrap(ErrInvalidAmount, "subtotal exceeds the settlement range: "+err.Error())
   213  	}
   214  	subtotal := int(subtotalCents)
   215  	tax := subtotal * taxRatePercent / percentDivisor
   216  	shipping := shippingFeeCents
   217  	total := subtotal + tax + shipping
   218  
   219  	return &Purchase{
   220  		id:             id,
   221  		code:           code,
   222  		userID:         userID,
   223  		status:         StatusUnprocessed,
   224  		subtotalAmount: subtotal,
   225  		taxAmount:      tax,
   226  		shippingFee:    shipping,
   227  		totalAmount:    total,
   228  		details:        details,
   229  	}, nil
   230  }

   232  // Reconstruct は、永続化済みの購入を再構築します（Repository の読み出し・書き込み後の再検証で使用）。
   233  // attrs.StatusCode は購入ステータスマスタで解決した現在状態、PaidAt / CanceledAt / ShippedAt / DeliveredAt
   234  // は各イベントの発生日時（未発生は nil）です。ID / Code / UserID / StatusID が nil、StatusCode が不正、
   235  // 金額が負、明細が空の場合は検証エラーを返します。StatusCode と timestamps の組み合わせが状態遷移では
   236  // 到達し得ないもの（発送後のキャンセル、発送を伴わない配達 等）の場合も ErrInvalidStatusID を返します。
   237  // 明細は New と同じ不変条件（ID / 商品 ID が設定済み、数量が最小値以上、同一商品 ID が重複しない）を課します。
   238  func Reconstruct(id uuid.UUID, attrs Attributes) (*Purchase, error) {
   239  	if id.IsNil() {
   240  		return nil, xerrors.Wrap(ErrInvalidID, "id is required")
   241  	}
   242  	if attrs.Code == "" {
   243  		return nil, xerrors.Wrap(ErrInvalidCode, "code is required")
   244  	}
   245  	if attrs.UserID.IsNil() {
   246  		return nil, xerrors.Wrap(ErrInvalidUserID, "userID is required")
   247  	}
   248  	if attrs.StatusID.IsNil() {
   249  		return nil, xerrors.Wrap(ErrInvalidStatusID, "statusID is required")
   250  	}
   251  	status, err := NewStatus(attrs.StatusCode)
   252  	if err != nil {
   253  		return nil, err
   254  	}
   255  	if err := validateStatusTimestamps(
   256  		status, attrs.PaidAt, attrs.CanceledAt, attrs.ShippedAt, attrs.DeliveredAt,
   257  	); err != nil {
   258  		return nil, err
   259  	}
   260  	if attrs.SubtotalAmount < 0 || attrs.TaxAmount < 0 || attrs.ShippingFee < 0 || attrs.TotalAmount < 0 {
   261  		return nil, xerrors.Wrap(ErrInvalidAmount, "amounts must not be negative")
   262  	}
   263  	if len(attrs.Details) == 0 {
   264  		return nil, ErrEmptyDetails
   265  	}
   266  
   267  	copied := make([]PurchaseDetail, len(attrs.Details))
   268  	copy(copied, attrs.Details)
   269  	if err := validateDetails(copied); err != nil {
   270  		return nil, err
   271  	}
   272  
   273  	return &Purchase{
   274  		id:             id,
   275  		code:           attrs.Code,
   276  		userID:         attrs.UserID,
   277  		statusID:       attrs.StatusID,
   278  		status:         status,
   279  		subtotalAmount: attrs.SubtotalAmount,
   280  		taxAmount:      attrs.TaxAmount,
   281  		shippingFee:    attrs.ShippingFee,
   282  		totalAmount:    attrs.TotalAmount,
   283  		details:        copied,
   284  		orderedAt:      attrs.OrderedAt,
   285  		paidAt:         ptr.Copy(attrs.PaidAt),
   286  		canceledAt:     ptr.Copy(attrs.CanceledAt),
   287  		shippedAt:      ptr.Copy(attrs.ShippedAt),
   288  		deliveredAt:    ptr.Copy(attrs.DeliveredAt),
   289  	}, nil
   290  }

   292  // validateStatusTimestamps は、statusCode と各イベント日時の組み合わせが状態遷移で到達し得るかを検証します。
   293  // 到達し得ない組み合わせ（発送後のキャンセル、発送を伴わない配達 等）の場合は ErrInvalidStatusID を返します。
   294  func validateStatusTimestamps(status Status, paidAt, canceledAt, shippedAt, deliveredAt *time.Time) error {
   295  	// キャンセル status と canceledAt は同時セットの不変条件を持つ。片方のみの矛盾した永続化状態を弾く。
   296  	if (status == StatusCanceled) != (canceledAt != nil) {
   297  		return xerrors.Wrap(ErrInvalidStatusID, "canceled status and canceledAt must be consistent")
   298  	}
   299  	// 支払い済み status は paidAt を必須とする（一方向）。paidAt は支払い後に残り続け、以降のキャンセル
   300  	// 等の遷移で status が変わっても保持されるため、キャンセルのような双条件にはしない。
   301  	if status == StatusPaid && paidAt == nil {
   302  		return xerrors.Wrap(ErrInvalidStatusID, "paid status requires paidAt")
   303  	}
   304  	// 発送済み status は shippedAt を必須とする（一方向）。shippedAt は発送後に残り続け、以降の
   305  	// 配達済み等の遷移で status が変わっても保持されるため、キャンセルのような双条件にはしない。
   306  	if status == StatusShipped && shippedAt == nil {
   307  		return xerrors.Wrap(ErrInvalidStatusID, "shipped status requires shippedAt")
   308  	}
   309  	// キャンセルは発送前にしか行えない（Cancel が発送済み・配達済みを拒否する）ため、キャンセル status と
   310  	// 発送の記録は同居しない。支払い後のキャンセルは正常なので paidAt は対象外。
   311  	if status == StatusCanceled && shippedAt != nil {
   312  		return xerrors.Wrap(ErrInvalidStatusID, "canceled status must not have shippedAt")
   313  	}
   314  	// 発送済みへは支払い済みからのみ遷移し、支払い済みは paidAt を必須とするため、paidAt を欠く発送済みは
   315  	// 到達不能な状態である。
   316  	if status == StatusShipped && paidAt == nil {
   317  		return xerrors.Wrap(ErrInvalidStatusID, "shipped status requires paidAt")
   318  	}
   319  	// 配達は発送済みからのみ到達するため、shippedAt を欠く deliveredAt は到達不能な状態である。
   320  	if deliveredAt != nil && shippedAt == nil {
   321  		return xerrors.Wrap(ErrInvalidStatusID, "deliveredAt requires shippedAt")
   322  	}
   323  	// 配達済み status と deliveredAt は同時セットの不変条件を持つ。配達済みは終端状態（IsTerminal）で
   324  	// あり、配達後に別の status へ遷移して deliveredAt だけが残ることがないため、paidAt / shippedAt のような
   325  	// 一方向ではなくキャンセルと同じ双条件になる。
   326  	if (status == StatusDelivered) != (deliveredAt != nil) {
   327  		return xerrors.Wrap(ErrInvalidStatusID, "delivered status and deliveredAt must be consistent")
   328  	}
   329  	return nil
   330  }
