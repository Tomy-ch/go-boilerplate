   114  // validateDetails は、購入集約が保持する明細の集合が満たすべき不変条件を検証します。
   115  // 明細 ID / 商品 ID が未設定、数量が最小値未満の場合はそれぞれ ErrInvalidID / ErrInvalidQuantity を、
   116  // 同一商品 ID が重複する場合は ErrDuplicateProductID を返します。
   117  // 空の集合は呼び出し元が ErrEmptyDetails で扱うため、ここでは許容します。
   118  func validateDetails(details []PurchaseDetail) error {
   119  	seen := make(map[uuid.UUID]struct{}, len(details))
   120  	for _, d := range details {
   121  		if d.id.IsNil() {
   122  			return xerrors.Wrap(ErrInvalidID, "detail id is required")
   123  		}
   124  		if d.productID.IsNil() {
   125  			return xerrors.Wrap(ErrInvalidID, "detail product id is required")
   126  		}
   127  		if d.quantity < minQuantity {
   128  			return xerrors.Wrap(
   129  				ErrInvalidQuantity, fmt.Sprintf("quantity must be %d or greater, got %d", minQuantity, d.quantity),
   130  			)
   131  		}
   132  		if _, dup := seen[d.productID]; dup {
   133  			return xerrors.Wrap(ErrDuplicateProductID, fmt.Sprintf("product %s appears more than once", d.productID))
   134  		}
   135  		seen[d.productID] = struct{}{}
   136  	}
   137  
   138  	return nil
   139  }

   141  // buildDetails は、明細の入力をロック済み在庫と突き合わせて購入明細を組み立て、単価スナップショット
   142  // による小計を価格スケールで返します。
   143  // 明細 ID が未設定、数量が最小値未満、同一商品 ID が重複、入力に対応するロック済み商品が無い場合は
   144  // それぞれ検証エラー（422）を、要求数量がロック済み在庫を超える場合は ErrInsufficientStock（409）を
   145  // 返します。単価は対応するロック済み商品の価格とし、入力の順序を保った明細を返します。
   146  func buildDetails(inputs []DetailInput, locked []LockedProduct) ([]PurchaseDetail, decimal.Decimal, error) {
   147  	lockedByID := make(map[uuid.UUID]LockedProduct, len(locked))
   148  	for _, l := range locked {
   149  		lockedByID[l.id] = l
   150  	}
   151  
   152  	seen := make(map[uuid.UUID]struct{}, len(inputs))
   153  	details := make([]PurchaseDetail, 0, len(inputs))
   154  	subtotalDollars := decimal.FromInt(0)
   155  	for _, in := range inputs {
   156  		l, err := resolveLocked(in, lockedByID, seen)
   157  		if err != nil {
   158  			return nil, decimal.FromInt(0), err
   159  		}
   160  		seen[in.ProductID] = struct{}{}
   161  
   162  		details = append(details, PurchaseDetail{
   163  			id:        in.ID,
   164  			productID: in.ProductID,
   165  			quantity:  in.Quantity,
   166  			unitPrice: l.price,
   167  		})
   168  		subtotalDollars = subtotalDollars.Add(l.price.Decimal().Mul(decimal.FromInt(int64(in.Quantity))))
   169  	}
   170  
   171  	return details, subtotalDollars, nil
   172  }

   174  // resolveLocked は、明細 1 件の入力を検証し、対応するロック済み商品を返します。
   175  // seen は既出の商品 ID の集合で、呼び出し元が返却後に in.ProductID を追加します。
   176  func resolveLocked(
   177  	in DetailInput,
   178  	lockedByID map[uuid.UUID]LockedProduct,
   179  	seen map[uuid.UUID]struct{},
   180  ) (LockedProduct, error) {
   181  	if in.ID.IsNil() {
   182  		return LockedProduct{}, xerrors.Wrap(ErrInvalidID, "detail id is required")
   183  	}
   184  	if in.Quantity < minQuantity {
   185  		return LockedProduct{}, xerrors.Wrap(
   186  			ErrInvalidQuantity, fmt.Sprintf("quantity must be %d or greater, got %d", minQuantity, in.Quantity),
   187  		)
   188  	}
   189  	if _, dup := seen[in.ProductID]; dup {
   190  		return LockedProduct{}, xerrors.Wrap(
   191  			ErrDuplicateProductID, fmt.Sprintf("product %s appears more than once", in.ProductID),
   192  		)
   193  	}
   194  
   195  	l, ok := lockedByID[in.ProductID]
   196  	if !ok {
   197  		return LockedProduct{}, xerrors.Wrap(ErrProductNotFound, fmt.Sprintf("product %s is not available", in.ProductID))
   198  	}
   199  	if in.Quantity > l.quantity {
   200  		return LockedProduct{}, xerrors.Wrap(
   201  			ErrInsufficientStock,
   202  			fmt.Sprintf("product %s: requested %d, in stock %d", in.ProductID, in.Quantity, l.quantity),
   203  		)
   204  	}
   205  
   206  	return l, nil
   207  }

   209  // New は、購入明細の入力とロック済み在庫から購入集約を生成します。
   210  // 明細が空、同一 productID が重複、数量が最小値未満、明細に対応するロック済み商品が無い場合はそれぞれ
   211  // 検証エラー（422）を返し、要求数量がロック済み在庫を超える場合は ErrInsufficientStock（409）を返します。
   212  // 単価は対応するロック済み商品の価格スナップショットとし、金額（小計・税・送料・合計）を整数で計算します
   213  // （税・送料の丸めは切り捨てで本メソッド内 1 箇所に集約）。ステータスは「未処理」で生成します。
   214  func New(
   215  	id uuid.UUID,
   216  	code string,
   217  	userID uuid.UUID,
   218  	inputs []DetailInput,
   219  	locked []LockedProduct,
   220  ) (*Purchase, error) {
   221  	if id.IsNil() {
   222  		return nil, xerrors.Wrap(ErrInvalidID, "id is required")
   223  	}
   224  	if code == "" {
   225  		return nil, xerrors.Wrap(ErrInvalidCode, "code is required")
   226  	}
   227  	if userID.IsNil() {
   228  		return nil, xerrors.Wrap(ErrInvalidUserID, "userID is required")
   229  	}
   230  	if len(inputs) == 0 {
   231  		return nil, ErrEmptyDetails
   232  	}
   233  
   234  	details, subtotalDollars, err := buildDetails(inputs, locked)
   235  	if err != nil {
   236  		return nil, err
   237  	}
   238  
   239  	// 価格スケール→決済スケールの丸め（切り捨て）。以降は整数セントで計算する（ADR-0038 (two-scale-quantity-model)）。
   240  	subtotalCents, err := subtotalDollars.Truncate(minorUnitDigits).ToScaledInt64(minorUnitDigits)
   241  	if err != nil {
   242  		return nil, xerrors.Join(ErrInvalidAmount, xerrors.Wrap(err, "subtotal exceeds the settlement range"))
   243  	}
   244  	if subtotalCents > maxSubtotalCents {
   245  		return nil, xerrors.Wrap(
   246  			ErrInvalidAmount,
   247  			fmt.Sprintf("subtotal %d exceeds the maximum that can carry tax and shipping (%d)", subtotalCents, maxSubtotalCents),
   248  		)
   249  	}
   250  	subtotal := int(subtotalCents)
   251  	tax := subtotal * taxRatePercent / percentDivisor
   252  	shipping := shippingFeeCents
   253  	total := subtotal + tax + shipping
   254  
   255  	return &Purchase{
   256  		id:             id,
   257  		code:           code,
   258  		userID:         userID,
   259  		status:         StatusUnprocessed,
   260  		subtotalAmount: subtotal,
   261  		taxAmount:      tax,
   262  		shippingFee:    shipping,
   263  		totalAmount:    total,
   264  		details:        details,
   265  	}, nil
   266  }

   268  // Reconstruct は、永続化済みの購入を再構築します（Repository の読み出し・書き込み後の再検証で使用）。
   269  // attrs.StatusCode は購入ステータスマスタで解決した現在状態、PaidAt / CanceledAt / ShippedAt / DeliveredAt
   270  // は各イベントの発生日時（未発生は nil）です。ID / Code / UserID / StatusID が nil、StatusCode が不正、
   271  // 金額が負、明細が空の場合は検証エラーを返します。StatusCode と timestamps の組み合わせが状態遷移では
   272  // 到達し得ないもの（発送後のキャンセル、発送を伴わない配達 等）の場合も ErrInvalidStatusID を返します。
   273  // 明細は New と同じ不変条件（ID / 商品 ID が設定済み、数量が最小値以上、同一商品 ID が重複しない）を課します。
   274  func Reconstruct(id uuid.UUID, attrs Attributes) (*Purchase, error) {
   275  	if id.IsNil() {
   276  		return nil, xerrors.Wrap(ErrInvalidID, "id is required")
   277  	}
   278  	if attrs.Code == "" {
   279  		return nil, xerrors.Wrap(ErrInvalidCode, "code is required")
   280  	}
   281  	if attrs.UserID.IsNil() {
   282  		return nil, xerrors.Wrap(ErrInvalidUserID, "userID is required")
   283  	}
   284  	if attrs.StatusID.IsNil() {
   285  		return nil, xerrors.Wrap(ErrInvalidStatusID, "statusID is required")
   286  	}
   287  	status, err := NewStatus(attrs.StatusCode)
   288  	if err != nil {
   289  		return nil, err
   290  	}
   291  	if err := validateStatusTimestamps(
   292  		status, attrs.PaidAt, attrs.CanceledAt, attrs.ShippedAt, attrs.DeliveredAt,
   293  	); err != nil {
   294  		return nil, err
   295  	}
   296  	if attrs.SubtotalAmount < 0 || attrs.TaxAmount < 0 || attrs.ShippingFee < 0 || attrs.TotalAmount < 0 {
   297  		return nil, xerrors.Wrap(ErrInvalidAmount, "amounts must not be negative")
   298  	}
   299  	if len(attrs.Details) == 0 {
   300  		return nil, ErrEmptyDetails
   301  	}
   302  
   303  	copied := make([]PurchaseDetail, len(attrs.Details))
   304  	copy(copied, attrs.Details)
   305  	if err := validateDetails(copied); err != nil {
   306  		return nil, err
   307  	}
   308  
   309  	return &Purchase{
   310  		id:             id,
   311  		code:           attrs.Code,
   312  		userID:         attrs.UserID,
   313  		statusID:       attrs.StatusID,
   314  		status:         status,
   315  		subtotalAmount: attrs.SubtotalAmount,
   316  		taxAmount:      attrs.TaxAmount,
   317  		shippingFee:    attrs.ShippingFee,
   318  		totalAmount:    attrs.TotalAmount,
   319  		details:        copied,
   320  		orderedAt:      attrs.OrderedAt,
   321  		paidAt:         ptr.Copy(attrs.PaidAt),
   322  		canceledAt:     ptr.Copy(attrs.CanceledAt),
   323  		shippedAt:      ptr.Copy(attrs.ShippedAt),
   324  		deliveredAt:    ptr.Copy(attrs.DeliveredAt),
   325  	}, nil
   326  }

   328  // validateStatusTimestamps は、statusCode と各イベント日時の組み合わせが状態遷移で到達し得るかを検証します。
   329  // 到達し得ない組み合わせ（発送後のキャンセル、発送を伴わない配達 等）の場合は ErrInvalidStatusID を返します。
   330  func validateStatusTimestamps(status Status, paidAt, canceledAt, shippedAt, deliveredAt *time.Time) error {
   331  	// キャンセルと canceledAt は同時セット（双条件。docs/spec/purchase/domain.md Cancel invariants）。
   332  	if (status == StatusCanceled) != (canceledAt != nil) {
   333  		return xerrors.Wrap(ErrInvalidStatusID, "canceled status and canceledAt must be consistent")
   334  	}
   335  	// 支払い済みは paidAt 必須（一方向。docs/spec/purchase/domain.md Pay invariants）。
   336  	if status == StatusPaid && paidAt == nil {
   337  		return xerrors.Wrap(ErrInvalidStatusID, "paid status requires paidAt")
   338  	}
   339  	// 発送済みは shippedAt 必須（一方向。docs/spec/purchase/domain.md Ship invariants）。
   340  	if status == StatusShipped && shippedAt == nil {
   341  		return xerrors.Wrap(ErrInvalidStatusID, "shipped status requires shippedAt")
   342  	}
   343  	// キャンセルは発送前にしか行えない（Cancel が発送済み・配達済みを拒否する）ため、キャンセル status と
   344  	// 発送の記録は同居しない。支払い後のキャンセルは正常なので paidAt は対象外。
   345  	if status == StatusCanceled && shippedAt != nil {
   346  		return xerrors.Wrap(ErrInvalidStatusID, "canceled status must not have shippedAt")
   347  	}
   348  	// 発送済みへは支払い済みからのみ遷移し、支払い済みは paidAt を必須とするため、paidAt を欠く発送済みは
   349  	// 到達不能な状態である。
   350  	if status == StatusShipped && paidAt == nil {
   351  		return xerrors.Wrap(ErrInvalidStatusID, "shipped status requires paidAt")
   352  	}
   353  	// 配達は発送済みからのみ到達するため、shippedAt を欠く deliveredAt は到達不能な状態である。
   354  	if deliveredAt != nil && shippedAt == nil {
   355  		return xerrors.Wrap(ErrInvalidStatusID, "deliveredAt requires shippedAt")
   356  	}
   357  	// 配達済みと deliveredAt は同時セット（双条件。docs/spec/purchase/domain.md Deliver invariants）。
   358  	if (status == StatusDelivered) != (deliveredAt != nil) {
   359  		return xerrors.Wrap(ErrInvalidStatusID, "delivered status and deliveredAt must be consistent")
   360  	}
   361  	return nil
   362  }

   444  // 支払い済み）からのみ遷移でき、statusCode をキャンセル（6）へ、canceledAt を now へ同時に更新します。
   445  // 既にキャンセル済みなら ErrAlreadyCanceled、完了・配達済み・発送済み（shippedAt）なら
   446  // ErrCancelNotAllowed をそれぞれ返します（いずれも 409）。
   447  // 遷移に成功したときだけキャンセルの事実（Event）を返します。
   448  func (p *Purchase) Cancel(now time.Time) (Event, error) {
   449  	if p.IsCanceled() {
   450  		return Event{}, ErrAlreadyCanceled
   451  	}
   452  	// 配達済みも deliveredAt と同値なので status で判定する。発送済みは status が配達済みへ進んでも
   453  	// 遷移不可であり続けるため、status ではなく残り続ける shippedAt で判定する。
