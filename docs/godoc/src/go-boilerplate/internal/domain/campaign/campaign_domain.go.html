    86  // newCampaign は、生成・再構築に共通の検証を行いキャンペーンエンティティを構築します。
    87  func newCampaign(id uuid.UUID, attrs Attributes, issuedCount int, suspendedAt *time.Time) (*Campaign, error) {
    88  	if id.IsNil() {
    89  		return nil, xerrors.Wrap(ErrInvalidID, "id is required")
    90  	}
    91  	if attrs.Code.IsZero() {
    92  		return nil, xerrors.Wrap(ErrInvalidCode, "code is required")
    93  	}
    94  	if attrs.Template.IsZero() {
    95  		return nil, ErrInvalidTemplate
    96  	}
    97  	if err := validatePeriod(attrs); err != nil {
    98  		return nil, err
    99  	}
   100  	if err := validateLimits(attrs, issuedCount); err != nil {
   101  		return nil, err
   102  	}
   103  
   104  	return &Campaign{
   105  		id:           id,
   106  		code:         attrs.Code,
   107  		template:     attrs.Template,
   108  		startsAt:     attrs.StartsAt,
   109  		endsAt:       attrs.EndsAt,
   110  		totalLimit:   attrs.TotalLimit,
   111  		perUserLimit: attrs.PerUserLimit,
   112  		issuedCount:  issuedCount,
   113  		suspendedAt:  ptr.Copy(suspendedAt),
   114  	}, nil
   115  }

   117  // validatePeriod は、配布期間と、配るクーポンの有効期限との関係を検証します。
   118  func validatePeriod(attrs Attributes) error {
   119  	if attrs.StartsAt.IsZero() {
   120  		return xerrors.Wrap(ErrInvalidPeriod, "startsAt is required")
   121  	}
   122  	if attrs.EndsAt.IsZero() {
   123  		return xerrors.Wrap(ErrInvalidPeriod, "endsAt is required")
   124  	}
   125  	if !attrs.EndsAt.After(attrs.StartsAt) {
   126  		return xerrors.Wrap(ErrInvalidPeriod, "endsAt must be after startsAt")
   127  	}
   128  	if !attrs.Template.ExpiresAt().After(attrs.EndsAt) {
   129  		return xerrors.Wrap(ErrInvalidCouponExpiresAt, "coupon expiresAt must be after endsAt")
   130  	}
   131  
   132  	return nil
   133  }

   135  // validateLimits は、上限と発行済み枚数を検証します。
   136  func validateLimits(attrs Attributes, issuedCount int) error {
   137  	if attrs.TotalLimit <= 0 {
   138  		return xerrors.Wrap(ErrInvalidTotalLimit, "totalLimit must be positive")
   139  	}
   140  	if attrs.PerUserLimit <= 0 {
   141  		return xerrors.Wrap(ErrInvalidPerUserLimit, "perUserLimit must be positive")
   142  	}
   143  	if attrs.PerUserLimit > attrs.TotalLimit {
   144  		return xerrors.Wrap(ErrInvalidPerUserLimit, "perUserLimit must not exceed totalLimit")
   145  	}
   146  	if issuedCount < 0 {
   147  		return xerrors.Wrap(ErrInvalidIssuedCount, "issuedCount must not be negative")
   148  	}
   149  	if issuedCount > attrs.TotalLimit {
   150  		return xerrors.Wrap(ErrInvalidIssuedCount, "issuedCount must not exceed totalLimit")
   151  	}
   152  
   153  	return nil
   154  }

   186  // IsDistributing は、渡された時点でキャンペーンが受け取りを認めるかどうかを返します。
   187  //
   188  // 開始日時ちょうどは認める側に、終了日時ちょうどは認めない側に含めます（有効期限の判定と同じ向きです）。
   189  // 停止済み、または総枚数上限に達している場合も認めません。
   190  func (c *Campaign) IsDistributing(now time.Time) bool {
   191  	return !c.IsSuspended() &&
   192  		!now.Before(c.startsAt) && now.Before(c.endsAt) &&
   193  		c.issuedCount < c.totalLimit
   194  }

   196  // Claim は、受け取りを 1 件受け付け、発行済み枚数を 1 増やし、受け取りの事実を返します。
   197  //
   198  // 停止済みなら ErrSuspended、配布期間の外なら ErrNotDistributing、総枚数上限に達していれば
   199  // ErrTotalLimitReached、その利用者が 1 人あたり上限に達していれば ErrPerUserLimitReached を返し、
   200  // いずれも状態を変えません。受け取り枚数が負の場合は ErrInvalidClaimedByUser を返します。
   201  //
   202  // **受け取りの事実（[Claim]）はこのメソッドだけが生みます。** 遷移に成功したときだけ記録を返すため、
   203  // 記録を得た呼び出し元は必ず遷移を通っています（理由は internal/domain/README.md の
   204  // Aggregate consistency と docs/spec/domain/campaign.md の Claim を参照）。
   205  //
   206  // 呼び出す前にキャンペーン行の排他ロックを取ること。ここが守る上限は「読んで判断して書く」形であり、
   207  // ロックを条件の評価より後に取ると直列化されません（ADR-0036 (ordered-pessimistic-row-locks) の決定 2）。
   208  func (c *Campaign) Claim(params ClaimParams) (*Claim, error) {
   209  	if params.ClaimedByUser < 0 {
   210  		return nil, xerrors.Wrap(ErrInvalidClaimedByUser, "claimedByUser must not be negative")
   211  	}
   212  	if c.IsSuspended() {
   213  		return nil, ErrSuspended
   214  	}
   215  	if params.ClaimedAt.Before(c.startsAt) || !params.ClaimedAt.Before(c.endsAt) {
   216  		return nil, ErrNotDistributing
   217  	}
   218  	if c.issuedCount >= c.totalLimit {
   219  		return nil, ErrTotalLimitReached
   220  	}
   221  	if params.ClaimedByUser >= c.perUserLimit {
   222  		return nil, ErrPerUserLimitReached
   223  	}
   224  
   225  	record, err := newClaim(params.ClaimID, ClaimAttributes{
   226  		CampaignID: c.id,
   227  		UserID:     params.UserID,
   228  		CouponID:   params.CouponID,
   229  		ClaimedAt:  params.ClaimedAt,
   230  	})
   231  	if err != nil {
   232  		return nil, err
   233  	}
   234  
   235  	c.issuedCount++
   236  
   237  	return record, nil
   238  }
