    49  //
    50  // 種別の名前が空、値引きの値が 0 以下、任意の金額が 0 以下、有効期限がゼロ値の場合は検証エラーを返します。
    51  // 利用開始日時は有効期限より前である必要があります。
    52  func NewTemplate(attrs TemplateAttributes) (Template, error) {
    53  	if attrs.DiscountKindName == "" {
    54  		return Template{}, xerrors.Wrap(ErrInvalidDiscountKind, "discountKindName is required")
    55  	}
    56  	if attrs.DiscountValue.Sign() <= 0 {
    57  		return Template{}, xerrors.Wrap(ErrInvalidDiscountValue, "discountValue must be positive")
    58  	}
    59  	if attrs.DiscountMaxAmount != nil && *attrs.DiscountMaxAmount <= 0 {
    60  		return Template{}, xerrors.Wrap(ErrInvalidMaxAmount, "discountMaxAmount must be positive")
    61  	}
    62  	if attrs.ScopeKindName == "" {
    63  		return Template{}, xerrors.Wrap(ErrInvalidScopeKind, "scopeKindName is required")
    64  	}
    65  	if attrs.MinPurchaseAmount != nil && *attrs.MinPurchaseAmount <= 0 {
    66  		return Template{}, xerrors.Wrap(ErrInvalidMinPurchaseAmount, "minPurchaseAmount must be positive")
    67  	}
    68  	if attrs.ExpiresAt.IsZero() {
    69  		return Template{}, xerrors.Wrap(ErrInvalidCouponExpiresAt, "expiresAt is required")
    70  	}
    71  	if attrs.UsableFrom != nil && !attrs.ExpiresAt.After(*attrs.UsableFrom) {
    72  		return Template{}, xerrors.Wrap(ErrInvalidUsableFrom, "usableFrom must be before expiresAt")
    73  	}
    74  
    75  	return Template{
    76  		discountKindName:  attrs.DiscountKindName,
    77  		discountValue:     attrs.DiscountValue,
    78  		discountMaxAmount: ptr.Copy(attrs.DiscountMaxAmount),
    79  		scopeKindName:     attrs.ScopeKindName,
    80  		scopeTargetID:     ptr.Copy(attrs.ScopeTargetID),
    81  		minPurchaseAmount: ptr.Copy(attrs.MinPurchaseAmount),
    82  		usableFrom:        ptr.Copy(attrs.UsableFrom),
    83  		expiresAt:         attrs.ExpiresAt,
    84  	}, nil
    85  }
