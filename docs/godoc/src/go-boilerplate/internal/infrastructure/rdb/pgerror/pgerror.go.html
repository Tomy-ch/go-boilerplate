    30  // NormalizeError は、PostgreSQLのエラーをアプリケーション固有のエラーに変換します。
    31  // 既に正規化済みの apperror はそのまま返します。
    32  func NormalizeError(err error) error {
    33  	if err == nil {
    34  		return nil
    35  	}
    36  
    37  	if apperror.IsAppError(err) {
    38  		return err
    39  	}
    40  
    41  	if xerrors.Is(err, pgx.ErrNoRows) {
    42  		return xerrors.Join(apperror.ErrNotFound, err)
    43  	}
    44  
    45  	var pgErr *pgconn.PgError
    46  	if xerrors.As(err, &pgErr) {
    47  		if appErr, ok := sqlstateToAppError[pgErr.Code]; ok {
    48  			return xerrors.Join(appErr, err)
    49  		}
    50  	}
    51  
    52  	if xerrors.Is(err, context.Canceled) {
    53  		return xerrors.Join(apperror.ErrCanceled, err)
    54  	}
    55  
    56  	if IsUnavailable(err) {
    57  		return xerrors.Join(apperror.ErrUnavailable, err)
    58  	}
    59  
    60  	return xerrors.Join(apperror.ErrInternal, err)
    61  }

    63  // NormalizeReconstructError は、DB 行からのエンティティ再構築（ドメインコンストラクタ）が
    64  // 返したエラーをデータ不整合として ErrInternal へ平坦化します。
    65  // 保存済みデータがドメイン不変条件を満たさないのはサーバ内部障害であり、入力検証エラー
    66  // （422 / details 付き）としてクライアントへ露出させないため、元エラーの分類センチネルと
    67  // apperror.Meta を意図的にチェーンから消します（load-bearing flatten）。理由文はメッセージに
    68  // 残るためログには出ます。err が nil の場合は nil を返します。
    69  func NormalizeReconstructError(err error) error {
    70  	if err == nil {
    71  		return nil
    72  	}
    73  	return xerrors.Wrap(apperror.ErrInternal, "entity reconstruction from stored row failed: "+err.Error())
    74  }

    76  // NormalizeExecResult は、影響行数を返す書き込み系クエリの結果を正規化します。
    77  // エラーは NormalizeError と同じ規則で変換し、エラーが無くても影響行数が 0 の場合は
    78  // 対象が存在しないとみなして ErrNotFound を返します（UPDATE / DELETE のサイレント成功を防ぐ）。
    79  func NormalizeExecResult(affected int64, err error) error {
    80  	if err != nil {
    81  		return NormalizeError(err)
    82  	}
    83  	if affected == 0 {
    84  		return xerrors.Wrap(apperror.ErrNotFound, "no rows affected")
    85  	}
    86  	return nil
    87  }

    89  // IsUnavailable は、DB が利用不可能な状態を示すエラーかを判定します。context.DeadlineExceeded・net.Error・PostgreSQL 接続例外クラス(08xxx) を対象とします。
    90  func IsUnavailable(err error) bool {
    91  	if err == nil {
    92  		return false
    93  	}
    94  
    95  	if xerrors.Is(err, context.DeadlineExceeded) {
    96  		return true
    97  	}
    98  
    99  	var ne net.Error
   100  	if xerrors.As(err, &ne) {
   101  		return true
   102  	}
   103  
   104  	return isPgConnectionError(err)
   105  }

   107  // isPgConnectionError は、与えられたエラーがPostgreSQLの接続エラーであるかを判定します。
   108  func isPgConnectionError(err error) bool {
   109  	var pgErr *pgconn.PgError
   110  	if !xerrors.As(err, &pgErr) {
   111  		return false
   112  	}
   113  	return strings.HasPrefix(pgErr.Code, "08")

   122  // IsRetryableTxError は、リトライで解消しうる tx エラーかを判定します。
   123  // 40001 = serialization_failure, 40P01 = deadlock_detected。
   124  // 写像後の sentinel（両者は ErrUnavailable へ写像される）ではなく生 SQLSTATE で判定し、
   125  // 接続断など他の ErrUnavailable をリトライ対象に巻き込まないようにします。
   126  func IsRetryableTxError(err error) bool {
   127  	var pgErr *pgconn.PgError
   128  	return xerrors.As(err, &pgErr) && (pgErr.Code == "40001" || pgErr.Code == "40P01")
   129  }
