    17  // queryMetrics は、DB クエリの duration / error を Prometheus メトリクスとして記録する driver.QueryRecorder 実装です。
    18  type queryMetrics struct {
    19  	duration    *prometheus.HistogramVec

    23  // NewQueryRecorder は、DB クエリメトリクス(rdb_query_duration_seconds / rdb_query_errors_total)を
    24  // 指定レジストリに登録し、recorder を返します。既に登録済みの場合は既存のコレクタを再利用します。
    25  func NewQueryRecorder(reg prometheus.Registerer) driver.QueryRecorder {
    26  	duration := prometheus.NewHistogramVec(prometheus.HistogramOpts{
    27  		Namespace: queryNamespace,
    28  		Subsystem: querySubsystem,
    29  		Name:      "duration_seconds",
    30  		Help:      "Duration of DB queries in seconds, partitioned by query name, operation and status.",
    31  		Buckets:   prometheus.DefBuckets,
    32  	}, []string{"query_name", "operation", "status"})
    33  
    34  	errorsTotal := prometheus.NewCounterVec(prometheus.CounterOpts{
    35  		Namespace: queryNamespace,
    36  		Subsystem: querySubsystem,
    37  		Name:      "errors_total",
    38  		Help:      "Total number of failed DB queries, partitioned by query name, operation and error class.",
    39  	}, []string{"query_name", "operation", "error_class"})
    40  
    41  	return &queryMetrics{
    42  		duration:    registerOrExisting(reg, duration),
    43  		errorsTotal: registerOrExisting(reg, errorsTotal),
    44  	}
    45  }
    46  
    47  // Observe は、1 クエリの duration を記録し、エラー時のみ error counter を増分します。
    48  // QueryAttrs には SQL 本文・bind 値・テーブル名などの高カーディナリティ／秘匿情報が含まれないため、
    49  // ラベルへ秘匿情報が漏れることはありません。
    50  func (m *queryMetrics) Observe(_ context.Context, attrs driver.QueryAttrs) {
    51  	m.duration.
    52  		WithLabelValues(attrs.QueryName, attrs.Operation, attrs.Status).
    53  		Observe(attrs.Duration.Seconds())
    54  
    55  	// ErrorClass が非空のときのみエラー（pgx.ErrNoRows を除く失敗）として計上します。
    56  	if attrs.ErrorClass != "" {
    57  		m.errorsTotal.WithLabelValues(attrs.QueryName, attrs.Operation, attrs.ErrorClass).Inc()
    58  	}
    59  }
    60  
    61  // registerOrExisting は、コレクタを登録します。既に同一コレクタが登録済みの場合は、
    62  // 新規生成したものではなく登録済みのコレクタを返し、複数回初期化されても同じメトリクスへ記録できるようにします。
    63  func registerOrExisting[T prometheus.Collector](reg prometheus.Registerer, c T) T {
    64  	err := reg.Register(c)
    65  	if err == nil {
    66  		return c
    67  	}
    68  
    69  	var alreadyRegisteredErr prometheus.AlreadyRegisteredError
    70  	if xerrors.As(err, &alreadyRegisteredErr) {
    71  		if existing, ok := alreadyRegisteredErr.ExistingCollector.(T); ok {
    72  			return existing
    73  		}
    74  	}
    75  	return c
    76  }
