    54  func Test_service_SummarizeByUserID(t *testing.T) {

    56

    65  		t.Run("本人の購入をステータス単位に集計しマスタの表示順で返す", func(t *testing.T) {
    66  			t.Parallel()
    67  
    68  			txm.WithinTx(func(ctx context.Context) {
    69  				drv := driver.New(ctx, testDB)

    74  				insertPurchase(ctx, t, drv, userA, mustParse(t, seedUnprocessedSID), "sm-unproc-2", 250)
    75

    77  				require.NoError(t, err)

    81  				assert.Equal(t, "未処理", got[0].StatusName)
    82  				assert.Equal(t, int64(2), got[0].Count)
    83  				assert.Equal(t, int64(400), got[0].TotalAmount)
    84  
    85  				assert.Equal(t, mustParse(t, seedPaidSID), got[1].StatusID)
