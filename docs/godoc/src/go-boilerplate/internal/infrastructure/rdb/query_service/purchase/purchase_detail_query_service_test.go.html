    88  
    89  	testDB := testkit.NewTestDB(t)
    90  	lt := observability.NewMockInfraLayerTracer(t)

    97  		t.Run("本人の購入を明細と商品名の結合込みで取得する", func(t *testing.T) {
    98  			t.Parallel()
    99  
   100  			txm.WithinTx(func(ctx context.Context) {
   101  				drv := driver.New(ctx, testDB)
   102  				userA := mustParse(t, seedUserA)
   103  				productA := insertProduct(ctx, t, drv, "e1000000-0000-4000-8000-000000000001", "商品A", 80000)
   104  				productB := insertProduct(ctx, t, drv, "e1000000-0000-4000-8000-000000000002", "商品B", 120000)
   105  				purchaseID := insertPurchase(ctx, t, drv, userA, mustParse(t, seedUnprocessedSID), "qs-code-ok", nil, nil)
   106  				// 明細 2 件を 1 回のリスト取得で束ねられること（固定 2 クエリ・N+1 でない）を確認する。
   107  				insertDetail(ctx, t, drv, purchaseID, productA, 2, 800)
   108  				insertDetail(ctx, t, drv, purchaseID, productB, 1, 1500)
   109  
   110  				got, err := svc.FindDetailByUserAndID(ctx, userA, purchaseID)
   111  				require.NoError(t, err)
   112  				assert.Equal(t, purchaseID, got.ID)
   113  				assert.Equal(t, userA, got.UserID)
   114  				assert.Equal(t, mustParse(t, seedUnprocessedSID), got.StatusID)
   115  				assert.Equal(t, "未処理", got.StatusName)
   116  				assert.Equal(t, int64(160000), got.SubtotalAmount)
   117  				assert.Equal(t, int64(16000), got.TaxAmount)
   118  				assert.Equal(t, int64(500), got.ShippingFee)
   119  				assert.Equal(t, int64(176500), got.TotalAmount)
   120  				assert.Nil(t, got.PaidAt)
   121  				assert.Nil(t, got.CanceledAt)
   122  
   123  				require.Len(t, got.Items, 2)
   124  				assert.Equal(t, productA, got.Items[0].ProductID)
   125  				assert.Equal(t, "商品A", got.Items[0].ProductName)
   126  				assert.Equal(t, 2, got.Items[0].Quantity)
   127  				assert.Equal(t, "800", got.Items[0].UnitPrice.String())
   128  				assert.Equal(t, productB, got.Items[1].ProductID)
   129  				assert.Equal(t, "商品B", got.Items[1].ProductName)
   130  				assert.Equal(t, "1500", got.Items[1].UnitPrice.String())
