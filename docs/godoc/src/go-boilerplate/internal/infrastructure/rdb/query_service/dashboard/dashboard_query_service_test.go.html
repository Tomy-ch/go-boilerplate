    58  // clearSeededPurchases は、呼び出したトランザクション内で購入と購入明細を空にします。
    59  // 売上集計は購入テーブル全体を期間で切って集計するため、seed が投入する購入履歴が期間に入ると
    60  // 金額と件数の期待値が崩れます。削除はロールバックされるので、コミット済みの seed は失われません。
    61  func clearSeededPurchases(ctx context.Context, t *testing.T, db driver.DBTX) {
    62  	t.Helper()
    63  	_, err := db.Exec(ctx, "DELETE FROM purchase_details")
    64  	require.NoError(t, err)
    65  	_, err = db.Exec(ctx, "DELETE FROM purchases")
    66  	require.NoError(t, err)
    67  }
