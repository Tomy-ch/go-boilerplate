    88  // resolveWindow は、集計対象期間の指定を SQL パラメータの半開区間 [after, before) へ変換します。
    89  // 現在時刻とタイムゾーンへの依存をインフラ層へ閉じ込め、暦日の境界は loc で解釈します。
    90  // loc は設定のタイムゾーンから構築された値であり、実行環境の time.Local には依存しません
    91  // （コンテナの既定は UTC のため、依存させると設定と異なる暦日で集計してしまいます）。
    92  // 区分が未知の場合は today として扱います（ユースケース側で正規化済みのため通常は到達しません）。
    93  func resolveWindow(period query.Period, now time.Time, loc *time.Location) (time.Time, time.Time) {
    94  	// 呼出側の変換有無に依存せず loc 基準の暦日を得るため、ここで現在時刻を loc へ移す。
    95  	// period.From / To は利用者が指定した暦日そのものを表すため、同じ変換をしてはならない
    96  	// （UTC より西のロケーションでは前日へずれる）。
    97  	now = now.In(loc)
    98  
    99  	switch period.Kind {
   100  	case query.PeriodMonth:
   101  		start := time.Date(now.Year(), now.Month(), 1, 0, 0, 0, 0, loc)
   102  		return start, start.AddDate(0, 1, 0)
   103  	case query.PeriodRange:
   104  		return startOfDay(period.From, loc), startOfDay(period.To, loc).AddDate(0, 0, 1)
   105  	case query.PeriodToday:
   106  		start := startOfDay(now, loc)
   107  		return start, start.AddDate(0, 0, 1)
   108  	default:
   109  		start := startOfDay(now, loc)
   110  		return start, start.AddDate(0, 0, 1)
   111  	}
   112  }
   113  
   114  // startOfDay は、t が表す年月日の開始時刻を loc のゾーンで返します。年月日は t 自身のロケーションで解釈し、
   115  // loc は返り値のゾーンとしてのみ用います（t を loc へ変換し直しません）。
   116  func startOfDay(t time.Time, loc *time.Location) time.Time {
   117  	return time.Date(t.Year(), t.Month(), t.Day(), 0, 0, 0, 0, loc)
   118  }
