    30  // buildLikeTokens は、キーワードを LIKE パターンへ変換します。空の場合は全件マッチの ["%"] を返します。
    31  func buildLikeTokens(keywords []string) []string {
    32  	if len(keywords) == 0 {
    33  		return []string{"%"}
    34  	}
    35  	tokens := make([]string, len(keywords))
    36  	for i, kw := range keywords {
    37  		escaped := sqlc.EscapeForLike(kw, sqlc.DefaultLikeEscapeChar)
    38  		tokens[i] = sqlc.WrapContainsLikePattern(escaped)
    39  	}
    40  	return tokens
    41  }

    60  // FindByFilter は、キーワード検索でユーザーの情報を取得します。
    61  func (s *service) FindByFilter(ctx context.Context, filter *query.UserSearchFilter, limit, offset int32) (query.UserSearchResults, error) {
    62  	ctx, endSpan := s.tracer.Start(ctx)
    63  	defer endSpan()
    64  
    65  	tokens := buildLikeTokens(filter.Keywords)
    66  	db := gen.New(s.db.NewLoggingDB(ctx))
    67  
    68  	switch {
    69  	case filter.Active == nil:
    70  		return fetchSearchAll(ctx, db, &gen.SearchUsersParams{
    71  			PatternsParam: tokens,
    72  			LimitParam:    limit,
    73  			OffsetParam:   offset,
    74  		})
    75  	case *filter.Active:
    76  		return fetchSearchActive(ctx, db, &gen.SearchActiveUsersParams{
    77  			PatternsParam: tokens,
    78  			LimitParam:    limit,
    79  			OffsetParam:   offset,
    80  		})
    81  	default:
    82  		return fetchSearchDeleted(ctx, db, &gen.SearchDeletedUsersParams{
    83  			PatternsParam: tokens,
    84  			LimitParam:    limit,
    85  			OffsetParam:   offset,
    86  		})
    87  	}
    88  }

    91  func fetchSearchAll(
    92  	ctx context.Context, db *gen.Queries, searchParams *gen.SearchUsersParams,
    93  ) (query.UserSearchResults, error) {
    94  	rows, err := db.SearchUsers(ctx, searchParams)
    95  	if err != nil {
    96  		return nil, pgerror.NormalizeError(err)
    97  	}
    98  	results := make(query.UserSearchResults, len(rows))
    99  	for i, row := range rows {
   100  		results[i] = toUserSearchResult(row.Users, row.PrefectureName)
   101  	}
   102  	return results, nil
   103  }

   136  func (s *service) CountByFilter(ctx context.Context, filter *query.UserSearchFilter) (int64, error) {
   137  	ctx, endSpan := s.tracer.Start(ctx)
