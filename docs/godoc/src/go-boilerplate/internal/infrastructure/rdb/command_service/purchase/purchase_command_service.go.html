    36  // CreatePurchase は、在庫減算・purchases INSERT・purchase_details INSERT を渡された tx 内で原子的に実行します。
    37  // 在庫減算は防御的に売り越しを弾き、更新 0 行の場合は ErrInsufficientStock（409）を返します。
    38  // この防御は domain の売り越し判定を言い換えた fail-closed の二重防御であり、独立した規則ではありません
    39  // （ADR-0027 § Derivation）。返すエラーも domain の sentinel をそのまま用います。
    40  func (c *commandService) CreatePurchase(ctx context.Context, p *purchase.Purchase) error {
    41  	ctx, endSpan := c.tracer.Start(ctx)
    42  	defer endSpan()
    43  
    44  	db := gen.New(driver.New(ctx, c.db))
    45  	details := p.Details()
    46  
    47  	quantities, err := toDetailQuantities(details)
    48  	if err != nil {
    49  		return err
    50  	}
    51  	statusCode, err := safecast.IntToInt16(p.StatusCode())
    52  	if err != nil {
    53  		return xerrors.Wrap(err, "invalid purchase status code")
    54  	}
    55  
    56  	for i, d := range details {
    57  		affected, derr := db.DecrementProductStock(ctx, &gen.DecrementProductStockParams{
    58  			QuantityParam:  quantities[i],
    59  			ProductIDParam: d.ProductID(),
    60  		})
    61  		if derr != nil {
    62  			return pgerror.NormalizeError(derr)
    63  		}
    64  		if affected == 0 {
    65  			return purchase.ErrInsufficientStock
    66  		}
    67  	}
    68  
    69  	if err := db.InsertPurchase(ctx, &gen.InsertPurchaseParams{
    70  		ID:             p.ID(),
    71  		Code:           p.Code(),
    72  		UserID:         p.UserID(),
    73  		StatusCode:     statusCode,
    74  		SubtotalAmount: int64(p.SubtotalAmount()),
    75  		TaxAmount:      int64(p.TaxAmount()),
    76  		ShippingFee:    int64(p.ShippingFee()),
    77  		TotalAmount:    int64(p.TotalAmount()),
    78  	}); err != nil {
    79  		return pgerror.NormalizeError(err)
    80  	}
    81  
    82  	for i, d := range details {
    83  		if err := db.InsertPurchaseDetail(ctx, &gen.InsertPurchaseDetailParams{
    84  			ID:         d.ID(),
    85  			PurchaseID: p.ID(),
    86  			ProductID:  d.ProductID(),
    87  			Quantity:   quantities[i],
    88  			UnitPrice:  d.UnitPrice().Decimal(),
    89  		}); err != nil {
    90  			return pgerror.NormalizeError(err)
    91  		}
    92  	}
    93  	return nil
    94  }

    99  func (c *commandService) LockPurchase(ctx context.Context, id uuid.UUID) (*purchase.Purchase, error) {
   100  	ctx, endSpan := c.tracer.Start(ctx)
   101  	defer endSpan()
   102  
   103  	db := gen.New(driver.New(ctx, c.db))
   104  
   105  	row, err := db.GetPurchaseByIDForUpdate(ctx, id)
   106  	if err != nil {
   107  		return nil, pgerror.NormalizeError(err)
   108  	}
   109  
   110  	detailRows, err := db.ListPurchaseDetailsByPurchaseID(ctx, id)
   111  	if err != nil {
   112  		return nil, pgerror.NormalizeError(err)

   114  
   115  	details := make([]purchase.PurchaseDetail, len(detailRows))
   116  	for i, dr := range detailRows {
   117  		d := dr.PurchaseDetails
   118  		unitPrice, perr := money.NewPrice(d.UnitPrice)
   119  		if perr != nil {
   120  			return nil, pgerror.NormalizeReconstructError(perr)
   121  		}
   122  		details[i] = purchase.NewPurchaseDetail(d.ID, purchase.PurchaseDetailAttributes{
   123  			ProductID: d.ProductID,
   124  			Quantity:  int(d.Quantity),
   125  			UnitPrice: unitPrice,
   126  		})
   127  	}
   128  
   129  	p := row.Purchases
   130  	entity, err := purchase.Reconstruct(p.ID, purchase.Attributes{
   131  		Code:           p.Code,
   132  		UserID:         p.UserID,
   133  		StatusID:       p.StatusID,
   134  		StatusCode:     int(row.StatusCode),
   135  		SubtotalAmount: int(p.SubtotalAmount),
   136  		TaxAmount:      int(p.TaxAmount),
   137  		ShippingFee:    int(p.ShippingFee),
   138  		TotalAmount:    int(p.TotalAmount),
   139  		Details:        details,
   140  		OrderedAt:      p.OrderedAt,
   141  		PaidAt:         p.PaidAt,
   142  		CanceledAt:     p.CanceledAt,
   143  		ShippedAt:      p.ShippedAt,
   144  		DeliveredAt:    p.DeliveredAt,
   145  	})
   146  	if err != nil {

   153  // 渡された tx 内で原子的に実行します。在庫加算は相対更新で売り越しを生まないため在庫不足ガードは不要です。
   154  func (c *commandService) CancelPurchase(ctx context.Context, p *purchase.Purchase) error {
   155  	ctx, endSpan := c.tracer.Start(ctx)
   156  	defer endSpan()
   157  
   158  	db := gen.New(driver.New(ctx, c.db))
   159  	details := p.Details()
   160  
   161  	quantities, err := toDetailQuantities(details)
   162  	if err != nil {
   163  		return err
   164  	}
   165  	statusCode, err := safecast.IntToInt16(p.StatusCode())
   166  	if err != nil {
   167  		return xerrors.Wrap(err, "invalid purchase status code")
   168  	}
   169  
   170  	for i, d := range details {
   171  		affected, ierr := db.IncrementProductStock(ctx, &gen.IncrementProductStockParams{
   172  			QuantityParam:  quantities[i],
   173  			ProductIDParam: d.ProductID(),
   174  		})
   175  		// 対象商品が不存在（影響 0 行）なら fail-closed で NotFound（減算側の売り越しガードと対称の二重防御）。
   176  		if nerr := pgerror.NormalizeExecResult(affected, ierr); nerr != nil {
   177  			return nerr
   178  		}
   179  	}
   180  
   181  	if err := db.UpdatePurchaseCanceled(ctx, &gen.UpdatePurchaseCanceledParams{
   182  		StatusCode: statusCode,
   183  		CanceledAt: p.CanceledAt(),
   184  		ID:         p.ID(),
   185  	}); err != nil {
   186  		return pgerror.NormalizeError(err)
   187  	}
   188  	return nil
   189  }

   191  // toDetailQuantities は、購入明細の数量を sqlc の int32（DB INTEGER 列）へ明細の順序どおりに変換します。
   192  // 数量が INTEGER 列に収まらない明細が 1 つでもあれば、部分的な書き込みを避けるため変換全体を失敗させます。
   193  func toDetailQuantities(details []purchase.PurchaseDetail) ([]int32, error) {
   194  	quantities := make([]int32, len(details))
   195  	for i, d := range details {
   196  		q, err := safecast.IntToInt32(d.Quantity())
   197  		if err != nil {
   198  			return nil, xerrors.Wrap(err, "invalid purchase detail quantity")
   199  		}
   200  		quantities[i] = q
   201  	}
   202  	return quantities, nil
   203  }
