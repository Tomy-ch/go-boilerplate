   129  // FindRolesByUserID が行→エンティティ変換で再構築エラーになる経路を検証するために使用します。
   130  func insertInvalidRole(ctx context.Context, t *testing.T, db driver.DBTX, userID, roleID uuid.UUID) {
   131  	t.Helper()
   132  
   133  	_, err := db.Exec(ctx,
   134  		"INSERT INTO roles (id, name, code) VALUES ($1, $2, $3)",

   136  	)
   137  	require.NoError(t, err)
   138  
   139  	_, err = db.Exec(ctx,
   140  		"INSERT INTO user_roles (user_id, role_id) VALUES ($1, $2)",
   141  		userID, roleID,
   142  	)
