    49  	testDB := testkit.NewTestDB(t)
    50  
    51  	// 期待値をシードの総件数で固定しているため、コミット済みの行を足し引きする他テスト
    52  	// （user_repository_lock_race_test.go は検証用ユーザーを実コミットする）と重なると件数がずれる。
    53  	// 読み取り専用だが直列化スイートを占有して重なりを防ぐ。
    54  	//
    55  	// このテスト関数だけ t.Parallel() を呼ばないのは、直列化ロックを保持したまま並列テストとして
    56  	// 待たされると、並列スロットを占有したまま自分のサブテストを開始できず停止するため。
    57  	// 逐次フェーズで走らせれば、他の並列テストが再開する前にロックを解放して終えられる。
    58  	testkit.HoldSuiteSerialization(t, testDB)
    59  
    60  	lt := observability.NewMockInfraLayerTracer(t)
    61  
    62  	repo := &repository{tracer: lt, db: testDB}
    63  
    64  	t.Run("正常系", func(t *testing.T) {
    65  		t.Parallel()
    66  
    67  		t.Run("activeがnilの場合、全ての状態のユーザーが対象になり作成日時の降順で返る", func(t *testing.T) {
    68  			t.Parallel()
    69  
    70  			ctx := context.Background()
    71  
    72  			// Grace(アクティブ) は Charlie(削除済み) より後に作成されており、created_at DESC で先頭に来る。
    73  			// = キーワードフィルタと並び順（created_at DESC）を同時に検証する。
    74  			actual, err := repo.SearchByKeyword(ctx, []string{"Grace", "Charlie"}, nil, 10, 0)
    75  			require.NoError(t, err)
    76  
    77  			require.Len(t, actual, 2)
    78  			assert.Equal(t, "Grace", actual[0].FirstName())
    79  			assert.Equal(t, "Lee", actual[0].LastName())

    81  			assert.Equal(t, "Davis", actual[1].LastName())
    82  		})
    83

    88

    90  			require.NoError(t, err)
    91  
    92  			require.Len(t, actual, 1)

    95  		})
