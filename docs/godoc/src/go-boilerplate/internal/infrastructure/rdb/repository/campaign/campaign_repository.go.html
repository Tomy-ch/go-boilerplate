   225  // rowToCampaign は、永続化された行からキャンペーンを再構築します。
   226  func rowToCampaign(row gen.Campaigns) (*campaign.Campaign, error) {
   227  	code, err := campaign.NewCode(row.Code)
   228  	if err != nil {
   229  		return nil, pgerror.NormalizeReconstructError(err)
   230  	}
   231  
   232  	template, err := campaign.NewTemplate(campaign.TemplateAttributes{
   233  		DiscountKindName:  row.DiscountKind,
   234  		DiscountValue:     row.DiscountValue,
   235  		DiscountMaxAmount: row.DiscountMaxAmount,
   236  		ScopeKindName:     row.ScopeKind,
   237  		ScopeTargetID:     row.ScopeTargetID,
   238  		MinPurchaseAmount: row.CouponMinPurchaseAmount,
   239  		UsableFrom:        row.CouponUsableFrom,
   240  		ExpiresAt:         row.CouponExpiresAt,
   241  	})
   242  	if err != nil {
   243  		return nil, pgerror.NormalizeReconstructError(err)
   244  	}
   245  
   246  	c, err := campaign.Reconstruct(row.ID, campaign.Attributes{
   247  		Code:         code,
   248  		Template:     template,
   249  		StartsAt:     row.StartsAt,
   250  		EndsAt:       row.EndsAt,
   251  		TotalLimit:   int(row.TotalLimit),
   252  		PerUserLimit: int(row.PerUserLimit),
   253  	}, int(row.IssuedCount), row.SuspendedAt)
   254  	if err != nil {
   255  		return nil, pgerror.NormalizeReconstructError(err)
   256  	}
   257  
   258  	return c, nil
   259  }
