    54  	t.Parallel()
    55  
    56  	testDB := testkit.NewTestDB(t)
    57  	lt := observability.NewMockInfraLayerTracer(t)
    58  	txm := testkit.NewTestTransactionRunner(t)
    59  
    60  	repo := &repository{tracer: lt, db: testDB}
    61  
    62  	// seed より確実に新しい固定時刻。id の大小でタイブレークを検証するため tie ペアを含む。
    63  	base := time.Date(2099, time.January, 1, 0, 0, 0, 0, time.UTC)
    64  	tieHigh := "eeeeeeee-0000-4000-8000-000000000002"
    65  	tieLow := "eeeeeeee-0000-4000-8000-000000000001"
    66  	mid := "eeeeeeee-0000-4000-8000-000000000003"  // base-1h
    67  	old := "eeeeeeee-0000-4000-8000-000000000004"  // base-2h
    68  	null := "eeeeeeee-0000-4000-8000-000000000005" // published_at=NULL
    69  
    70  	// insertProbeSet は、tie ペア + 2 件 + 未公開(NULL) を probeKeyword 付きで挿入します。
    71  	insertProbeSet := func(ctx context.Context, t *testing.T, drv driver.DBTX) {
    72  		t.Helper()
    73  		insertProduct(ctx, t, drv, tieHigh, probeKeyword+"-A", nil, 1000, statusInStock, categoryElectronics, ptr.To(base))
    74  		insertProduct(ctx, t, drv, tieLow, probeKeyword+"-B", nil, 1000, statusInStock, categoryElectronics, ptr.To(base))
    75  		insertProduct(ctx, t, drv, mid, probeKeyword+"-C", nil, 1000, statusInStock, categoryElectronics, ptr.To(base.Add(-time.Hour)))
    76  		insertProduct(ctx, t, drv, old, probeKeyword+"-D", nil, 1000, statusInStock, categoryElectronics, ptr.To(base.Add(-2*time.Hour)))
    77  		insertProduct(ctx, t, drv, null, probeKeyword+"-NULL", nil, 1000, statusInStock, categoryElectronics, nil)
    78  	}
    79  
    80  	t.Run("正常系", func(t *testing.T) {
    81  		t.Parallel()

    83  		t.Run("降順で先頭ページとafter境界がkeyset安定順に次ページを返し未公開行は除外される", func(t *testing.T) {
    84  			t.Parallel()
    85  
    86  			txm.WithinTx(func(ctx context.Context) {
    87  				drv := driver.New(ctx, testDB)
    88  				insertProbeSet(ctx, t, drv)
    89  
    90  				// 先頭ページ(limit=2): 公開日時降順・id降順で tieHigh, tieLow。
    91  				firstPage, err := repo.FindPublishedList(ctx, domainproduct.ListParams{
    92  					Limit: 2, Ascending: false, Keyword: ptr.To(probeKeyword),
    93  				})
    94  				require.NoError(t, err)
    95  				require.Len(t, firstPage, 2)
    96  				assert.Equal(t, mustParse(t, tieHigh), firstPage[0].ID())
    97  				assert.Equal(t, mustParse(t, tieLow), firstPage[1].ID())
    98  
    99  				// 次ページ: 末尾行(tieLow)を境界に keyset を進めると mid → old。
   100  				last := firstPage[len(firstPage)-1]
   101  				secondPage, err := repo.FindPublishedList(ctx, domainproduct.ListParams{
   102  					Limit: 2, Ascending: false, Keyword: ptr.To(probeKeyword),
   103  					AfterPublishedAt: ptr.To(last.PublishedAt()), AfterID: ptr.To(last.ID()),
   104  				})
   105  				require.NoError(t, err)
   106  				require.Len(t, secondPage, 2)
   107  				assert.Equal(t, mustParse(t, mid), secondPage[0].ID())
   108  				assert.Equal(t, mustParse(t, old), secondPage[1].ID())
   109  
   110  				// 未公開(NULL)行はどのページにも現れない。
   111  				all, err := repo.FindPublishedList(ctx, domainproduct.ListParams{
   112  					Limit: 100, Ascending: false, Keyword: ptr.To(probeKeyword),
   113  				})
   114  				require.NoError(t, err)
   115  				for _, p := range all {
   116  					assert.NotEqual(t, mustParse(t, null), p.ID())
   117  				}
   118  			})
   119  		})
