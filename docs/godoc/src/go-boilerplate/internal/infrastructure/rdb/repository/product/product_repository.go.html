    47  
    48  	db := gen.New(driver.New(ctx, r.db))
    49  	hasAfter := params.AfterPublishedAt != nil && params.AfterID != nil
    50  
    51  	// keyset は並び順ごとに「先頭ページ用」「カーソル以降用」の 2 本の固定クエリへ分ける。
    52  	// 1 本へ畳んでオプショナル述語にすると、実行計画が入力に依存する単一のステートメントになる。

    54  		return productRow{p: p, statusName: statusName, categoryName: categoryName}
    55  	}
    56  
    57  	switch {
    58  	case params.Ascending && hasAfter:
    59  		rows, err := db.ListPublishedProductsAscAfter(ctx, &gen.ListPublishedProductsAscAfterParams{
    60  			CategoryID:       params.CategoryID,
    61  			StatusID:         params.StatusID,
    62  			Keyword:          params.Keyword,
    63  			AfterPublishedAt: params.AfterPublishedAt,
    64  			AfterID:          *params.AfterID,
    65  			LimitParam:       params.Limit,
    66  		})
    67  		if err != nil {
    68  			return nil, pgerror.NormalizeError(err)
    69  		}
    70  		return rowsToProducts(rows, func(row *gen.ListPublishedProductsAscAfterRow) productRow {
    71  			return toRow(row.Products, row.StatusName, row.CategoryName)
    72  		})
    73  	case params.Ascending:
    74  		rows, err := db.ListPublishedProductsAscFirst(ctx, &gen.ListPublishedProductsAscFirstParams{
    75  			CategoryID: params.CategoryID,
    76  			StatusID:   params.StatusID,
    77  			Keyword:    params.Keyword,
    78  			LimitParam: params.Limit,
    79  		})
    80  		if err != nil {
    81  			return nil, pgerror.NormalizeError(err)
    82  		}
    83  		return rowsToProducts(rows, func(row *gen.ListPublishedProductsAscFirstRow) productRow {
    84  			return toRow(row.Products, row.StatusName, row.CategoryName)
    85  		})
    86  	case hasAfter:
    87  		rows, err := db.ListPublishedProductsDescAfter(ctx, &gen.ListPublishedProductsDescAfterParams{
    88  			CategoryID:       params.CategoryID,
    89  			StatusID:         params.StatusID,
    90  			Keyword:          params.Keyword,
    91  			AfterPublishedAt: params.AfterPublishedAt,
    92  			AfterID:          *params.AfterID,
    93  			LimitParam:       params.Limit,
    94  		})
    95  		if err != nil {
    96  			return nil, pgerror.NormalizeError(err)
    97  		}

    99  			return toRow(row.Products, row.StatusName, row.CategoryName)
   100  		})
   101  	default:
   102  		rows, err := db.ListPublishedProductsDescFirst(ctx, &gen.ListPublishedProductsDescFirstParams{

   104  			StatusID:   params.StatusID,

   197  // UpdateStock は、p が保持するバージョンを条件に在庫数を更新し、採番後のバージョンを返します。
   198  func (r *repository) UpdateStock(ctx context.Context, p *product.Product) (int, error) {
   199  	ctx, endSpan := r.tracer.Start(ctx)
   200  	defer endSpan()
   201  
   202  	quantity, err := safecast.IntToInt32(p.Quantity())
   203  	if err != nil {
   204  		return 0, xerrors.Wrap(err, "invalid product quantity")
   205  	}
   206  	currentVersion, err := safecast.IntToInt32(p.Version())
   207  	if err != nil {
   208  		return 0, xerrors.Wrap(err, "invalid product version")
   209  	}
   210  
   211  	db := gen.New(driver.New(ctx, r.db))
   212  	lockVersion, err := db.UpdateProductStock(ctx, &gen.UpdateProductStockParams{
   213  		Quantity:       quantity,
   214  		ID:             p.ID(),
   215  		CurrentVersion: currentVersion,
   216  	})
   217  	if err != nil {
   218  		// 対象行なしは、取得後に他トランザクションが更新しバージョンが進んだことを意味します
   219  		// （存在は同一トランザクション内の取得で確認済みです）。
   220  		// 行ロックを取っている経路では起こりませんが、ロックなしで呼ばれた場合に在庫を上書きしないための
   221  		// 二重防御として衝突を返します。
   222  		if pgerror.IsNoRows(err) {
   223  			return 0, xerrors.Wrap(product.ErrVersionConflict, "product was updated by another transaction")
   224  		}
   225  		return 0, pgerror.NormalizeError(err)
   226  	}
   227  
   228  	return int(lockVersion), nil
   229  }

   265  func (r *repository) Update(ctx context.Context, p *product.Product) (int, error) {
   266  	ctx, endSpan := r.tracer.Start(ctx)

   290  		CategoryID:            p.Category().ID(),
   291  		PublishedAt:           p.PublishedAt(),
   292  		ImagePath:             p.ImagePath(),
   293  		ID:                    p.ID(),
   294  		CurrentVersion:        currentVersion,
   295  	})
   296  	if err != nil {
   297  		// 対象行なしは、読み込み後に他トランザクションが更新しバージョンが進んだことを意味します
   298  		// （存在は同一トランザクション内の読み込みで確認済みです）。
   299  		// tx.Manager が透過リトライする一時障害（serialization_failure）と異なり同じ内容の再送では

   301  		if pgerror.IsNoRows(err) {
   302  			return 0, xerrors.Wrap(product.ErrVersionConflict, "product was updated by another transaction")
   303  		}
   304  		return 0, pgerror.NormalizeError(err)

   327  	ctx, endSpan := r.tracer.Start(ctx)

   352  // rowToProduct は、sqlc が返す商品行（マスタ JOIN 込み）をドメインエンティティへ変換します。
   353  // 再構築時の検証失敗はデータ不整合として ErrInternal へ正規化します（422 / details にしない）。
   354  func rowToProduct(row productRow) (*product.Product, error) {
   355  	price, err := money.NewPrice(row.p.Price)
   356  	if err != nil {
   357  		return nil, pgerror.NormalizeReconstructError(err)
   358  	}
   359  	status, err := product.NewStatusRef(row.p.StatusID, row.statusName)
   360  	if err != nil {
   361  		return nil, pgerror.NormalizeReconstructError(err)
   362  	}
   363  	category, err := product.NewCategoryRef(row.p.CategoryID, row.categoryName)
   364  	if err != nil {
   365  		return nil, pgerror.NormalizeReconstructError(err)
   366  	}
   367  
   368  	entity, err := product.Reconstruct(row.p.ID, product.Attributes{
   369  		Name:                  row.p.Name,
   370  		Description:           row.p.Description,
   371  		Price:                 price,
   372  		Quantity:              int(row.p.Quantity),
   373  		StockWarningThreshold: ptr.Map(row.p.StockWarningThreshold, func(v int32) int { return int(v) }),
   374  		Status:                status,
   375  		Category:              category,
   376  		PublishedAt:           row.p.PublishedAt,
   377  		ImagePath:             row.p.ImagePath,
   378  	}, int(row.p.LockVersion))
   379  	if err != nil {
   380  		return nil, pgerror.NormalizeReconstructError(err)
   381  	}
   382  	return entity, nil
   383  }

   385  // rowsToProducts は、行スライスをドメインエンティティ列へ変換します。
   386  func rowsToProducts[T any](rows []T, extract func(T) productRow) (product.Products, error) {
   387  	products := make(product.Products, len(rows))
   388  	for i, row := range rows {
   389  		p, err := rowToProduct(extract(row))
   390  		if err != nil {
   391  			return nil, err
   392  		}
   393  		products[i] = p
   394  	}
   395  	return products, nil
   396  }
