   259  // UpdateStock は、p が保持するバージョンを条件に在庫数を更新し、採番後のバージョンを返します。
   260  func (r *repository) UpdateStock(ctx context.Context, p *product.Product) (int, error) {
   261  	ctx, endSpan := r.tracer.Start(ctx)
   262  	defer endSpan()
   263  
   264  	quantity, err := safecast.IntToInt32(p.Quantity())
   265  	if err != nil {
   266  		return 0, xerrors.Wrap(err, "invalid product quantity")
   267  	}
   268  	currentVersion, err := safecast.IntToInt32(p.Version())
   269  	if err != nil {
   270  		return 0, xerrors.Wrap(err, "invalid product version")
   271  	}
   272  
   273  	db := gen.New(driver.New(ctx, r.db))
   274  	lockVersion, err := db.UpdateProductStock(ctx, &gen.UpdateProductStockParams{
   275  		Quantity:       quantity,
   276  		ID:             p.ID(),
   277  		CurrentVersion: currentVersion,
   278  	})
   279  	if err != nil {
   280  		// 対象行なしは衝突として返します。行ロックを取っている経路では起こりませんが、ロックなしで
   281  		// 呼ばれた場合に在庫を上書きしないための二重防御なので、到達しないからと外してはなりません。
   282  		if pgerror.IsNoRows(err) {
   283  			return 0, xerrors.Wrap(product.ErrVersionConflict, "product was updated by another transaction")
   284  		}
   285  		return 0, pgerror.NormalizeError(err)
   286  	}
   287  
   288  	return int(lockVersion), nil
   289  }

   324  // Update は、p が保持するバージョンを条件に商品を更新し、採番後のバージョンを返します。
   325  // 画像も p が保持する集合へ一致させます。
   326  func (r *repository) Update(ctx context.Context, p *product.Product) (int, error) {
   327  	ctx, endSpan := r.tracer.Start(ctx)
   328  	defer endSpan()
   329  
   330  	quantity, err := safecast.IntToInt32(p.Quantity())
   331  	if err != nil {
   332  		return 0, xerrors.Wrap(err, "invalid product quantity")
   333  	}
   334  	stockWarningThreshold, err := safecast.IntPtrToInt32Ptr(p.StockWarningThreshold())
   335  	if err != nil {
   336  		return 0, xerrors.Wrap(err, "invalid product stockWarningThreshold")
   337  	}
   338  	currentVersion, err := safecast.IntToInt32(p.Version())
   339  	if err != nil {
   340  		return 0, xerrors.Wrap(err, "invalid product version")
   341  	}
   342  
   343  	db := gen.New(driver.New(ctx, r.db))
   344  	lockVersion, err := db.UpdateProduct(ctx, &gen.UpdateProductParams{
   345  		Name:                  p.Name(),
   346  		Description:           p.Description(),
   347  		Price:                 p.Price().Decimal(),
   348  		Quantity:              quantity,
   349  		StockWarningThreshold: stockWarningThreshold,
   350  		StatusID:              p.Status().ID(),
   351  		CategoryID:            p.Category().ID(),
   352  		PublishedAt:           p.PublishedAt(),
   353  		ID:                    p.ID(),
   354  		CurrentVersion:        currentVersion,
   355  	})
   356  	if err != nil {
   357  		// 対象行なしは衝突として返します。tx.Manager が透過リトライする一時障害と異なり同じ内容の
   358  		// 再送では解消しないため、リトライ対象と混同されてはなりません。
   359  		if pgerror.IsNoRows(err) {
   360  			return 0, xerrors.Wrap(product.ErrVersionConflict, "product was updated by another transaction")
   361  		}
   362  		return 0, pgerror.NormalizeError(err)
   363  	}
   364  
   365  	if err = r.syncImages(ctx, db, p); err != nil {
   366  		return 0, err
   367  	}
   368  
   369  	return int(lockVersion), nil
   370  }

   426  // syncImages は、商品が現在参照している画像を p が保持する集合へ一致させます。
   427  // 集合から外れた行を論理削除してから、まだ無い行を登録します。生存行の (product_id, display_sort) は部分
   428  // UNIQUE インデックスが一意に保つため、この順序でなければ表示順を使い回した登録が 23505 で失敗します。
   429  func (r *repository) syncImages(ctx context.Context, db *gen.Queries, p *product.Product) error {
   430  	images := p.Images()
   431  	ids := make([]uuid.UUID, len(images))
   432  	paths := make([]string, len(images))
   433  	displaySorts := make([]int16, len(images))
   434  	for i, img := range images {
   435  		displaySort, err := safecast.IntToInt16(img.DisplaySort())
   436  		if err != nil {
   437  			return xerrors.Wrap(err, "invalid product image displaySort")
   438  		}
   439  
   440  		ids[i] = img.ID()
   441  		paths[i] = img.ImagePath()
   442  		displaySorts[i] = displaySort
   443  	}
   444  
   445  	if err := db.SoftDeleteProductImagesNotIn(ctx, &gen.SoftDeleteProductImagesNotInParams{
   446  		ProductID: p.ID(),
   447  		ImageIds:  ids,
   448  	}); err != nil {
   449  		return pgerror.NormalizeError(err)
   450  	}
   451  
   452  	if err := db.CreateProductImagesIfAbsent(ctx, &gen.CreateProductImagesIfAbsentParams{
   453  		ProductID:    p.ID(),
   454  		Ids:          ids,
   455  		ImagePaths:   paths,
   456  		DisplaySorts: displaySorts,
   457  	}); err != nil {
   458  		return pgerror.NormalizeError(err)
   459  	}
   460  
   461  	return nil
   462  }

   526  func rowToProduct(row productRow, images []product.Image) (*product.Product, error) {
   527  	price, err := money.NewPrice(row.p.Price)
   528  	if err != nil {
   529  		return nil, pgerror.NormalizeReconstructError(err)
   530  	}
   531  	status, err := product.NewStatusRef(row.p.StatusID, row.statusName)
   532  	if err != nil {

   534  	}
   535  	category, err := product.NewCategoryRef(row.p.CategoryID, row.categoryName)
   536  	if err != nil {
   537  		return nil, pgerror.NormalizeReconstructError(err)
   538  	}
   539  
   540  	entity, err := product.Reconstruct(row.p.ID, product.Attributes{
   541  		Name:                  row.p.Name,
   542  		Description:           row.p.Description,
   543  		Price:                 price,
   544  		Quantity:              int(row.p.Quantity),
   545  		StockWarningThreshold: ptr.Map(row.p.StockWarningThreshold, func(v int32) int { return int(v) }),
   546  		Status:                status,
   547  		Category:              category,
   548  		PublishedAt:           row.p.PublishedAt,
   549  		Images:                images,
   550  	}, int(row.p.LockVersion))
   551  	if err != nil {
   552  		return nil, pgerror.NormalizeReconstructError(err)
   553  	}
