    93  func rowsToProducts[T any](rows []T, extract func(T) gen.Products) (product.Products, error) {
    94  	products := make(product.Products, len(rows))
    95  	for i, row := range rows {

    97  		if err != nil {
    98  			return nil, err
    99  		}
   100  		products[i] = p
   101  	}
   102  	return products, nil
   103  }
