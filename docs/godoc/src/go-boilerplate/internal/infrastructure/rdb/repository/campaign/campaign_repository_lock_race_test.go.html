    56  // Test_lockByCodeSerializesConcurrentClaimsAgainstTotalLimit は、総枚数上限の最後の 1 枚へ 2 つの受け取りが
    57  // 同時に到達したとき、LockByCode が両者を直列化し、待たされた側が「上限に達した」状態を観測することを検証します。
    58  //
    59  // これが DoD「総枚数上限が並行引き換えの下で守られる」の機械的な証明です。判定の前にロックを取らなければ、
    60  // 両者とも「まだ 1 枚残っている」を読んで 2 枚配ってしまいます
    61  // （ADR-0036 (ordered-pessimistic-row-locks) の決定 2）。
    62  //
    63  // tx を 2 本同時に生かす必要があるため testkit.WithinTx（tx 1 本 + ロールバック）では表現できません。
    64  // 先行役をコミットさせる必要から、検証用の行は後始末で物理削除します。
    65  func Test_lockByCodeSerializesConcurrentClaimsAgainstTotalLimit(t *testing.T) {
    66  	t.Parallel()
    67  
    68  	testDB := testkit.NewTestDB(t)
    69  	// 検証用キャンペーンの作成から 2 本の tx の完了までを、他パッケージの CASCADE TRUNCATE から守る。
    70  	testkit.HoldSuiteSerialization(t, testDB)
    71  
    72  	repo := &repository{db: testDB, tracer: observability.NewMockInfraLayerTracer(t)}
    73  	ctx := context.Background()
    74  
    75  	// 残り 1 枚のキャンペーンを立てる。総枚数上限 1 枚に対して発行済み 0 枚。
    76  	target := newRaceCampaign(t, "RACE-TOTAL-LIMIT", 1, 1)
    77  	require.NoError(t, repo.Create(ctx, target))
    78  	couponID := insertTestCoupon(ctx, t, testDB)
    79  	t.Cleanup(func() {
    80  		cleanupRaceRows(ctx, t, testDB, raceRowIDs{CampaignID: target.ID(), CouponID: couponID})
    81  	})
    82  
    83  	firstLocked := make(chan struct{})
    84  	secondDone := make(chan struct{})
    85  	secondResult := make(chan error, 1)
    86  	secondAcquiredAt := make(chan time.Time, 1)
    87  	var firstReleasedAt time.Time
    88  
    89  	// 入力は goroutine の外で組み立てる。testify のアサーションはテスト本体の goroutine でしか使えない。
    90  	secondParams := raceClaimParams(t, mustParse(t, seedAliceUserID), couponID, 0)
    91  
    92  	// 後続役: 先行役がキャンペーン行を押さえている間にロックへ入り、先行役の確定まで待たされる。
    93  	go func() {
    94  		defer close(secondDone)
    95  		<-firstLocked
    96  		secondResult <- newRaceTxManager(t, testDB).Do(ctx, func(txCtx context.Context) error {
    97  			locked, lockErr := repo.LockByCode(txCtx, target.Code())
    98  			secondAcquiredAt <- time.Now()
    99  			if lockErr != nil {
   100  				return xerrors.Join(errRollbackRaceTx, lockErr)
   101  			}
   102  			// ロックを取れた時点で上限に達していることが、直列化の成立を示す。
   103  			if _, claimErr := locked.Claim(secondParams); claimErr != nil {
   104  				return xerrors.Join(errRollbackRaceTx, errLimitObserved, claimErr)
   105  			}
   106  
   107  			return errRollbackRaceTx
   108  		})
   109  	}()
   110  
   111  	// 先行役: キャンペーン行を排他ロックしてから、最後の 1 枚の発行を確定させる。
   112  	require.NoError(t, newRaceTxManager(t, testDB).Do(ctx, func(txCtx context.Context) error {
   113  		locked, lockErr := repo.LockByCode(txCtx, target.Code())
   114  		if lockErr != nil {
   115  			return lockErr
   116  		}
   117  		close(firstLocked)
   118  
   119  		// ロックを握ったまま、後続役が素通りしないことを確かめる。
   120  		select {
   121  		case <-secondDone:
   122  			t.Error("後続の受け取り役が先行役のロックを待たずに完了した")
   123  		case <-time.After(raceBlockedGracePeriod):
   124  		}
   125  		firstReleasedAt = time.Now()
   126  
   127  		record, claimErr := locked.Claim(raceClaimParams(t, mustParse(t, seedAliceUserID), couponID, 0))
   128  		if claimErr != nil {
   129  			return claimErr
   130  		}
   131  
   132  		return repo.RecordClaim(txCtx, record)
   133  	}))
   134  
   135  	<-secondDone
   136  	// 先行役が最後の 1 枚を配った以上、待たされていた後続役は上限到達を観測する。
   137  	secondErr := <-secondResult
   138  	require.ErrorIs(t, secondErr, errLimitObserved)
   139  	require.ErrorIs(t, secondErr, domaincampaign.ErrTotalLimitReached)
   140  	// 上限到達を観測できたのがロック待ちの結果であることを、取得時刻の前後で裏づける。
   141  	assert.True(t, (<-secondAcquiredAt).After(firstReleasedAt))
   142  }

   144  // Test_lockByCodeSerializesConcurrentClaimsAgainstPerUserLimit は、同じ利用者による 2 つの受け取りが
   145  // 1 人あたり上限の最後の 1 枚へ同時に到達したとき、LockByCode が両者を直列化し、待たされた側が
   146  // 「その利用者は上限に達した」状態を観測することを検証します。
   147  //
   148  // これが DoD「1 人あたり上限が並行引き換えの下で守られる」の機械的な証明です。総枚数上限と違い、
   149  // 判定に必要な枚数は別表（campaign_claims）を数えて得ますが、数えるのはキャンペーン行の
   150  // ロックを取ったあとであるため、同じ直列化に乗ります。
   151  func Test_lockByCodeSerializesConcurrentClaimsAgainstPerUserLimit(t *testing.T) {
   152  	t.Parallel()
   153  
   154  	testDB := testkit.NewTestDB(t)
   155  	testkit.HoldSuiteSerialization(t, testDB)
   156  
   157  	repo := &repository{db: testDB, tracer: observability.NewMockInfraLayerTracer(t)}
   158  	ctx := context.Background()
   159  	userID := mustParse(t, seedAliceUserID)
   160  
   161  	// 総枚数には余裕があり、1 人あたり上限だけが 1 枚のキャンペーンを立てる。
   162  	target := newRaceCampaign(t, "RACE-PER-USER", 10, 1)
   163  	require.NoError(t, repo.Create(ctx, target))
   164  	couponID := insertTestCoupon(ctx, t, testDB)
   165  	t.Cleanup(func() {
   166  		cleanupRaceRows(ctx, t, testDB, raceRowIDs{CampaignID: target.ID(), CouponID: couponID})
   167  	})
   168  
   169  	firstLocked := make(chan struct{})
   170  	secondDone := make(chan struct{})
   171  	secondResult := make(chan error, 1)
   172  	secondAcquiredAt := make(chan time.Time, 1)
   173  	var firstReleasedAt time.Time
   174  
   175  	// 入力は goroutine の外で組み立てる（既受け取り枚数だけはロック下で数え直す）。
   176  	secondParams := raceClaimParams(t, userID, couponID, 0)
   177  
   178  	go func() {
   179  		defer close(secondDone)
   180  		<-firstLocked
   181  		secondResult <- newRaceTxManager(t, testDB).Do(ctx, func(txCtx context.Context) error {
   182  			locked, lockErr := repo.LockByCode(txCtx, target.Code())
   183  			secondAcquiredAt <- time.Now()
   184  			if lockErr != nil {
   185  				return xerrors.Join(errRollbackRaceTx, lockErr)
   186  			}
   187  
   188  			// ロックを取ったあとで数えるため、先行役が挿入した記録が見える。
   189  			claimed, countErr := repo.CountClaims(txCtx, domaincampaign.ClaimCountParams{
   190  				CampaignID: target.ID(),
   191  				UserID:     userID,
   192  			})
   193  			if countErr != nil {
   194  				return xerrors.Join(errRollbackRaceTx, countErr)
   195  			}
   196  			params := secondParams
   197  			params.ClaimedByUser = claimed
   198  			if _, claimErr := locked.Claim(params); claimErr != nil {
   199  				return xerrors.Join(errRollbackRaceTx, errLimitObserved, claimErr)
   200  			}
   201  
   202  			return errRollbackRaceTx
   203  		})
   204  	}()
   205  
   206  	require.NoError(t, newRaceTxManager(t, testDB).Do(ctx, func(txCtx context.Context) error {
   207  		locked, lockErr := repo.LockByCode(txCtx, target.Code())
   208  		if lockErr != nil {
   209  			return lockErr
   210  		}
   211  		close(firstLocked)
   212  
   213  		select {
   214  		case <-secondDone:
   215  			t.Error("後続の受け取り役が先行役のロックを待たずに完了した")
   216  		case <-time.After(raceBlockedGracePeriod):
   217  		}
   218  		firstReleasedAt = time.Now()
   219  
   220  		claimed, countErr := repo.CountClaims(txCtx, domaincampaign.ClaimCountParams{
   221  			CampaignID: target.ID(),
   222  			UserID:     userID,
   223  		})
   224  		if countErr != nil {
   225  			return countErr
   226  		}
   227  		record, claimErr := locked.Claim(raceClaimParams(t, userID, couponID, claimed))
   228  		if claimErr != nil {
   229  			return claimErr
   230  		}
   231  
   232  		return repo.RecordClaim(txCtx, record)
   233  	}))
   234  
   235  	<-secondDone
   236  	secondErr := <-secondResult
   237  	require.ErrorIs(t, secondErr, errLimitObserved)
   238  	require.ErrorIs(t, secondErr, domaincampaign.ErrPerUserLimitReached)
   239  	assert.True(t, (<-secondAcquiredAt).After(firstReleasedAt))
   240  }
