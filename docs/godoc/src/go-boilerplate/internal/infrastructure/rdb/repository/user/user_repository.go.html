   128  // buildLikeTokens は、キーワードを部分一致の LIKE パターンへ変換します。空の場合は全件マッチの ["%"] を返します。
   129  func buildLikeTokens(keywords []string) []string {
   130  	if len(keywords) == 0 {
   131  		return []string{"%"}
   132  	}
   133  	tokens := make([]string, len(keywords))
   134  	for i, kw := range keywords {
   135  		escaped := sqlc.EscapeForLike(kw, sqlc.DefaultLikeEscapeChar)
   136  		tokens[i] = sqlc.WrapContainsLikePattern(escaped)
   137  	}
   138  	return tokens
   139  }

   166  // rowsToUsers は、行スライスをドメインエンティティ列へ変換します。
   167  func rowsToUsers[T any](rows []T, extract func(T) gen.Users) (user.Users, error) {
   168  	users := make(user.Users, len(rows))
   169  	for i, row := range rows {
   170  		u, err := rowToUser(extract(row))
   171  		if err != nil {
   172  			return nil, err
   173  		}
   174  		users[i] = u
   175  	}
   176  	return users, nil
   177  }
