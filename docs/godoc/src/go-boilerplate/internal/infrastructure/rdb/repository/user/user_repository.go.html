    78  // rowsToUsers は、行スライスをドメインエンティティ列へ変換します。
    79  func rowsToUsers[T any](rows []T, extract func(T) gen.Users) (user.Users, error) {
    80  	users := make(user.Users, len(rows))
    81  	for i, row := range rows {
    82  		u, err := rowToUser(extract(row))
    83  		if err != nil {
    84  			return nil, err
    85  		}
    86  		users[i] = u
    87  	}
    88  	return users, nil
    89  }
