   107  func rowsToUsers[T any](rows []T, extract func(T) gen.Users) (user.Users, error) {
   108  	users := make(user.Users, len(rows))
   109  	for i, row := range rows {
   110  		u, err := rowToUser(extract(row))
   111  		if err != nil {
   112  			return nil, err
   113  		}
   114  		users[i] = u
   115  	}
   116  	return users, nil
