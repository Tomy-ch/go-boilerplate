   130  // buildLikeTokens は、キーワードを部分一致の LIKE パターンへ変換します。空の場合は全件マッチの ["%"] を返します。
   131  func buildLikeTokens(keywords []string) []string {
   132  	if len(keywords) == 0 {
   133  		return []string{"%"}
   134  	}
   135  	tokens := make([]string, len(keywords))
   136  	for i, kw := range keywords {
   137  		escaped := sqlc.EscapeForLike(kw, sqlc.DefaultLikeEscapeChar)
   138  		tokens[i] = sqlc.WrapContainsLikePattern(escaped)
   139  	}
   140  	return tokens
   141  }

   169  func rowsToUsers[T any](rows []T, extract func(T) gen.Users) (user.Users, error) {
   170  	users := make(user.Users, len(rows))
   171  	for i, row := range rows {
   172  		u, err := rowToUser(extract(row))
   173  		if err != nil {
   174  			return nil, err
   175  		}
   176  		users[i] = u
   177  	}
   178  	return users, nil

   339  // 子行を先に消すことで FK 違反を避けます。3 クエリは渡された ctx のトランザクションで実行されます
   340  // （トランザクションの開始は usecase の責務）。

   342  // そのユーザーは子行を含めて一切削除されず、返る件数が ids の件数を下回ります。
   343  func (r *repository) PurgeByIDs(ctx context.Context, ids []uuid.UUID) (int64, error) {
   344  	ctx, endSpan := r.tracer.Start(ctx)
   345  	defer endSpan()
   346  
   347  	if len(ids) == 0 {
   348  		return 0, nil
   349  	}
   350  
   351  	db := gen.New(driver.New(ctx, r.db))
   352  	if err := db.DeleteUserIdentitiesByUserIDs(ctx, ids); err != nil {
   353  		return 0, pgerror.NormalizeError(err)
   354  	}
   355  	if err := db.DeleteUserRolesByUserIDs(ctx, ids); err != nil {
   356  		return 0, pgerror.NormalizeError(err)
   357  	}
   358  	purged, err := db.DeleteUsersByIDs(ctx, ids)
   359  	if err != nil {
   360  		return 0, pgerror.NormalizeError(err)
   361  	}
   362  	return purged, nil
