   162  // Update は、カートを明細込みで現在の状態へ一致させます。存在しない場合は NotFound を返します。
   163  //
   164  // 複数の文に分かれるため、呼び出し元は ctx にトランザクションを載せてください
   165  // （internal/infrastructure/rdb/README.md の Centralized Transaction Boundary Management）。
   166  // 境界が無いまま呼ぶと、親行だけが確定した状態が残りえます。
   167  func (r *repository) Update(ctx context.Context, c *cart.Cart) error {
   168  	ctx, endSpan := r.tracer.Start(ctx)
   169  	defer endSpan()
   170  
   171  	db := gen.New(driver.New(ctx, r.db))
   172  
   173  	affected, err := db.UpdateCart(ctx, &gen.UpdateCartParams{
   174  		UserID:       c.OwnerID(),
   175  		SessionToken: sessionTokenValue(c),
   176  		ExpiresAt:    c.ExpiresAt(),
   177  		ID:           c.ID(),
   178  	})
   179  	if err := pgerror.NormalizeExecResult(affected, err); err != nil {
   180  		return err
   181  	}
   182  
   183  	items := c.Items()
   184  	productIDs := make([]uuid.UUID, 0, len(items))
   185  	for _, item := range items {
   186  		quantity, cerr := safecast.IntToInt32(item.Quantity())
   187  		if cerr != nil {
   188  			return xerrors.Wrap(cerr, "invalid cart item quantity")
   189  		}
   190  		if uerr := db.UpsertCartItem(ctx, &gen.UpsertCartItemParams{
   191  			ID:            item.ID(),
   192  			CartID:        c.ID(),
   193  			ProductID:     item.ProductID(),
   194  			Quantity:      quantity,
   195  			LastSeenPrice: lastSeenPriceValue(item),
   196  			AddedAt:       item.AddedAt(),
   197  		}); uerr != nil {
   198  			return pgerror.NormalizeError(uerr)
   199  		}
   200  		productIDs = append(productIDs, item.ProductID())
   201  	}
   202  
   203  	if derr := db.DeleteCartItemsNotIn(ctx, &gen.DeleteCartItemsNotInParams{
   204  		CartID:     c.ID(),
   205  		ProductIds: productIDs,
   206  	}); derr != nil {
   207  		return pgerror.NormalizeError(derr)
   208  	}
   209  	return nil
   210  }

   244  func (r *repository) reconstruct(ctx context.Context, db *gen.Queries, row gen.Carts) (*cart.Cart, error) {
   245  	itemRows, err := db.ListCartItemsByCartID(ctx, row.ID)
   246  	if err != nil {
   247  		return nil, pgerror.NormalizeError(err)
   248  	}
   249  
   250  	items, err := toCartItems(itemRows)
   251  	if err != nil {
   252  		return nil, err
   253  	}
   254  
   255  	token, err := toSessionToken(row.SessionToken)
   256  	if err != nil {
   257  		return nil, pgerror.NormalizeReconstructError(err)
   258  	}
   259  
   260  	entity, err := cart.Reconstruct(row.ID, cart.Attributes{
   261  		OwnerID:      row.UserID,
   262  		SessionToken: token,
   263  		Items:        items,
   264  		ExpiresAt:    row.ExpiresAt,
   265  		CreatedAt:    row.CreatedAt,
   266  		UpdatedAt:    row.UpdatedAt,
   267  	})
   268  	if err != nil {
   269  		return nil, pgerror.NormalizeReconstructError(err)
   270  	}
   271  	return entity, nil
   272  }

   275  func toCartItems(rows []*gen.ListCartItemsByCartIDRow) ([]cart.CartItem, error) {
   276  	items := make([]cart.CartItem, 0, len(rows))
   277  	for _, row := range rows {
   278  		ci := row.CartItems
   279  
   280  		price, err := toPrice(ci.LastSeenPrice)
   281  		if err != nil {
   282  			return nil, pgerror.NormalizeReconstructError(err)
   283  		}
   284  
   285  		items = append(items, cart.NewCartItem(ci.ID, cart.CartItemAttributes{
   286  			ProductID:     ci.ProductID,
   287  			Quantity:      int(ci.Quantity),
   288  			AddedAt:       ci.AddedAt,
   289  			LastSeenPrice: price,
   290  		}))
   291  	}
   292  	return items, nil
   293  }
