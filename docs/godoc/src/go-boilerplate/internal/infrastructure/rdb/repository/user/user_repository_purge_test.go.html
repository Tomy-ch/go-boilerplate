    26  func insertPurgeUser(ctx context.Context, t *testing.T, db driver.DBTX, id string, deletedAt *time.Time) uuid.UUID {

    42  	)

    79  	txm := testkit.NewTestTransactionRunner(t)
    80  	repo := &repository{tracer: lt, db: testDB}
    81  
    82  	// ID の昇順が keyset の前進順になるよう、末尾だけが異なる固定 ID を用意する。
    83  	oldest := "eeeeeeee-0000-4000-8000-000000000001"
    84  	middle := "eeeeeeee-0000-4000-8000-000000000002"
    85  	newest := "eeeeeeee-0000-4000-8000-000000000003"
    86  	withinRetention := "eeeeeeee-0000-4000-8000-000000000004"
    87  	notDeleted := "eeeeeeee-0000-4000-8000-000000000005"
    88  	atCutoff := "eeeeeeee-0000-4000-8000-000000000006"
    89  
    90  	deletedAt := time.Date(2000, time.January, 1, 0, 0, 0, 0, time.UTC)
    91  	afterCutoff := time.Date(2010, time.January, 1, 0, 0, 0, 0, time.UTC)
    92  
    93  	insertFixtures := func(ctx context.Context, t *testing.T) (uuid.UUID, uuid.UUID, uuid.UUID) {
    94  		t.Helper()
    95  		drv := driver.New(ctx, testDB)
    96  		oldestID := insertPurgeUser(ctx, t, drv, oldest, &deletedAt)
    97  		middleID := insertPurgeUser(ctx, t, drv, middle, &deletedAt)
    98  		newestID := insertPurgeUser(ctx, t, drv, newest, &deletedAt)
    99  		insertPurgeUser(ctx, t, drv, withinRetention, &afterCutoff)
   100  		insertPurgeUser(ctx, t, drv, notDeleted, nil)
   101  		return oldestID, middleID, newestID
   102  	}
   103

   211  func Test_repository_PurgeByIDs(t *testing.T) {
   212  	t.Parallel()
   213  
   214  	testDB := testkit.NewTestDB(t)
   215  	lt := observability.NewMockInfraLayerTracer(t)
   216  	txm := testkit.NewTestTransactionRunner(t)
   217  	repo := &repository{tracer: lt, db: testDB}
   218  
   219  	const (
   220  		countIdentities = "SELECT COUNT(*) FROM user_identities WHERE user_id = $1"
   221  		countRoles      = "SELECT COUNT(*) FROM user_roles WHERE user_id = $1"
   222  		countUsers      = "SELECT COUNT(*) FROM users WHERE id = $1"
   223  	)
   224  
   225  	target := "eeeeeeee-1111-4000-8000-000000000001"
   226  	survivor := "eeeeeeee-1111-4000-8000-000000000002"
   227  	alive := "eeeeeeee-1111-4000-8000-000000000003"
   228  	deletedAt := time.Date(2000, time.January, 1, 0, 0, 0, 0, time.UTC)
   229  
   230  	t.Run("正常系", func(t *testing.T) {
   231  		t.Parallel()
   232  
   233  		t.Run("対象ユーザーを従属データごと削除し削除件数を返す", func(t *testing.T) {
   234  			t.Parallel()
   235  
   236  			txm.WithinTx(func(ctx context.Context) {
   237  				drv := driver.New(ctx, testDB)
   238  				targetID := insertPurgeUser(ctx, t, drv, target, &deletedAt)
   239  				insertPurgeUserDependents(ctx, t, drv, targetID, "eeeeeeee-2222-4000-8000-000000000001")
   240  
   241  				got, err := repo.PurgeByIDs(ctx, []uuid.UUID{targetID})
   242  				require.NoError(t, err)
   243  
   244  				assert.Equal(t, int64(1), got)
   245  				assert.Equal(t, int64(0), countPurgeRows(ctx, t, drv, countIdentities, targetID))
   246  				assert.Equal(t, int64(0), countPurgeRows(ctx, t, drv, countRoles, targetID))
   247  				assert.Equal(t, int64(0), countPurgeRows(ctx, t, drv, countUsers, targetID))
   248  			})
   249  		})
   250  
   251  		t.Run("対象外のユーザーと従属データは残る", func(t *testing.T) {
   252  			t.Parallel()
   253  
   254  			txm.WithinTx(func(ctx context.Context) {
   255  				drv := driver.New(ctx, testDB)
   256  				targetID := insertPurgeUser(ctx, t, drv, target, &deletedAt)
   257  				insertPurgeUserDependents(ctx, t, drv, targetID, "eeeeeeee-2222-4000-8000-000000000002")
   258  				survivorID := insertPurgeUser(ctx, t, drv, survivor, &deletedAt)
   259  				insertPurgeUserDependents(ctx, t, drv, survivorID, "eeeeeeee-2222-4000-8000-000000000003")
   260  
   261  				_, err := repo.PurgeByIDs(ctx, []uuid.UUID{targetID})
   262  				require.NoError(t, err)
   263  
   264  				assert.Equal(t, int64(1), countPurgeRows(ctx, t, drv, countIdentities, survivorID))
   265  				assert.Equal(t, int64(1), countPurgeRows(ctx, t, drv, countRoles, survivorID))
   266  				assert.Equal(t, int64(1), countPurgeRows(ctx, t, drv, countUsers, survivorID))
   267  			})
   268  		})
   269  
   270  		t.Run("論理削除されていないユーザーは従属データを含め削除しない", func(t *testing.T) {
   271  			t.Parallel()
   272  
   273  			// 物理削除は取り消せないため、候補列挙を誤って現役ユーザーの ID が届いても消えないことを
   274  			// SQL 側の最終防壁として固定する。users にだけガードを付けると、生存したまま
   275  			// 認証アイデンティティとロールだけを失ったアカウントが残るため、従属データも併せて検証する。
   276  			txm.WithinTx(func(ctx context.Context) {
   277  				drv := driver.New(ctx, testDB)
   278  				aliveID := insertPurgeUser(ctx, t, drv, alive, nil)
   279  				insertPurgeUserDependents(ctx, t, drv, aliveID, "eeeeeeee-2222-4000-8000-000000000005")
   280  
   281  				got, err := repo.PurgeByIDs(ctx, []uuid.UUID{aliveID})
   282  				require.NoError(t, err)
   283  
   284  				assert.Equal(t, int64(0), got)
   285  				assert.Equal(t, int64(1), countPurgeRows(ctx, t, drv, countIdentities, aliveID))
   286  				assert.Equal(t, int64(1), countPurgeRows(ctx, t, drv, countRoles, aliveID))
   287  				assert.Equal(t, int64(1), countPurgeRows(ctx, t, drv, countUsers, aliveID))
   288  			})
   289  		})
