    33  	t.Parallel()
    34  
    35  	testDB := testkit.NewTestDB(t)
    36  	lt := observability.NewMockInfraLayerTracer(t)
    37  
    38  	txm := testkit.NewTestTransactionRunner(t)
    39  
    40  	repo := &repository{
    41  		tracer: lt,
    42  		db:     testDB,
    43  	}
    44  
    45  	t.Run("正常系", func(t *testing.T) {
    46  		t.Parallel()
    47  
    48  		t.Run("全商品ステータスが10件sortKey昇順で取得できる", func(t *testing.T) {
    49  			t.Parallel()
    50  
    51  			// seed の sortKey は code と非連動。sortKey=1 は「検討中」(code=8)。
    52  			reviewingID, err := uuid.Parse("bdf44f06-227c-4549-b2c8-e57b32f06321")
    53  			require.NoError(t, err)
    54  
    55  			txm.WithinTx(func(ctx context.Context) {
    56  				expectedReviewing, err := status.New(reviewingID, status.Attributes{Name: "検討中", Code: 8, SortKey: 1})
    57  				require.NoError(t, err)
    58  
    59  				actual, err := repo.FindAll(ctx)
    60  				require.NoError(t, err)
    61  				assert.Len(t, actual, 10)
    62  
    63  				// sort_key は UNIQUE のため厳密昇順が正。
    64  				for i := 1; i < len(actual); i++ {
    65  					assert.Less(t, actual[i-1].SortKey(), actual[i].SortKey())
    66  				}
    67  
    68  				// sortKey 昇順の先頭は「検討中」(sortKey=1)。
    69  				assert.Equal(t, expectedReviewing, actual[0])
    70  			})
    71  		})
    72  
    73  		t.Run("テーブルが空の場合、nilではない空一覧を返す", func(t *testing.T) {
    74  			t.Parallel()

    76  			txm.WithinTx(func(ctx context.Context) {
    77  				// product_statuses を参照する依存行（products など）ごと空にする（tx はロールバックされる）。
