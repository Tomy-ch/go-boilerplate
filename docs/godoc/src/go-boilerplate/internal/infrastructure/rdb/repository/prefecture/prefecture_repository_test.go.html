   145  	t.Parallel()
   146  
   147  	testDB := testkit.NewTestDB(t)
   148  	lt := observability.NewMockInfraLayerTracer(t)
   149  
   150  	txm := testkit.NewTestTransactionRunner(t)
   151  
   152  	repo := &repository{
   153  		tracer: lt,
   154  		db:     testDB,
   155  	}
   156  
   157  	t.Run("正常系", func(t *testing.T) {
   158  		t.Parallel()
   159  
   160  		t.Run("全都道府県が47件、既定seedの順で取得できる", func(t *testing.T) {
   161  			t.Parallel()
   162  
   163  			tokyoID, err := uuid.Parse("101caa1e-84e7-4ceb-9108-50d40b6be1a3")
   164  			require.NoError(t, err)
   165  
   166  			txm.WithinTx(func(ctx context.Context) {
   167  				expectedTokyo, err := prefecture.New(tokyoID, "東京都", 8)
   168  				require.NoError(t, err)
   169  
   170  				actual, err := repo.FindAll(ctx)
   171  				require.NoError(t, err)
   172  				assert.Len(t, actual, 47)
   173  
   174  				// seed は sort_key と code を同じ値で入れているため、既定の並びは code 昇順にも一致する。
   175  				// 並び順の出所が sort_key であることは、直後のケースが両者を食い違わせて確かめる。
   176  				for i := 1; i < len(actual); i++ {
   177  					assert.Less(t, actual[i-1].Code(), actual[i].Code())
   178  				}
   179  
   180  				var actualTokyo *prefecture.Prefecture
   181  				for _, p := range actual {
   182  					if p.ID() == tokyoID {
   183  						actualTokyo = p
   184  						break
   185  					}
   186  				}
   187  				require.NotNil(t, actualTokyo)
   188  				assert.Equal(t, expectedTokyo, actualTokyo)
   189  			})
   190  		})
   191  
   192  		t.Run("並び順はcodeではなくsort_keyが決める", func(t *testing.T) {
   193  			t.Parallel()
   194  
   195  			hokkaidoID, err := uuid.Parse("faba7bb2-f5a0-4a51-adae-1564929077b2")
   196  			require.NoError(t, err)
   197  
   198  			txm.WithinTx(func(ctx context.Context) {
   199  				// seed では code=1 / sort_key=1 で先頭に来る北海道の sort_key だけを最後尾へ動かす。
   200  				// code は動かさないため、並びが code に従っていればこの更新では何も変わらない。
   201  				_, execErr := driver.New(ctx, testDB).Exec(ctx,
   202  					"UPDATE prefectures SET sort_key = $1 WHERE id = $2", 999, hokkaidoID,
   203  				)
   204  				require.NoError(t, execErr)
   205  
   206  				actual, err := repo.FindAll(ctx)
   207  				require.NoError(t, err)
   208  				require.Len(t, actual, 47)
   209  
   210  				assert.Equal(t, hokkaidoID, actual[len(actual)-1].ID())
   211  				assert.NotEqual(t, hokkaidoID, actual[0].ID())
   212  			})
   213  		})

   215  		t.Run("テーブルが空の場合、nilではない空一覧を返す", func(t *testing.T) {
   216  			t.Parallel()
   217  
   218  			// TRUNCATE は依存表ごと ACCESS EXCLUSIVE を取るため、直列化の外側で走る tx と
   219  			// deadlock（40P01）しうる。require で即死させるとトランザクションマネージャーが
   220  			// 戻り値を受け取れず、リトライ可能と宣言済みのエラーが恒久的な失敗になるため、
   221  			// エラーは返して再試行に委ねる（WithinTxE の doc 参照）。
   222  			txm.WithinTxE(func(ctx context.Context) error {
   223  				// prefectures を参照する依存行（users など）ごと空にする（tx はロールバックされる）。
   224  				if _, execErr := driver.New(ctx, testDB).Exec(ctx, "TRUNCATE prefectures CASCADE"); execErr != nil {
   225  					return execErr
   226  				}
   227  
   228  				actual, err := repo.FindAll(ctx)
   229  				require.NoError(t, err)
   230  				assert.NotNil(t, actual)
   231  				assert.Empty(t, actual)
   232  
   233  				return nil
   234  			})
   235  		})
   236  	})
