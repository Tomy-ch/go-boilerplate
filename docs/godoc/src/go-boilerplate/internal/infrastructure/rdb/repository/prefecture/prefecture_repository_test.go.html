   167  
   168  			expectedIDs := []uuid.UUID{

   172  			expectedNames := []string{

   174  				"大阪府",
   175  			}
   176  			expectedCodes := []int{
   177  				8,
   178  				27,
   179  			}
   180  
   181  			txm.WithinTx(func(ctx context.Context) {
   182  				expected := make(prefecture.Prefectures, len(expectedIDs))
   183  				ids := make([]uuid.UUID, len(expectedIDs))
   184  				for i := range expectedIDs {
   185  					e, err := prefecture.New(
   186  						expectedIDs[i],
   187  						expectedNames[i],
   188  						expectedCodes[i],
   189  					)
   190  					require.NoError(t, err)
   191  					expected[i] = e
   192  
   193  					ids[i] = expectedIDs[i]
   194  				}
   195  
   196  				actual, err := repo.FindByIDs(ctx, ids)
   197  				require.NoError(t, err)
