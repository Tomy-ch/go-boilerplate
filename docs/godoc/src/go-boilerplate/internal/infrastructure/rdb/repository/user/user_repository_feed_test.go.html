    66  func Test_repository_FindFeed(t *testing.T) {
    67  	t.Parallel()
    68  
    69  	testDB := testkit.NewTestDB(t)
    70  	lt := observability.NewMockInfraLayerTracer(t)
    71  	txm := testkit.NewTestTransactionRunner(t)
    72  
    73  	repo := &repository{tracer: lt, db: testDB}
    74  
    75  	// created_at 降順で安定した順序を作るための固定 ID（タイブレーク検証のため tie ペアを含む）。
    76  	// base は十分未来の時刻にし、seed データより必ず後ろ（より新しい）に来るようにする。
    77  	base := time.Date(2099, time.January, 1, 0, 0, 0, 0, time.UTC)
    78  
    79  	// tie ペア: 同一 created_at（base）で id だけが異なる。
    80  	// ORDER BY created_at DESC, id DESC のため、id が大きい tieHigh が先に来る。
    81  	tieHigh := "ffffffff-0000-4000-8000-000000000002"
    82  	tieLow := "ffffffff-0000-4000-8000-000000000001"
    83  	// それより古い 2 件。
    84  	mid := "ffffffff-0000-4000-8000-000000000003" // base-1h
    85  	old := "ffffffff-0000-4000-8000-000000000004" // base-2h
    86  
    87  	mustParse := func(s string) uuid.UUID {
    88  		id, err := uuid.Parse(s)
    89  		require.NoError(t, err)
    90  		return id
    91  	}
    92  
    93  	t.Run("正常系", func(t *testing.T) {
    94  		t.Parallel()
    95  
    96  		t.Run("先頭ページとafterカーソルでkeysetが安定順に次ページを返す", func(t *testing.T) {
    97  			t.Parallel()
    98  
    99  			txm.WithinTx(func(ctx context.Context) {
   100  				drv := driver.New(ctx, testDB)
   101  				insertFeedUser(ctx, t, drv, tieHigh, base)
   102  				insertFeedUser(ctx, t, drv, tieLow, base)
   103  				insertFeedUser(ctx, t, drv, mid, base.Add(-time.Hour))
   104  				insertFeedUser(ctx, t, drv, old, base.Add(-2*time.Hour))
   105  
   106  				// 先頭ページ（after=nil, limit=2）: 最新の 2 件 = tieHigh, tieLow。
   107  				// 同一 created_at では id DESC のため tieHigh(...0002) が tieLow(...0001) より先。
   108  				firstPage, err := repo.FindFeed(ctx, nil, 2)
   109  				require.NoError(t, err)
   110  				require.Len(t, firstPage, 2)
   111  				assert.Equal(t, mustParse(tieHigh), firstPage[0].ID())
   112  				assert.Equal(t, mustParse(tieLow), firstPage[1].ID())
   113  
   114  				// 次ページ: 先頭ページ末尾行(tieLow)を境界に keyset を進める。
   115  				// (created_at, id) < (base, tieLow) のため、同一 created_at の tieHigh は除外され、
   116  				// より古い mid → old が返る。
   117  				last := firstPage[len(firstPage)-1]
   118  				afterCursor := user.NewFeedCursor(last.CreatedAt(), last.ID())
   119  				after := &afterCursor
   120  
   121  				secondPage, err := repo.FindFeed(ctx, after, 2)
   122  				require.NoError(t, err)
   123  				require.Len(t, secondPage, 2)
   124  				assert.Equal(t, mustParse(mid), secondPage[0].ID())
   125  				assert.Equal(t, mustParse(old), secondPage[1].ID())
   126  			})
   127  		})
   128  
   129  		t.Run("afterの境界がtieペアの先頭行の場合、同一created_atのもう一方が次ページ先頭に来る", func(t *testing.T) {
   130  			t.Parallel()
   131  
   132  			txm.WithinTx(func(ctx context.Context) {
   133  				drv := driver.New(ctx, testDB)
   134  				insertFeedUser(ctx, t, drv, tieHigh, base)
   135  				insertFeedUser(ctx, t, drv, tieLow, base)
   136  
   137  				// 境界を tieHigh（同一 created_at の大きい id）にすると、
   138  				// (created_at, id) < (base, tieHigh) により tieLow のみが次ページに残る
   139  				// = created_at が同値でも id で正しくタイブレークされることの検証。
   140  				afterCursor := user.NewFeedCursor(base, mustParse(tieHigh))
   141  				after := &afterCursor
   142  
   143  				page, err := repo.FindFeed(ctx, after, 10)
   144  				require.NoError(t, err)
   145  				require.NotEmpty(t, page)
   146  				assert.Equal(t, mustParse(tieLow), page[0].ID())
   147  				// 自身（tieHigh）および新しい行は含まれない。
   148  				for _, u := range page {
   149  					assert.NotEqual(t, mustParse(tieHigh), u.ID())
   150  				}
   151  			})
   152  		})

   165  			t.Parallel()
   166  
   167  			txm.WithinTx(func(ctx context.Context) {
   168  				// seed を含む全ユーザーより確実に古い行を挿入し、それをカーソル境界にする。
   169  				// (created_at, id) < 境界 を満たす行が存在しないため、末尾到達で空ページになる。

   173  
   174  				afterCursor := user.NewFeedCursor(oldestAt, mustParse(oldest))
   175  				page, err := repo.FindFeed(ctx, &afterCursor, 10)
   176  				require.NoError(t, err)
   177  				assert.Empty(t, page)
   178  			})
   179  		})
