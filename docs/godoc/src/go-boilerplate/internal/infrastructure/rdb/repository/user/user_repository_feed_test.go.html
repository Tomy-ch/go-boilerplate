    41  func Test_repository_FindFeed(t *testing.T) {
    42  	t.Parallel()
    43  
    44  	loggingDB := testkit.NewTestLoggingProvider(t)
    45  	db := testkit.NewTestDB(t)
    46  	lt := observability.NewMockInfraLayerTracer(t)
    47  	txm := testkit.NewTestTransactionRunner(t)
    48  
    49  	repo := &repository{tracer: lt, db: loggingDB}
    50  
    51  	// created_at 降順で安定した順序を作るための固定 ID（タイブレーク検証のため tie ペアを含む）。
    52  	// base は十分未来の時刻にし、seed データより必ず後ろ（より新しい）に来るようにする。
    53  	base := time.Date(2099, time.January, 1, 0, 0, 0, 0, time.UTC)
    54  
    55  	// tie ペア: 同一 created_at（base）で id だけが異なる。
    56  	// ORDER BY created_at DESC, id DESC のため、id が大きい tieHigh が先に来る。
    57  	tieHigh := "ffffffff-0000-4000-8000-000000000002"
    58  	tieLow := "ffffffff-0000-4000-8000-000000000001"
    59  	// それより古い 2 件。
    60  	mid := "ffffffff-0000-4000-8000-000000000003" // base-1h
    61  	old := "ffffffff-0000-4000-8000-000000000004" // base-2h
    62  
    63  	mustParse := func(s string) uuid.UUID {
    64  		id, err := uuid.Parse(s)
    65  		require.NoError(t, err)
    66  		return id
    67  	}
    68  
    69  	t.Run("正常系", func(t *testing.T) {
    70  		t.Parallel()
    71  
    72  		t.Run("先頭ページとafterカーソルでkeysetが安定順に次ページを返す", func(t *testing.T) {
    73  			t.Parallel()
    74  
    75  			txm.WithinTx(func(ctx context.Context) {
    76  				drv := driver.New(ctx, db)
    77  				insertFeedUser(ctx, t, drv, tieHigh, base)
    78  				insertFeedUser(ctx, t, drv, tieLow, base)
    79  				insertFeedUser(ctx, t, drv, mid, base.Add(-time.Hour))
    80  				insertFeedUser(ctx, t, drv, old, base.Add(-2*time.Hour))
    81  
    82  				// 先頭ページ（after=nil, limit=2）: 最新の 2 件 = tieHigh, tieLow。
    83  				// 同一 created_at では id DESC のため tieHigh(...0002) が tieLow(...0001) より先。
    84  				firstPage, err := repo.FindFeed(ctx, nil, 2)
    85  				require.NoError(t, err)
    86  				require.Len(t, firstPage, 2)
    87  				assert.Equal(t, mustParse(tieHigh), firstPage[0].ID())
    88  				assert.Equal(t, mustParse(tieLow), firstPage[1].ID())
    89  
    90  				// 次ページ: 先頭ページ末尾行(tieLow)を境界に keyset を進める。
    91  				// (created_at, id) < (base, tieLow) のため、同一 created_at の tieHigh は除外され、
    92  				// より古い mid → old が返る。
    93  				last := firstPage[len(firstPage)-1]
    94  				afterCursor := user.NewFeedCursor(last.CreatedAt(), last.ID())
    95  				after := &afterCursor
    96  
    97  				secondPage, err := repo.FindFeed(ctx, after, 2)
    98  				require.NoError(t, err)
    99  				require.Len(t, secondPage, 2)
   100  				assert.Equal(t, mustParse(mid), secondPage[0].ID())
   101  				assert.Equal(t, mustParse(old), secondPage[1].ID())
   102  			})
   103  		})

   105  		t.Run("afterの境界がtieペアの先頭行の場合、同一created_atのもう一方が次ページ先頭に来る", func(t *testing.T) {
   106  			t.Parallel()
   107  
   108  			txm.WithinTx(func(ctx context.Context) {
   109  				drv := driver.New(ctx, db)
   110  				insertFeedUser(ctx, t, drv, tieHigh, base)
   111  				insertFeedUser(ctx, t, drv, tieLow, base)
   112  
   113  				// 境界を tieHigh（同一 created_at の大きい id）にすると、
   114  				// (created_at, id) < (base, tieHigh) により tieLow のみが次ページに残る
   115  				// = created_at が同値でも id で正しくタイブレークされることの検証。
   116  				afterCursor := user.NewFeedCursor(base, mustParse(tieHigh))
   117  				after := &afterCursor
   118  
   119  				page, err := repo.FindFeed(ctx, after, 10)
   120  				require.NoError(t, err)
   121  				require.NotEmpty(t, page)
   122  				assert.Equal(t, mustParse(tieLow), page[0].ID())
   123  				// 自身（tieHigh）および新しい行は含まれない。
   124  				for _, u := range page {
   125  					assert.NotEqual(t, mustParse(tieHigh), u.ID())
   126  				}
   127  			})
   128  		})
