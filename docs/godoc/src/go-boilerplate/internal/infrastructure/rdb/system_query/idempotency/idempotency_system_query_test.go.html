    37  func Test_store_lifecycle(t *testing.T) {
    38  	t.Parallel()
    39  
    40  	loggingDB := testkit.NewTestLoggingProvider(t)
    41  	lt := observability.NewMockInfraLayerTracer(t)
    42  	txm := testkit.NewTestTransactionRunner(t)
    43  	s := &store{tracer: lt, db: loggingDB}
    44  
    45  	t.Run("正常系", func(t *testing.T) {
    46  		t.Parallel()
    47  
    48  		t.Run("claim→get(claimed)→complete→get(completed) の一連が成立する", func(t *testing.T) {
    49  			t.Parallel()
    50  
    51  			txm.WithinTx(func(ctx context.Context) {
    52  				const scope, key = "test-scope-lifecycle", "key-1"
    53  				fp := newFingerprint(0xAB)
    54  
    55  				// 未claimは nil, nil。
    56  				before, err := s.Get(ctx, scope, key)
    57  				require.NoError(t, err)
    58  				assert.Nil(t, before)
    59  
    60  				// 新規claim → true。
    61  				claimed, err := s.Claim(ctx, idempotencybndry.ClaimParams{
    62  					Scope: scope, Key: key, Method: "POST", Path: "/v1/users",
    63  					Fingerprint: fp, ExpiresAt: time.Now().Add(24 * time.Hour),
    64  				})
    65  				require.NoError(t, err)
    66  				assert.True(t, claimed)
    67  
    68  				// claim直後は claimed 状態・結果未保存。
    69  				rec, err := s.Get(ctx, scope, key)
    70  				require.NoError(t, err)
    71  				require.NotNil(t, rec)
    72  				assert.Equal(t, idempotencybndry.StatusClaimed, rec.Status)
    73  				assert.Nil(t, rec.ResponseStatus)
    74  				assert.Equal(t, fp, rec.Fingerprint)
    75  
    76  				// 同一(scope,key)の再claimは false(一意制約で衝突)。
    77  				again, err := s.Claim(ctx, idempotencybndry.ClaimParams{
    78  					Scope: scope, Key: key, Method: "POST", Path: "/v1/users",
    79  					Fingerprint: fp, ExpiresAt: time.Now().Add(24 * time.Hour),
    80  				})
    81  				require.NoError(t, err)
    82  				assert.False(t, again)
    83  
    84  				// complete で結果保存。
    85  				payload := []byte(`{"v":1}`)
    86  				require.NoError(t, s.Complete(ctx, idempotencybndry.CompleteParams{
    87  					Scope: scope, Key: key, ResponseStatus: 201, ResponsePayload: payload,
    88  				}))
    89  
    90  				// completed 状態・結果が読める。
    91  				done, err := s.Get(ctx, scope, key)
    92  				require.NoError(t, err)
    93  				require.NotNil(t, done)
    94  				assert.Equal(t, idempotencybndry.StatusCompleted, done.Status)
    95  				require.NotNil(t, done.ResponseStatus)
    96  				assert.Equal(t, int32(201), *done.ResponseStatus)
    97  				assert.Equal(t, payload, done.ResponsePayload)
    98  			})
    99  		})
   100  	})

   118  				const scope, key = "test-scope-gc", "key-1"

   121  				claimed, err := s.Claim(ctx, idempotencybndry.ClaimParams{
   122  					Scope: scope, Key: key, Method: "POST", Path: "/v1/users",

   128  				// cutoff=now で失効行が削除される。

   135  				require.NoError(t, err)
   136  				assert.Nil(t, rec)
   137  			})
   138  		})
