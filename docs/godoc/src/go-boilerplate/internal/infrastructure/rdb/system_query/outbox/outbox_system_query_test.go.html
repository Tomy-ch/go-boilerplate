    38  func Test_store_lifecycle(t *testing.T) {
    39  	t.Parallel()
    40  
    41  	testDB := testkit.NewTestDB(t)
    42  	lt := observability.NewMockInfraLayerTracer(t)
    43  	txm := testkit.NewTestTransactionRunner(t)
    44  	s := &store{tracer: lt, db: testDB}
    45  
    46  	t.Run("正常系", func(t *testing.T) {
    47  		t.Parallel()
    48  
    49  		t.Run("Insert→claim→publish の一連が成立し published 後は claim されない", func(t *testing.T) {
    50  			t.Parallel()
    51  
    52  			txm.WithinTx(func(ctx context.Context) {
    53  				msgID, err := s.Insert(ctx, emitParams())
    54  				require.NoError(t, err)
    55  				assert.False(t, msgID.IsNil())
    56  
    57  				// pending を claim すると挿入行が返る。
    58  				msgs, err := s.ClaimPending(ctx, 10)
    59  				require.NoError(t, err)
    60  				require.Len(t, msgs, 1)
    61  				assert.Equal(t, msgID, msgs[0].MessageID)
    62  				assert.Equal(t, "purchase.created.v1", msgs[0].EventType)
    63  				assert.JSONEq(t, `{"v":1}`, string(msgs[0].Payload))
    64  				assert.Equal(t, int32(0), msgs[0].Attempts)
    65  
    66  				// published へ遷移すると再 claim では返らない。
    67  				require.NoError(t, s.MarkPublished(ctx, msgs[0].ID))
    68  				after, err := s.ClaimPending(ctx, 10)
    69  				require.NoError(t, err)
    70  				assert.Empty(t, after)
    71  			})
    72  		})
