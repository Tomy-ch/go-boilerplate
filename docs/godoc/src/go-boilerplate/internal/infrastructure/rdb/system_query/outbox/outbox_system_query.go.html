    62  	defer endSpan()
    63

    67  		return nil, pgerror.NormalizeError(err)

    69  
    70  	messages := make([]outboxbndry.PendingMessage, 0, len(rows))
    71  	for _, row := range rows {
    72  		messages = append(messages, outboxbndry.PendingMessage{
    73  			ID:        row.ID,
    74  			MessageID: row.MessageID,
    75  			EventType: row.EventType,
    76  			Payload:   row.Payload,
    77  			Headers:   row.Headers,
    78  			Attempts:  row.Attempts,
    79  		})
    80  	}
    81  	return messages, nil
