    85  func (t *testTxRunner) WithinTx(fn func(ctx context.Context)) {
    86  	t.t.Helper()

    88  	txLock.Lock()
    89  	defer txLock.Unlock()

    91  	baseCtx := context.Background()

    93  	err := t.inner.Do(baseCtx, func(txCtx context.Context) error {
    94  		fn(txCtx)
    95  		return rollbackForTestError
    96  	})
    97  
    98  	if xerrors.Is(err, rollbackForTestError) {
    99  		return
   100  	}
   101  	require.NoError(t.t, err)
   102  }
