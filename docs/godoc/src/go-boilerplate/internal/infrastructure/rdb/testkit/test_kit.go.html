    71  // WithinTx は、テスト用のトランザクションマネージャーでトランザクションを開始し、引数で渡されたfnを実行し、最後にロールバックします。
    72  //
    73  // 使用例:
    74  //
    75  //	txm.WithinTx(func(ctx context.Context) {
    76  //	  // トランザクション内の処理
    77  //	})
    78  func (t *testTxRunner) WithinTx(fn func(ctx context.Context)) {
    79  	t.t.Helper()
    80  
    81  	txLock.Lock()
    82  	defer txLock.Unlock()
    83  
    84  	baseCtx := context.Background()
    85  
    86  	err := t.inner.Do(baseCtx, func(txCtx context.Context) error {
    87  		if lockErr := lockSuiteSerialization(txCtx, driver.New(txCtx, t.db)); lockErr != nil {
    88  			return lockErr
    89  		}
    90  		fn(txCtx)
    91  		return errRollbackForTest
    92  	})
    93  
    94  	if xerrors.Is(err, errRollbackForTest) {
    95  		return
    96  	}
    97  	require.NoError(t.t, err)
    98  }

   100  // HoldSuiteSerialization は、呼び出したテストが終わるまでスイート全体の直列化を占有します。
   101  // 占有している間、他パッケージのテスト（別プロセス）が張る CASCADE TRUNCATE は走りません。
   102  // 解放は t.Cleanup で行われます。
   103  //
   104  // 占有は専用のトランザクションで行い、テスト自身のトランザクションはこの直列化に参加しません。
   105  // 参加させると自分の占有を待つことになり、進みません。
   106  //
   107  // 通常のテストは WithinTx が同じ直列化を内部で行うため、これを呼ぶ必要はありません。呼ぶのは、
   108  // トランザクションを 2 本同時に生かす検証（ロック競合の再現など）のように、WithinTx の
   109  // 「1 本 + ロールバック」では表現できないテストに限られます。
   110  func HoldSuiteSerialization(t *testing.T, db driver.DatabaseDriver) {
   111  	t.Helper()
   112  
   113  	txLock.Lock()
   114  	ctx := context.Background()
   115  
   116  	holder, err := db.Begin(ctx)
   117  	require.NoError(t, err)
   118  	t.Cleanup(func() {
   119  		_ = holder.Rollback(ctx)
   120  		txLock.Unlock()
   121  	})
   122  
   123  	require.NoError(t, lockSuiteSerialization(ctx, holder))
   124  }
   125  
   126  // lockSuiteSerialization は、q が属するトランザクションでスイート直列化用の advisory lock を取ります。
   127  func lockSuiteSerialization(ctx context.Context, q driver.DBTX) error {
   128  	// advisory lock 待ちは lock_timeout(10s) の対象で、高負荷で超過すると 55P03（非リトライ）で
   129  	// 落ちる。取得の間だけ無効化する（直列化ゆえテーブルロック競合は起きず待ち詰まりの懸念はない）。
   130  	if _, err := q.Exec(ctx, "SET LOCAL lock_timeout = 0"); err != nil {
   131  		return err
   132  	}
   133  	if _, err := q.Exec(ctx, "SELECT pg_advisory_xact_lock($1)", txAdvisoryLockKey); err != nil {
   134  		return err
   135  	}
   136  	return nil
   137  }
