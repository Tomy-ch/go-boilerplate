    26  // TestTracedDB_QueryMetricsInstrumentation は、実 DB を用いた黒箱テストで、
    27  // recorder（生成モック MockQueryRecorder）にクエリ属性が記録されること、
    28  // および成功/失敗で status / error_class が適切に分類されることを検証する。
    29  // ラベルに SQL 本文やテーブル名が混入しないことも併せて確認する。
    30  func TestTracedDB_QueryMetricsInstrumentation(t *testing.T) {
    31  	t.Parallel()
    32  
    33  	cfg := config.MockConfigForTest(t)
    34  	dbCfg := config.NewDatabaseConfig(cfg)
    35  	osCfg := config.NewOperatingSystemConfig(cfg)
    36  	dbConnCfg := config.NewDBConnectionConfig(cfg)
    37  	obsCfg := config.NewObservabilityConfig(cfg)
    38  
    39  	ctrl := gomock.NewController(t)
    40  	mockLogger := mock_logging.NewMockLogger(ctrl)
    41  	mockLogger.EXPECT().Named(gomock.Any()).Return(mockLogger).AnyTimes()
    42  	mockLogger.EXPECT().Info(gomock.Any(), gomock.Any(), gomock.Any()).AnyTimes()
    43  	mockLogger.EXPECT().Error(gomock.Any(), gomock.Any(), gomock.Any()).AnyTimes()
    44  	mockLogger.EXPECT().Warn(gomock.Any(), gomock.Any(), gomock.Any()).AnyTimes()
    45  	lf := logging.NewTestLogFieldBuilder(t)
    46  
    47  	recorder := mock_driver.NewMockQueryRecorder(ctrl)
    48  	var observed []driver.QueryAttrs
    49  	// db.Exec は逐次実行されるため Observe も逐次呼び出しとなる（追加の排他は不要）。
    50  	recorder.EXPECT().
    51  		Observe(gomock.Any(), gomock.Any()).
    52  		MinTimes(2).
    53  		Do(func(_ context.Context, attrs driver.QueryAttrs) {
    54  			observed = append(observed, attrs)
    55  		})
    56  
    57  	tracer := driver.NewQueryTracer(dbCfg, obsCfg, otelpgx.NewTracer(), recorder, mockLogger, lf)
    58  
    59  	db, err := driver.NewTracedDB(dbCfg, osCfg, dbConnCfg, tracer)
    60  	require.NoError(t, err)
    61  	defer func() { _ = db.Close() }()
    62  
    63  	ctx := driver.WithQueryName(context.Background(), "test.select_one")
    64  
    65  	// 正常クエリ。
    66  	_, err = db.Exec(ctx, "SELECT 1")
    67  	require.NoError(t, err)
    68  
    69  	// 失敗クエリ。
    70  	_, err = db.Exec(ctx, "SELECT 1 FROM no_such_table_for_test")
    71  	require.Error(t, err)
    72  
    73  	require.GreaterOrEqual(t, len(observed), 2)
    74  
    75  	var sawSuccess, sawError bool
    76  	for _, attrs := range observed {
    77  		// query_name / operation は明示名・固定 enum のみで、SQL 本文やテーブル名は混入しない。
    78  		assert.Equal(t, "test.select_one", attrs.QueryName)
    79  		assert.Equal(t, testOperationSelect, attrs.Operation)
    80  		assert.NotContains(t, attrs.QueryName, "SELECT")
    81  		assert.NotContains(t, attrs.Operation, "no_such_table_for_test")
    82  
    83  		switch attrs.Status {
    84  		case testStatusSuccess:
    85  			sawSuccess = true
    86  			assert.Empty(t, attrs.ErrorClass)
    87  		case testStatusError:
    88  			sawError = true
    89  			assert.NotEmpty(t, attrs.ErrorClass)
    90  		}
    91  	}
    92  	assert.True(t, sawSuccess, "成功クエリの属性が記録されていること")
    93  	assert.True(t, sawError, "失敗クエリの属性が記録されていること")
    94  }
