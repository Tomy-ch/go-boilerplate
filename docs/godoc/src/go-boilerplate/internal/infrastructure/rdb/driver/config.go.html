    10  // buildDSN は、sslmode と追加クエリパラメータから接続URLを組み立てます。
    11  func buildDSN(dbCfg *config.DatabaseConfig, extra url.Values) *url.URL {
    12  	q := url.Values{}
    13  	q.Set("sslmode", dbCfg.SSLMode())
    14  	for key, values := range extra {
    15  		for _, v := range values {
    16  			q.Set(key, v)
    17  		}
    18  	}
    19  
    20  	return &url.URL{
    21  		Scheme:   "postgres",
    22  		User:     url.UserPassword(dbCfg.User(), dbCfg.Password()),
    23  		Host:     fmt.Sprintf("%s:%d", dbCfg.Host(), dbCfg.Port()),
    24  		Path:     dbCfg.DBName(),
    25  		RawQuery: q.Encode(),
    26  	}
    27  }
