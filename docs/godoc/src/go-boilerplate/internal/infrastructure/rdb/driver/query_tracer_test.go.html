   132  	t.Parallel()
   133  
   134  	t.Run("正常系", func(t *testing.T) {
   135  		t.Parallel()
   136  
   137  		t.Run("開始時のspanIDがparentSpanIDとして終了処理経路まで引き継がれる", func(t *testing.T) {
   138  			t.Parallel()
   139  
   140  			ctx, end := observability.NewStubSpanContext(t)
   141  			defer end()
   142  
   143  			wantSpanID := observability.ExtractTraceContext(ctx).SpanID()
   144  			require.NotEmpty(t, wantSpanID)
   145  
   146  			qt, mockLogger := newTestQueryTracer(t)
   147  			qt.slowThreshold = time.Second
   148  			mockLogger.EXPECT().Info("DB query completed", gomock.Any()).Times(1)
   149  
   150  			// TraceQueryStart が開始時点の spanID を queryLogData.parentSpanID に取り込む。
   151  			startedCtx := qt.TraceQueryStart(ctx, nil, pgx.TraceQueryStartData{SQL: "SELECT 1"})
   152  
   153  			ld, ok := startedCtx.Value(queryLogKey{}).(queryLogData)
   154  			require.True(t, ok)
   155  			assert.Equal(t, wantSpanID, ld.parentSpanID)
   156  
   157  			// 手組みの context を使わず TraceQueryStart の出力 context をそのまま
   158  			// TraceQueryEnd へ渡し、parentSpanID が終了処理経路へ引き継がれることを確認する。
   159  			qt.TraceQueryEnd(startedCtx, nil, pgx.TraceQueryEndData{})
   160  		})
   161  	})
