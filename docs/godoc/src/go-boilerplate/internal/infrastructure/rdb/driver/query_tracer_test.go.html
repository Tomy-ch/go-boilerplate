   192  func Test_queryTracer_TraceQueryStart(t *testing.T) {
   193  	t.Parallel()
   194  
   195  	t.Run("正常系", func(t *testing.T) {
   196  		t.Parallel()
   197  
   198  		t.Run("開始時のspanIDがparentSpanIDとして終了処理経路まで引き継がれる", func(t *testing.T) {
   199  			t.Parallel()
   200  
   201  			ctx, end := observability.NewStubSpanContext(t)
   202  			defer end()
   203  
   204  			wantSpanID := observability.ExtractTraceContext(ctx).SpanID()
   205  			require.NotEmpty(t, wantSpanID)
   206  
   207  			qt, mockLogger := newTestQueryTracer(t)
   208  			qt.slowThreshold = time.Second
   209  			mockLogger.EXPECT().Info(gomock.Any(), "DB query completed", gomock.Any()).Times(1)
   210  
   211  			// TraceQueryStart が開始時点の spanID を queryLogData.parentSpanID に取り込む。
   212  			startedCtx := qt.TraceQueryStart(ctx, nil, pgx.TraceQueryStartData{SQL: "SELECT 1"})
   213  
   214  			ld, ok := startedCtx.Value(queryLogKey{}).(queryLogData)
   215  			require.True(t, ok)
   216  			assert.Equal(t, wantSpanID, ld.parentSpanID)
   217  
   218  			// 手組みの context を使わず TraceQueryStart の出力 context をそのまま
   219  			// TraceQueryEnd へ渡し、parentSpanID が終了処理経路へ引き継がれることを確認する。
   220  			qt.TraceQueryEnd(startedCtx, nil, pgx.TraceQueryEndData{})
   221  		})
   222  
   223  		t.Run("正常終了ログに parent_span_id フィールドが乗る", func(t *testing.T) {
   224  			t.Parallel()
   225  
   226  			ctx, end := observability.NewStubSpanContext(t)
   227  			defer end()
   228  
   229  			// 開始時点の span ID が parent_span_id としてログに乗ることを値レベルで検証する。
   230  			wantParentSpanID := observability.ExtractTraceContext(ctx).SpanID()
   231  			require.NotEmpty(t, wantParentSpanID)
   232  
   233  			logger, observed := logging.NewObservedTestLogger(t)
   234  			qt := newObservedQueryTracer(t, logger)
   235  			qt.slowThreshold = time.Second
   236  
   237  			startedCtx := qt.TraceQueryStart(ctx, nil, pgx.TraceQueryStartData{SQL: "SELECT 1"})
   238  			qt.TraceQueryEnd(startedCtx, nil, pgx.TraceQueryEndData{})
   239  
   240  			entries := observed.FilterMessage("DB query completed").All()
   241  			require.Len(t, entries, 1)
   242  			assert.Equal(t, wantParentSpanID, entries[0].ContextMap()[logging.ParentSpanIDKey])
   243  		})
   244  	})

   250  	t.Run("正常系", func(t *testing.T) {
   251  		t.Parallel()
   252  
   253  		t.Run("maskArgs が true の場合は args 件数フィールドが付かない", func(t *testing.T) {
   254  			t.Parallel()
   255  
   256  			ld := queryLogData{
   257  				sql:   "SELECT $1",
   258  				args:  []any{"secret"},
   259  				start: time.Now(),
   260  			}
   261  
   262  			maskedLogger, maskedLogs := logging.NewObservedTestLogger(t)
   263  			qtMasked := newObservedQueryTracer(t, maskedLogger)
   264  			qtMasked.maskArgs = true
   265  			qtMasked.slowThreshold = time.Second
   266  			qtMasked.TraceQueryEnd(
   267  				context.WithValue(context.Background(), queryLogKey{}, ld), nil, pgx.TraceQueryEndData{})
   268  
   269  			plainLogger, plainLogs := logging.NewObservedTestLogger(t)
   270  			qtPlain := newObservedQueryTracer(t, plainLogger)
   271  			qtPlain.maskArgs = false
   272  			qtPlain.slowThreshold = time.Second
   273  			qtPlain.TraceQueryEnd(
   274  				context.WithValue(context.Background(), queryLogKey{}, ld), nil, pgx.TraceQueryEndData{})
   275  
   276  			maskedEntries := maskedLogs.FilterMessage("DB query completed").All()
   277  			require.Len(t, maskedEntries, 1)
   278  			plainEntries := plainLogs.FilterMessage("DB query completed").All()
   279  			require.Len(t, plainEntries, 1)
   280  
   281  			// マスク時は args 件数キーが付かず、非マスク時は付く。
   282  			assert.NotContains(t, maskedEntries[0].ContextMap(), logging.QueryArgsCountKey)
   283  			assert.Contains(t, plainEntries[0].ContextMap(), logging.QueryArgsCountKey)
   284  		})
   285  	})
