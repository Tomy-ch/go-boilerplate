    75  // buildQueryAttrs は、クエリ終了時の情報から QueryAttrs を組み立てます。

    77  func buildQueryAttrs(ctx context.Context, sql string, duration time.Duration, err error) QueryAttrs {
    78  	status := statusSuccess
    79  	errorClass := ""

    94  // classifyOperation は、SQL の先頭トークンのみを見て低カーディナリティな operation へ分類します。
    95  // SQL 本文・テーブル名は一切ラベルに含めません。先頭コメントや WITH 句は select / other に丸めます。
    96  func classifyOperation(sql string) string {
    97  	switch firstSQLToken(sql) {
    98  	case "select", "with":
    99  		return operationSelect
   100  	case "insert":
   101  		return operationInsert
   102  	case "update":
   103  		return operationUpdate
   104  	case "delete":
   105  		return operationDelete
   106  	case "begin", "start":
   107  		return operationBegin
   108  	case "commit":
   109  		return operationCommit
   110  	case "rollback":
   111  		return operationRollback
   112  	case "copy":
   113  		return operationCopy
   114  	default:
   115  		return operationOther
   116  	}
   117  }

   119  // firstSQLToken は、先頭コメントと記号を取り除いた最初のトークンを小文字で返します。
   120  func firstSQLToken(sql string) string {
   121  	sql = stripLeadingSQLComments(sql)
   122  	sql = strings.TrimLeft(sql, " \t\r\n(")
   123  
   124  	end := strings.IndexFunc(sql, func(r rune) bool {
   125  		switch r {
   126  		case ' ', '\t', '\r', '\n', '(', ';':
   127  			return true
   128  		default:
   129  			return false
   130  		}
   131  	})
   132  	if end >= 0 {
   133  		sql = sql[:end]
   134  	}
   135  
   136  	return strings.ToLower(sql)
   137  }

   139  // stripLeadingSQLComments は、SQL 先頭の行コメント(--)・ブロックコメント(/* */)を取り除きます。
   140  func stripLeadingSQLComments(sql string) string {
   141  	for {
   142  		sql = strings.TrimLeft(sql, " \t\r\n")
   143  		switch {
   144  		case strings.HasPrefix(sql, "--"):
   145  			i := strings.IndexByte(sql, '\n')
   146  			if i < 0 {
   147  				return ""
   148  			}
   149  			sql = sql[i+1:]
   150  		case strings.HasPrefix(sql, "/*"):
   151  			i := strings.Index(sql, "*/")
   152  			if i < 0 {
   153  				return ""
   154  			}
   155  			sql = sql[i+2:]
   156  		default:
   157  			return sql
   158  		}
   159  	}
   160  }

   162  // classifyErrorClass は、pgerror の判定を用いてエラーを固定 enum へ丸めます。
   163  // raw error message / SQLSTATE 詳細 / 制約名などはラベルに含めません。
   164  // pgx.ErrNoRows は呼び出し前に除外して渡すこと（success として扱うため）。
   165  func classifyErrorClass(err error) string {
   166  	switch {
   167  	case err == nil:
   168  		return ""
   169  	case isConstraintViolation(err):
   170  		return errorClassConstraint
   171  	case isTimeout(err):
   172  		return errorClassTimeout
   173  	case pgerror.IsRetryableTxError(err):
   174  		return errorClassRetryable
   175  	case pgerror.IsUnavailable(err):
   176  		return errorClassConnection
   177  	default:
   178  		return errorClassUnknown
   179  	}
   180  }

   182  // isConstraintViolation は、整合性制約違反(SQLSTATE 23xxx)であるかを判定します。
   183  func isConstraintViolation(err error) bool {
   184  	var pgErr *pgconn.PgError
   185  	return errors.As(err, &pgErr) && strings.HasPrefix(pgErr.Code, "23")
   186  }
   187  
   188  // isTimeout は、タイムアウト系のエラー（context 期限切れ / lock_timeout 失効 / statement_timeout）であるかを判定します。
   189  // SQLSTATE 57014(query_canceled) は statement_timeout 失効でも発生するため timeout に分類します。
   190  func isTimeout(err error) bool {
   191  	return errors.Is(err, context.DeadlineExceeded) ||
   192  		pgerror.IsLockNotAvailable(err) ||
   193  		isStatementTimeout(err)
   194  }
   195  
   196  // isStatementTimeout は、statement_timeout 失効(SQLSTATE 57014 query_canceled)であるかを判定します。
   197  func isStatementTimeout(err error) bool {
   198  	var pgErr *pgconn.PgError
   199  	return errors.As(err, &pgErr) && pgErr.Code == "57014"
   200  }
