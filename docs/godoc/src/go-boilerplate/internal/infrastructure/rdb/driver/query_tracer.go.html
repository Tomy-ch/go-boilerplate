    76  // TraceQueryEnd は、span を終了し、正常終了(Info)・スロー(Warn)・エラー(Error)のログを出力します。
    77  func (t *queryTracer) TraceQueryEnd(
    78  	ctx context.Context, conn *pgx.Conn, data pgx.TraceQueryEndData,
    79  ) {
    80  	t.Tracer.TraceQueryEnd(ctx, conn, data)
    81  
    82  	ld, ok := ctx.Value(queryLogKey{}).(queryLogData)
    83  	if !ok {
    84  		return
    85  	}
    86  	duration := time.Since(ld.start)
    87  
    88  	t.recordQueryMetric(ctx, ld.sql, duration, data.Err)
    89  
    90  	switch {
    91  	case data.Err != nil:
    92  		t.logger.Error("DB query failed", t.endFields(ctx, ld, duration, data.Err)...)
    93  	case t.slowThreshold > 0 && duration > t.slowThreshold:
    94  		t.logger.Warn("DB slow query", t.endFields(ctx, ld, duration, nil)...)
    95  	default:
    96  		t.logger.Info("DB query completed", t.endFields(ctx, ld, duration, nil)...)
    97  	}
    98  }
