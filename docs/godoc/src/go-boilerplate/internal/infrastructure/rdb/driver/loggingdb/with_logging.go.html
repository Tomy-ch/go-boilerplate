    36  func (dwl *dbWithLogging) Exec(ctx context.Context, sql string, args ...any) (pgconn.CommandTag, error) {
    37  	start := time.Now()
    38  
    39  	spanName := observability.BuildSpanName(layer, pkg, execFunc)
    40  	tc, _, end := observability.StartSpanWithParent(ctx, dwl.provider.tracer, spanName)
    41  	defer end()
    42  
    43  	dwl.logQueryStart(tc, execFunc, spanName, sqlExec)
    44  
    45  	res, err := dwl.db.Exec(ctx, sql, args...)
    46  	duration := time.Since(start)
    47  
    48  	fields := dwl.buildSQLEndLogFields(tc, execFunc, spanName, sql, duration, args, err)
    49  	dwl.logQueryResult(sqlExec, duration, fields, err)
    50  	return res, err
    51  }

   147  // logQueryResult は、SQLクエリの実行結果をログ出力します。
   148  func (dwl *dbWithLogging) logQueryResult(
   149  	msg string, duration time.Duration, fields []*logging.Field, err error,
   150  ) {
   151  	logger := dwl.logger()
   152  	threshold := dwl.provider.dbCfg.SlowQueryWarnThreshold()
   153  	switch {
   154  	case err != nil:
   155  		logger.Error(msg, fields...)
   156  	case threshold > 0 && duration > threshold:
   157  		logger.Warn(msg, fields...)
   158  	default:
   159  		logger.Info(msg, fields...)
   160  	}
   161  }
