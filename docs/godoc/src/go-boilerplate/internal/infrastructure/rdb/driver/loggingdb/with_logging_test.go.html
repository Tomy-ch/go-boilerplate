    51  func Test_dbWithLogging_callerSkip(t *testing.T) {
    52  	t.Parallel()
    53  
    54  	cfg := config.MockConfigForTest(t)
    55  	dbCfg := config.NewDatabaseConfig(cfg)
    56  	obsCfg := config.NewObservabilityConfig(cfg)
    57  	lf := logging.NewTestLogFieldBuilder(t)
    58  	lg, observed := logging.NewObservedTestLoggerWithCaller(t)
    59  	noopLayerTracer := observability.NewNoopLayerTracer(t)
    60  
    61  	ctrl := gomock.NewController(t)
    62  
    63  	md := mock_driver.NewMockDBTX(ctrl)
    64  	md.EXPECT().Exec(gomock.Any(), gomock.Any(), gomock.Any()).Return(pgconn.CommandTag{}, nil)
    65  
    66  	dwl := &dbWithLogging{
    67  		db:       md,
    68  		provider: &provider{l: lg, lf: lf, dbCfg: dbCfg, obsCfg: obsCfg, tracer: noopLayerTracer},
    69  	}
    70  
    71  	// repository → sqlc gen → Exec の呼び出し段数を再現する。
    72  	sqlcGen := func() {
    73  		_, err := dwl.Exec(context.Background(), "INSERT INTO users (name) VALUES ($1)", "alice")
    74  		require.NoError(t, err)
    75  	}
    76  	repository := func() { sqlcGen() }
    77  	repository()
    78  
    79  	entries := observed.All()
    80  	require.Len(t, entries, 2) // 開始ログ + 終了ログ
    81  
    82  	startCaller := entries[0].Caller
    83  	endCaller := entries[1].Caller
    84  	require.True(t, startCaller.Defined)
    85  	require.True(t, endCaller.Defined)
    86  	// 開始・終了とも同一の呼び出し元（repository 層相当＝本テストファイル）を指すこと。
    87  	assert.Equal(t, startCaller.TrimmedPath(), endCaller.TrimmedPath())
    88  	assert.Contains(t, startCaller.TrimmedPath(), "with_logging_test.go")
    89  }
