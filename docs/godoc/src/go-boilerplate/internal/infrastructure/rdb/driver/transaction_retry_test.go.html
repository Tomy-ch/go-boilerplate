    20  func TestTxManager_Do_Retry(t *testing.T) {
    21  	t.Parallel()
    22  
    23  	// Begin 経路で生 SQLSTATE を注入し、pgx.Tx を mock せずにリトライ挙動を検証する。
    24  	retryablePgErr := &pgconn.PgError{Code: "40001"}    // serialization_failure
    25  	nonRetryablePgErr := &pgconn.PgError{Code: "23505"} // unique_violation
    26  
    27  	t.Run("正常系", func(t *testing.T) {
    28  		t.Parallel()
    29  
    30  		t.Run("1回目のリトライ可能エラー後に2回目の試行でコミットが成功する", func(t *testing.T) {
    31  			t.Parallel()
    32  
    33  			// pgx.Tx のモックが存在しないため Begin を mock で差し替える方式は採れない。
    34  			// 実 DB を用い fn 内でカウンタにより1回目のみリトライ可能エラーを返す方式とする。
    35  			cfg := config.MockConfigForTest(t)
    36  			dbCfg := config.NewDatabaseConfig(cfg)
    37  			dbConnCfg := config.NewDBConnectionConfig(cfg)
    38  			osCfg := config.NewOperatingSystemConfig(cfg)
    39  			dbCfg.SetDatabaseHost(t, "localhost")
    40  
    41  			realDB, err := driver.NewDB(dbCfg, osCfg, dbConnCfg)
    42  			require.NoError(t, err)
    43  			t.Cleanup(func() {
    44  				require.NoError(t, realDB.Close())
    45  			})
    46  
    47  			ctrl := gomock.NewController(t)
    48  			sleeper := mock_clock.NewMockSleeper(ctrl)
    49  			// 試行間の sleep は1回。
    50  			sleeper.EXPECT().Sleep(gomock.Any(), gomock.Any()).Return(nil).Times(1)
    51  			m := driver.NewTransactionManager(realDB, dbCfg, logging.NewTestLogger(t), sleeper)
    52  
    53  			attempts := 0
    54  			err = m.Do(context.Background(), func(_ context.Context) error {
    55  				attempts++
    56  				if attempts == 1 {
    57  					return retryablePgErr
    58  				}
    59  				return nil
    60  			})
    61  
    62  			require.NoError(t, err)
    63  			assert.Equal(t, 2, attempts)
    64  		})
    65  	})
    66  
    67  	t.Run("異常系", func(t *testing.T) {
    68  		t.Parallel()
    69  
    70  		t.Run("serialization_failureが続くとmaxAttemptsまで再試行しErrUnavailableを返す", func(t *testing.T) {
    71  			t.Parallel()
    72  
    73  			ctrl := gomock.NewController(t)
    74  			db := mock_driver.NewMockDatabaseDriver(ctrl)
    75  			dbCfg := config.NewDatabaseConfig(config.MockConfigForTest(t))
    76  			// maxAttempts は config 値（DB_TX_MAX_RETRIES）から導出される。
    77  			maxAttempts := dbCfg.TxMaxRetries()
    78  			db.EXPECT().Begin(gomock.Any()).Return(nil, retryablePgErr).Times(maxAttempts)
    79  			sleeper := mock_clock.NewMockSleeper(ctrl)
    80  			// 試行間の sleep は maxAttempts-1 回。
    81  			sleeper.EXPECT().Sleep(gomock.Any(), gomock.Any()).Return(nil).Times(maxAttempts - 1)
    82  
    83  			m := driver.NewTransactionManager(db, dbCfg, logging.NewTestLogger(t), sleeper)
    84  			err := m.Do(context.Background(), func(context.Context) error { return nil })
    85  
    86  			require.ErrorIs(t, err, apperror.ErrUnavailable)
    87  		})
    88  
    89  		t.Run("リトライ不可エラーは1回で返し待機しない", func(t *testing.T) {
    90  			t.Parallel()
    91  
    92  			ctrl := gomock.NewController(t)
    93  			db := mock_driver.NewMockDatabaseDriver(ctrl)
    94  			db.EXPECT().Begin(gomock.Any()).Return(nil, nonRetryablePgErr).Times(1)
    95  			sleeper := mock_clock.NewMockSleeper(ctrl)
    96  			// リトライしないため待機は発生しない。
    97  			sleeper.EXPECT().Sleep(gomock.Any(), gomock.Any()).Times(0)
    98  
    99  			dbCfg := config.NewDatabaseConfig(config.MockConfigForTest(t))
   100  			m := driver.NewTransactionManager(db, dbCfg, logging.NewTestLogger(t), sleeper)
   101  			err := m.Do(context.Background(), func(context.Context) error { return nil })
   102  
   103  			require.ErrorIs(t, err, apperror.ErrConflict)
   104  		})
   105  
   106  		t.Run("リトライ待機中にcontextがキャンセルされると直前のエラーを正規化して返す", func(t *testing.T) {
   107  			t.Parallel()
   108  
   109  			ctrl := gomock.NewController(t)
   110  			db := mock_driver.NewMockDatabaseDriver(ctrl)
   111  			// 1回目のリトライ可能エラー後、待機（Sleep）が ctx 打ち切りで失敗するため再試行されない。
   112  			db.EXPECT().Begin(gomock.Any()).Return(nil, retryablePgErr).Times(1)
   113  			sleeper := mock_clock.NewMockSleeper(ctrl)
   114  			sleeper.EXPECT().Sleep(gomock.Any(), gomock.Any()).Return(context.Canceled).Times(1)
   115  
   116  			dbCfg := config.NewDatabaseConfig(config.MockConfigForTest(t))
   117  			m := driver.NewTransactionManager(db, dbCfg, logging.NewTestLogger(t), sleeper)
   118  			err := m.Do(context.Background(), func(context.Context) error { return nil })
   119  
   120  			// 待機の context.Canceled ではなく、リトライ対象だった元の失敗（40001）を
   121  			// 正規化した ErrUnavailable が返る。
   122  			require.ErrorIs(t, err, apperror.ErrUnavailable)
   123  		})
   124  	})
