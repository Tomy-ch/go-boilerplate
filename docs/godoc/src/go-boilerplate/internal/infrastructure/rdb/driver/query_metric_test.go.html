   194  func Test_buildQueryAttrs(t *testing.T) {
   195  	t.Parallel()
   196  
   197  	t.Run("正常系", func(t *testing.T) {
   198  		t.Parallel()
   199  
   200  		t.Run("成功時はstatus=successでerror_classは空", func(t *testing.T) {
   201  			t.Parallel()
   202  
   203  			ctx := WithQueryName(context.Background(), "user.find_by_id")
   204  			got := buildQueryAttrs(ctx, "SELECT 1", 5*time.Millisecond, nil)
   205  
   206  			assert.Equal(t, "user.find_by_id", got.QueryName)
   207  			assert.Equal(t, operationSelect, got.Operation)
   208  			assert.Equal(t, statusSuccess, got.Status)
   209  			assert.Empty(t, got.ErrorClass)
   210  			assert.Equal(t, 5*time.Millisecond, got.Duration)
   211  		})
   212  
   213  		t.Run("ErrNoRowsはsuccess扱いでerrorに数えない", func(t *testing.T) {
   214  			t.Parallel()
   215  
   216  			got := buildQueryAttrs(context.Background(), "SELECT 1", time.Millisecond, pgx.ErrNoRows)
   217  
   218  			assert.Equal(t, queryNameUnknown, got.QueryName)
   219  			assert.Equal(t, statusSuccess, got.Status)
   220  			assert.Empty(t, got.ErrorClass)
   221  		})
   222  
   223  		t.Run("ラベルにSQL本文やbind値を含めない", func(t *testing.T) {
   224  			t.Parallel()
   225  
   226  			ctx := WithQueryName(context.Background(), "user.find_by_email")
   227  			sql := "SELECT id FROM users WHERE email = $1"
   228  			got := buildQueryAttrs(ctx, sql, time.Millisecond, nil)
   229  
   230  			// operation / query_name は固定 enum / 明示名のみで、SQL 本文や bind 値は混入しない。
   231  			assert.NotContains(t, got.Operation, "users")
   232  			assert.NotContains(t, got.Operation, "email")
   233  			assert.NotContains(t, got.QueryName, "SELECT")
   234  			assert.Equal(t, operationSelect, got.Operation)
   235  		})
   236  	})
   237  
   238  	t.Run("異常系", func(t *testing.T) {
   239  		t.Parallel()
   240  
   241  		t.Run("失敗時はstatus=errorでerror_classが入る", func(t *testing.T) {
