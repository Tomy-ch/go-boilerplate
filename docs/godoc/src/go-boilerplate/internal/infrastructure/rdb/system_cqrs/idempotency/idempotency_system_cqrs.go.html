    38  // Claim は、claimed 行を作ります。新規に作れたら true、同一 (scope, key) の既存行があれば
    39  // false を返します。ロック競合タイムアウト時は ErrLockTimeout を返します。
    40  // 業務 tx 内から呼ぶこと。claim 待ち用の lock_timeout は claim 直後に既定へ戻すため、
    41  // 同一 tx で続く業務クエリへ 3s 上限が波及しない。
    42  func (s *store) Claim(ctx context.Context, p idempotencybndry.ClaimParams) (bool, error) {
    43  	ctx, endSpan := s.tracer.Start(ctx)
    44  	defer endSpan()
    45  
    46  	ldb := driver.New(ctx, s.db)
    47  	if _, err := ldb.Exec(ctx, "SET LOCAL lock_timeout = '"+claimLockTimeout+"'"); err != nil {
    48  		return false, pgerror.NormalizeError(err)
    49  	}
    50  
    51  	db := gen.New(ldb)
    52  	_, err := db.ClaimIdempotencyKey(ctx, &gen.ClaimIdempotencyKeyParams{
    53  		Scope:              p.Scope,
    54  		IdempotencyKey:     p.Key,
    55  		RequestMethod:      p.Method,
    56  		RequestPath:        p.Path,
    57  		RequestFingerprint: p.Fingerprint,
    58  		ExpiresAt:          p.ExpiresAt,
    59  	})
    60  
    61  	// claim が成功（新規）または既存キー検出で tx が健全なうちに、claim 待ち限定の
    62  	// lock_timeout を既定へ戻す。ロック競合等で claim が失敗した場合は tx が abort 済みで
    63  	// 後続コマンドを受け付けないため戻さない（呼び出し側が rollback する）。
    64  	existing := xerrors.Is(err, pgx.ErrNoRows)
    65  	if err == nil || existing {
    66  		if _, rerr := ldb.Exec(ctx, "SET LOCAL lock_timeout TO DEFAULT"); rerr != nil {
    67  			return false, pgerror.NormalizeError(rerr)
    68  		}
    69  	}
    70  
    71  	switch {
    72  	case err == nil:
    73  		return true, nil
    74  	case existing:
    75  		// ON CONFLICT DO NOTHING で 0 行 = 既存キーあり。
    76  		return false, nil
    77  	case pgerror.IsLockNotAvailable(err):
    78  		return false, idempotencybndry.ErrLockTimeout
    79  	default:
    80  		return false, pgerror.NormalizeError(err)
    81  	}
    82  }

    85  func (s *store) Get(ctx context.Context, scope, key string) (*idempotencybndry.Record, error) {
    86  	ctx, endSpan := s.tracer.Start(ctx)

   109  // Complete は、claimed → completed へ遷移し結果を保存します。
   110  func (s *store) Complete(ctx context.Context, p idempotencybndry.CompleteParams) error {
   111  	ctx, endSpan := s.tracer.Start(ctx)
   112  	defer endSpan()
   113  
   114  	db := gen.New(driver.New(ctx, s.db))
   115  	status := p.ResponseStatus
   116  	affected, err := db.CompleteIdempotencyKey(ctx, &gen.CompleteIdempotencyKeyParams{
   117  		Scope:           p.Scope,
   118  		IdempotencyKey:  p.Key,
   119  		ResponseStatus:  &status,
   120  		ResponsePayload: p.ResponsePayload,
   121  	})
   122  	if err != nil {
   123  		return pgerror.NormalizeError(err)
   124  	}
   125  	if affected == 0 {
   126  		// 同一 tx で claim 済みの行が complete 対象にならない＝前提崩壊（404 ではなく内部エラー）。
   127  		return xerrors.Wrap(apperror.ErrInternal, "complete: claimed row not found in the same transaction")
   128  	}
   129  	return nil
   130  }
