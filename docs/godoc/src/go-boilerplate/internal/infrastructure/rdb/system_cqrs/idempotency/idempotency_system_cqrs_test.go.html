   152  // assertClaimLockTimeoutRestored は、claim 時の SET LOCAL lock_timeout(3s) が業務フェーズへ
   153  // 波及していないことを、保持側 tx の advisory lock を claim の 3s 上限を超えて待たせても
   154  // 55P03 にならない、という振る舞いで検証するテストヘルパーです。WithinTx は txLock で
   155  // 直列化するため使わず、同一 TransactionManager の Do を 2 本並行させます
   156  // （Do ごとに別コネクションの tx を張る）。
   157  func assertClaimLockTimeoutRestored(t *testing.T, s *store, testDB driver.DatabaseDriver) {
   158  	t.Helper()
   159  
   160  	claimTimeout, err := time.ParseDuration(claimLockTimeout)
   161  	require.NoError(t, err)
   162  
   163  	dbCfg := config.NewDatabaseConfig(config.MockConfigForTest(t))
   164  	txm := driver.NewTransactionManager(testDB, dbCfg, logging.NewTestLogger(t), system.NewSleeper())
   165  
   166  	const advisoryLockKey = int64(732100000000000091)
   167  
   168  	holderLocked := make(chan struct{})
   169  	releaseHolder := make(chan struct{})
   170  	holderDone := make(chan error, 1)
   171  
   172  	var releaseOnce sync.Once
   173  	release := func() { releaseOnce.Do(func() { close(releaseHolder) }) }
   174  	t.Cleanup(release) // 失敗時も保持側 goroutine をリークさせない
   175  
   176  	// 保持側 tx: advisory xact lock を取得し、claim の lock_timeout(3s)を超えて保持する。
   177  	go func() {
   178  		holderDone <- txm.Do(context.Background(), func(ctx context.Context) error {
   179  			if _, err := driver.New(ctx, testDB).Exec(ctx, "SELECT pg_advisory_xact_lock($1)", advisoryLockKey); err != nil {
   180  				return err
   181  			}
   182  			close(holderLocked)
   183  			<-releaseHolder
   184  			return errHolderRollback // rollback で advisory lock を解放する。
   185  		})
   186  	}()
   187  
   188  	select {
   189  	case <-holderLocked:
   190  	case err := <-holderDone:
   191  		require.NoError(t, err, "保持側 tx がロック取得前に失敗した")
   192  		return
   193  	}
   194  
   195  	// 業務側 tx: claim（ここで lock_timeout=3s がセットされ直後に既定へ戻る）後、
   196  	// 保持側が握る advisory lock を待つ。ロック待ちがブロックするため goroutine で実行する。
   197  	subjectDone := make(chan struct{})
   198  	var subjectTxErr, bizPhaseErr error
   199  	go func() {
   200  		defer close(subjectDone)
   201  		subjectTxErr = txm.Do(context.Background(), func(ctx context.Context) error {
   202  			claimed, err := s.Claim(ctx, idempotencybndry.ClaimParams{
   203  				Scope: "lock-timeout-restore-scope", Key: "key-1", Method: "POST", Path: "/v1/users",
   204  				Fingerprint: newFingerprint(0x06), ExpiresAt: time.Now().Add(time.Hour),
   205  			})
   206  			if err != nil {
   207  				return err
   208  			}
   209  			if !claimed {
   210  				return xerrors.New("subject: expected fresh claim to succeed")
   211  			}
   212  			// 業務フェーズのロック待ち。lock_timeout が claim に閉じていれば 55P03 にならない。
   213  			_, bizPhaseErr = driver.New(ctx, testDB).Exec(ctx, "SELECT pg_advisory_xact_lock($1)", advisoryLockKey)
   214  			return errSubjectRollback // 業務クエリの成否は bizPhaseErr で見て、行は残さない。
   215  		})
   216  	}()
   217  
   218  	// 業務側が実際に advisory lock 待ちへ入るまで待つ。goroutine 起動直後に保持タイマーを
   219  	// 始めると、CI 遅延でロック待ち突入前に解放してしまい偽陽性で通る窓があるため。
   220  	waitForAdvisoryLockWait(t, testDB, advisoryLockKey)
   221  
   222  	// ロック待ちを確認できたので、claim の 3s 上限を確実に超える時間だけ保持してから解放する。
   223  	// lock_timeout が未復元なら業務側は待機中に 3s で 55P03 になっている。
   224  	time.Sleep(claimTimeout + time.Second)
   225  	release()
   226  
   227  	<-subjectDone
   228  	require.NoError(t, bizPhaseErr)
   229  	require.False(t, pgerror.IsLockNotAvailable(bizPhaseErr))
   230  	require.ErrorIs(t, subjectTxErr, errSubjectRollback)
   231  	require.ErrorIs(t, <-holderDone, errHolderRollback)
   232  }

   234  // assertClaimLockContentionTimeout は、同一 (scope,key) を claim する 2 本の業務 tx を
   235  // 並行競合させ、競合側が 55P03 → ErrLockTimeout へ正規化されることを検証するテストヘルパーです。
   236  // testkit の WithinTx は txLock で直列化するため再現できないので、TransactionManager を
   237  // 2 本（= 2 コネクション）並行で走らせて再現します。
   238  func assertClaimLockContentionTimeout(t *testing.T, s *store, testDB driver.DatabaseDriver) {
   239  	t.Helper()
   240  
   241  	dbCfg := config.NewDatabaseConfig(config.MockConfigForTest(t))
   242  	txm := driver.NewTransactionManager(testDB, dbCfg, logging.NewTestLogger(t), system.NewSleeper())
   243  
   244  	params := idempotencybndry.ClaimParams{
   245  		Scope: "lock-timeout-scope", Key: "key-1", Method: "POST", Path: "/v1/users",
   246  		Fingerprint: newFingerprint(0x04), ExpiresAt: time.Now().Add(time.Hour),
   247  	}
   248  
   249  	holderClaimed := make(chan struct{})
   250  	releaseHolder := make(chan struct{})
   251  	holderDone := make(chan error, 1)
   252  
   253  	var releaseOnce sync.Once
   254  	release := func() { releaseOnce.Do(func() { close(releaseHolder) }) }
   255  	t.Cleanup(release) // 失敗時も保持側 goroutine をリークさせない
   256  
   257  	// 保持側 tx: 同一キーを claim し、競合側がタイムアウトするまで未コミットで保持する。
   258  	go func() {
   259  		holderDone <- txm.Do(context.Background(), func(ctx context.Context) error {
   260  			claimed, err := s.Claim(ctx, params)
   261  			if err != nil {
   262  				return err
   263  			}
   264  			if !claimed {
   265  				return xerrors.New("holder: expected fresh claim to succeed")
   266  			}
   267  			close(holderClaimed)
   268  			<-releaseHolder
   269  			// 行を残さないようエラー返却で rollback させる。
   270  			return errHolderRollback
   271  		})
   272  	}()
   273  
   274  	// 保持側が claim 済み（行ロック確立）になるまで待つ。失敗時は holderDone で検知。
   275  	select {
   276  	case <-holderClaimed:
   277  	case err := <-holderDone:
   278  		require.NoError(t, err, "保持側 tx が claim 前に失敗した")
   279  		return
   280  	}
   281  
   282  	// 競合側 tx: 同一キーの claim はロック待ち→ lock_timeout で 55P03 → ErrLockTimeout。
   283  	contenderErr := txm.Do(context.Background(), func(ctx context.Context) error {
   284  		_, err := s.Claim(ctx, params)
   285  		return err
   286  	})
   287  	require.ErrorIs(t, contenderErr, idempotencybndry.ErrLockTimeout)
   288  
   289  	release()
   290  	require.ErrorIs(t, <-holderDone, errHolderRollback)
   291  }

   293  // waitForAdvisoryLockWait は、指定キーの advisory lock 待ち（granted=false）が pg_locks に
   294  // 現れるまで待ちます。bigint 版 advisory lock は classid=上位32bit / objid=下位32bit /
   295  // objsubid=1 に分解されるため、キーを再構成して照合します。
   296  func waitForAdvisoryLockWait(t *testing.T, db driver.DatabaseDriver, key int64) {
   297  	t.Helper()
   298  	require.Eventually(t, func() bool {
   299  		var waiting bool
   300  		if err := driver.New(context.Background(), db).QueryRow(context.Background(),
   301  			`SELECT EXISTS (SELECT 1 FROM pg_locks WHERE locktype = 'advisory' AND NOT granted
   302  				AND (classid::bigint << 32) + objid::bigint = $1 AND objsubid = 1)`,
   303  			key).Scan(&waiting); err != nil {
   304  			return false
   305  		}
   306  		return waiting
   307  	}, 10*time.Second, 50*time.Millisecond, "advisory lock 待ちに入らなかった")
   308  }
