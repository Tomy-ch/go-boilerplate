    47  func Test_store_lifecycle(t *testing.T) {
    48  	t.Parallel()
    49  
    50  	testDB := testkit.NewTestDB(t)
    51  	lt := observability.NewMockInfraLayerTracer(t)
    52  	txm := testkit.NewTestTransactionRunner(t)
    53  	s := &store{tracer: lt, db: testDB}
    54  
    55  	t.Run("正常系", func(t *testing.T) {
    56  		t.Parallel()
    57  
    58  		t.Run("claim→get(claimed)→complete→get(completed) の一連が成立する", func(t *testing.T) {
    59  			t.Parallel()
    60  
    61  			txm.WithinTx(func(ctx context.Context) {
    62  				const scope, key = "test-scope-lifecycle", "key-1"
    63  				fp := newFingerprint(0xAB)
    64  
    65  				// 未claimは nil, nil。
    66  				before, err := s.Get(ctx, scope, key)
    67  				require.NoError(t, err)
    68  				assert.Nil(t, before)
    69  
    70  				// 新規claim → true。
    71  				claimed, err := s.Claim(ctx, idempotencybndry.ClaimParams{
    72  					Scope: scope, Key: key, Method: "POST", Path: "/v1/users",
    73  					Fingerprint: fp, ExpiresAt: time.Now().Add(24 * time.Hour),
    74  				})
    75  				require.NoError(t, err)
    76  				assert.True(t, claimed)
    77  
    78  				// claim直後は claimed 状態・結果未保存。
    79  				rec, err := s.Get(ctx, scope, key)
    80  				require.NoError(t, err)
    81  				require.NotNil(t, rec)
    82  				assert.Equal(t, idempotencybndry.StatusClaimed, rec.Status)
    83  				assert.Nil(t, rec.ResponseStatus)
    84  				assert.Equal(t, fp, rec.Fingerprint)
    85  
    86  				// 同一(scope,key)の再claimは false(一意制約で衝突)。
    87  				again, err := s.Claim(ctx, idempotencybndry.ClaimParams{
    88  					Scope: scope, Key: key, Method: "POST", Path: "/v1/users",
    89  					Fingerprint: fp, ExpiresAt: time.Now().Add(24 * time.Hour),
    90  				})
    91  				require.NoError(t, err)
    92  				assert.False(t, again)
    93  
    94  				// complete で結果保存。
    95  				payload := []byte(`{"v":1}`)
    96  				require.NoError(t, s.Complete(ctx, idempotencybndry.CompleteParams{
    97  					Scope: scope, Key: key, ResponseStatus: 201, ResponsePayload: payload,
    98  				}))
    99  
   100  				// completed 状態・結果が読める。
   101  				done, err := s.Get(ctx, scope, key)
   102  				require.NoError(t, err)
   103  				require.NotNil(t, done)
   104  				assert.Equal(t, idempotencybndry.StatusCompleted, done.Status)
   105  				require.NotNil(t, done.ResponseStatus)
   106  				assert.Equal(t, int32(201), *done.ResponseStatus)
   107  				assert.Equal(t, payload, done.ResponsePayload)
   108  			})
   109  		})
   110  	})

   114  	t.Parallel()
   115  
   116  	testDB := testkit.NewTestDB(t)
   117  	lt := observability.NewMockInfraLayerTracer(t)
   118  	txm := testkit.NewTestTransactionRunner(t)
   119  	s := &store{tracer: lt, db: testDB}
   120  
   121  	t.Run("正常系", func(t *testing.T) {
   122  		t.Parallel()
   123  
   124  		t.Run("cutoffより古い失効行を削除する", func(t *testing.T) {
   125  			t.Parallel()
   126  
   127  			txm.WithinTx(func(ctx context.Context) {
   128  				const scope, key = "test-scope-gc", "key-1"
   129  
   130  				// expires_at を過去にして claim（= 失効済み）。
   131  				claimed, err := s.Claim(ctx, idempotencybndry.ClaimParams{
   132  					Scope: scope, Key: key, Method: "POST", Path: "/v1/users",
   133  					Fingerprint: newFingerprint(0x01), ExpiresAt: time.Now().Add(-1 * time.Hour),
   134  				})
   135  				require.NoError(t, err)
   136  				require.True(t, claimed)
   137  
   138  				// cutoff=now で失効行が削除される。
   139  				deleted, err := s.DeleteExpired(ctx, time.Now(), 100)
   140  				require.NoError(t, err)
   141  				// DeleteExpired は scope 非限定でテーブル全体を対象とするため、共有DB上では他の失効行も
   142  				// 含まれ得る。厳密件数に依存せず「1件以上削除された」ことのみ検証し、対象行の削除は
   143  				// 後続の Get==nil で担保する。
   144  				assert.GreaterOrEqual(t, deleted, int64(1))
   145  
   146  				// 削除後は取得できない。
   147  				rec, err := s.Get(ctx, scope, key)
   148  				require.NoError(t, err)
   149  				assert.Nil(t, rec)
   150  			})
   151  		})
   152  
   153  		t.Run("失効行が無ければ0件", func(t *testing.T) {
   154  			t.Parallel()
   155  
   156  			txm.WithinTx(func(ctx context.Context) {
   157  				const scope, key = "test-scope-gc-fresh", "key-1"
   158  
   159  				// 未失効(将来失効)で claim。
   160  				claimed, err := s.Claim(ctx, idempotencybndry.ClaimParams{
   161  					Scope: scope, Key: key, Method: "POST", Path: "/v1/users",
   162  					Fingerprint: newFingerprint(0x02), ExpiresAt: time.Now().Add(24 * time.Hour),
   163  				})
   164  				require.NoError(t, err)
   165  				require.True(t, claimed)
   166  
   167  				// cutoff を過去に置けば、この行は対象外。
   168  				deleted, err := s.DeleteExpired(ctx, time.Now().Add(-48*time.Hour), 100)
   169  				require.NoError(t, err)
   170  				assert.Equal(t, int64(0), deleted)

   176  func Test_store_errors(t *testing.T) {
   177  	t.Parallel()
   178  
   179  	testDB := testkit.NewTestDB(t)
   180  	lt := observability.NewMockInfraLayerTracer(t)
   181  	txm := testkit.NewTestTransactionRunner(t)
   182  	s := &store{tracer: lt, db: testDB}
   183  
   184  	t.Run("異常系", func(t *testing.T) {
   185  		t.Parallel()
   186  
   187  		t.Run("存在しないキーへのcompleteはエラーになる", func(t *testing.T) {
   188  			t.Parallel()
   189  
   190  			txm.WithinTx(func(ctx context.Context) {
   191  				err := s.Complete(ctx, idempotencybndry.CompleteParams{
   192  					Scope: "missing-scope", Key: "missing-key",
   193  					ResponseStatus: 201, ResponsePayload: []byte(`{}`),
   194  				})
   195  				require.ErrorIs(t, err, apperror.ErrInternal)
   196  			})
   197  		})
   198  
   199  		t.Run("ロック競合タイムアウト時はErrLockTimeoutを返す", func(t *testing.T) {
   200  			t.Parallel()
   201  
   202  			// Claim の 55P03(lock_not_available) 分岐は、同一 (scope,key) を SET LOCAL
   203  			// lock_timeout 付きで claim する 2 本の業務 tx が並行に競合して初めて発生する。
   204  			// testkit の WithinTx は txLock で直列化するため再現できないので、ここでは
   205  			// TransactionManager を 2 本（= 2 コネクション）並行で走らせて再現する。
   206  			dbCfg := config.NewDatabaseConfig(config.MockConfigForTest(t))
   207  			txm := driver.NewTransactionManager(testDB, dbCfg, logging.NewTestLogger(t), system.NewSleeper())
   208  
   209  			params := idempotencybndry.ClaimParams{
   210  				Scope: "lock-timeout-scope", Key: "key-1", Method: "POST", Path: "/v1/users",
   211  				Fingerprint: newFingerprint(0x04), ExpiresAt: time.Now().Add(time.Hour),
   212  			}
   213  
   214  			holderClaimed := make(chan struct{})
   215  			releaseHolder := make(chan struct{})
   216  			holderDone := make(chan error, 1)
   217  
   218  			var releaseOnce sync.Once
   219  			release := func() { releaseOnce.Do(func() { close(releaseHolder) }) }
   220  			t.Cleanup(release) // 失敗時も保持側 goroutine をリークさせない
   221  
   222  			// 保持側 tx: 同一キーを claim し、競合側がタイムアウトするまで未コミットで保持する。
   223  			go func() {
   224  				holderDone <- txm.Do(context.Background(), func(ctx context.Context) error {
   225  					claimed, err := s.Claim(ctx, params)
   226  					if err != nil {
   227  						return err
   228  					}
   229  					if !claimed {
   230  						return xerrors.New("holder: expected fresh claim to succeed")
   231  					}
   232  					close(holderClaimed)
   233  					<-releaseHolder
   234  					// 行を残さないようエラー返却で rollback させる。
   235  					return errHolderRollback
   236  				})
   237  			}()
   238  
   239  			// 保持側が claim 済み（行ロック確立）になるまで待つ。失敗時は holderDone で検知。
   240  			select {
   241  			case <-holderClaimed:
   242  			case err := <-holderDone:
   243  				require.NoError(t, err, "保持側 tx が claim 前に失敗した")
   244  				return
   245  			}
   246  
   247  			// 競合側 tx: 同一キーの claim はロック待ち→ lock_timeout で 55P03 → ErrLockTimeout。
   248  			contenderErr := txm.Do(context.Background(), func(ctx context.Context) error {
   249  				_, err := s.Claim(ctx, params)
   250  				return err
   251  			})
   252  			require.ErrorIs(t, contenderErr, idempotencybndry.ErrLockTimeout)
   253  
   254  			release()
   255  			require.ErrorIs(t, <-holderDone, errHolderRollback)
   256  		})
   257  
   258  		t.Run("claim挿入が汎用DBエラーで失敗した場合はErrInternalへ正規化される", func(t *testing.T) {
   259  			t.Parallel()
   260  
   261  			txm.WithinTx(func(ctx context.Context) {
   262  				// NUL バイトを含む scope は PostgreSQL の TEXT に格納できず、ロック競合でも既存行でもない
   263  				// 汎用エラーとなるため、Claim の default 分岐（ErrInternal 正規化）へ落ちる。
   264  				params := idempotencybndry.ClaimParams{
   265  					Scope: "bad\x00scope", Key: "key-1", Method: "POST", Path: "/v1/users",
   266  					Fingerprint: newFingerprint(0x05), ExpiresAt: time.Now().Add(time.Hour),
   267  				}
   268  				_, err := s.Claim(ctx, params)
   269  				require.ErrorIs(t, err, apperror.ErrInternal)
   270  			})
   271  		})
   272  
   273  		t.Run("キャンセル済みコンテキストでは各操作がErrCanceledへ正規化して返す", func(t *testing.T) {
   274  			t.Parallel()
   275  
   276  			ctx, cancel := context.WithCancel(context.Background())
   277  			cancel()
   278  
   279  			params := idempotencybndry.ClaimParams{
   280  				Scope: "ctx-cancel", Key: "key-1", Method: "POST", Path: "/v1/users",
   281  				Fingerprint: newFingerprint(0x03), ExpiresAt: time.Now().Add(time.Hour),
   282  			}
   283  			_, claimErr := s.Claim(ctx, params)
   284  			require.ErrorIs(t, claimErr, apperror.ErrCanceled)
   285  
   286  			_, getErr := s.Get(ctx, "ctx-cancel", "key-1")
   287  			require.ErrorIs(t, getErr, apperror.ErrCanceled)
   288  
   289  			completeErr := s.Complete(ctx, idempotencybndry.CompleteParams{
   290  				Scope: "ctx-cancel", Key: "key-1", ResponseStatus: 201, ResponsePayload: []byte(`{}`),
   291  			})
   292  			require.ErrorIs(t, completeErr, apperror.ErrCanceled)
   293  
   294  			_, delErr := s.DeleteExpired(ctx, time.Now(), 100)
   295  			require.ErrorIs(t, delErr, apperror.ErrCanceled)
   296  		})
   297  	})
