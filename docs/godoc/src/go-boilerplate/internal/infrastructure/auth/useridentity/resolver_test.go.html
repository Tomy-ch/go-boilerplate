    45  	t.Parallel()
    46  
    47  	testDB := testkit.NewTestDB(t)
    48  	lt := observability.NewMockInfraLayerTracer(t)

    51  	repo := &resolver{

    55  
    56  	newAuthn := func(t *testing.T, issuer, subject string) *authbd.Authn {
    57  		t.Helper()
    58  		authn, err := authbd.New(subject, issuer, nil, nil)
    59  		require.NoError(t, err)
    60  		return authn
    61  	}
    62  
    63  	t.Run("正常系", func(t *testing.T) {
    64  		t.Parallel()
    65  
    66  		t.Run("JWT issuer と subject に対応する未削除ユーザーの UserID を解決する", func(t *testing.T) {
    67  			t.Parallel()
    68  
    69  			txm.WithinTx(func(ctx context.Context) {
    70  				resolved, err := repo.Resolve(ctx, newAuthn(t, testIssuerJWT, "user-john-doe"))
    71  				require.NoError(t, err)
    72  				require.True(t, resolved.HasUserID())
    73  				userID, err := resolved.UserID()
    74  				require.NoError(t, err)
    75  				assert.Equal(t, johnUserID, userID.String())
    76  			})
    77  		})
    78  
    79  		t.Run("mock issuer と subject でも同じユーザーの UserID を解決する", func(t *testing.T) {
    80  			t.Parallel()
    81  
    82  			txm.WithinTx(func(ctx context.Context) {
    83  				resolved, err := repo.Resolve(ctx, newAuthn(t, testIssuerMock, johnUserID))
    84  				require.NoError(t, err)
    85  				require.True(t, resolved.HasUserID())
    86  				userID, err := resolved.UserID()
    87  				require.NoError(t, err)
    88  				assert.Equal(t, johnUserID, userID.String())
    89  			})
    90  		})
    91  	})
