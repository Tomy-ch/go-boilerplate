    57  // signToken は指定した署名方式・typ・クレームでトークン文字列を組み立てます。
    58  func signToken(t *testing.T, key *rsa.PrivateKey, method jwtlib.SigningMethod, typ string, claims jwtlib.MapClaims) string {
    59  	t.Helper()
    60  	token := jwtlib.NewWithClaims(method, claims)
    61  	if typ != "" {
    62  		token.Header["typ"] = typ
    63  	}
    64  
    65  	var (
    66  		signed string
    67  		err    error
    68  	)
    69  	switch method {
    70  	case jwtlib.SigningMethodNone:
    71  		signed, err = token.SignedString(jwtlib.UnsafeAllowNoneSignatureType)
    72  	case jwtlib.SigningMethodHS256:
    73  		signed, err = token.SignedString([]byte("shared-secret"))
    74  	default:
    75  		signed, err = token.SignedString(key)
    76  	}
    77  	require.NoError(t, err)
    78  	return signed
    79  }
