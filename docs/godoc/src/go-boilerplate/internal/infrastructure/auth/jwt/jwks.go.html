   110  // ResolveKey は、kid に対応する署名検証用公開鍵を返します（KeyResolver 実装）。
   111  // キャッシュに無い / 期限切れの場合は JWKS を再取得します。
   112  func (r *jwksResolver) ResolveKey(ctx context.Context, kid string) (crypto.PublicKey, error) {
   113  	if key := r.lookup(kid); key != nil {
   114  		return key, nil
   115  	}
   116  	// 現世代で不在が確定済みの kid は再取得せず即座に拒否する（再取得連打の抑止）。
   117  	if r.negativelyCached(kid) {
   118  		return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidToken, "kid known-absent in current JWKS")
   119  	}
   120  	fetched, err := r.refresh(ctx)
   121  	if err != nil {
   122  		return nil, xerrors.Join(ErrJWTAuthenticatorInvalidToken, err)
   123  	}
   124  	if key := r.lookup(kid); key != nil {
   125  		return key, nil
   126  	}
   127  	// 実際に取得した世代でのみ不在を確定する。cooldown で未取得（throttle）のときに記録すると、
   128  	// その後 cooldown が明けても cacheTTL 満了まで再取得がブロックされ、回転追加の kid を取りこぼす。
   129  	if fetched {
   130  		r.recordAbsent(kid)
   131  	}
   132  	return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidToken, "no matching JWKS key for kid")
   133  }

   135  // negativelyCached は、現世代（鮮度内）で kid が不在確定として記録済みかを返します。
   136  // キャッシュが空 / 期限切れのとき、および kid が negative に未記録のときも false を返します。
   137  // false は「再取得で解決を試みるべき」を意味し kid の存在は保証しません（退役後の再評価・回転追加の取りこぼし防止）。
   138  func (r *jwksResolver) negativelyCached(kid string) bool {
   139  	r.mu.RLock()
   140  	defer r.mu.RUnlock()
   141  	if r.fetchedAt.IsZero() || r.clk.Now().Sub(r.fetchedAt) >= r.cacheTTL {
   142  		return false
   143  	}
   144  	_, ok := r.negative[kid]
   145  	return ok
   146  }
   147  
   148  // recordAbsent は、現世代で不在が確定した kid を negative へ記録します。
   149  // 鮮度外（fetchedAt ゼロ / cacheTTL 超過）では記録しません — 次の世代交代で negative がクリアされるまで残り、
   150  // その間に回転追加された kid を誤って拒否し続けるためです。
   151  func (r *jwksResolver) recordAbsent(kid string) {
   152  	r.mu.Lock()
   153  	defer r.mu.Unlock()
   154  	if r.fetchedAt.IsZero() || r.clk.Now().Sub(r.fetchedAt) >= r.cacheTTL {
   155  		return
   156  	}
   157  	if r.negative == nil {
   158  		r.negative = map[string]struct{}{}
   159  	}
   160  	r.negative[kid] = struct{}{}
   161  }

   163  // lookup は、鮮度内のキャッシュから kid に対応する鍵を返します（無ければ nil）。
   164  func (r *jwksResolver) lookup(kid string) crypto.PublicKey {
   165  	r.mu.RLock()
   166  	defer r.mu.RUnlock()
   167  	if r.fetchedAt.IsZero() || r.clk.Now().Sub(r.fetchedAt) >= r.cacheTTL {
   168  		return nil
   169  	}
   170  	return r.keys[kid]
   171  }

   173  // refresh は、cooldown を尊重して JWKS を再取得します。HTTP I/O 中は mu を保持しません。
   174  // 第 1 戻り値は実際に HTTP 取得を行ったかを表し、cooldown 中の throttle・ctx キャンセルでは false になります
   175  // （呼び出し元が「取得済み世代での不在確定」と「未取得」を区別し、後者を negative へ誤記録しないため）。
   176  func (r *jwksResolver) refresh(ctx context.Context) (bool, error) {
   177  	r.fetchMu.Lock()
   178  	defer r.fetchMu.Unlock()
   179  
   180  	r.mu.RLock()
   181  	throttled := !r.lastAttempt.IsZero() && r.clk.Now().Sub(r.lastAttempt) < r.cooldown
   182  	lastErr := r.lastErr
   183  	r.mu.RUnlock()
   184  	if throttled {
   185  		// cooldown 中は再取得せず直近取得の結果を伝播する。
   186  		// 直近が失敗なら原因を返し（インフラ障害を「無効トークン」と誤認させない）、成功なら nil。
   187  		return false, lastErr
   188  	}
   189  
   190  	keys, err := r.fetch(ctx)
   191  
   192  	// 呼び出し元 ctx のキャンセルは共有キャッシュ（lastErr/lastAttempt）へ載せない。
   193  	// 1 リクエストの切断が cooldown 中の全リクエストへ「無効トークン」として波及するのを防ぐ。
   194  	if err != nil && xerrors.Is(err, apperror.ErrCanceled) {
   195  		return false, err
   196  	}
   197  
   198  	now := r.clk.Now()
   199  	r.mu.Lock()
   200  	r.lastAttempt = now
   201  	r.lastErr = err
   202  	if err == nil {
   203  		// 公開鍵集合が変わった取得成功では、不在確定（negative）を破棄して再評価する（回転追加の取りこぼし防止）。
   204  		// 集合が不変なら negative は現世代の判断として保持する（同一 bogus kid の再取得抑止を維持）。
   205  		if !sameKeySet(r.keys, keys) {
   206  			r.negative = nil
   207  		}
   208  		r.keys = keys
   209  		r.fetchedAt = now
   210  	}
   211  	r.mu.Unlock()
   212  
   213  	return true, err
   214  }

   217  func sameKeySet(a, b map[string]crypto.PublicKey) bool {
   218  	if len(a) != len(b) {
   219  		return false
   220  	}
   221  	for kid := range a {
   222  		if _, ok := b[kid]; !ok {
   223  			return false
   224  		}
   225  	}

   229  // fetch は、httpclient substrate 経由で JWKS を取得し、kid→公開鍵のマップへ変換します。
   230  // タイムアウト・キャンセルは ctx と substrate の profile が担います。
   231  func (r *jwksResolver) fetch(ctx context.Context) (map[string]crypto.PublicKey, error) {
   232  	url, err := r.urlFn(ctx)
   233  	if err != nil {
   234  		return nil, err
   235  	}
   236  	resp, err := r.client.Do(ctx, httpclient.NewRequest(httpclient.MethodGet(), jwksDownstream, url))
   237  	if err != nil {
   238  		return nil, err
   239  	}
   240  	return parseJWKSKeys(resp.Body, r.allowedAlgs)

   243  // parseJWKSKeys は、JWKS(JSON) を kid→公開鍵のマップへパースします。
   244  // 署名用途（use=sig）の KeyID 付き RSA 公開鍵のみ採用し、JWK が alg を宣言している場合は
   245  // allowedAlgs（許可署名アルゴリズム）に含まれるものだけ採用します（RFC 7517 §4.4）。
   246  // 1 件も無ければエラーを返します。
   247  func parseJWKSKeys(data []byte, allowedAlgs []string) (map[string]crypto.PublicKey, error) {
   248  	var set jose.JSONWebKeySet
   249  	if err := json.Unmarshal(data, &set); err != nil {
   250  		return nil, xerrors.Join(errJWKSMalformed, err)
   251  	}
   252  
   253  	keys := make(map[string]crypto.PublicKey, len(set.Keys))
   254  	for i := range set.Keys {
   255  		k := set.Keys[i]
   256  		// 署名用途以外（use=="enc" 等）は採用しない。use 未指定は RFC 7517 上 optional のため許容する。
   257  		if k.Use != "" && k.Use != "sig" {
   258  			continue
   259  		}
   260  		// JWK が alg を宣言している場合は許可アルゴリズムに限定する（未宣言は許容）。
   261  		if k.Algorithm != "" && !slices.Contains(allowedAlgs, k.Algorithm) {
   262  			continue
   263  		}
   264  		pub, ok := k.Key.(*rsa.PublicKey)
   265  		if k.KeyID == "" || !ok {
   266  			continue
   267  		}
   268  		// 重複 kid は silent な last-wins を避け、文書ごと不採用にする（攻撃・設定事故の兆候として loud に失敗する）。
   269  		if _, dup := keys[k.KeyID]; dup {
   270  			return nil, xerrors.Wrap(errJWKSDuplicateKID, k.KeyID)
   271  		}
   272  		keys[k.KeyID] = pub
   273  	}
   274  	if len(keys) == 0 {
   275  		return nil, errJWKSNoKeys
   276  	}
   277  	return keys, nil
   278  }
