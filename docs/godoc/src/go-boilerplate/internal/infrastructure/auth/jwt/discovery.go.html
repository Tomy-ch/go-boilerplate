    85  // cached は、鮮度内のキャッシュから jwks_uri を返します（無ければ空文字）。
    86  func (d *discoveryResolver) cached() string {
    87  	d.mu.RLock()
    88  	defer d.mu.RUnlock()
    89  	if d.fetchedAt.IsZero() || d.clk.Now().Sub(d.fetchedAt) >= d.ttl {
    90  		return ""
    91  	}
    92  	return d.jwksURI
    93  }

    95  // refresh は、discovery を取得・検証し jwks_uri をキャッシュします。HTTP I/O 中は mu を保持しません。
    96  func (d *discoveryResolver) refresh(ctx context.Context) (string, error) {
    97  	d.fetchMu.Lock()
    98  	defer d.fetchMu.Unlock()
    99  
   100  	// 待機中に他 goroutine が更新済みの可能性があるため、取得前に再確認する（二重取得の回避）。
   101  	if u := d.cached(); u != "" {
   102  		return u, nil
   103  	}
   104  
   105  	jwksURI, err := d.fetch(ctx)
   106  	if err != nil {
   107  		return "", err
   108  	}
   109  
   110  	now := d.clk.Now()
   111  	d.mu.Lock()
   112  	d.jwksURI = jwksURI
   113  	d.fetchedAt = now
   114  	d.mu.Unlock()
   115  	return jwksURI, nil
   116  }

   118  // fetch は、discovery 文書を取得し、issuer 厳密一致 / https / 同一オリジンを検証して jwks_uri を返します。
   119  func (d *discoveryResolver) fetch(ctx context.Context) (string, error) {
   120  	if err := requireSecureURL(d.discoveryURL, d.allowInsecure); err != nil {
   121  		return "", err
   122  	}
   123  
   124  	resp, err := d.client.Do(ctx, httpclient.NewRequest(httpclient.MethodGet(), jwksDownstream, d.discoveryURL))
   125  	if err != nil {
   126  		return "", err
   127  	}
   128  
   129  	var doc openidConfiguration
   130  	if err := json.Unmarshal(resp.Body, &doc); err != nil {
   131  		return "", xerrors.Join(errDiscoveryMalformed, err)
   132  	}
   133  
   134  	// issuer 厳密一致（OpenID Connect Discovery §4.3 MUST）。応答侵害時に別 issuer を信頼させない。
   135  	if doc.Issuer != d.issuer {
   136  		return "", xerrors.Wrap(errDiscoveryIssuerMismatch, doc.Issuer)
   137  	}
   138  	if strings.TrimSpace(doc.JWKSURI) == "" {
   139  		return "", errDiscoveryNoJWKSURI
   140  	}
   141  	if err := requireSecureURL(doc.JWKSURI, d.allowInsecure); err != nil {
   142  		return "", err
   143  	}
   144  	// jwks_uri は応答由来のため、issuer と同一オリジンに限定して任意 URL への誘導を防ぐ。
   145  	if err := d.requireSameOrigin(doc.JWKSURI); err != nil {
   146  		return "", err
   147  	}
   148  	return doc.JWKSURI, nil
   149  }

   151  // requireSecureURL は、URL が https であることを要求します（allowInsecure が true の場合は検証しません）。
   152  func requireSecureURL(rawURL string, allowInsecure bool) error {
   153  	if allowInsecure {
   154  		return nil
   155  	}
   156  	u, err := url.Parse(rawURL)
   157  	if err != nil {
   158  		return xerrors.Join(errDiscoveryInsecureURL, err)
   159  	}
   160  	if u.Scheme != "https" {
   161  		return xerrors.Wrap(errDiscoveryInsecureURL, rawURL)
   162  	}
   163  	return nil
   164  }

   166  // requireSameOrigin は、jwks_uri が issuer と同一オリジン（scheme + host）であることを要求します。
   167  func (d *discoveryResolver) requireSameOrigin(rawJWKS string) error {
   168  	iss, err := url.Parse(d.issuer)
   169  	if err != nil {
   170  		return xerrors.Join(errDiscoveryUntrustedJWKS, err)
   171  	}
   172  	j, err := url.Parse(rawJWKS)
   173  	if err != nil {
   174  		return xerrors.Join(errDiscoveryUntrustedJWKS, err)
   175  	}
   176  	if iss.Scheme != j.Scheme || iss.Host != j.Host {
   177  		return xerrors.Wrap(errDiscoveryUntrustedJWKS, rawJWKS)
   178  	}
   179  	return nil
   180  }
