    98  // cached は、鮮度内のキャッシュから jwks_uri を返します（無ければ空文字）。
    99  func (d *discoveryResolver) cached() string {
   100  	d.mu.RLock()
   101  	defer d.mu.RUnlock()
   102  	if d.fetchedAt.IsZero() || d.clk.Now().Sub(d.fetchedAt) >= d.ttl {
   103  		return ""
   104  	}
   105  	return d.jwksURI
   106  }

   108  // refresh は、discovery を取得・検証し jwks_uri をキャッシュします。HTTP I/O 中は mu を保持しません。
   109  func (d *discoveryResolver) refresh(ctx context.Context) (string, error) {
   110  	d.fetchMu.Lock()
   111  	defer d.fetchMu.Unlock()
   112  
   113  	// 待機中に他 goroutine が更新済みの可能性があるため、取得前に再確認する（二重取得の回避）。
   114  	if u := d.cached(); u != "" {
   115  		return u, nil
   116  	}
   117  
   118  	jwksURI, err := d.fetch(ctx)
   119  	if err != nil {
   120  		return "", err
   121  	}
   122  
   123  	now := d.clk.Now()
   124  	d.mu.Lock()
   125  	d.jwksURI = jwksURI
   126  	d.fetchedAt = now
   127  	d.mu.Unlock()
   128  	return jwksURI, nil
   129  }

   131  // fetch は、discovery 文書を取得し、issuer 厳密一致 / https / 同一オリジンを検証して jwks_uri を返します。
   132  func (d *discoveryResolver) fetch(ctx context.Context) (string, error) {
   133  	ctx, endSpan := d.tracer.Start(ctx)
   134  	defer endSpan()
   135  
   136  	if err := requireSecureURL(d.discoveryURL, d.allowInsecure); err != nil {
   137  		return "", err
   138  	}
   139  
   140  	resp, err := d.client.Do(ctx, httpclient.NewRequest(httpclient.MethodGet(), jwksDownstream, d.discoveryURL))
   141  	if err != nil {
   142  		return "", err
   143  	}
   144  
   145  	var doc openidConfiguration
   146  	if err := json.Unmarshal(resp.Body, &doc); err != nil {
   147  		return "", xerrors.Join(errDiscoveryMalformed, err)
   148  	}
   149  
   150  	// issuer 厳密一致（OpenID Connect Discovery §4.3 MUST）。応答侵害時に別 issuer を信頼させない。
   151  	if doc.Issuer != d.issuer {
   152  		return "", xerrors.Wrap(errDiscoveryIssuerMismatch, doc.Issuer)
   153  	}
   154  	if strings.TrimSpace(doc.JWKSURI) == "" {
   155  		return "", errDiscoveryNoJWKSURI
   156  	}
   157  	if err := requireSecureURL(doc.JWKSURI, d.allowInsecure); err != nil {
   158  		return "", err
   159  	}
   160  	// jwks_uri は応答由来のため、issuer と同一オリジンに限定して任意 URL への誘導を防ぐ。
   161  	if err := d.requireSameOrigin(doc.JWKSURI); err != nil {
   162  		return "", err
   163  	}
   164  	return doc.JWKSURI, nil
   165  }

   167  // requireSecureURL は、URL が https であることを要求します（allowInsecure が true の場合は検証しません）。
   168  func requireSecureURL(rawURL string, allowInsecure bool) error {
   169  	if allowInsecure {
   170  		return nil
   171  	}
   172  	u, err := url.Parse(rawURL)
   173  	if err != nil {
   174  		return xerrors.Join(errDiscoveryInsecureURL, err)
   175  	}
   176  	if u.Scheme != "https" {
   177  		return xerrors.Wrap(errDiscoveryInsecureURL, rawURL)
   178  	}
   179  	return nil
   180  }

   182  // requireSameOrigin は、jwks_uri が issuer と同一オリジン（scheme + host）であることを要求します。
   183  func (d *discoveryResolver) requireSameOrigin(rawJWKS string) error {
   184  	iss, err := url.Parse(d.issuer)
   185  	if err != nil {
   186  		return xerrors.Join(errDiscoveryUntrustedJWKS, err)
   187  	}
   188  	j, err := url.Parse(rawJWKS)
   189  	if err != nil {
   190  		return xerrors.Join(errDiscoveryUntrustedJWKS, err)
   191  	}
   192  	if iss.Scheme != j.Scheme || iss.Host != j.Host {
   193  		return xerrors.Wrap(errDiscoveryUntrustedJWKS, rawJWKS)
   194  	}
   195  	return nil
   196  }
