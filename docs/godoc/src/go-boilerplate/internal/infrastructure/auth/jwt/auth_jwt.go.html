    69  // New は JWT Authenticator のコンストラクタです。
    70  // 公開鍵 PEM のパース失敗・必須パラメータ不足は設定エラーとして返します（認証エラーとは区別）。
    71  func New(params Params) (authbd.Authenticator, error) {
    72  	if params.Clock == nil {
    73  		return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidParams, "clock is required")
    74  	}
    75  	if strings.TrimSpace(params.Issuer) == "" {
    76  		return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidParams, "issuer is required")
    77  	}
    78  	if strings.TrimSpace(params.Audience) == "" {
    79  		return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidParams, "audience is required")
    80  	}
    81  
    82  	publicKey, err := jwtlib.ParseRSAPublicKeyFromPEM([]byte(params.PublicKeyPEM))
    83  	if err != nil {
    84  		return nil, xerrors.Join(ErrJWTAuthenticatorInvalidPublicKey, err)
    85  	}
    86  
    87  	algs := params.AllowedAlgs
    88  	if len(algs) == 0 {
    89  		algs = defaultAllowedAlgs
    90  	}
    91  
    92  	leeway := params.Leeway
    93  	if leeway <= 0 {
    94  		leeway = defaultLeeway
    95  	}
    96  
    97  	parser := jwtlib.NewParser(
    98  		jwtlib.WithValidMethods(algs),
    99  		jwtlib.WithIssuer(params.Issuer),
   100  		jwtlib.WithAudience(params.Audience),
   101  		jwtlib.WithExpirationRequired(),
   102  		jwtlib.WithLeeway(leeway),
   103  		jwtlib.WithTimeFunc(params.Clock.Now),
   104  	)
   105  
   106  	return &authenticator{
   107  		parser:       parser,
   108  		publicKey:    publicKey,
   109  		expectedType: strings.TrimSpace(params.ExpectedType),
   110  	}, nil
   111  }

   113  // Authenticate は access token（JWT）を検証し、検証済みの Authn を返します。
   114  // cred が nil の場合は無効トークンとして扱います。
   115  // 検証失敗はすべて ErrJWTAuthenticatorInvalidToken（= apperror.ErrUnauthenticated）へ正規化します。
   116  func (a *authenticator) Authenticate(_ context.Context, cred *authbd.Credential) (*authbd.Authn, error) {
   117  	if cred == nil {
   118  		return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidToken, "credential is nil")
   119  	}
   120  
   121  	claims := jwtlib.MapClaims{}
   122  	token, err := a.parser.ParseWithClaims(cred.AccessToken(), claims, a.keyFunc)
   123  	if err != nil || !token.Valid {
   124  		return nil, xerrors.Join(ErrJWTAuthenticatorInvalidToken, err)
   125  	}
   126  
   127  	if err := a.verifyType(token); err != nil {
   128  		return nil, err
   129  	}
   130  
   131  	subject, err := claims.GetSubject()
   132  	if err != nil || strings.TrimSpace(subject) == "" {
   133  		return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidToken, "subject missing")
   134  	}
   135  
   136  	return authbd.New(subject, authbd.ProviderJWT, extractScopes(claims), map[string]any(claims))
   137  }

   139  // keyFunc は署名検証に用いる固定公開鍵を返します。
   140  // 保持する鍵は RSA 公開鍵であるため、RSA 系（RS* / PS*）以外の署名方式は
   141  // alg allowlist をすり抜けても鍵種別不一致として拒否します（鍵混同の防御）。
   142  func (a *authenticator) keyFunc(token *jwtlib.Token) (any, error) {
   143  	switch token.Method.(type) {
   144  	case *jwtlib.SigningMethodRSA, *jwtlib.SigningMethodRSAPSS:
   145  		return a.publicKey, nil
   146  	default:
   147  		return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidToken, "unexpected signing method type")
   148  	}
   149  }

   151  // verifyType は expectedType が設定されている場合に typ ヘッダを検証し、ID Token 等の誤用を拒否します。
   152  func (a *authenticator) verifyType(token *jwtlib.Token) error {
   153  	if a.expectedType == "" {
   154  		return nil
   155  	}
   156  	typ, _ := token.Header["typ"].(string)
   157  	if !strings.EqualFold(typ, a.expectedType) {
   158  		return xerrors.Wrap(ErrJWTAuthenticatorInvalidToken, "unexpected token type")
   159  	}
   160  	return nil
   161  }

   163  // extractScopes は OAuth2 標準 scope クレーム（スペース区切り文字列）を []string に分割します。
   164  // scope クレームが無い場合、文字列型でない場合、または空の場合は nil を返します。
   165  func extractScopes(claims jwtlib.MapClaims) []string {
   166  	raw, ok := claims[scopeClaim].(string)
   167  	if !ok {
   168  		return nil
   169  	}
   170  	return strings.Fields(raw)
   171  }
