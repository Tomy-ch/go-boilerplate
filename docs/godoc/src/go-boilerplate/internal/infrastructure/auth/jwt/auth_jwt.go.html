   118  // New は固定 RSA 公開鍵で検証する JWT Authenticator を生成します。
   119  // 公開鍵 PEM のパース失敗・必須パラメータ不足は設定エラーとして返します（認証エラーとは区別）。
   120  func New(params Params) (authbd.Authenticator, error) {
   121  	publicKey, err := jwtlib.ParseRSAPublicKeyFromPEM([]byte(params.PublicKeyPEM))
   122  	if err != nil {
   123  		return nil, xerrors.Join(ErrJWTAuthenticatorInvalidPublicKey, err)
   124  	}
   125  
   126  	return buildAuthenticator(params, fixedKeyResolver{key: publicKey})
   127  }

   139  // NewJWKS は、JWKS エンドポイントから kid で公開鍵を解決する JWT Authenticator を生成します。
   140  // JWKSURL を明示した場合はそれを直接用い、空の場合は Issuer から OIDC discovery で jwks_uri を導出します。
   141  // JWKS / discovery の取得は httpclient substrate（timeout / retry / breaker / budget / o11y）に委ね、遅延取得します。
   142  func NewJWKS(params JWKSParams, client httpclient.Client) (authbd.Authenticator, error) {
   143  	if client == nil {
   144  		return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidParams, "http client is required")
   145  	}
   146  
   147  	urlFn, err := buildJWKSURLProvider(params, client)
   148  	if err != nil {
   149  		return nil, err
   150  	}
   151  
   152  	resolver := newJWKSResolver(client, urlFn, params.CacheTTL, params.AllowedAlgs, params.UnknownKidCooldown, params.Clock, params.TracerFactory)
   153  	return NewWithKeyResolver(params.Params, resolver)
   154  }

   156  // buildJWKSURLProvider は、JWKS URL の供給関数を返します。
   157  // JWKSURL を明示した場合は静的にそれを返し（discovery / same-origin をスキップ、split-horizon を温存）、
   158  // 空の場合は Issuer から discovery で jwks_uri を導出する解決器を返します。
   159  func buildJWKSURLProvider(params JWKSParams, client httpclient.Client) (func(context.Context) (string, error), error) {
   160  	if url := strings.TrimSpace(params.JWKSURL); url != "" {
   161  		// override は same-origin を免除するが、https は環境ゲート（AllowInsecureURL）で構築時に強制する。
   162  		if err := requireSecureURL(url, params.AllowInsecureURL); err != nil {
   163  			return nil, xerrors.Join(ErrJWTAuthenticatorInvalidParams, err)
   164  		}
   165  		return func(context.Context) (string, error) { return url, nil }, nil
   166  	}
   167  
   168  	if strings.TrimSpace(params.Issuer) == "" {
   169  		return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidParams, "either jwks url or issuer is required")
   170  	}
   171  
   172  	// issuer 由来の discovery URL のスキームも構築時に検証し、設定ミスを起動時 fail-fast にする。
   173  	discovery := newDiscoveryResolver(client, params.Issuer, params.DiscoveryTTL, params.Clock, params.AllowInsecureURL, params.TracerFactory)
   174  	if err := requireSecureURL(discovery.discoveryURL, params.AllowInsecureURL); err != nil {
   175  		return nil, xerrors.Join(ErrJWTAuthenticatorInvalidParams, err)
   176  	}
   177  	return discovery.jwksURL, nil
   178  }

   180  // buildAuthenticator は共通の検証パラメータを検証して authenticator を生成します。
   181  func buildAuthenticator(params Params, keyResolver KeyResolver) (authbd.Authenticator, error) {
   182  	if params.Clock == nil {
   183  		return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidParams, "clock is required")
   184  	}
   185  	if strings.TrimSpace(params.Issuer) == "" {
   186  		return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidParams, "issuer is required")
   187  	}
   188  	if strings.TrimSpace(params.Audience) == "" {
   189  		return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidParams, "audience is required")
   190  	}
   191  
   192  	algs := params.AllowedAlgs
   193  	if len(algs) == 0 {
   194  		algs = defaultAllowedAlgs
   195  	}
   196  
   197  	leeway := params.Leeway
   198  	if leeway <= 0 {
   199  		leeway = defaultLeeway
   200  	}
   201  
   202  	parser := jwtlib.NewParser(
   203  		jwtlib.WithValidMethods(algs),
   204  		jwtlib.WithIssuer(params.Issuer),
   205  		jwtlib.WithAudience(params.Audience),
   206  		jwtlib.WithExpirationRequired(),
   207  		jwtlib.WithLeeway(leeway),
   208  		jwtlib.WithTimeFunc(params.Clock.Now),
   209  	)
   210  
   211  	return &authenticator{
   212  		parser:       parser,
   213  		keyResolver:  keyResolver,
   214  		expectedType: strings.TrimSpace(params.ExpectedType),
   215  		issuer:       strings.TrimSpace(params.Issuer),
   216  	}, nil
   217  }

   219  // Authenticate は access token（JWT）を検証し、検証済みの Authn を返します。
   220  // cred が nil の場合は無効トークンとして扱います。
   221  // 検証失敗はすべて ErrJWTAuthenticatorInvalidToken（= apperror.ErrUnauthenticated）へ正規化します。
   222  func (a *authenticator) Authenticate(ctx context.Context, cred *authbd.Credential) (*authbd.Authn, error) {
   223  	if cred == nil {
   224  		return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidToken, "credential is nil")
   225  	}
   226  
   227  	claims := jwtlib.MapClaims{}
   228  	token, err := a.parser.ParseWithClaims(cred.Token(), claims, a.keyFunc(ctx))
   229  	if err != nil || !token.Valid {
   230  		return nil, xerrors.Join(ErrJWTAuthenticatorInvalidToken, err)
   231  	}
   232  
   233  	if err := a.verifyType(token); err != nil {
   234  		return nil, err
   235  	}
   236  
   237  	subject, err := claims.GetSubject()
   238  	if err != nil || strings.TrimSpace(subject) == "" {
   239  		return nil, xerrors.Wrap(ErrJWTAuthenticatorInvalidToken, "subject missing")
   240  	}
   241  
   242  	// issuer は parser が iss クレームを params.Issuer と一致検証済みのため、その期待値を採用する。
   243  	return authbd.New(subject, a.issuer, extractScopes(claims), map[string]any(claims))
   244  }

   246  // keyFunc は、ctx を捕捉した jwtlib.Keyfunc を返します。署名検証に用いる公開鍵を KeyResolver 経由で解決します。
   247  // 解決する鍵は RSA 公開鍵であるため、RSA 系（RS* / PS*）以外の署名方式は
   248  // alg allowlist をすり抜けても鍵種別不一致として拒否します（鍵混同の防御）。
   249  // keyFunc / KeyResolver が返すエラーは ParseWithClaims 経由で Authenticate の境界へ伝播し、
   250  // そこで ErrJWTAuthenticatorInvalidToken へ一括正規化するため、ここでは原因のみを持つ素の error を返します。
   251  func (a *authenticator) keyFunc(ctx context.Context) jwtlib.Keyfunc {
   252  	return func(token *jwtlib.Token) (any, error) {
   253  		switch token.Method.(type) {
   254  		case *jwtlib.SigningMethodRSA, *jwtlib.SigningMethodRSAPSS:
   255  			kid, _ := token.Header["kid"].(string)
   256  			return a.keyResolver.ResolveKey(ctx, kid)
   257  		default:
   258  			return nil, errUnexpectedSigningMethod
   259  		}
   260  	}
   261  }

   263  // verifyType は expectedType が設定されている場合に typ ヘッダを検証し、ID Token 等の誤用を拒否します。
   264  func (a *authenticator) verifyType(token *jwtlib.Token) error {
   265  	if a.expectedType == "" {
   266  		return nil
   267  	}
   268  	typ, _ := token.Header["typ"].(string)
   269  	if !strings.EqualFold(typ, a.expectedType) {
   270  		return xerrors.Wrap(ErrJWTAuthenticatorInvalidToken, "unexpected token type")
   271  	}
   272  	return nil
   273  }

   275  // extractScopes は OAuth2 標準 scope クレーム（スペース区切り文字列）を []string に分割します。
   276  // scope クレームが無い場合、文字列型でない場合、または空の場合は nil を返します。
   277  func extractScopes(claims jwtlib.MapClaims) []string {
   278  	raw, ok := claims[scopeClaim].(string)
   279  	if !ok {
   280  		return nil
   281  	}
   282  	return strings.Fields(raw)
   283  }
