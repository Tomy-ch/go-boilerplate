  1279  func newBlockedFetch(t *testing.T, client *mock_httpclient.MockClient, pub *rsa.PublicKey) blockedFetch {

  1282  	body := jwksPublicJSON(t, pub)
  1283  	client.EXPECT().Do(gomock.Any(), gomock.Any()).DoAndReturn(
  1284  		func(ctx context.Context, _ *httpclient.Request) (*httpclient.Response, error) {
  1285  			close(b.entered)
  1286  			select {
  1287  			case <-ctx.Done():
  1288  				return nil, xerrors.Wrap(apperror.ErrCanceled, "canceled")
  1289  			case <-b.release:

  1291  			}
  1292  		}).Times(1)
