    13  // newQueueConfig は、config から SQS publisher の設定を解決して返します。
    14  // 未設定のまま起動すると全 publish が失敗し、気付かぬうちに全メッセージが dead 化するため、
    15  // HTTP 側の NewEndpoint と同じく relay 起動時点で弾きます（サイレント障害の防止）。
    16  func newQueueConfig(cfg *config.OutboxConfig) (sqs.PublisherConfig, error) {
    17  	if cfg.QueueURL() == "" {
    18  		return sqs.PublisherConfig{}, xerrors.Wrap(ErrInvalidQueue, "OUTBOX_QUEUE_URL must not be empty")
    19  	}
    20  	// region は SigV4 署名に必須で、空のまま送ると署名不一致として送信時に落ちる。
    21  	if cfg.QueueRegion() == "" {
    22  		return sqs.PublisherConfig{}, xerrors.Wrap(ErrInvalidQueue, "OUTBOX_QUEUE_REGION must not be empty")
    23  	}
    24  	// 資格情報はここで見ない。両方空は SDK 既定の credential chain（IAM ロール等）へ委ねる正当な指定で、
    25  	// 解決できるかどうかは sqs.NewClient が起動時に確かめる。
    26  	return sqs.PublisherConfig{QueueURL: cfg.QueueURL()}, nil
    27  }
