    27  // NewDownstreamProfile は、outbox publish 向けの resilient プロファイルを返します。
    28  // relay の poll ループ自体が at-least-once の retry 本体であるため、transport 層 retry は
    29  // 二重になる。これを避けるため MaxAttempts=1 で transport retry を無効化する（D10）。
    30  // traceparent は emit 時に capture した値を headers で明示伝搬するため、ここでの自動 inject は抑止する。
    31  // 送信先はクラスター外部の受信エンドポイントを想定するため、private/loopback 宛てを拒否する（SSRF 抑止）。
    32  func NewDownstreamProfile() httpclient.DownstreamProfile {
    33  	p := httpclient.DefaultProfile()
    34  	p.MaxAttempts = 1
    35  	p.PropagateTrace = false
    36  	p.AllowPrivateNetwork = false
    37  	return httpclient.DownstreamProfile{Name: downstream, Profile: p}
    38  }

    60  // Publish は、メッセージを受信エンドポイントへ POST します。
    61  // message_id を Idempotency-Key として伝搬し、非冪等メソッドだが retry は明示的に無効化します
    62  // （再送は relay の次 poll が担う）。非 2xx / transport 失敗は substrate が apperror へ写像して返します。
    63  func (p *httpPublisher) Publish(ctx context.Context, m boundary.Message) error {
    64  	ctx, endSpan := p.tracer.Start(ctx)
    65  	defer endSpan()
    66  
    67  	header := httpclient.Header{"Content-Type": {"application/json"}}
    68  	for k, v := range m.Headers {
    69  		// emit 側でも同じ判定を行うが、emit を経由せず INSERT された行に対する真の egress 境界での防御。
    70  		if httpheader.IsSensitive(k) {
    71  			continue
    72  		}
    73  		header[k] = []string{v}
    74  	}
    75  
    76  	// AllowRetry は付与しない（at-least-once は relay の poll ループが担う）。
    77  	// 受信側 dedup 用に IdempotencyKey（message_id）のみ伝搬する。
    78  	_, err := p.client.Do(ctx, httpclient.NewRequest(
    79  		httpclient.MethodPost(), downstream, string(p.endpoint),
    80  		httpclient.WithHeader(header),
    81  		httpclient.WithBody(m.Payload),
    82  		httpclient.WithIdempotencyKey(m.MessageID.String()),
    83  	))
    84  	return err
    85  }
