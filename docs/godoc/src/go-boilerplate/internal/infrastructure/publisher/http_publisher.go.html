    26  // NewDownstreamProfile は、outbox publish 向けの resilient プロファイルを返します。
    27  // relay の poll ループ自体が at-least-once の retry 本体であるため、transport 層 retry は
    28  // 二重になる。これを避けるため MaxAttempts=1 で transport retry を無効化する（D10）。
    29  // traceparent は emit 時に capture した値を headers で明示伝搬するため、ここでの自動 inject は抑止する。
    30  // 送信先はクラスター外部の受信エンドポイントを想定するため、private/loopback 宛てを拒否する（SSRF 抑止）。
    31  func NewDownstreamProfile() httpclient.DownstreamProfile {
    32  	p := httpclient.DefaultProfile()
    33  	p.MaxAttempts = 1
    34  	p.PropagateTrace = false
    35  	p.AllowPrivateNetwork = false
    36  	return httpclient.DownstreamProfile{Name: downstream, Profile: p}
    37  }

    52  // Publish は、メッセージを受信エンドポイントへ POST します。
    53  // message_id を Idempotency-Key として伝搬し、非冪等メソッドだが retry は明示的に無効化します
    54  // （再送は relay の次 poll が担う）。非 2xx / transport 失敗は substrate が apperror へ写像して返します。
    55  func (p *httpPublisher) Publish(ctx context.Context, m boundary.Message) error {
    56  	ctx, endSpan := p.tracer.Start(ctx)
    57  	defer endSpan()
    58  
    59  	header := httpclient.Header{"Content-Type": {"application/json"}}
    60  	for k, v := range m.Headers {
    61  		header[k] = []string{v}
    62  	}
    63  
    64  	_, err := p.client.Do(ctx, &httpclient.Request{
    65  		Downstream:     downstream,
    66  		Method:         httpclient.MethodPost,
    67  		URL:            string(p.endpoint),
    68  		Header:         header,
    69  		Body:           m.Payload,
    70  		IdempotencyKey: m.MessageID.String(),
    71  		AllowRetry:     false,
    72  	})
    73  	return err
    74  }
