    27  func TestBreakerAllowAndRecord(t *testing.T) {
    28  	t.Parallel()
    29  
    30  	base := time.Unix(1000, 0)
    31  
    32  	t.Run("正常系", func(t *testing.T) {
    33  		t.Parallel()
    34  
    35  		t.Run("MinRequests未満では失敗が続いてもclosedのまま", func(t *testing.T) {
    36  			t.Parallel()
    37  
    38  			b := newBreaker(testBreakerConfig())
    39  			recordClosedFailures(b, 3, base)
    40  
    41  			assert.Equal(t, breakerClosed, b.currentState())
    42  			allowed, _ := b.allow(base)
    43  			assert.True(t, allowed)
    44  		})
    45  
    46  		t.Run("MinRequests以上かつ失敗率が閾値以上でopenに倒れfail-fastする", func(t *testing.T) {
    47  			t.Parallel()
    48  
    49  			b := newBreaker(testBreakerConfig())
    50  			recordClosedFailures(b, 4, base)
    51  
    52  			assert.Equal(t, breakerOpen, b.currentState())
    53  			allowed, _ := b.allow(base) // OpenDuration 未経過は通さない
    54  			assert.False(t, allowed)
    55  		})
    56  
    57  		t.Run("OpenDuration経過でhalf-openへ遷移しプローブを通す", func(t *testing.T) {
    58  			t.Parallel()
    59  
    60  			b := newBreaker(testBreakerConfig())
    61  			recordClosedFailures(b, 4, base)
    62  
    63  			allowed, _ := b.allow(base.Add(10 * time.Second))
    64  			assert.True(t, allowed)
    65  			assert.Equal(t, breakerHalfOpen, b.currentState())
    66  		})
    67  
    68  		t.Run("half-openでHalfOpenProbes件成功するとclosedへ戻る", func(t *testing.T) {
    69  			t.Parallel()
    70  
    71  			b := newBreaker(testBreakerConfig())
    72  			recordClosedFailures(b, 4, base)
    73  			now := base.Add(10 * time.Second)
    74  
    75  			allowed, gen := b.allow(now)
    76  			require.True(t, allowed)
    77  			b.record(true, now, gen)
    78  			allowed, gen = b.allow(now)
    79  			require.True(t, allowed)
    80  			b.record(true, now, gen)
    81  
    82  			assert.Equal(t, breakerClosed, b.currentState())
    83  		})
    84  	})
    85

    87  		t.Parallel()
    88

   103  		t.Run("half-openでHalfOpenProbesを超えるプローブは通さない", func(t *testing.T) {
   104  			t.Parallel()
   105  
   106  			b := newBreaker(testBreakerConfig())

   111  			a2, _ := b.allow(now)
   112  			a3, _ := b.allow(now)
   113  			assert.True(t, a1)  // probe 1
   114  			assert.True(t, a2)  // probe 2
   115  			assert.False(t, a3) // 上限超過
   116  		})

   118  		t.Run("旧エポックの遅延成功は新エポックのクローズ判定に混入しない", func(t *testing.T) {
   119  			t.Parallel()
   120  
   121  			b := newBreaker(testBreakerConfig())
   122  			recordClosedFailures(b, 4, base)
   123  			now := base.Add(10 * time.Second)
   124  
   125  			// エポック1: 2 プローブを承認し、1 つ目を失敗させて再び open へ倒す。
   126  			_, gen1 := b.allow(now)
   127  			_, genStale := b.allow(now) // 同エポックの遅延プローブ枠
   128  			b.record(false, now, gen1)
   129  			require.Equal(t, breakerOpen, b.currentState())
   130  
   131  			// エポック2へ遷移し、新たに 1 プローブを承認する。
   132  			later := now.Add(10 * time.Second)
   133  			allowed, gen3 := b.allow(later)
   134  			require.True(t, allowed)
   135  			require.Equal(t, breakerHalfOpen, b.currentState())
   136  
   137  			// 旧エポックの遅延成功はエポック不一致で無視され、新エポックの成功にカウントされない。
   138  			b.record(true, later, genStale)
   139  			assert.Equal(t, breakerHalfOpen, b.currentState())
   140  
   141  			// 新エポックの成功は 1 件のみのため、まだ closed へは戻らない。
   142  			b.record(true, later, gen3)
   143  			assert.Equal(t, breakerHalfOpen, b.currentState())
   144  		})
   145  	})
