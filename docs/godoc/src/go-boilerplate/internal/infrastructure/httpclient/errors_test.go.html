    20  		t.Parallel()

    23  			status  int
    24  			want    error

    27  			"200はエラーなし":                     {status: http.StatusOK, wantNil: true},

    31  			"400はErrInvalidArgument":        {status: http.StatusBadRequest, want: apperror.ErrInvalidArgument},
    32  			"401はErrUnauthenticated":        {status: http.StatusUnauthorized, want: apperror.ErrUnauthenticated},
    33  			"403はErrPermissionDenied":       {status: http.StatusForbidden, want: apperror.ErrPermissionDenied},
    34  			"404はErrNotFound":               {status: http.StatusNotFound, want: apperror.ErrNotFound},
    35  			"409はErrConflict":               {status: http.StatusConflict, want: apperror.ErrConflict},
    36  			"422はErrValidation":             {status: http.StatusUnprocessableEntity, want: apperror.ErrValidation},
    37  			"429はErrTooManyRequests":        {status: http.StatusTooManyRequests, want: apperror.ErrTooManyRequests},
    38  			"418など他の4xxはErrInvalidArgument": {status: http.StatusTeapot, want: apperror.ErrInvalidArgument},
    39  			"500はErrUnavailable":            {status: http.StatusInternalServerError, want: apperror.ErrUnavailable},
    40  			"503はErrUnavailable":            {status: http.StatusServiceUnavailable, want: apperror.ErrUnavailable},
    41  		}
    42  
    43  		for name, tc := range cases {
    44  			t.Run(name, func(t *testing.T) {
    45  				t.Parallel()
    46  				got := statusToAppError(tc.status)
    47  				if tc.wantNil {
    48  					require.NoError(t, got)
    49  					return
    50  				}
    51  				require.ErrorIs(t, got, tc.want)
    52  			})
    53  		}
    54  	})
