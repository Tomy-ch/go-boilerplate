    49  func TestRequest(t *testing.T) {
    50  	t.Parallel()
    51  
    52  	t.Run("正常系", func(t *testing.T) {
    53  		t.Parallel()
    54  
    55  		t.Run("各フィールドを保持できる", func(t *testing.T) {
    56  			t.Parallel()
    57  
    58  			req := httpclient.NewRequest(
    59  				httpclient.MethodPost(), "payment", "https://example.com/charge",
    60  				httpclient.WithHeader(httpclient.Header{"Content-Type": {"application/json"}}),
    61  				httpclient.WithBody([]byte(`{"amount":100}`)),
    62  				httpclient.WithRetry("key-1"),
    63  			)
    64  
    65  			assert.Equal(t, httpclient.Downstream("payment"), req.Downstream())
    66  			assert.Equal(t, httpclient.MethodPost(), req.Method())
    67  			assert.Equal(t, "https://example.com/charge", req.URL())
    68  			assert.Equal(t, []string{"application/json"}, req.Header()["Content-Type"])
    69  			assert.Equal(t, []byte(`{"amount":100}`), req.Body())
    70  			assert.Equal(t, "key-1", req.IdempotencyKey())
    71  			assert.True(t, req.AllowRetry())
    72  		})
    73  
    74  		t.Run("HeaderとBodyのゲッターは防御的コピーを返し内部状態は不変", func(t *testing.T) {
    75  			t.Parallel()
    76  
    77  			req := httpclient.NewRequest(
    78  				httpclient.MethodPost(), "payment", "https://example.com/charge",
    79  				httpclient.WithHeader(httpclient.Header{"X-Token": {"secret"}}),
    80  				httpclient.WithBody([]byte("original")),
    81  			)
    82  
    83  			// 返り値を破壊的に変更しても、Request 内部には影響しないこと。
    84  			got := req.Header()
    85  			got["X-Token"][0] = "tampered"
    86  			got["X-Injected"] = []string{"x"}
    87  			body := req.Body()
    88  			body[0] = 'X'
    89  
    90  			assert.Equal(t, []string{"secret"}, req.Header()["X-Token"], "Header の値スライスが共有されていないこと")
    91  			assert.NotContains(t, req.Header(), "X-Injected", "Header の map が共有されていないこと")
    92  			assert.Equal(t, []byte("original"), req.Body(), "Body スライスが共有されていないこと")
    93  		})
    94  	})
