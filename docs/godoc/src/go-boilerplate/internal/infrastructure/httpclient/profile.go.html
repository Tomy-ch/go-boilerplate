   122  // NewRegistryFromProfiles は、各 gateway が寄与した DownstreamProfile 群から Registry を生成します。
   123  // 未登録の Downstream には DefaultProfile が適用されます。
   124  // 同一 Name が重複した場合は、silent な last-wins 上書きを避けるためエラーを返します。
   125  func NewRegistryFromProfiles(profiles []DownstreamProfile) (Registry, error) {
   126  	m := make(map[Downstream]Profile, len(profiles))
   127  	for _, p := range profiles {
   128  		if _, dup := m[p.Name]; dup {
   129  			return nil, xerrors.Wrap(ErrDuplicateProfile, fmt.Sprintf("%q", p.Name))
   130  		}
   131  		m[p.Name] = p.Profile
   132  	}
   133  	return NewRegistry(m), nil
   134  }

   136  // MissingDownstreams は、required のうち profiles に Profile が登録されていない Downstream を返します。
   137  // 起動時の網羅チェックに使い、空スライスなら全ての required が登録済みであることを表します。
   138  // これにより gateway 追加時の profile 登録漏れ・改名が、silent な DefaultProfile fallback ではなく
   139  // loud な起動失敗として顕在化します。
   140  func MissingDownstreams(profiles []DownstreamProfile, required []Downstream) []Downstream {
   141  	registered := make(map[Downstream]struct{}, len(profiles))
   142  	for _, p := range profiles {
   143  		registered[p.Name] = struct{}{}
   144  	}
   145  
   146  	var missing []Downstream
   147  	for _, d := range required {
   148  		if _, ok := registered[d]; !ok {
   149  			missing = append(missing, d)
   150  		}
   151  	}
   152  	return missing
   153  }
