    26  	t.Helper()
    27  	var hits atomic.Int32
    28  	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
    29  		hits.Add(1)
    30  		w.WriteHeader(status)
    31  	}))
    32  	t.Cleanup(srv.Close)
    33  	return srv, &hits
    34  }

   462  				DoAndReturn(func(_ context.Context, d time.Duration) error {

   479  				upperBound := min(profile.BaseBackoff<<i, profile.MaxBackoff)
