    24  // countingServer は、リクエスト回数を数えつつ固定ステータスを返すテストサーバを生成します。
    25  func countingServer(t *testing.T, status int) (*httptest.Server, *atomic.Int32) {
    26  	t.Helper()
    27  	var hits atomic.Int32
    28  	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
    29  		hits.Add(1)
    30  		w.WriteHeader(status)
    31  	}))
    32  	t.Cleanup(srv.Close)
    33  	return srv, &hits
    34  }

   378  			t.Parallel()
   379  
   380  			srv, _ := countingServer(t, http.StatusServiceUnavailable)
   381  
   382  			profile := httpclient.DefaultProfile()
   383  			profile.MaxAttempts = 4
   384  			profile.BaseBackoff = 10 * time.Millisecond
   385  			profile.MaxBackoff = time.Second
   386  			profile.RetryBudgetRatio = 100 // budget で retry が絞られないよう十分に確保
   387  			registry := httpclient.NewRegistry(map[httpclient.Downstream]httpclient.Profile{"retry": profile})
   388  
   389  			var slept []time.Duration
   390  			ctrl := gomock.NewController(t)
   391  			sleeper := mock_clock.NewMockSleeper(ctrl)
   392  			sleeper.EXPECT().Sleep(gomock.Any(), gomock.Any()).
   393  				DoAndReturn(func(_ context.Context, d time.Duration) error {
   394  					slept = append(slept, d)
   395  					return nil
   396  				}).AnyTimes()
   397  
   398  			client := httpclient.New(observability.NewNoopHTTPClientTransport(t), sleeper, registry, observability.NewNoopHTTPClientMetrics(t))
   399  			_, err := client.Do(context.Background(), httpclient.NewRequest(httpclient.MethodGet(), "retry", srv.URL))
   400  
   401  			require.ErrorIs(t, err, apperror.ErrUnavailable)
   402  			require.Len(t, slept, 3) // MaxAttempts-1 回のリトライ
   403  			for i, d := range slept {
   404  				upperBound := min(profile.BaseBackoff<<i, profile.MaxBackoff)
   405  				assert.GreaterOrEqual(t, d, time.Duration(0))
   406  				assert.LessOrEqual(t, d, upperBound)
   407  			}
   408  		})
   409  	})
