    48  // allow は、now 時点でリクエストを通してよいかと、その試行が属するエポックを返します。
    49  // 返したエポックは record にそのまま渡します。
    50  // open は OpenDuration 経過で half-open へ遷移し、half-open は HalfOpenProbes 件までプローブを通します。
    51  func (b *breaker) allow(now time.Time) (bool, uint64) {
    52  	b.mu.Lock()
    53  	defer b.mu.Unlock()
    54  
    55  	switch b.state {
    56  	case breakerOpen:
    57  		if now.Sub(b.openedAt) >= b.config.OpenDuration {
    58  			b.toHalfOpen()
    59  			b.halfOpenProbes++
    60  			return true, b.generation
    61  		}
    62  		return false, b.generation
    63  	case breakerHalfOpen:
    64  		if b.halfOpenProbes < b.config.HalfOpenProbes {
    65  			b.halfOpenProbes++
    66  			return true, b.generation
    67  		}
    68  		return false, b.generation
    69  	default: // breakerClosed
    70  		return true, b.generation
    71  	}
    72  }

    74  // record は、試行結果を記録し状態遷移を行います。success=false は downstream 起因の失敗です。
    75  func (b *breaker) record(success bool, now time.Time, generation uint64) {
    76  	b.mu.Lock()
    77  	defer b.mu.Unlock()
    78  
    79  	switch b.state {
    80  	case breakerHalfOpen:
    81  		// half-open は厳密にプローブ数で回復判定するため、別エポックの遅延結果を無視する。
    82  		if generation != b.generation {
    83  			return
    84  		}
    85  		if !success {
    86  			b.toOpen(now)
    87  			return
    88  		}
    89  		b.halfOpenSuccess++
    90  		if b.halfOpenSuccess >= b.config.HalfOpenProbes {
    91  			b.toClosed()
    92  		}
    93  	case breakerClosed:
    94  		// closed の集計は状態遷移時にリセットされるため、競合した遅延 record も無害に計上できる。
    95  		b.requests++
    96  		if !success {
    97  			b.failures++
    98  		}
    99  		if b.shouldOpen() {
   100  			b.toOpen(now)
   101  		}
   102  	default: // breakerOpen（allow が遷移を司るため記録は無視）
   103  	}
   104  }
