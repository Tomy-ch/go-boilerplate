    82  // Lookup は、外部 API から郵便番号に対応する住所候補を取得し、境界 DTO へ変換して返します。
    83  // 外部レスポンスの型・エラーは本メソッド内で完結し、外側へは boundary.Candidate と apperror のみ露出します。
    84  func (g *gateway) Lookup(ctx context.Context, postalCode string) ([]*boundary.Candidate, error) {
    85  	ctx, endSpan := g.tracer.Start(ctx)
    86  	defer endSpan()
    87  
    88  	reqURL := fmt.Sprintf("%s/api/search?zipcode=%s", g.endpoint, url.QueryEscape(postalCode))
    89  
    90  	resp, err := g.client.Do(ctx, httpclient.NewRequest(httpclient.MethodGet(), downstream, reqURL))
    91  	if err != nil {
    92  		return nil, err // substrate が apperror sentinel へ写像済み
    93  	}
    94  
    95  	var body zipcloudResponse
    96  	if uerr := json.Unmarshal(resp.Body, &body); uerr != nil {
    97  		return nil, xerrors.Wrap(apperror.ErrUnavailable, "invalid address lookup response: "+uerr.Error())
    98  	}
    99  	if body.Status != zipcloudSuccessStatus {
   100  		msg := ""
   101  		if body.Message != nil {
   102  			msg = *body.Message
   103  		}
   104  		return nil, xerrors.Wrap(apperror.ErrUnavailable,
   105  			fmt.Sprintf("address lookup returned status %d: %s", body.Status, msg))
   106  	}
   107  
   108  	// results が null（該当なし）でも障害ではないため、空スライス + nil error で返す（degrade 対象外）。
   109  	results := body.Results
   110  	if len(results) > maxCandidates {
   111  		results = results[:maxCandidates]
   112  	}
   113  	candidates := make([]*boundary.Candidate, 0, len(results))
   114  	for _, r := range results {
   115  		candidates = append(candidates, &boundary.Candidate{
   116  			PrefectureName: r.Address1,
   117  			City:           r.Address2,
   118  			Town:           r.Address3,
   119  		})
   120  	}
   121  	return candidates, nil
   122  }
