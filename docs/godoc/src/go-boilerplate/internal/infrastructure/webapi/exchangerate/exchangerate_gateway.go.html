    58  func (g *gateway) GetRate(ctx context.Context, base, quote string) (*boundary.Rate, error) {
    59  	ctx, endSpan := g.tracer.Start(ctx)
    60  	defer endSpan()
    61  
    62  	reqURL := fmt.Sprintf("%s/rates?base=%s&quote=%s",
    63  		g.endpoint, url.QueryEscape(base), url.QueryEscape(quote))
    64  
    65  	resp, err := g.client.Do(ctx, httpclient.NewRequest(httpclient.MethodGet(), downstream, reqURL))
    66  	if err != nil {
    67  		return nil, err // substrate が apperror sentinel へ写像済み
    68  	}
    69  
    70  	var body rateResponse
    71  	if uerr := json.Unmarshal(resp.Body, &body); uerr != nil {
    72  		return nil, xerrors.Wrap(apperror.ErrUnavailable, "invalid exchangerate response: "+uerr.Error())
    73  	}
    74  	if body.Rate <= 0 {
    75  		return nil, xerrors.Wrap(apperror.ErrUnavailable, "exchangerate response has non-positive rate")
    76  	}
    77  
    78  	return &boundary.Rate{Base: base, Quote: quote, Value: body.Rate}, nil
    79  }
