    26  // NewQueueStatsProvider は、SQS の滞留量を取得する QueueStatsProvider を生成します。

    32  // QueueStats は、source queue と（DLQURL があれば）DLQ の滞留量を取得します。
    33  // DLQURL が空の場合は DLQ の取得をスキップし、DLQ は nil のままにします。
    34  func (p *statsProvider) QueueStats(ctx context.Context) (worker.QueueStats, error) {
    35  	ctx, endSpan := p.tracer.Start(ctx)
    36  	defer endSpan()
    37  
    38  	source, err := p.queueDepth(ctx, p.cfg.QueueURL)
    39  	if err != nil {
    40  		return worker.QueueStats{}, err
    41  	}
    42  	stats := worker.QueueStats{Source: source}
    43  
    44  	if p.cfg.DLQURL != "" {
    45  		dlq, err := p.queueDepth(ctx, p.cfg.DLQURL)
    46  		if err != nil {
    47  			return worker.QueueStats{}, err
    48  		}
    49  		stats.DLQ = &dlq
    50  	}
    51  
    52  	return stats, nil
    53  }

    55  // queueDepth は、GetQueueAttributes で approximate な滞留量を取得し QueueDepth へ正規化します。
    56  func (p *statsProvider) queueDepth(ctx context.Context, queueURL string) (worker.QueueDepth, error) {
    57  	out, err := p.api.GetQueueAttributes(ctx, &sqs.GetQueueAttributesInput{
    58  		QueueUrl: aws.String(queueURL),
    59  		AttributeNames: []types.QueueAttributeName{
    60  			types.QueueAttributeNameApproximateNumberOfMessages,
    61  			types.QueueAttributeNameApproximateNumberOfMessagesNotVisible,
    62  			types.QueueAttributeNameApproximateNumberOfMessagesDelayed,
    63  		},
    64  	})
    65  	if err != nil {
    66  		return worker.QueueDepth{}, normalizeError(err)
    67  	}
    68  
    69  	return worker.QueueDepth{
    70  		Visible:  parseApproxCount(out.Attributes[string(types.QueueAttributeNameApproximateNumberOfMessages)]),
    71  		InFlight: parseApproxCount(out.Attributes[string(types.QueueAttributeNameApproximateNumberOfMessagesNotVisible)]),
    72  		Delayed:  parseApproxCount(out.Attributes[string(types.QueueAttributeNameApproximateNumberOfMessagesDelayed)]),
    73  	}, nil
    74  }
    75  
    76  // parseApproxCount は、SQS の attribute 値を int64 へ変換します。
    77  // attribute 欠落（空文字）や parse 不能は、滞留傾向の gauge として 0 件扱いにします。
    78  func parseApproxCount(s string) int64 {
    79  	n, err := strconv.ParseInt(s, 10, 64)
    80  	if err != nil {
    81  		return 0
    82  	}
    83  	return n
    84  }
