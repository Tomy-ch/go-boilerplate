    66  // Publish は、メッセージを SendMessage でキューへ送ります。
    67  // outbox の message_id と伝搬対象ヘッダは MessageAttributes へ載せます。本文は payload そのままで、
    68  // 受信側が本文を解釈せずに冪等キーを取り出せるようにするためです。
    69  // 上限超過は送信前に ErrTooManyAttributes として返します。超過分を落とすと、どのヘッダが残るかが
    70  // map の反復順に左右され、traceparent を失ったことにも気付けないためです。
    71  // 送信失敗は apperror へ正規化して返し、再送は relay の次 poll が担います（at-least-once）。
    72  func (p *publisher) Publish(ctx context.Context, m boundary.Message) error {
    73  	ctx, endSpan := p.tracer.Start(ctx)
    74  	defer endSpan()
    75  
    76  	// SQS は空値の属性を拒むため、送れないメッセージは送信前に弾く。ヘッダと違って落として済ませられない。
    77  	// 種別を欠いたメッセージは受信側が自分宛かを判定できず、黙って読み捨てられる。
    78  	if m.EventType == "" {
    79  		return xerrors.Wrap(ErrMissingEventType, "event type must not be empty")
    80  	}
    81  
    82  	attrs := p.messageAttributes(m)
    83  	if len(attrs) > maxMessageAttributes {
    84  		return xerrors.Wrap(ErrTooManyAttributes,
    85  			fmt.Sprintf("%d attributes exceed the SQS limit of %d", len(attrs), maxMessageAttributes))
    86  	}
    87  
    88  	_, err := p.api.SendMessage(ctx, &sqs.SendMessageInput{
    89  		QueueUrl:          aws.String(p.cfg.QueueURL),
    90  		MessageBody:       aws.String(string(m.Payload)),
    91  		MessageAttributes: attrs,
    92  	})
    93  	return normalizeError(err)
    94  }

    96  // messageAttributes は、message_id / event_type と伝搬対象ヘッダを MessageAttributes へ組み立てます。
    97  // 空値のヘッダは SQS が InvalidParameterValue で拒否するため落とします。
    98  func (p *publisher) messageAttributes(m boundary.Message) map[string]types.MessageAttributeValue {
    99  	attrs := make(map[string]types.MessageAttributeValue, len(m.Headers)+reservedAttributes)
   100  	attrs[AttrMessageID] = types.MessageAttributeValue{
   101  		DataType:    aws.String(attrTypeString),
   102  		StringValue: aws.String(m.MessageID.String()),
   103  	}
   104  	attrs[AttrEventType] = types.MessageAttributeValue{
   105  		DataType:    aws.String(attrTypeString),
   106  		StringValue: aws.String(m.EventType),
   107  	}
   108  
   109  	for k, v := range m.Headers {
   110  		// emit 側でも同じ判定を行いますが、emit を経由せず INSERT された行に対する egress 境界での防御です。
   111  		if httpheader.IsSensitive(k) {
   112  			continue
   113  		}
   114  		// 同名ヘッダに outbox 由来の値を上書きさせません。受信側の選別が本文と食い違うためです。
   115  		if k == AttrMessageID || k == AttrEventType || v == "" {
   116  			continue
   117  		}
   118  		attrs[k] = types.MessageAttributeValue{
   119  			DataType:    aws.String(attrTypeString),
   120  			StringValue: aws.String(v),
   121  		}
   122  	}
   123  	return attrs
   124  }
