    49  // Receive は、ReceiveMessage で long-poll し、broker 非依存の Message へ変換して返します。
    50  func (c *consumer) Receive(ctx context.Context, maxMessages int) ([]worker.Message, error) {
    51  	ctx, endSpan := c.tracer.Start(ctx)
    52  	defer endSpan()
    53  
    54  	limit := maxMessages
    55  	if c.cfg.MaxMessages > 0 && limit > int(c.cfg.MaxMessages) {
    56  		limit = int(c.cfg.MaxMessages)
    57  	}
    58  	if limit > maxSQSBatch {
    59  		limit = maxSQSBatch
    60  	}
    61  	if limit < 1 {
    62  		limit = 1
    63  	}
    64  	n := int32(limit) // limit は 1..maxSQSBatch(10) に丸め済み
    65  
    66  	out, err := c.api.ReceiveMessage(ctx, &sqs.ReceiveMessageInput{
    67  		QueueUrl:            aws.String(c.cfg.QueueURL),
    68  		MaxNumberOfMessages: n,
    69  		WaitTimeSeconds:     c.cfg.WaitTimeSeconds,
    70  		VisibilityTimeout:   c.cfg.VisibilityTimeout,
    71  		MessageSystemAttributeNames: []types.MessageSystemAttributeName{
    72  			types.MessageSystemAttributeNameApproximateReceiveCount,
    73  			types.MessageSystemAttributeNameMessageGroupId,
    74  		},
    75  		MessageAttributeNames: []string{"All"},
    76  	})
    77  	if err != nil {
    78  		return nil, normalizeError(err)
    79  	}
    80  
    81  	msgs := make([]worker.Message, 0, len(out.Messages))
    82  	for _, m := range out.Messages {
    83  		msgs = append(msgs, toMessage(m))
    84  	}
    85  	return msgs, nil
    86  }

   144  // visibilitySeconds は、Duration を SQS の可視性秒数（int32）へ丸めます。
   145  // サブ秒は切り捨てると 0 秒＝即時再配送になり処理中メッセージが重複配送されるため、
   146  // 切り上げたうえで 1 秒を下限とします。
   147  func visibilitySeconds(d time.Duration) int32 {
   148  	return max(1, int32(math.Ceil(d.Seconds())))
   149  }

   151  // toMessage は、SQS のメッセージを broker 非依存の Message へ正規化します。
   152  //   - MessageAttributes（traceparent 等）→ Attributes
   153  //   - ReceiptHandle → Attributes の予約キー（engine は解釈しない）
   154  //   - ApproximateReceiveCount → ReceiveCount / MessageGroupId → PartitionKey
   155  func toMessage(m types.Message) worker.Message {
   156  	attrs := make(map[string]string, len(m.MessageAttributes)+1)
   157  	for k, v := range m.MessageAttributes {
   158  		if v.StringValue != nil {
   159  			attrs[k] = *v.StringValue
   160  		}
   161  	}
   162  	if m.ReceiptHandle != nil {
   163  		attrs[worker.AttrReceiptHandle] = *m.ReceiptHandle
   164  	}
   165  
   166  	receiveCount := 0
   167  	if s, ok := m.Attributes[string(types.MessageSystemAttributeNameApproximateReceiveCount)]; ok {
   168  		receiveCount, _ = strconv.Atoi(s)
   169  	}
   170  
   171  	return worker.Message{
   172  		ID:           aws.ToString(m.MessageId),
   173  		Body:         []byte(aws.ToString(m.Body)),
   174  		Attributes:   attrs,
   175  		ReceiveCount: receiveCount,
   176  		PartitionKey: m.Attributes[string(types.MessageSystemAttributeNameMessageGroupId)],
   177  	}
   178  }
