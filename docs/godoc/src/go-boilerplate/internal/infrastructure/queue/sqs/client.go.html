    12  // ClientConfig は、SQS クライアントの接続設定です。
    13  // endpoint / 資格情報の差し替えだけで ElasticMQ・LocalStack・本番 SQS のいずれにも接続できます。
    14  type ClientConfig struct {
    15  	// Endpoint は、SQS 互換エンドポイントです（例 ElasticMQ: "http://elasticmq:9324"）。
    16  	// 空の場合は SDK 既定のエンドポイント解決に委ねます（本番 AWS SQS 等）。
    17  	Endpoint string
    18  	// Region は、署名に用いるリージョンです。
    19  	Region string
    20  	// AccessKeyID / SecretAccessKey は、明示注入する静的資格情報です。両方空なら
    21  	// SDK 既定の credential chain（IAM ロール等）へ委ねます。詳細は awsclient.Resolve を参照。
    22  	AccessKeyID     string
    23  	SecretAccessKey string
    24  	// HTTPClient は、SDK が API 呼び出しに使う HTTP クライアントです。SSRF ガード付きの実装を DI が
    25  	// 注入します。nil を渡すと SDK 既定のトランスポートになり、ガードを素通りします。
    26  	HTTPClient aws.HTTPClient
    27  }

    29  // NewClient は、設定から SQS クライアントを生成します。
    30  // 資格情報を解決できない場合はエラーを返し、認証エラーが最初の送受信まで隠れないようにします。
    31  func NewClient(ctx context.Context, cfg ClientConfig) (*sqs.Client, error) {
    32  	awsCfg, err := awsclient.Resolve(ctx, awsclient.Config{
    33  		Region:          cfg.Region,
    34  		AccessKeyID:     cfg.AccessKeyID,
    35  		SecretAccessKey: cfg.SecretAccessKey,
    36  		HTTPClient:      cfg.HTTPClient,
    37  	})
    38  	if err != nil {
    39  		return nil, err
    40  	}
    41  
    42  	return sqs.NewFromConfig(awsCfg, func(o *sqs.Options) {
    43  		if cfg.Endpoint != "" {
    44  			o.BaseEndpoint = aws.String(cfg.Endpoint)
    45  		}
    46  	}), nil
    47  }
