     1  // Package awsclient は、AWS SDK v2 クライアントに共通する組み立て — 資格情報の解決と、
     2  // API 呼び出し用 / 資格情報解決用に分けた HTTP クライアントの割り当て — を提供します。
     3  // SQS・S3 などの adapter はこの結果の aws.Config からサービスクライアントを生成します。
     4  package awsclient

     6  import (
     7  	"context"
     8  	"time"

    18  // resolveTimeout は、起動時の資格情報の解決に許す上限時間です。
    19  // chain は複数のプロバイダを順に試すため、個々のプロバイダの既定タイムアウトだけでは全体の上限が
    20  // 決まりません。応答しない解決先で起動が止まったままにならないよう、全体へ締切を与えます。
    21  const resolveTimeout = 10 * time.Second
    22  
    23  // ErrInvalidCredentials は、資格情報の指定が解決できる形になっていないことを示すエラーです。
    24  var ErrInvalidCredentials = xerrors.Wrap(apperror.ErrInvalidArgument, "invalid aws credentials")
    25  
    26  // Config は、AWS サービスクライアントを組み立てるための共通設定です。
    27  type Config struct {
    28  	// Region は、署名に用いるリージョンです。空の場合は SDK 既定の解決（環境変数・共有プロファイル）に委ねます。
    29  	Region string
    30  	// AccessKeyID / SecretAccessKey は、明示注入する静的資格情報です。
    31  	// 両方空なら SDK 既定の credential chain（環境変数・共有プロファイル・web identity・
    32  	// コンテナ / IMDS）へ委ね、両方非空ならその chain を上書きします。
    33  	AccessKeyID     string
    34  	SecretAccessKey string
    35  	// HTTPClient は、サービス API の呼び出しに使う HTTP クライアントです。
    36  	// SSRF ガード付きの実装を DI が注入します。nil を渡すと SDK 既定のトランスポートになり、
    37  	// ガードを素通りします。
    38  	HTTPClient aws.HTTPClient
    39  }

    41  // Resolve は、資格情報を解決した aws.Config を返します。
    42  //
    43  // 資格情報は「明示注入か chain か」を設定で選ばせません。SDK 既定の chain は静的資格情報
    44  // （環境変数プロバイダ）を自身の一部として含むため、static / chain の二者択一は SDK の標準形に
    45  // 存在せず、判別子を足すのは標準が決めている所へノブを置くことになるためです。明示注入は
    46  // chain の上書きとして扱います。
    47  //
    48  // 解決可否は起動時に一度だけ確かめます。誤設定のまま起動すると、最初の API 呼び出しまで
    49  // 認証エラーが顕在化しません。確認には resolveTimeout の締切を与え、応答しない解決先で
    50  // 起動が止まったままにならないようにします。
    51  //
    52  // cfg.HTTPClient（SSRF ガード付き）はサービス API の呼び出しにだけ割り当て、資格情報の解決は
    53  // SDK 既定のトランスポートで行います。IMDS（169.254.169.254）と ECS の
    54  // task metadata（169.254.170.2）は link-local にあり、ガードは link-local を常に拒否するため、
    55  // 同じクライアントを共有すると EC2 / ECS のロール運用だけが解決不能になります。守る対象が
    56  // 違う — ガードが防ぐのは外部サービスへの egress であって、自身の実行基盤への資格情報の
    57  // 問い合わせではありません。
    58  //
    59  // 適用外になるのは metadata 経由だけではなく、STS の web identity 交換や SSO も含む credential
    60  // chain の通信すべてです。宛先 IP の検査も、ガードが行う proxy 環境変数の無効化も効きません。
    61  // chain の接続先を差し替えられる（AWS_EC2_METADATA_SERVICE_ENDPOINT / HTTPS_PROXY 等）のは
    62  // プロセスの環境変数を書ける主体だけで、その主体は資格情報そのものにも手が届きます。
    63  func Resolve(ctx context.Context, cfg Config) (aws.Config, error) {
    64  	if err := ctx.Err(); err != nil {
    65  		return aws.Config{}, xerrors.Wrap(ErrInvalidCredentials, err.Error())
    66  	}
    67  
    68  	opts := []func(*awsconfig.LoadOptions) error{awsconfig.WithRegion(cfg.Region)}
    69  
    70  	switch {
    71  	case cfg.AccessKeyID != "" && cfg.SecretAccessKey != "":
    72  		opts = append(opts, awsconfig.WithCredentialsProvider(
    73  			awscreds.NewStaticCredentialsProvider(cfg.AccessKeyID, cfg.SecretAccessKey, ""),
    74  		))
    75  	case cfg.AccessKeyID != "" || cfg.SecretAccessKey != "":
    76  		// 片方だけの指定は chain 委譲とも明示注入とも読めない。SDK は access key ID さえあれば
    77  		// 署名を作ってしまい、認証エラーが呼び出しまで出ないため、ここで弾く。
    78  		return aws.Config{}, xerrors.Wrap(ErrInvalidCredentials,
    79  			"access key id and secret access key must be set together, or both left empty to use the default chain")
    80  	}
    81  
    82  	awsCfg, err := awsconfig.LoadDefaultConfig(ctx, opts...)
    83  	if err != nil {
    84  		return aws.Config{}, xerrors.Wrap(ErrInvalidCredentials, err.Error())
    85  	}
    86  
    87  	retrieveCtx, cancel := context.WithTimeout(ctx, resolveTimeout)
    88  	defer cancel()
    89  	if _, err := awsCfg.Credentials.Retrieve(retrieveCtx); err != nil {
    90  		return aws.Config{}, xerrors.Wrap(ErrInvalidCredentials, err.Error())
    91  	}
    92  
    93  	awsCfg.HTTPClient = cfg.HTTPClient
    94  	return awsCfg, nil
    95  }
