    10  // Meta は、エラー発生箇所がレスポンス向けに動的に付与できるメタ情報です。
    11  // [NewMeta] で構築します。空の項目は transport 層(controller)の既定値にフォールバックします。
    12  // apperror センチネルによる分類(HTTP ステータス解決)には一切影響しません。
    13  type Meta struct {
    14  	code    string
    15  	message string
    16  	details []string
    17  }
    18  
    19  // MetaError は、Meta を運ぶラッパーエラーです。
    20  // Unwrap により元エラーのセンチネル分類([xerrors.Is] / [IsAppError])を保持します。
    21  type MetaError struct {
    22  	meta Meta
    23  	err  error
    24  }
    25  
    26  // NewMeta は、機械可読なエラーコード code と詳細識別子 details から Meta を構築します。
    27  // code が空の場合はステータス由来の既定コードが使われます。details はレスポンスに
    28  // そのまま公開されるため、識別子(例: 不正フィールド名)のみを入れ、理由文や入力値
    29  // そのものを入れてはいけません。details は防御的コピーされます。
    30  // 利用者向け文言は持ちません(文言の正は controller のカタログ。上書きは [Meta.WithMessage])。
    31  func NewMeta(code string, details ...string) Meta {
    32  	return Meta{code: code, details: append([]string(nil), details...)}
    33  }
    34  
    35  // WithMessage は、利用者向け文言を上書きした Meta のコピーを返します。
    36  // 文言の正は controller のカタログにあるため、呼び出しは controller 層に限ります
    37  // (domain / usecase では使わない)。
    38  func (m Meta) WithMessage(message string) Meta {
    39  	m.message = message
    40  	return m
    41  }

    44  func (m Meta) Code() string {
    45  	return m.code
    46  }
    47  
    48  // Message は利用者向け文言を返します。
    49  func (m Meta) Message() string {
    50  	return m.message
    51  }

    53  // Details は詳細識別子の防御的コピーを返します。
    54  func (m Meta) Details() []string {
    55  	if m.details == nil {
    56  		return nil
    57  	}
    58  	return append([]string(nil), m.details...)
    59  }

    61  // WithMeta は、err に meta を付与したエラーを返します。err が nil の場合は nil を返します。
    62  func WithMeta(err error, meta Meta) error {
    63  	if err == nil {
    64  		return nil
    65  	}
    66  	return &MetaError{meta: meta, err: err}
    67  }
    68  
    69  // WithDetails は、err に details のみを付与したエラーを返します。[WithMeta] の糖衣です。
    70  func WithDetails(err error, details ...string) error {
    71  	return WithMeta(err, NewMeta("", details...))
    72  }

    74  // MetaFrom は、err のチェーンから最も外側の [MetaError] の Meta を抽出します。
    75  // 多重に付与されている場合は外側が勝ちます。見つからない場合は ok=false を返します。
    76  func MetaFrom(err error) (Meta, bool) {
    77  	var metaErr *MetaError
    78  	if !xerrors.As(err, &metaErr) {
    79  		return Meta{}, false
    80  	}
    81  	return metaErr.Meta(), true
    82  }

    84  // Error は error インターフェースを実装します。メッセージは元エラーのまま変えません。
    85  func (e *MetaError) Error() string {
    86  	return e.err.Error()
    87  }

    89  // Unwrap は元エラーを返します。シグネチャは errors.Is / errors.As がチェーンを辿るための
    90  // 標準ライブラリ契約であり、改名するとセンチネル分類がこのラッパーを貫通できなくなります。
    91  func (e *MetaError) Unwrap() error {
    92  	return e.err
    93  }

    95  // Meta は付与されたメタ情報を返します。フィールドは非公開のため、取り出した値経由で
    96  // 内部状態を変更することはできません([Meta.Details] は防御的コピーを返します)。
    97  func (e *MetaError) Meta() Meta {
    98  	return e.meta
    99  }

   101  // Format は fmt.Formatter を実装し、元エラーの表現(スタックトレース含む)へ委譲します。
   102  // これがないと %+v(ログ出力)で元エラーのスタックトレースが失われます。
