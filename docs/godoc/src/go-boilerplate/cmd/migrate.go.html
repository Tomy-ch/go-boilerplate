    79  func buildMigrateInstance(database string) (climigrate.Migrator, error) {
    80  	if err := config.Load(); err != nil {
    81  		return nil, xerrors.Wrap(err, "failed to load config")
    82  	}
    83  	if database != "" {
    84  		restore, err := envutil.Override("DB_NAME", database)
    85  		if err != nil {
    86  			return nil, xerrors.Wrap(err, "failed to override DB_NAME env var")
    87  		}
    88  		defer restore()
    89  	}
    90  	cfg, err := config.New()
    91  	if err != nil {
    92  		return nil, xerrors.Wrap(err, "failed to build config")
    93  	}
    94  	dbCfg := config.NewDatabaseConfig(cfg)
    95  	osCfg := config.NewOperatingSystemConfig(cfg)
    96  
    97  	src, err := iofs.New(root.FS, migrateFilePlace)
    98  	if err != nil {
    99  		return nil, xerrors.Wrap(err, "failed to create migration source")
   100  	}
   101  
   102  	m, err := migrate.NewWithSourceInstance("iofs", src, driver.DSNWithTimeZoneString(dbCfg, osCfg))
   103  	if err != nil {
   104  		return nil, xerrors.Wrap(err, "failed to create migrate instance")
   105  	}
   106  	return m, nil
   107  }
