    41  func dbSeedRun(database string) error {
    42  	logger := logging.NewJSONLogger(logging.LevelInfo(), logging.LevelError(), nil)
    43  
    44  	vars, err := seedVars(logger, database)
    45  	if err != nil {
    46  		return err
    47  	}
    48  
    49  	if err := seed.RunDBSeed(logger, fs.OS{}, database, vars, openSeedDB); err != nil {
    50  		return err
    51  	}
    52  
    53  	endpoint, put, err := openSeedObjectStorage(logger, database)
    54  	if err != nil {
    55  		return err
    56  	}
    57  
    58  	return seed.RunObjectSeed(logger, fs.OS{}, endpoint, put)

    75  // openSeedObjectStorage は、seed 用設定を読み込みオブジェクトストレージ実装を組み立てる実依存の口です。
    76  // 併せて接続先エンドポイントを返し、実環境のバケットへ投入しない判断を呼び出し先へ委ねます。
    77  func openSeedObjectStorage(logger logging.Logger, database string) (string, seed.PutObjectFunc, error) {
    78  	cfg, err := newConfigForSeed(logger, database)
    79  	if err != nil {
    80  		return "", nil, err
    81  	}
    82  	osCfg := config.NewObjectStorageConfig(cfg)
    83  	// private 網の可否はサーバ本体と同じ env 基準で決める。seed は staging でも実行されうるため、
    84  	// ローカル前提で常時許可にするとその環境だけ SSRF ガードが緩くなる。
    85  	appEnv := config.NewApplicationConfig(cfg).Env()
    86  	storage, err := objectstorageinfra.New(
    87  		context.Background(),
    88  		osCfg,
    89  		observability.NewDisabledOutboundHTTPClient(config.IsLocalClassEnv(appEnv)),
    90  		observability.NewDisabledTracerFactory(),
    91  	)
    92  	if err != nil {
    93  		return "", nil, err
    94  	}
    95  
    96  	put := func(ctx context.Context, obj seed.ObjectToPut) error {
    97  		_, perr := storage.Put(ctx, objectstorage.PutObject{
    98  			Key:         obj.Key,
    99  			Body:        obj.Body,
   100  			ContentType: obj.ContentType,
   101  		})
   102  		return perr
   103  	}
   104  
   105  	return osCfg.Endpoint(), put, nil
   106  }

   122  func newConfigForSeed(logger logging.Logger, database string) (*config.Config, error) {
   123  	cfg, err := newCLIConfig(database)
