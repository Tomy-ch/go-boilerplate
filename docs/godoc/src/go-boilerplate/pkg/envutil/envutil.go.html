    21  func Override(key, value string) (func(), error) {
    22  	prev, existed := os.LookupEnv(key)
    23  	if err := os.Setenv(key, value); err != nil {
    24  		return func() {}, fmt.Errorf("failed to set env %q: %w", key, err)
    25  	}
    26  	restore := func() {
    27  		if existed {
    28  			_ = os.Setenv(key, prev)
    29  			return
    30  		}
    31  		_ = os.Unsetenv(key)
    32  	}
    33  	return restore, nil
    34  }
