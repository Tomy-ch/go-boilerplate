    10  // Exponential は、指数バックオフの設定です。
    11  type Exponential struct {
    12  	// Initial は、attempt=0 のときの待機時間です。
    13  	Initial time.Duration
    14  	// Max は、待機時間の上限です（0 以下なら上限なし）。
    15  	Max time.Duration
    16  	// Multiplier は、attempt ごとの倍率です（1 未満は 1 として扱います）。
    17  	Multiplier float64
    18  }
    19  
    20  // Duration は、attempt 回目（0 起算）の待機時間を返します。
    21  // Initial * Multiplier^attempt を計算し、Max が正なら Max で上限を設けます。
    22  // Initial が 0 以下なら常に 0 を返し、attempt が負の場合は 0 として扱います。
    23  func (e Exponential) Duration(attempt int) time.Duration {
    24  	if e.Initial <= 0 {
    25  		return 0
    26  	}
    27  	if attempt < 0 {
    28  		attempt = 0
    29  	}
    30  
    31  	mult := e.Multiplier
    32  	if mult < 1 {
    33  		mult = 1
    34  	}
    35  
    36  	d := float64(e.Initial)
    37  	for range attempt {
    38  		d *= mult
    39  		if e.Max > 0 && d >= float64(e.Max) {
    40  			return e.Max
    41  		}
    42  	}
    43  	if e.Max > 0 && d > float64(e.Max) {
    44  		return e.Max
    45  	}
    46  	// Max 上限なし（Max<=0）かつ高 attempt で d が +Inf / int64 範囲外になりうる。
    47  	// time.Duration(+Inf) は負値になり cooldown が即発火するため、MaxInt64 で頭打ちにする。
    48  	if d > float64(math.MaxInt64) {
    49  		return time.Duration(math.MaxInt64)
    50  	}
    51  	return time.Duration(d)
    52  }
