    15  // fakeSleeper は、Sleeper のテスト用 fake です。
    16  // 呼び出し回数を記録し、待機は即時化します。errAt 指定時はその回数目で error を返します。
    17  type fakeSleeper struct {
    18  	calls int
    19  	errAt int   // 0 なら常に nil。N なら N 回目の Sleep で sleepErr を返す。
    20  	err   error // errAt 到達時に返す error。
    21  }

    23  func (s *fakeSleeper) Sleep(_ context.Context, _ time.Duration) error {
    24  	s.calls++
    25  	if s.errAt > 0 && s.calls == s.errAt {
    26  		return s.err
    27  	}
    28  	return nil
    29  }
