    14  // 呼び出し回数を記録し、待機は即時化します。errAt 指定時はその回数目で error を返します。
    15  type fakeSleeper struct {
    16  	calls int
    17  	errAt int   // 0 なら常に nil。N なら N 回目の Sleep で sleepErr を返す。
    18  	err   error // errAt 到達時に返す error。

    21  func (s *fakeSleeper) Sleep(_ context.Context, _ time.Duration) error {
    22  	s.calls++
    23  	if s.errAt > 0 && s.calls == s.errAt {
    24  		return s.err
    25  	}
    26  	return nil
    27  }
