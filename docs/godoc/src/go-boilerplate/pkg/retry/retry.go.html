     1  // Package retry は、失敗分類を消費する有限リトライの行動層です。
     2  //
     3  // classify → bounded attempts → backoff + full jitter → deadline-aware という
     4  // 共通ループを 1 度だけ実装し、tx リトライや外部 HTTP の retry が共有します。
     5  // バックオフの待機時間は呼び出し側が [Policy.Backoff]（attempt → 基本待機時間）で
     6  // 与え、本パッケージはそこへ full jitter（[Full]）を重畳します。乱数（math/rand/v2）
     7  // 依存を本パッケージ側へ閉じることで、待機時間算出を担う純粋な pkg/backoff の
     8  // 「現在時刻や乱数に依存しない」純粋性を保ちます。
     9  //
    10  // pkg は相互独立（pkg→pkg import 禁止）のため、Backoff は pkg/backoff の型ではなく
    11  // 関数値として受け取ります。
    12  package retry
    13  
    14  import (
    15  	"context"
    16  	"time"
    17  )
    18  
    19  // Sleeper は、リトライ間の待機を抽象化するインターフェースです。
    20  // ctx のキャンセル / deadline を尊重し、待機が打ち切られた場合は error を返します。
    21  //
    22  // 本パッケージは internal 非依存のため独自に定義します。Sleep(ctx context.Context, d time.Duration) error を持つ任意の型を注入できます。
    23  type Sleeper interface {
    24  	// Sleep は、d 経過まで待機します。ctx 打ち切り時は error を返します。
    25  	Sleep(ctx context.Context, d time.Duration) error
    26  }
    27  
    28  // Policy は、有限リトライの設定です。
    29  type Policy struct {
    30  	// MaxAttempts は、fn の最大試行回数です。1 未満は 1 として扱います（最低 1 回試行）。
    31  	MaxAttempts int
    32  	// Backoff は、attempt（0 起算）に対する jitter 適用前の基本待機時間を返します。
    33  	// nil の場合は待機 0（即時リトライ）。実待機は本値に [Full] を重畳したものです。
    34  	// 呼び出し側は backoff.Exponential.Duration 等をそのまま渡せます。
    35  	Backoff func(attempt int) time.Duration
    36  }
    37  
    38  // Do は、fn を有限リトライで実行します。
    39  //
    40  // fn の返した error が isRetryable を満たす限り、policy.MaxAttempts まで再試行し、
    41  // 試行間は policy.Backoff（+ full jitter）だけ sleeper で待機します。最後に観測した
    42  // error（成功時は nil）を返します。
    43  //
    44  //   - fn が nil を返した、isRetryable が false、または最終試行に達した場合は即座に返します。
    45  //   - sleeper.Sleep が error を返した（ctx 打ち切り）場合は、sleep の error ではなく
    46  //     直前の fn の error を返します（リトライ対象だった元の失敗を呼び出し側へ伝えるため）。
    47  //
    48  // isRetryable は fn が返した非 nil error に対してのみ呼ばれます。isRetryable 自体が nil の
    49  // 場合は全ての error をリトライ不可として扱い、最初の error で即座に返します。
    50  func Do(
    51  	ctx context.Context,
    52  	sleeper Sleeper,
    53  	policy Policy,
    54  	isRetryable func(error) bool,
    55  	fn func(context.Context) error,
    56  ) error {
    57  	if isRetryable == nil {
    58  		isRetryable = func(error) bool { return false }
    59  	}
    60  	attempts := max(1, policy.MaxAttempts)
    61  
    62  	var err error
    63  	for attempt := 1; attempt <= attempts; attempt++ {
    64  		err = fn(ctx)
    65  		if err == nil || !isRetryable(err) || attempt == attempts {
    66  			return err
    67  		}
    68  
    69  		var base time.Duration
    70  		if policy.Backoff != nil {
    71  			base = policy.Backoff(attempt - 1)
    72  		}
    73  		if serr := sleeper.Sleep(ctx, Full(base)); serr != nil {
    74  			return err
    75  		}
    76  	}
    77  	return err
    78  }
