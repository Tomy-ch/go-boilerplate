type AddressCandidate ¶

AddressCandidate 郵便番号 lookup による住所候補 1 件です。都道府県名は prefectures マスタで解決し、 解決できた場合のみ prefectureId を埋めます（解決不能時は null で、市区町村・町域の部分補完は生かします）。 

type AddressCandidate struct {
    // City 市区町村
    City string `json:"city"`

    // PrefectureId 都道府県ID。県名を prefectures マスタで完全一致解決できなかった場合は null。
    PrefectureId *openapi_types.UUID `json:"prefectureId"`

    // PrefectureName 都道府県名（外部 lookup が返すフル表記、例 "東京都"）
    PrefectureName string `json:"prefectureName"`

    // Town 町域
    Town string `json:"town"`
}

type AddressCandidatesResponse ¶

AddressCandidatesResponse 郵便番号 lookup による住所候補のレスポンスです。 外部 lookup サービスの障害時は 503 ではなく `candidates: []` + `isFallback: true` で degrade し、 登録フローを止めません（外部が落ちても縦一本のジャーニーを進めるためのフォールバック）。 

type AddressCandidatesResponse struct {
    // Candidates 住所候補の一覧。該当なし・degrade 時は空配列。
    Candidates []AddressCandidate `json:"candidates"`

    // IsFallback lookup 機構が機能しなかった（外部不通・不正応答）場合に true。true のときフロントは全項目手入力へ倒します。
    // 正常応答で候補が 0 件（該当なし）の場合や、候補はあるが県名解決に失敗した場合は false です。
    IsFallback bool `json:"isFallback"`
}
