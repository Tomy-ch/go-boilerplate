type CartMergeResponse struct {
    // Clamped 数量を合算した結果が 1 明細あたりの上限を超え、上限へ丸めた商品の ID。
    // 引き継ぎは失敗させないため、超過分は捨てて成功として返します。
    //
    //
    // Example: ["b1d4e0f2-3c5a-4b6d-8e7f-1a2b3c4d5e6f"]
    Clamped []openapi_types.UUID `json:"clamped"`

    // Dropped 明細数が上限を超えたため取り込まれなかった商品の ID。
    // 先にカートへ入っていたものを優先して残し、後から来たものを切り捨てます。
    //
    //
    // Example: []
    Dropped []openapi_types.UUID `json:"dropped"`
}
