type PurchaseDetailItemResponse ¶

PurchaseDetailItemResponse 購入詳細（GET）専用の購入明細レスポンス。productName は products との結合で解決した現在の商品名です。 unitPrice は購入時点の単価スナップショットです。 

type PurchaseDetailItemResponse struct {
    // ProductId 商品ID
    ProductId openapi_types.UUID `json:"productId"`

    // ProductName 商品名（products との結合で解決した現在名）
    ProductName string `json:"productName"`

    // Quantity 購入数量
    Quantity int32 `json:"quantity"`

    // UnitPrice 購入時点の単価スナップショット。サブセント精度を保持する decimal 文字列（USD ドル）で表します（例 "800.00"）。
    // 購入成立後に商品価格が変わっても本値は不変です。
    UnitPrice string `json:"unitPrice"`
}

type PurchaseGetDetailResponse ¶

PurchaseGetDetailResponse 購入詳細（GET）のレスポンススキーマ。status は ID・名称を事前解決済みで、明細には商品名 （products との結合で解決した現在名）を含みます。paidAt は未支払いなら null、canceledAt は 未キャンセルなら null です。金額（subtotalAmount / taxAmount / shippingFee / totalAmount）は すべて USD セント単位の整数、明細 unitPrice は購入時点の単価スナップショット（USD ドルの decimal 文字列）です。 

type PurchaseGetDetailResponse struct {
    // CanceledAt キャンセル日時。未キャンセルの場合は null です。
    CanceledAt *time.Time `json:"canceledAt,omitempty"`

    // Code 購入コード（UUIDv7 文字列・一意）
    Code string `json:"code"`

    // Details 購入明細の配列（商品名込み）
    Details []PurchaseDetailItemResponse `json:"details"`

    // Id 購入ID
    Id openapi_types.UUID `json:"id"`

    // OrderedAt 注文日時
    OrderedAt time.Time `json:"orderedAt"`

    // PaidAt 支払日時。未支払いの場合は null です。
    PaidAt *time.Time `json:"paidAt,omitempty"`

    // ShippingFee 送料。USD セント単位の整数です。
    ShippingFee int64 `json:"shippingFee"`

    // Status 購入に紐づくステータス（ID と名称）。name は事前解決済みで、別途の名称解決 API 呼び出しは不要です。
    Status PurchaseStatusRef `json:"status"`

    // SubtotalAmount 小計。明細（単価×数量）の合計。USD セント単位の整数です。
    SubtotalAmount int64 `json:"subtotalAmount"`

    // TaxAmount 税額（国内消費税）。USD セント単位の整数です。
    TaxAmount int64 `json:"taxAmount"`

    // TotalAmount 合計（小計 + 税額 + 送料）。USD セント単位の整数です。
    TotalAmount int64 `json:"totalAmount"`

    // UserId 購入したユーザーのID
    UserId openapi_types.UUID `json:"userId"`
}
