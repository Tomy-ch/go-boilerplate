type CampaignResponse struct {
    // Code 正規化済みのキャンペーンコード。前後の空白を落とし、大文字に揃えた値です。
    // 受け取り時のコードも同じ正規化を経るため、大文字・小文字の違いは同じコードとして扱われます。
    //
    //
    // Example: WELCOME-2026
    Code string `json:"code"`

    // CouponExpiresAt 配るクーポンの有効期限。配布期間の終了より後である必要があります。
    // 期間の終わりぎわに受け取った利用者へ、その時点で既に失効しているクーポンを渡さないためです。
    //
    //
    // Example: 2026-12-31T14:59:59Z
    CouponExpiresAt time.Time `json:"couponExpiresAt"`

    // Discount クーポンの値引き。いくら引くかを表します。適用範囲（どの明細が対象か）とは独立した軸です。
    Discount CouponDiscount `json:"discount"`

    // EndsAt 配布期間の終了日時。この時刻ちょうどを含めて以降は受け取れません。
    //
    // Example: 2026-11-30T14:59:59Z
    EndsAt time.Time `json:"endsAt"`

    // Id キャンペーンID
    //
    // Example: 0193a1c0-0002-7000-8000-000000000003
    Id openapi_types.UUID `json:"id"`

    // IssuedCount 発行済み枚数。購入のキャンセルでクーポンが未使用へ戻っても減りません。
    // 配った事実は取り消されないためです。
    //
    //
    // Example: 42
    IssuedCount int32 `json:"issuedCount"`

    // MinPurchaseAmount 配るクーポンの最低購入金額。USD セント単位の整数です。条件が無い場合は null です。
    //
    // Example: 5000
    MinPurchaseAmount *int64 `json:"minPurchaseAmount"`

    // PerUserLimit 1 人あたり上限。1 人の利用者が受け取れる上限枚数です。
    //
    // Example: 1
    PerUserLimit int32 `json:"perUserLimit"`

    // Scope クーポンの適用範囲。どの明細が対象かを表します。値引き（いくら引くか）とは独立した軸です。
    Scope CouponScope `json:"scope"`

    StartsAt time.Time `json:"startsAt"`

type CouponDiscount ¶

CouponDiscount クーポンの値引き。いくら引くかを表します。適用範囲（どの明細が対象か）とは独立した軸です。 

type CouponDiscount struct {
    // Kind 値引きの決まり方。`flat` は金額を直接差し引く定額、`rate` は対象額に率を掛ける定率です。
    //
    //
    // Example: rate
    Kind CouponDiscountKind `json:"kind"`

    // MaxAmount 定率の値引きが 1 回に引ける額の上限。USD セント単位の整数です。
    // 定率（`rate`）にのみ意味を持ち、定額（`flat`）では常に null です。
    // 上限が無い場合も null で、そのときは対象額に率を掛けた額がそのまま引かれます。
    //
    //
    // Example: 2000
    MaxAmount *int64 `json:"maxAmount"`

    // Value 種別における値。定額なら差し引く金額、定率なら対象額に掛ける率を、
    // 正確な十進量を保つ decimal 文字列で表します（例 `"0.10"` は 10% 引き）。
    // 定額の金額は商品の price と同じ USD のドル建てで（例 `"5.00"` は 5 ドル引き）、
    // USD セント単位の整数で表す maxAmount とは桁が違います。
    // 定額で引く額は対象額が上限で、セント未満の桁は値引き額を求める時点で切り捨てられます
    // （四捨五入ではありません）。
    // JSON number は IEEE754 double として復元され精度を失うため、文字列で表現します。
    //
    //
    // Example: 0.10
    Value string `json:"value"`
}

type PostCampaignsSuspendParams struct {
    // IdempotencyKey 冪等キー。非冪等な変更操作（作成系など）で、同一キーによるリトライを重複実行なく
    // 安全にするための識別子。サーバは同一 (認証主体, キー) の再送を初回結果のリプレイとして扱う。
    // このヘッダを宣言する操作は、ハンドラが必ず idempotency.Run 経由で処理する契約（完全性テストで機械検証）。
    //
    // キーの一意性は認証主体ごとに判定し、全操作で共通です。再送の扱いは次のとおりです。
    // - 初回が成功していれば、同じキーと同じ本文の再送には初回の応答をそのまま返します。
    // - 同じキーを別の操作や別の本文に使うと 422（code は IDEMPOTENCY_KEY_REUSED）です。
    // - 失敗した応答（4xx / 5xx）はキーを消費しません。本文を直して同じキーで再送すれば、新しく実行されます。
    // - 初回の処理中に同じキーで再送すると 409（code は IDEMPOTENCY_KEY_IN_PROGRESS）です。時間をおいて再送してください。
    // - キーは最初の要求を受け付けてから少なくとも 24 時間保持します。保持期間を過ぎて掃除された後の再送は、新しい実行として扱います。
    // - 形式に違反するキーは 400 です。
    IdempotencyKey *IdempotencyKeyParam `json:"Idempotency-Key,omitempty"`
